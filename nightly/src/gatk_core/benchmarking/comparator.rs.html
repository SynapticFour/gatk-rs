4/// Roll-up statistics across many suite-to-suite benchmark comparisons.
5/// # Invariants
6/// Rates are fractions in `[0.0, 1.0]` when `total_comparisons > 0`.
7/// `meets_all_targets` uses 2× speedup and 2× memory reduction heuristics.
8/// # Ownership
9/// Plain scalars; serde clone.
10/// # Mutation
11/// Immutable output of [`SuiteComparator::compare_suites`].
12/// # Biological assumptions
13/// None (infrastructure).
14/// # Java equivalence
15/// None / Rust-native.
16#[derive(Debug, Clone, Serialize, Deserialize)]
17pub struct OverallStats {
18    pub total_comparisons: usize,
19    pub average_speedup: f64,
20    pub average_memory_reduction: f64,
21    pub average_overall_score: f64,
22    pub bitwise_identical_rate: f64,
23    pub functional_equivalent_rate: f64,
24    pub meets_all_targets: bool,
25}

27/// Comparison of two [`BenchmarkSuite`] instances with aggregated stats.
28/// # Invariants
29/// Pairs GATK and GATK-RS results by zip index; empty suites yield zeroed stats.
30/// # Ownership
31/// Owns [`OverallStats`]; cheap clone.
32/// # Mutation
33/// Immutable comparison product.
34/// # Biological assumptions
35/// None (infrastructure).
36/// # Java equivalence
37/// None / Rust-native.
38#[derive(Debug, Clone, Serialize, Deserialize)]
39pub struct SuiteComparison {
40    pub overall_stats: OverallStats,
41}

67    pub fn compare_suites(
68        &self,
69        gatk: &BenchmarkSuite,
70        gatk_rs: &BenchmarkSuite,
71    ) -> SuiteComparison {
72        let pairs = gatk
73            .gatk_results
74            .iter()
75            .zip(gatk_rs.gatk_rs_results.iter())
76            .collect::<Vec<_>>();
77        let total = pairs.len();
78        if total == 0 {
79            return SuiteComparison {
80                overall_stats: OverallStats {
81                    total_comparisons: 0,
82                    average_speedup: 0.0,
83                    average_memory_reduction: 0.0,
84                    average_overall_score: 0.0,
85                    bitwise_identical_rate: 0.0,
86                    functional_equivalent_rate: 0.0,
87                    meets_all_targets: false,
88                },
89            };
90        }
91
92        let mut speedups = Vec::new();
93        let mut mem_red = Vec::new();
94        let mut scores = Vec::new();
95        let mut functional = 0usize;
96        for (a, b) in pairs {
97            let c = ComparisonResult::new(a, b);
98            speedups.push(c.speedup);
99            mem_red.push(c.memory_reduction);
100            scores.push((c.speedup.min(4.0) / 4.0 + c.memory_reduction.min(4.0) / 4.0) * 50.0);
101            if c.both_succeeded {
102                functional += 1;
103            }
104        }
105        let avg = |v: &[f64]| v.iter().sum::<f64>() / v.len() as f64;
106        SuiteComparison {
107            overall_stats: OverallStats {
108                total_comparisons: total,
109                average_speedup: avg(&speedups),
110                average_memory_reduction: avg(&mem_red),
111                average_overall_score: avg(&scores),
112                bitwise_identical_rate: 0.0,
113                functional_equivalent_rate: functional as f64 / total as f64,
114                meets_all_targets: avg(&speedups) >= 2.0 && avg(&mem_red) >= 2.0,
115            },
116        }
117    }
118}
