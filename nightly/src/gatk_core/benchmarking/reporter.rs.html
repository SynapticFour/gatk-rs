3/// Output serialization format for benchmark reports.
4/// # Invariants
5/// Each variant maps to a distinct file extension in [`BenchmarkReporter::generate_report`].
6/// # Ownership
7/// `Copy` enum.
8/// # Mutation
9/// N/A.
10/// # Biological assumptions
11/// None (infrastructure).
12/// # Java equivalence
13/// None / Rust-native.
14#[derive(Debug, Clone, Copy)]
15pub enum ReportFormat {
16    Html,
17    Json,
18    Markdown,
19    Csv,
20}
21
22/// Writes multi-format benchmark reports from a [`BenchmarkSuite`].
23/// # Invariants
24/// Creates `output_dir` if missing; one primary artifact per `generate_report` call.
25/// # Ownership
26/// Owns output directory path and format selection.
27/// # Mutation
28/// Report generation writes files; reporter instance reusable.
29/// # Biological assumptions
30/// None (infrastructure).
31/// # Java equivalence
32/// None / Rust-native.
33pub struct BenchmarkReporter {
34    output_dir: String,
35    format: ReportFormat,
36}
37
38impl BenchmarkReporter {
39    pub fn new(output_dir: String, format: ReportFormat) -> Self {
40        Self { output_dir, format }
41    }
42
43    pub fn generate_report(&self, suite: &BenchmarkSuite) -> gatk_common::GatkResult<()> {
44        std::fs::create_dir_all(&self.output_dir)
45            .map_err(|e| gatk_common::GatkError::io("Failed to create report directory", e))?;
46
47        let path = match self.format {
48            ReportFormat::Html => format!("{}/report.html", self.output_dir),
49            ReportFormat::Json => format!("{}/report.json", self.output_dir),
50            ReportFormat::Markdown => format!("{}/report.md", self.output_dir),
51            ReportFormat::Csv => format!("{}/report.csv", self.output_dir),
52        };
53
54        let body = match self.format {
55            ReportFormat::Json => serde_json::to_string_pretty(suite)
56                .map_err(|e| gatk_common::GatkError::generic(format!("Failed to serialize report: {e}")))?,
57            ReportFormat::Markdown => format!(
58                "# Benchmark Report\n\n- Comparisons: {}\n- Average speedup: {:.2}x\n- Average memory reduction: {:.2}x\n",
59                suite.comparisons.len(),
60                suite.average_speedup(),
61                suite.average_memory_reduction()
62            ),
63            ReportFormat::Csv => {
64                let mut out = "tool,time_s,memory_mb,success\n".to_string();
65                for r in &suite.gatk_rs_results {
66                    out.push_str(&format!(
67                        "{},{:.3},{:.2},{}\n",
68                        r.tool,
69                        r.execution_time_seconds(),
70                        r.peak_memory_mb(),
71                        r.success
72                    ));
73                }
74                out
75            }
76            ReportFormat::Html => format!(
77                "<html><body><h1>Benchmark Report</h1><p>Comparisons: {}</p><p>Average speedup: {:.2}x</p><p>Average memory reduction: {:.2}x</p></body></html>",
78                suite.comparisons.len(),
79                suite.average_speedup(),
80                suite.average_memory_reduction()
81            ),
82        };
83
84        std::fs::write(path, body)
85            .map_err(|e| gatk_common::GatkError::io("Failed to write report", e))?;
86        Ok(())
87    }
88
89    pub fn generate_json_report(&self, suite: &BenchmarkSuite) -> gatk_common::GatkResult<()> {
90        Self::new(self.output_dir.clone(), ReportFormat::Json).generate_report(suite)
91    }
92
93    pub fn generate_markdown_report(&self, suite: &BenchmarkSuite) -> gatk_common::GatkResult<()> {
94        Self::new(self.output_dir.clone(), ReportFormat::Markdown).generate_report(suite)
95    }
96
97    pub fn generate_csv_report(&self, suite: &BenchmarkSuite) -> gatk_common::GatkResult<()> {
98        Self::new(self.output_dir.clone(), ReportFormat::Csv).generate_report(suite)
99    }
100}

102/// Writes a short executive summary markdown file for stakeholders.
103/// # Invariants
104/// Produces `{output_dir}/summary.md` with target/speedup highlights.
105/// # Ownership
106/// Owns output directory string.
107/// # Mutation
108/// File write on `generate_summary`.
109/// # Biological assumptions
110/// None (infrastructure).
111/// # Java equivalence
112/// None / Rust-native.
113pub struct SummaryReporter {
114    output_dir: String,
115}
116
117impl SummaryReporter {
118    pub fn new(output_dir: String) -> Self {
119        Self { output_dir }
120    }
121
122    pub fn generate_summary(&self, suite: &BenchmarkSuite) -> gatk_common::GatkResult<()> {
123        std::fs::create_dir_all(&self.output_dir)
124            .map_err(|e| gatk_common::GatkError::io("Failed to create summary directory", e))?;
125        let path = format!("{}/summary.md", self.output_dir);
126        let content = format!(
127            "# Executive Summary\n\n- Meets all targets: {}\n- Average speedup: {:.2}x\n- Average memory reduction: {:.2}x\n",
128            suite.meets_all_targets(),
129            suite.average_speedup(),
130            suite.average_memory_reduction()
131        );
132        std::fs::write(path, content)
133            .map_err(|e| gatk_common::GatkError::io("Failed to write summary", e))?;
134        Ok(())
135    }
