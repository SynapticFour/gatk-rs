1use crate::parallel::ParallelConfig;
2use futures::future::try_join_all;

4/// Concurrent file read/write helper built on Tokio.
5/// # Invariants
6/// One async task per path in batch APIs; failures fail entire batch.
7/// # Ownership
8/// Owns config; returns owned `(path, bytes)` tuples.
9/// # Mutation
10/// Async methods borrow `&self`.
11/// # Biological assumptions
12/// None (generic I/O); paths typically FASTA/BAM/VCF inputs.
13/// # Java equivalence
14/// None / Rust-native.
15pub struct TokioProcessor {
16    _config: ParallelConfig,
17}
18
19impl TokioProcessor {
20    pub fn new(config: ParallelConfig) -> gatk_common::GatkResult<Self> {
21        Ok(Self { _config: config })
22    }
23
24    pub async fn read_files_concurrent(
25        &self,
26        paths: Vec<String>,
27    ) -> gatk_common::GatkResult<Vec<(String, Vec<u8>)>> {
28        let tasks = paths.into_iter().map(|path| async move {
29            let bytes = tokio::fs::read(&path).await.map_err(|e| {
30                gatk_common::GatkError::io(format!("Failed to read file {path}"), e)
31            })?;
32            Ok::<_, gatk_common::GatkError>((path, bytes))
33        });
34        try_join_all(tasks).await
35    }
36
37    pub async fn write_files_concurrent(
38        &self,
39        files: Vec<(String, Vec<u8>)>,
40    ) -> gatk_common::GatkResult<()> {
41        let tasks = files.into_iter().map(|(path, bytes)| async move {
42            tokio::fs::write(&path, bytes)
43                .await
44                .map_err(|e| gatk_common::GatkError::io(format!("Failed to write file {path}"), e))
45        });
46        try_join_all(tasks).await?;
47        Ok(())
48    }
49}
