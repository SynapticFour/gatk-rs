4/// Tabular variant row for Polars dataframe export.
5/// # Invariants
6/// One row per variant; chromosome/position uniquely identify row in typical frames.
7/// # Ownership
8/// Owns string fields; clone for dataframe building.
9/// # Mutation
10/// Public fields for ETL staging.
11/// # Biological assumptions
12/// VCF-like variant summary with depth and allele frequency.
13/// # Java equivalence
14/// Approximates htsjdk `VariantContext` columns in a dataframe (Rust-native).
15#[derive(Debug, Clone)]
16pub struct VariantData {
17    pub chromosome: String,
18    pub position: u64,
19    pub id: String,
20    pub reference: String,
21    pub alternate: String,
22    pub quality: f64,
23    pub filter: String,
24    pub depth: u32,
25    pub allele_frequency: f64,
26}

28/// High-level QC summary computed from variant dataframe analytics.
29/// # Invariants
30/// `ti_tv_ratio` meaningful only when transition/transversion counts are populated by caller.
31/// # Ownership
32/// Plain scalars; default zeroed.
33/// # Mutation
34/// Immutable summary output.
35/// # Biological assumptions
36/// Ti/Tv and high-quality counts used for SNP QC heuristics.
37/// # Java equivalence
38/// None / Rust-native analytics.
39#[derive(Debug, Clone, Default)]
40pub struct GenomicAnalysisSummary {
41    pub high_quality_count: usize,
42    pub ti_tv_ratio: f64,
43}

65    pub fn create_variant_dataframe(
66        &self,
67        variants: Vec<VariantData>,
68    ) -> gatk_common::GatkResult<DataFrame> {
69        let chr = Series::new(
70            "chromosome",
71            variants
72                .iter()
73                .map(|v| v.chromosome.as_str())
74                .collect::<Vec<_>>(),
75        );
76        let pos = Series::new(
77            "position",
78            variants.iter().map(|v| v.position).collect::<Vec<_>>(),
79        );
80        let id = Series::new(
81            "id",
82            variants.iter().map(|v| v.id.as_str()).collect::<Vec<_>>(),
83        );
84        let r = Series::new(
85            "reference",
86            variants
87                .iter()
88                .map(|v| v.reference.as_str())
89                .collect::<Vec<_>>(),
90        );
91        let a = Series::new(
92            "alternate",
93            variants
94                .iter()
95                .map(|v| v.alternate.as_str())
96                .collect::<Vec<_>>(),
97        );
98        let q = Series::new(
99            "quality",
100            variants.iter().map(|v| v.quality).collect::<Vec<_>>(),
101        );
102        let filt = Series::new(
103            "filter",
104            variants
105                .iter()
106                .map(|v| v.filter.as_str())
107                .collect::<Vec<_>>(),
108        );
109        let dp = Series::new(
110            "depth",
111            variants.iter().map(|v| v.depth).collect::<Vec<_>>(),
112        );
113        let af = Series::new(
114            "allele_frequency",
115            variants
116                .iter()
117                .map(|v| v.allele_frequency)
118                .collect::<Vec<_>>(),
119        );
120        DataFrame::new(vec![chr, pos, id, r, a, q, filt, dp, af])
121            .map_err(|e| gatk_common::GatkError::generic(format!("Failed to build DataFrame: {e}")))
122    }

124    pub fn filter_variants_by_quality(
125        &self,
126        df: &DataFrame,
127        min_quality: f64,
128        min_depth: u32,
129    ) -> gatk_common::GatkResult<DataFrame> {
130        let out = df
131            .clone()
132            .lazy()
133            .filter(
134                col("quality")
135                    .gt_eq(lit(min_quality))
136                    .and(col("depth").gt_eq(lit(min_depth))),
137            )
138            .collect()
139            .map_err(|e| {
140                gatk_common::GatkError::generic(format!("Failed to filter variants: {e}"))
141            })?;
142        Ok(out)
143    }

145    pub fn group_by_chromosome(&self, df: &DataFrame) -> gatk_common::GatkResult<DataFrame> {
146        df.clone()
147            .lazy()
148            .group_by([col("chromosome")])
149            .agg([col("position").count().alias("count")])
150            .collect()
151            .map_err(|e| gatk_common::GatkError::generic(format!("Failed to group variants: {e}")))
152    }

154    pub fn calculate_ti_tv_ratio(&self, df: &DataFrame) -> gatk_common::GatkResult<f64> {
155        let ref_col = df.column("reference").map_err(|e| {
156            gatk_common::GatkError::generic(format!("Missing reference column: {e}"))
157        })?;
158        let alt_col = df.column("alternate").map_err(|e| {
159            gatk_common::GatkError::generic(format!("Missing alternate column: {e}"))
160        })?;
161
162        let refs = ref_col
163            .str()
164            .map_err(|e| gatk_common::GatkError::generic(format!("{e}")))?;
165        let alts = alt_col
166            .str()
167            .map_err(|e| gatk_common::GatkError::generic(format!("{e}")))?;
168
169        let mut transitions = 0usize;
170        let mut transversions = 0usize;
171        for (r, a) in refs.into_iter().zip(alts) {
172            if let (Some(r), Some(a)) = (r, a) {
173                let pair = (r.as_bytes().first().copied(), a.as_bytes().first().copied());
174                match pair {
175                    (Some(b'A'), Some(b'G'))
176                    | (Some(b'G'), Some(b'A'))
177                    | (Some(b'C'), Some(b'T'))
178                    | (Some(b'T'), Some(b'C')) => transitions += 1,
179                    (Some(_), Some(_)) => transversions += 1,
180                    _ => {}
181                }
182            }
183        }
184        Ok(if transversions == 0 {
185            transitions as f64
186        } else {
187            transitions as f64 / transversions as f64
188        })
189    }

191    pub fn genomic_analysis_pipeline(
192        &self,
193        variants_df: &DataFrame,
194        _reads_df: &DataFrame,
195    ) -> gatk_common::GatkResult<GenomicAnalysisSummary> {
196        let filtered = self.filter_variants_by_quality(variants_df, 30.0, 10)?;
197        let ti_tv = self.calculate_ti_tv_ratio(&filtered)?;
198        Ok(GenomicAnalysisSummary {
199            high_quality_count: filtered.height(),
200            ti_tv_ratio: ti_tv,
201        })
202    }
