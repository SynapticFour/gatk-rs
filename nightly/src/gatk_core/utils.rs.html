6/// Reverse complement a DNA sequence
7pub fn reverse_complement(sequence: &[Base]) -> Vec<Base> {
8    sequence
9        .iter()
10        .rev()
11        .map(|base| base.complement())
12        .collect()
13}

15/// Calculate GC content of a sequence
16pub fn gc_content(sequence: &[Base]) -> f64 {
17    if sequence.is_empty() {
18        return 0.0;
19    }
20
21    let gc_count = sequence
22        .iter()
23        .filter(|&&base| matches!(base, Base::G | Base::C))
24        .count();
25
26    gc_count as f64 / sequence.len() as f64
27}

34/// Merge overlapping intervals
35pub fn merge_intervals(intervals: &[GenomicInterval]) -> Vec<GenomicInterval> {
36    if intervals.is_empty() {
37        return Vec::new();
38    }
39
40    let mut sorted_intervals = intervals.to_vec();
41    sorted_intervals.sort_by(|a, b| {
42        a.contig
43            .cmp(&b.contig)
44            .then(a.start.cmp(&b.start))
45            .then(a.end.cmp(&b.end))
46    });
47
48    // Lifetime: `sorted_intervals` is function-local; move intervals into `current`
49    // / `merged` instead of cloning while scanning.
50    let mut merged = Vec::new();
51    let mut iter = sorted_intervals.into_iter();
52    let mut current = iter.next().expect("non-empty");
53
54    for interval in iter {
55        if interval.contig == current.contig && interval.start <= current.end + 1 {
56            // Overlapping or adjacent intervals - merge them
57            current.end = current.end.max(interval.end);
58        } else {
59            // Non-overlapping interval - push current and start new one
60            merged.push(current);
61            current = interval;
62        }
63    }
64    merged.push(current);
65
66    merged
67}

69/// Calculate Hamming distance between two sequences
70pub fn hamming_distance(seq1: &[Base], seq2: &[Base]) -> Option<usize> {
71    if seq1.len() != seq2.len() {
72        return None;
73    }
74
75    Some(seq1.iter().zip(seq2.iter()).filter(|(a, b)| a != b).count())
76}

78/// Find all unique bases in a sequence
79pub fn unique_bases(sequence: &[Base]) -> HashSet<Base> {
80    sequence.iter().cloned().collect()

83/// Check if a sequence contains only valid DNA bases (A, C, G, T)
84pub fn is_valid_dna(sequence: &[Base]) -> bool {
85    sequence
86        .iter()
87        .all(|&base| matches!(base, Base::A | Base::C | Base::G | Base::T))
88}

90/// Calculate base composition of a sequence
91pub fn base_composition(sequence: &[Base]) -> HashMap<Base, usize> {
92    let mut composition = HashMap::new();
93    for &base in sequence {
94        *composition.entry(base).or_insert(0) += 1;
95    }
96    composition
97}

99/// Convert Phred quality score to probability
100pub fn phred_to_probability(quality: BaseQuality) -> f64 {
101    quality.error_probability()
102}
103
104/// Convert probability to Phred quality score
105pub fn probability_to_phred(prob: f64) -> BaseQuality {
106    let quality = if prob <= 0.0 {
107        93.0 // Maximum quality
108    } else if prob >= 1.0 {
109        0.0
110    } else {
111        (-10.0 * prob.log10()).clamp(0.0, 93.0)
112    };
113    BaseQuality::new(quality as u8)
114}

116/// Calculate average base quality
117pub fn average_base_quality(qualities: &[BaseQuality]) -> f64 {
118    if qualities.is_empty() {
119        return 0.0;
120    }
121
122    let sum: f64 = qualities.iter().map(|q| q.value() as f64).sum();
123    sum / qualities.len() as f64
124}

126/// Filter bases by minimum quality
127pub fn filter_by_quality(
128    sequence: &[Base],
129    qualities: &[BaseQuality],
130    min_quality: u8,
131) -> Vec<(Base, BaseQuality)> {
132    sequence
133        .iter()
134        .zip(qualities.iter())
135        .filter_map(|(&base, &quality)| {
136            if quality.value() >= min_quality {
137                Some((base, quality))
138            } else {
139                None
140            }
141        })
142        .collect()
143}

160#[cfg(test)]

164    #[test]
165    fn test_reverse_complement() {
166        let seq = vec![Base::A, Base::C, Base::G, Base::T];
167        let rc = reverse_complement(&seq);
168        assert_eq!(rc, vec![Base::A, Base::C, Base::G, Base::T]); // ATGC -> GCAT -> reverse complement
169
170        let seq = vec![Base::A, Base::T, Base::G, Base::C];
171        let rc = reverse_complement(&seq);
172        assert_eq!(rc, vec![Base::G, Base::C, Base::A, Base::T]);
173    }
174
175    #[test]
176    fn test_gc_content() {
177        let seq = vec![Base::G, Base::C, Base::A, Base::T];
178        assert_eq!(gc_content(&seq), 0.5);
179
180        let seq = vec![Base::G, Base::C, Base::G, Base::C];
181        assert_eq!(gc_content(&seq), 1.0);
182
183        let seq = vec![Base::A, Base::T, Base::A, Base::T];
184        assert_eq!(gc_content(&seq), 0.0);
185    }
186
187    #[test]
188    fn test_hamming_distance() {
189        let seq1 = vec![Base::A, Base::C, Base::G, Base::T];
190        let seq2 = vec![Base::A, Base::C, Base::A, Base::T];
191        assert_eq!(hamming_distance(&seq1, &seq2), Some(1));
192
193        let seq3 = vec![Base::A, Base::C];
194        assert_eq!(hamming_distance(&seq1, &seq3), None);
195    }

202            GenomicInterval::new(0, 300, 400),

205        let merged = merge_intervals(&intervals);

217        let q = BaseQuality::new(20);
218        let prob = phred_to_probability(q);
219        assert!((prob - 0.01).abs() < 0.001);

221        let q_back = probability_to_phred(prob);
