1//! GATK `NearbyKmerErrorCorrector` — k-mer spectrum read correction before assembly.
2
3use gatk_common::GatkResult;
4use rust_htslib::bam::Record;
5use std::collections::{BTreeMap, HashMap};
6
7/// GATK `AssemblerArgumentCollection` defaults for k-mer read error correction.
8pub const GATK_KMER_LENGTH_FOR_READ_ERROR_CORRECTION: usize = 25;
9pub const GATK_MIN_OBSERVATIONS_FOR_KMER_TO_BE_SOLID: usize = 20;
10/// GATK `HaplotypeCallerEngine.MIN_TAIL_QUALITY_WITH_ERROR_CORRECTION`.
11pub const GATK_MIN_TAIL_QUALITY_WITH_ERROR_CORRECTION: u8 = 6;
12
13/// Configuration matching Java `NearbyKmerErrorCorrector` simple constructor.
14/// # Invariants
15/// `kmer_length` must be positive for k-mer counting; reads shorter than this are skipped.
16/// `min_observations_for_kmer_to_be_solid` gates whether a k-mer is trusted for correction.
17/// # Ownership
18/// [`Clone`] config passed into [`correct_reads_nearby_kmer`]; does not own reads or reference.
19/// # Mutation
20/// Immutable per correction pass; read bases are mutated in the caller's slice.
21/// # Biological assumptions
22/// Illumina-like errors cluster in read tails; solid k-mers come from local read pileup agreement.
23/// # Java equivalence
24/// GATK `NearbyKmerErrorCorrector` simple constructor + `AssemblerArgumentCollection` k-mer defaults.
25#[derive(Debug, Clone)]
26pub struct NearbyKmerErrorCorrectorConfig {
27    pub kmer_length: usize,
28    /// GATK constructor field; tail soft-clip before correction not yet wired here.
29    #[allow(dead_code)]
30    pub min_tail_quality: u8,
31    pub min_observations_for_kmer_to_be_solid: usize,
32    pub debug: bool,
33}
34
35impl NearbyKmerErrorCorrectorConfig {
36    pub fn gatk_defaults() -> Self {
37        Self {
38            kmer_length: GATK_KMER_LENGTH_FOR_READ_ERROR_CORRECTION,
39            min_tail_quality: GATK_MIN_TAIL_QUALITY_WITH_ERROR_CORRECTION,
40            min_observations_for_kmer_to_be_solid: GATK_MIN_OBSERVATIONS_FOR_KMER_TO_BE_SOLID,
41            debug: false,
42        }
43    }
44}

46struct KmerCounter {
47    kmer_length: usize,
48    /// Ordered map so solid-kmer ties resolve by lexicographic sequence.
49    counts: BTreeMap<Vec<u8>, u32>,
50}
51
52impl KmerCounter {
53    fn new(kmer_length: usize) -> Self {
54        Self {
55            kmer_length,
56            counts: BTreeMap::new(),
57        }
58    }
59
60    fn add_read(&mut self, bases: &[u8]) {
61        if bases.len() < self.kmer_length {
62            return;
63        }
64        for offset in 0..=bases.len() - self.kmer_length {
65            let kmer = bases[offset..offset + self.kmer_length].to_vec();
66            *self.counts.entry(kmer).or_insert(0) += 1;
67        }
68    }
69}
70
71fn max_homopolymer_run(bases: &[u8]) -> usize {
72    let mut best = 1usize;
73    let mut run = 1usize;
74    let mut prev = bases.first().copied();
75    for &b in bases.iter().skip(1) {
76        if Some(b) == prev {
77            run += 1;
78            best = best.max(run);
79        } else {
80            run = 1;
81            prev = Some(b);
82        }
83    }
84    best
85}
86
87fn hamming_distance(a: &[u8], b: &[u8]) -> usize {
88    a.iter().zip(b.iter()).filter(|(x, y)| x != y).count()
89}

91/// Correct reads when `errorCorrectReads` is enabled (GATK `NearbyKmerErrorCorrector.correctReads`).
92pub fn correct_reads_nearby_kmer(
93    reads: &mut [Record],
94    full_reference_with_padding: &[u8],
95    cfg: &NearbyKmerErrorCorrectorConfig,
96) -> GatkResult<()> {
97    if max_homopolymer_run(full_reference_with_padding) > 12 {
98        return Ok(());
99    }
100    let mut counter = KmerCounter::new(cfg.kmer_length);
101    for rec in reads.iter() {
102        let bases = rec.seq().as_bytes();
103        counter.add_read(&bases);
104    }
105    let solid_threshold = cfg.min_observations_for_kmer_to_be_solid as u32;
106
107    let mut insolid_to_solid: HashMap<Vec<u8>, Vec<u8>> = HashMap::new();
108    for (kmer, &count) in &counter.counts {
109        if count >= solid_threshold {
110            continue;
111        }
112        let mut best: Option<(&[u8], u32)> = None;
113        for (solid, &sc) in &counter.counts {
114            if sc < solid_threshold {
115                continue;
116            }
117            let dist = hamming_distance(kmer, solid);
118            if dist > 0 && dist <= 2 {
119                let better = match best {
120                    None => true,
121                    Some((prev, c)) => sc > c || (sc == c && solid.as_slice() < prev),
122                };
123                if better {
124                    best = Some((solid.as_slice(), sc));
125                }
126            }
127        }
128        if let Some((solid, _)) = best {
129            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
130            insolid_to_solid.insert(kmer.clone(), solid.to_vec());
131        }
132    }
133
134    let mut reads_corrected = 0usize;
135    for rec in reads.iter_mut() {
136        let mut bases = rec.seq().as_bytes().to_vec();
137        let mut quals = rec.qual().to_vec();
138        let mut corrections: Vec<(usize, u8)> = Vec::new();
139        for offset in 0..bases.len().saturating_sub(cfg.kmer_length) {
140            let kmer = bases[offset..offset + cfg.kmer_length].to_vec();
141            let Some(solid) = insolid_to_solid.get(&kmer) else {
142                continue;
143            };
144            for (i, (&ob, &sb)) in kmer.iter().zip(solid.iter()).enumerate() {
145                if ob != sb {
146                    corrections.push((offset + i, sb));
147                }
148            }
149        }
150        if corrections.is_empty() {
151            continue;
152        }
153        for (pos, base) in corrections {
154            bases[pos] = base;
155            quals[pos] = 30;
156        }
157        let qname = rec.qname().to_vec();
158        let cigar = rec.cigar();
159        rec.set(&qname, Some(&cigar), &bases, &quals);
160        reads_corrected += 1;
161    }
162    if cfg.debug {
163        eprintln!(
164            "[NearbyKmerErrorCorrector] corrected {reads_corrected}/{} reads ({} kmer mappings)",
165            reads.len(),
166            insolid_to_solid.len()
167        );
168    }
169    Ok(())
170}
