1//! Fixed-width traversal tiles over closed genomic intervals.
2//! GATK’s adaptive active regions come later; this only splits each closed interval into
3//! non-overlapping tiles for engine wiring and logging.
4
5/// Default tile size (bp) for scaffolding traversal; real assembly regions will be adaptive.
6pub const DEFAULT_TRAVERSAL_TILE_BP: u64 = 100;
7
8/// One contiguous 1-based inclusive tile on a contig.
9/// # Invariants
10/// `start` / `end` are **1-based inclusive**; `end >= start` for non-empty tiles.
11/// Tiles from [`tile_closed_interval`] are non-overlapping and cover the closed interval.
12/// # Ownership
13/// Owns contig name; coordinates are plain `u64`.
14/// # Mutation
15/// Immutable scaffold record; engine traversal does not mutate tiles in place.
16/// # Biological assumptions
17/// Fixed-width scaffold only — not adaptive GATK active regions.
18/// # Java equivalence
19/// Rust-native Phase 2 scaffold; adaptive `AssemblyRegion` discovery comes later.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct TraversalTile {
22    pub contig: String,
23    pub start: u64,
24    pub end: u64,
25}
26
27/// Split `[start, end]` into non-overlapping tiles of at most `tile_bp` bases.
28pub fn tile_closed_interval(
29    contig: &str,
30    start: u64,
31    end: u64,
32    tile_bp: u64,
33) -> Vec<TraversalTile> {
34    if tile_bp == 0 {
35        return Vec::new();
36    }
37    let mut out = Vec::new();
38    let mut pos = start;
39    while pos <= end {
40        let region_end = (pos + tile_bp - 1).min(end);
41        out.push(TraversalTile {
42            contig: contig.to_string(),
43            start: pos,
44            end: region_end,
45        });
46        pos = region_end.saturating_add(1);
47    }
48    out
49}

51#[cfg(test)]
52mod tests {

55    #[test]
56    fn tiles_short_interval() {
57        let t = tile_closed_interval("chr1", 1, 32, 100);
58        assert_eq!(t.len(), 1);
59        assert_eq!(t[0].start, 1);
60        assert_eq!(t[0].end, 32);
61    }

63    #[test]

67        assert_eq!(t[0].end, 100);
68        assert_eq!(t[1].start, 101);
