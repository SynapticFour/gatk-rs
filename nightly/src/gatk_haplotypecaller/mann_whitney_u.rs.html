8const NORMAL_SD: f64 = 1.0;

10/// Which tail of the Mann–Whitney U distribution to use (one- or two-sided).
11/// # Invariants
12/// Drives U statistic selection and p-value halving in [`MannWhitneyU::test`].
13/// # Ownership
14/// [`Copy`] enum.
15/// # Mutation
16/// Immutable test configuration per call.
17/// # Biological assumptions
18/// Used to compare two numeric samples (e.g., ref vs alt base qualities).
19/// # Java equivalence
20/// GATK `MannWhitneyU.TestType`.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum TestType {
23    FirstDominates,
24    SecondDominates,
25    TwoSided,
26}

28/// Mann–Whitney U test statistics (U, z, p, median shift).
29/// # Invariants
30/// When either input series is empty, `u`, `z`, `p`, and `median_shift` are NaN.
31/// # Ownership
32/// [`Copy`] result bundle returned from [`MannWhitneyU::test`].
33/// # Mutation
34/// Immutable statistics snapshot.
35/// # Biological assumptions
36/// Compares two independent numeric samples (rank-sum style QC annotations).
37/// # Java equivalence
38/// GATK `MannWhitneyU` result object (`org.broadinstitute.hellbender.utils.MannWhitneyU`).
39#[derive(Debug, Clone, Copy)]
40pub struct MannWhitneyResult {
41    pub u: f64,
42    pub z: f64,
43    pub p: f64,
44    pub median_shift: f64,
45}

65/// Mann–Whitney U rank-sum test engine (BaseQ rank sum and related annotations).
66/// # Invariants
67/// `minimum_normal_n` controls normal approximation vs exact permutation path.
68/// # Ownership
69/// Stateless calculator except for `minimum_normal_n`; input series are copied/sorted internally.
70/// # Mutation
71/// [`Self::test`] may sort input copies; the engine itself is not mutated.
72/// # Biological assumptions
73/// Two-sample comparison of per-read or per-base numeric scores at a variant site.
74/// # Java equivalence
75/// GATK `org.broadinstitute.hellbender.utils.MannWhitneyU`.
76pub struct MannWhitneyU {
77    minimum_normal_n: usize,
78}

88impl MannWhitneyU {
89    pub fn test(
90        &self,
91        series1: &[f64],
92        series2: &[f64],
93        which_side: TestType,
94    ) -> MannWhitneyResult {
95        let mut s1 = series1.to_vec();
96        let mut s2 = series2.to_vec();
97        let n1 = s1.len();
98        let n2 = s2.len();
99        if n1 == 0 || n2 == 0 {
100            return MannWhitneyResult {
101                u: f64::NAN,
102                z: f64::NAN,
103                p: f64::NAN,
104                median_shift: f64::NAN,
105            };
106        }
107        let (u, nties) = match which_side {
108            TestType::TwoSided => {
109                let stat = self.calculate_u1_and_u2(&mut s1, &mut s2);
110                (stat.u1.min(stat.u2), stat.num_of_ties)
111            }
112            TestType::FirstDominates => {
113                let stat = self.calculate_one_sided_u(&mut s1, &mut s2, TestType::FirstDominates);
114                (stat.true_u, stat.num_of_ties)
115            }
116            TestType::SecondDominates => {
117                let stat = self.calculate_one_sided_u(&mut s1, &mut s2, TestType::SecondDominates);
118                (stat.true_u, stat.num_of_ties)
119            }
120        };
121        let (z, p) = if n1 >= self.minimum_normal_n || n2 >= self.minimum_normal_n {
122            let z = self.calculate_z(u, n1, n2, nties, which_side);
123            let normal = Normal::new(NORMAL_MEAN, NORMAL_SD).expect("standard normal");
124            let mut p = 2.0 * normal.cdf(NORMAL_MEAN + z * NORMAL_SD);
125            if which_side != TestType::TwoSided {
126                p /= 2.0;
127            }
128            (z, p)
129        } else {
130            let p = self.permutation_test(&mut s1, &mut s2, u);
131            let normal = Normal::new(NORMAL_MEAN, NORMAL_SD).expect("standard normal");
132            let z = normal.inverse_cdf(p);
133            (z, p)
134        };
135        MannWhitneyResult {
136            u,
137            z,
138            p,
139            median_shift: (median(&s1) - median(&s2)).abs(),
140        }
141    }

143    fn calculate_rank(&self, series1: &mut [f64], series2: &mut [f64]) -> RankedData {
144        series1.sort_by(|a, b| a.partial_cmp(b).unwrap_or(Ordering::Equal));
145        series2.sort_by(|a, b| a.partial_cmp(b).unwrap_or(Ordering::Equal));
146        let mut ranks = Vec::with_capacity(series1.len() + series2.len());
147        let mut i = 0usize;
148        let mut j = 0usize;
149        while ranks.len() < series1.len() + series2.len() {
150            let r = ranks.len() + 1;
151            if i >= series1.len() {
152                ranks.push(Rank {
153                    value: series2[j],
154                    rank: r as f32,
155                    series: 2,
156                });
157                j += 1;
158            } else if j >= series2.len() {
159                ranks.push(Rank {
160                    value: series1[i],
161                    rank: r as f32,
162                    series: 1,
163                });
164                i += 1;
165            } else if series1[i] <= series2[j] {
166                ranks.push(Rank {
167                    value: series1[i],
168                    rank: r as f32,
169                    series: 1,
170                });
171                i += 1;
172            } else {
173                ranks.push(Rank {
174                    value: series2[j],
175                    rank: r as f32,
176                    series: 2,
177                });
178                j += 1;
179            }
180        }
181        let mut num_of_ties = Vec::new();
182        let mut idx = 0usize;
183        while idx < ranks.len() {
184            let mut rank_sum = ranks[idx].rank;
185            let mut count = 1usize;
186            let value = ranks[idx].value;
187            let mut j = idx + 1;
188            while j < ranks.len() && ranks[j].value == value {
189                rank_sum += ranks[j].rank;
190                count += 1;
191                j += 1;
192            }
193            if count > 1 {
194                let avg = rank_sum / count as f32;
195                for r in &mut ranks[idx..idx + count] {
196                    r.rank = avg;
197                }
198                num_of_ties.push(count as i32);
199            }
200            idx += count;
201        }
202        RankedData { ranks, num_of_ties }
203    }

205    fn calculate_u1_and_u2(&self, series1: &mut [f64], series2: &mut [f64]) -> TestStatistic {
206        let ranked = self.calculate_rank(series1, series2);
207        let length = ranked.ranks.len();
208        let nties = transform_ties(length, &ranked.num_of_ties);
209        let mut r1 = 0.0f32;
210        let mut r2 = 0.0f32;
211        for rank in &ranked.ranks {
212            if rank.series == 1 {
213                r1 += rank.rank;
214            } else {
215                r2 += rank.rank;
216            }
217        }
218        let n1 = series1.len() as f64;
219        let n2 = series2.len() as f64;
220        let u1 = f64::from(r1) - (n1 * (n1 + 1.0)) / 2.0;
221        let u2 = f64::from(r2) - (n2 * (n2 + 1.0)) / 2.0;
222        TestStatistic {
223            u1,
224            u2,
225            true_u: f64::NAN,
226            num_of_ties: nties,
227        }
228    }

230    fn calculate_one_sided_u(
231        &self,
232        series1: &mut [f64],
233        series2: &mut [f64],
234        which: TestType,
235    ) -> TestStatistic {
236        let stat = self.calculate_u1_and_u2(series1, series2);
237        let true_u = if which == TestType::FirstDominates {
238            stat.u1
239        } else {
240            stat.u2
241        };
242        TestStatistic {
243            u1: stat.u1,
244            u2: stat.u2,
245            true_u,
246            num_of_ties: stat.num_of_ties,
247        }
248    }

250    fn calculate_z(&self, u: f64, n1: usize, n2: usize, nties: f64, which_side: TestType) -> f64 {
251        let n1f = n1 as f64;
252        let n2f = n2 as f64;
253        let m = (n1f * n2f) / 2.0;
254        let correction = match which_side {
255            TestType::TwoSided => {
256                if (u - m) >= 0.0 {
257                    0.5
258                } else {
259                    -0.5
260                }
261            }
262            TestType::FirstDominates => -0.5,
263            TestType::SecondDominates => 0.5,
264        };
265        let correction = if nties == 0.0 { 0.0 } else { correction };
266        let sigma = ((n1f * n2f / 12.0)
267            * ((n1f + n2f + 1.0) - nties / ((n1f + n2f) * (n1f + n2f - 1.0))))
268            .sqrt();
269        (u - m - correction) / sigma
270    }

272    fn permutation_test(&self, series1: &mut [f64], series2: &mut [f64], test_stat_u: f64) -> f64 {
273        let n1 = series1.len();
274        let n2 = series2.len();
275        let ranked = self.calculate_rank(series1, series2);
276        let ranks = ranked.ranks;
277        let mut first_perm: Vec<u8> = vec![0; n1 + n2];
278        for i in n1..n1 + n2 {
279            first_perm[i] = 1;
280        }
281        let perms = get_permutations(&first_perm);
282        let mut histo: HashMap<i64, f64> = HashMap::new();
283        let mut new_series1 = vec![0.0; n1];
284        let mut new_series2 = vec![0.0; n2];
285        for perm in perms {
286            let mut s1_end = 0usize;
287            let mut s2_end = 0usize;
288            for (i, &grouping) in perm.iter().enumerate() {
289                if grouping == 0 {
290                    new_series1[s1_end] = f64::from(ranks[i].rank);
291                    s1_end += 1;
292                } else {
293                    new_series2[s2_end] = f64::from(ranks[i].rank);
294                    s2_end += 1;
295                }
296            }
297            let new_u = new_series1.iter().sum::<f64>() - (n1 as f64 * (n1 as f64 + 1.0)) / 2.0;
298            let key = (new_u * 2.0).round() as i64;
299            *histo.entry(key).or_insert(0.0) += 1.0;
300        }
301        let test_key = (test_stat_u * 2.0).round() as i64;
302        let mut sum_smaller = histo.get(&test_key).copied().unwrap_or(0.0) / 2.0;
303        for (key, val) in &histo {
304            if *key < test_key {
305                sum_smaller += val;
306            }
307        }
308        let total: f64 = histo.values().sum();
309        sum_smaller / total
310    }
311}

313fn transform_ties(num_of_ranks: usize, num_of_ties: &[i32]) -> f64 {
314    let mut total = 0.0;
315    for &count in num_of_ties {
316        if count as usize != num_of_ranks {
317            total += f64::from(count).powi(3) - f64::from(count);
318        }
319    }
320    total
321}

323fn median(data: &[f64]) -> f64 {
324    let len = data.len();
325    let mid = len / 2;
326    if len % 2 == 0 {
327        (data[mid] + data[mid - 1]) / 2.0
328    } else {
329        data[mid]
330    }
331}

333fn get_permutations(first_perm: &[u8]) -> Vec<Vec<u8>> {
334    let mut temp: Vec<u8> = first_perm.to_vec();
335    let mut all = Vec::new();
336    // CLONE: needed because owned element into collection.
337    all.push(temp.clone());
338    loop {
339        let mut k: Option<usize> = None;
340        for i in (0..temp.len().saturating_sub(1)).rev() {
341            if temp[i] < temp[i + 1] {
342                k = Some(i);
343                break;
344            }
345        }
346        let k = match k {
347            Some(v) => v,
348            None => break,
349        };
350        let mut l: Option<usize> = None;
351        for i in (k + 1..temp.len()).rev() {
352            if temp[k] < temp[i] {
353                l = Some(i);
354                break;
355            }
356        }
357        let l = l.expect("l exists");
358        temp.swap(k, l);
359        let mut end = temp.len() - 1;
360        let mut begin = k + 1;
361        while begin < end {
362            temp.swap(begin, end);
363            begin += 1;
364            end -= 1;
365        }
366        // CLONE: needed because owned element into collection.
367        all.push(temp.clone());
368    }
369    all
370}
