17/// Every 1-based position in merged closed intervals `[start, end]`.
18/// # Invariants
19/// Yields every position in each closed span in increasing order, including empty pileup loci.
20/// # Ownership
21/// Owns span list and cursor state.
22/// # Mutation
23/// Advances span/position cursors via [`Iterator::next`].
24/// # Biological assumptions
25/// None — interval enumeration for pileup walks.
26/// # Java equivalence
27/// GATK `IntervalLocusIterator` (every base in merged intervals).
28#[derive(Debug, Clone)]
29pub struct IntervalLocusIterator {
30    spans: Vec<(u64, u64)>,
31    span_idx: usize,
32    pos: Option<u64>,
33}
34
35impl IntervalLocusIterator {
36    pub fn from_spans(spans: Vec<(u64, u64)>) -> Self {
37        Self {
38            spans,
39            span_idx: 0,
40            pos: None,
41        }
42    }
43
44    pub fn from_closed_interval(start1: u64, end1: u64) -> Self {
45        Self::from_spans(vec![(start1, end1)])
46    }
47}
48
49impl Iterator for IntervalLocusIterator {
50    type Item = u64;
51
52    fn next(&mut self) -> Option<Self::Item> {
53        loop {
54            if self.span_idx >= self.spans.len() {
55                return None;
56            }
57            let (s, e) = self.spans[self.span_idx];
58            let next = match self.pos {
59                None => s,
60                Some(p) if p < e => p + 1,
61                Some(_) => {
62                    self.span_idx += 1;
63                    self.pos = None;
64                    continue;
65                }
66            };
67            self.pos = Some(next);
68            return Some(next);
69        }
70    }
71}

73fn reference_span0(
74    record: &bam::Record,
75    header: &bam::HeaderView,
76    filters: &ReadFilterParams,
77) -> Option<(i64, i64)> {
78    if !passes_hc_read_filters_with_header(record, header, filters) {
79        return None;
80    }
81    let start = record.pos();
82    let end = record.cigar().end_pos();
83    if end <= start {
84        return None;
85    }
86    Some((start, end))
87}

89/// Owned LIBS-style pileup cursor (no borrow of `records`; safe inside [`crate::assembly_region_iterator::AssemblyRegionIterator`]).
90/// # Invariants
91/// Active-read set advances monotonically in BAM order for increasing locus positions.
92/// Records are indexed; filtered-out records have `ref_end0 == -1`.
93/// # Ownership
94/// Owns header view, index lists, and caches; borrows record slices only during pileup queries.
95/// # Mutation
96/// Advances active-set cursor and last-position tracking as loci are visited.
97/// # Biological assumptions
98/// Builds per-locus pileup observations for activity / genotyping evidence.
99/// # Java equivalence
100/// GATK `LocusIteratorByState` (LIBS) style pileup walk.
101#[derive(Debug)]
102pub struct LocusPileupState {
103    /// Retained for Java LIBS-style state; contig names are resolved at construction.
104    _header: bam::HeaderView,
105    sorted_indices: Vec<usize>,
106    pub(crate) active: Vec<usize>,
107    next_sorted: usize,
108    pub(crate) last_pos1: Option<u64>,
109    /// When true, pileup observations use Java `isAltAfterAssembly` (post-realign genotyping reads).
110    pub(crate) reads_were_realigned: bool,
111    /// Exclusive reference end (0-based) per record index; `-1` if filtered out at construction.
112    ref_end0: Vec<i64>,
113    /// Cached RCM HQ soft-clip counts (R4-1 — avoid recounting every locus).
114    hq_soft_clip_cache: Vec<Option<u32>>,
115}

147    fn from_records_inner(
148        records: &[bam::Record],
149        header: &bam::HeaderView,
150        contig: &str,
151        filters: &ReadFilterParams,
152        reads_were_realigned: bool,
153    ) -> Self {
154        let mut sorted_indices: Vec<usize> = (0..records.len())
155            .filter(|&i| {
156                let rec = &records[i];
157                let rn = String::from_utf8_lossy(header.tid2name(rec.tid() as u32));
158                rn == contig && reference_span0(rec, header, filters).is_some()
159            })
160            .collect();
161        sorted_indices.sort_by(|&a, &b| {
162            let ra = &records[a];
163            let rb = &records[b];
164            ra.pos()
165                .cmp(&rb.pos())
166                .then_with(|| ra.qname().cmp(rb.qname()))
167        });
168        let mut ref_end0 = vec![-1i64; records.len()];
169        for &i in &sorted_indices {
170            if let Some((_, r1)) = reference_span0(&records[i], header, filters) {
171                ref_end0[i] = r1;
172            }
173        }
174        Self {
175            _header: header.clone(),
176            sorted_indices,
177            active: Vec::new(),
178            next_sorted: 0,
179            last_pos1: None,
180            reads_were_realigned,
181            ref_end0,
182            hq_soft_clip_cache: vec![None; records.len()],
183        }
184    }
185
186    #[inline]
187    fn hq_soft_clip_cached(&mut self, records: &[bam::Record], idx: usize) -> u32 {
188        if let Some(n) = self.hq_soft_clip_cache[idx] {
189            return n;
190        }
191        let n = crate::hq_soft_clip::count_high_quality_soft_clip_bases_rcm(&records[idx]);
192        self.hq_soft_clip_cache[idx] = Some(n);
193        n
194    }

196    /// Clear active-read cursor (e.g. when jumping backward across disjoint spans).
197    pub fn reset_cursor(&mut self) {
198        self.active.clear();
199        self.next_sorted = 0;
200        self.last_pos1 = None;
201    }
202
203    pub fn advance_to(
204        &mut self,
205        records: &[bam::Record],
206        filters: &ReadFilterParams,
207        pos1: u64,
208    ) -> GatkResult<()> {
209        if let Some(prev) = self.last_pos1 {
210            if pos1 < prev {
211                return Err(GatkError::argument(format!(
212                    "LocusPileupState: pos {pos1} < previous {prev}"
213                )));
214            }
215            if pos1 == prev {
216                return Ok(());
217            }
218        }
219
220        let ref_pos0 = pos1.saturating_sub(1) as i64;
221        if self.last_pos1.is_none() {
222            self.bootstrap_active(records, filters, ref_pos0);
223        } else {
224            // R4-1: use cached exclusive ref ends (no re-filter / CIGAR per locus).
225            self.active.retain(|&idx| self.ref_end0[idx] > ref_pos0);
226            self.ingest_reads_through(records, filters, ref_pos0);
227        }
228        self.last_pos1 = Some(pos1);
229        Ok(())
230    }
231
232    fn bootstrap_active(
233        &mut self,
234        records: &[bam::Record],
235        filters: &ReadFilterParams,
236        ref_pos0: i64,
237    ) {
238        self.active.clear();
239        self.next_sorted = 0;
240        self.ingest_reads_through(records, filters, ref_pos0);
241    }
242
243    fn ingest_reads_through(
244        &mut self,
245        records: &[bam::Record],
246        _filters: &ReadFilterParams,
247        ref_pos0: i64,
248    ) {
249        while self.next_sorted < self.sorted_indices.len() {
250            let idx = self.sorted_indices[self.next_sorted];
251            let r1 = self.ref_end0[idx];
252            if r1 < 0 {
253                self.next_sorted += 1;
254                continue;
255            }
256            let r0 = records[idx].pos();
257            if r0 > ref_pos0 {
258                break;
259            }
260            // Each index is visited once via `next_sorted` — no membership scan.
261            if r1 > ref_pos0 {
262                self.active.push(idx);
263            }
264            self.next_sorted += 1;
265        }
266    }

268    pub fn pileup_observations(
269        &mut self,
270        records: &[bam::Record],
271        ref_base: u8,
272    ) -> GatkResult<Vec<PileupObservation>> {
273        let ref_pos0 = self
274            .last_pos1
275            .ok_or_else(|| GatkError::argument("advance_to before pileup_observations"))?
276            .saturating_sub(1) as i64;
277        let mut obs = Vec::with_capacity(self.active.len());
278        let mut seen_qname = if self.reads_were_realigned {
279            Some(std::collections::BTreeSet::new())
280        } else {
281            None
282        };
283        for i in 0..self.active.len() {
284            let idx = self.active[i];
285            let hq = self.hq_soft_clip_cached(records, idx);
286            let rec = &records[idx];
287            if let Some(ref mut seen) = seen_qname {
288                if !seen.insert(rec.qname().to_owned()) {
289                    continue;
290                }
291            }
292            if let Some(o) = pileup_observation_from_record_with_hq(
293                rec,
294                ref_pos0,
295                ref_base,
296                self.reads_were_realigned,
297                hq,
298            )? {
299                obs.push(o);
300            }
301        }
302        Ok(obs)
303    }

305    /// Non-empty per-sample pileups in **pileup visitation order**, matching encounter order from
306    /// `ReadPileup#getSamples` on the active read list (excluding empty strata like Java splits).
307    pub fn nonempty_stratified_sample_pileups_ordered(
308        &mut self,
309        records: &[bam::Record],
310        header_semantics: &ReadHeaderSemantics,
311        ref_base: u8,
312    ) -> GatkResult<Vec<Vec<PileupObservation>>> {
313        let ref_pos0 = self
314            .last_pos1
315            .ok_or_else(|| GatkError::argument("advance_to before stratified piles"))?
316            .saturating_sub(1) as i64;
317        let mut by_sample: HashMap<String, Vec<PileupObservation>> = HashMap::new();
318        let mut encounter: Vec<String> = Vec::new();
319        for i in 0..self.active.len() {
320            let idx = self.active[i];
321            let hq = self.hq_soft_clip_cached(records, idx);
322            let rec = &records[idx];
323            let sample = rg_sm_from_record(rec, header_semantics)?;
324            if let Some(obs) =
325                pileup_observation_from_record_with_hq(rec, ref_pos0, ref_base, false, hq)?
326            {
327                use std::collections::hash_map::Entry;
328                match by_sample.entry(sample) {
329                    Entry::Vacant(v) => {
330                        // CLONE: needed — encounter order list and HashMap both own sample id.
331                        encounter.push(v.key().clone());
332                        v.insert(vec![obs]);
333                    }
334                    Entry::Occupied(mut o) => {
335                        o.get_mut().push(obs);
336                    }
337                }
338            }
339        }
340
341        Ok(encounter
342            .into_iter()
343            .filter_map(|s| {
344                let p = by_sample.remove(&s)?;
345                if !p.is_empty() {
346                    Some(p)
347                } else {
348                    None
349                }
350            })
351            .collect())
352    }

371pub(crate) fn pileup_observation_from_record(
372    rec: &bam::Record,
373    ref_pos0: i64,
374    ref_base: u8,
375    reads_were_realigned: bool,
376) -> GatkResult<Option<PileupObservation>> {
377    let hq = crate::hq_soft_clip::count_high_quality_soft_clip_bases_rcm(rec);
378    pileup_observation_from_record_with_hq(rec, ref_pos0, ref_base, reads_were_realigned, hq)
379}

381pub(crate) fn pileup_observation_from_record_with_hq(
382    rec: &bam::Record,
383    ref_pos0: i64,
384    ref_base: u8,
385    reads_were_realigned: bool,
386    read_hq_soft_clip_base_count: u32,
387) -> GatkResult<Option<PileupObservation>> {
388    let cigar: Vec<_> = rec.cigar().iter().copied().collect();
389    let seq = rec.seq();
390    let qual = rec.qual();
391    let Some(flags) =
392        pileup_element_flags_at_ref(rec.pos(), &cigar, &seq.as_bytes(), qual, ref_pos0)
393    else {
394        return Ok(None);
395    };
396    let is_alt = if reads_were_realigned {
397        is_alt_after_assembly(flags.read_base, ref_base, flags.is_deletion)
398    } else {
399        is_alt_before_assembly(
400            flags.read_base,
401            ref_base,
402            flags.is_deletion,
403            flags.is_before_deletion_start,
404            flags.is_after_deletion_end,
405            flags.is_before_insertion,
406            flags.is_after_insertion,
407            flags.is_next_to_soft_clip,
408        )
409    };
410    let qual = if flags.is_deletion {
411        REF_MODEL_DELETION_QUAL
412    } else {
413        flags.qual
414    };
415    Ok(Some(PileupObservation {
416        read_base: flags.read_base,
417        qual,
418        is_deletion: flags.is_deletion,
419        is_alt,
420        is_next_to_soft_clip: flags.is_next_to_soft_clip,
421        read_hq_soft_clip_base_count,
422    }))
423}

425fn rg_sm_from_record(rec: &bam::Record, sem: &ReadHeaderSemantics) -> GatkResult<String> {
426    let aux = rec
427        .aux(b"RG")
428        .map_err(|_| GatkError::read("read missing RG auxiliary field"))?;
429    let rg_id: String = match aux {
430        Aux::String(s) => s.to_string(),
431        _ => return Err(GatkError::read("RG tag has unexpected htslib Aux type")),
432    };
433    let resolved = sem.validate_record_links(Some(rg_id.as_str()), None)?;
434    resolved
435        .sample_name
436        .ok_or_else(|| GatkError::read("RG is missing SM sample name in header"))
437}

439/// Borrowing wrapper over [`LocusPileupState`] for short-lived dump scopes.
440/// # Invariants
441/// Lifetime ties records and filter params to the walker; state is owned.
442/// # Ownership
443/// Borrows records/filters; owns inner [`LocusPileupState`].
444/// # Mutation
445/// [`Self::pileup_at`] mutates the inner pileup cursor.
446/// # Biological assumptions
447/// Same as [`LocusPileupState`].
448/// # Java equivalence
449/// Thin Rust wrapper around LIBS-style pileup for dumps.
450#[derive(Debug)]
451pub struct LocusPileupWalker<'a> {
452    records: &'a [bam::Record],
453    filters: &'a ReadFilterParams,
454    state: LocusPileupState,
455}
456
457impl<'a> LocusPileupWalker<'a> {
458    pub fn new(
459        records: &'a [bam::Record],
460        header: &bam::HeaderView,
461        contig: &str,
462        filters: &'a ReadFilterParams,
463    ) -> Self {
464        Self {
465            records,
466            filters,
467            state: LocusPileupState::from_records(records, header, contig, filters),
468        }
469    }
470
471    pub fn pileup_at(&mut self, pos1: u64, ref_base: u8) -> GatkResult<Vec<PileupObservation>> {
472        self.state
473            .pileup_at(self.records, self.filters, pos1, ref_base)
474    }
475}

477/// True when a record overlaps a single 1-based reference position.
478pub fn record_overlaps_locus_1based(
479    record: &bam::Record,
480    header: &bam::HeaderView,
481    contig: &str,
482    pos1: u64,
483    filters: &ReadFilterParams,
484) -> bool {
485    let rn = String::from_utf8_lossy(header.tid2name(record.tid() as u32));
486    if rn != contig {
487        return false;
488    }
489    let Some((r0, r1)) = reference_span0(record, header, filters) else {
490        return false;
491    };
492    let (i0, i1) = closed_interval_1based_to_ref_span0(pos1, pos1);
493    r0 < i1 && i0 < r1
494}

509        let path =

511        if !path.exists() {
512            return;
513        }
514        let filters = ReadFilterParams::gatk_standard_hc();

516        let header = reader.header().clone();
517        let tid = header.tid(b"chr1").expect("chr1") as i32;
518        let mut records = Vec::new();
519        for res in reader.records() {
520            let rec = res.expect("rec");
521            if rec.tid() == tid {
522                records.push(rec);
523            }
524        }

526        let mut prev = 0u64;
527        for pos1 in [5u64, 6, 7, 8] {
528            let pile = walker.pileup_at(pos1, b'N').expect("pileup");
529            assert!(pos1 > prev || prev == 0);
530            prev = pos1;
531            let _ = pile;
532        }
