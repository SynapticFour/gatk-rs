1//! GATK `LoglessPairHMM` — linear-space DP, log10 only on the final sum.
2//! Observable Java contract (`LoglessPairHMM.subComputeReadLikelihoodGivenHaplotypeLog10`):
3//! `INITIAL_CONDITION = 2^1020`; free leading deletions = `INITIAL / haplen`.
4//! Match/Ins/Del updates multiply (not log-sum); priors use `qualToProb` / `qualToErrorProb/3`.
5//! Result = `log10(sum last-row M+I) − log10(INITIAL_CONDITION)`.
6//! This is the scalar reference for SIMD kernels. Exact `Log10PairHMM` remains in
7//! [`crate::pairhmm_log10`] for F.1/F.2 bit-identical dumps.
8
9use gatk_common::{GatkError, GatkResult};
10use std::cell::RefCell;
11use std::sync::OnceLock;
12
13const MAX_QUAL: usize = 127;
14/// `2^1020` — GATK `LoglessPairHMM.INITIAL_CONDITION` (IEEE bits of `ldexp(1, 1020)`).
15/// Do not replace with a rounded decimal literal — a prior `1.07…e307` was ~4.6% low.
16pub const INITIAL_CONDITION: f64 = f64::from_bits(0x7fb0_0000_0000_0000);
17/// `log10(2^1020) = 1020 · log10(2)`. A prior literal `307.652…` was high by `log10(4)`
18/// and shifted every logless score by ≈ −0.602 vs Log10PairHMM.
19pub const INITIAL_CONDITION_LOG10: f64 = 1020.0 * std::f64::consts::LOG10_2;
20
21const MATCH_TO_MATCH: usize = 0;
22const INDEL_TO_MATCH: usize = 1;
23const MATCH_TO_INSERTION: usize = 2;
24const INSERTION_TO_INSERTION: usize = 3;
25const MATCH_TO_DELETION: usize = 4;
26const DELETION_TO_DELETION: usize = 5;
27
28/// GKL-style underflow threshold for f32→f64 retry (linear-space final sum before log10).
29pub const MIN_ACCEPTED_LINEAR_SUM: f64 = 1e-28;

31#[inline]
32pub(crate) fn logless_match_mismatch_prior(qual: u8) -> (f64, f64) {
33    let mismatch = qual_to_error_prob(qual) / 3.0;
34    (qual_to_prob(qual), mismatch)
35}
36
37#[inline]
38fn qual_to_error_prob(qual: u8) -> f64 {
39    10f64.powf(-(qual as f64) / 10.0)
40}
41
42#[inline]
43fn qual_to_prob(qual: u8) -> f64 {
44    1.0 - qual_to_error_prob(qual)
45}

47fn approximate_log10_sum_log10(a: f64, b: f64) -> f64 {
48    let (x, y) = if a > b { (b, a) } else { (a, b) };
49    if x.is_infinite() && x.is_sign_negative() {
50        return y;
51    }
52    y + (1.0 + 10f64.powf(x - y)).log10()
53}

55fn match_to_match_prob(ins_qual: u8, del_qual: u8) -> f64 {
56    let (min_q, max_q) = if ins_qual <= del_qual {
57        (ins_qual as usize, del_qual as usize)
58    } else {
59        (del_qual as usize, ins_qual as usize)
60    };
61    if max_q > MAX_QUAL {
62        let log10_sum = approximate_log10_sum_log10(-0.1 * min_q as f64, -0.1 * max_q as f64);
63        1.0 - 10f64.powf(log10_sum).min(1.0)
64    } else {
65        match_to_match_prob_table()[((max_q * (max_q + 1)) >> 1) + min_q]
66    }
67}
68
69fn match_to_match_prob_table() -> &'static [f64; ((MAX_QUAL + 1) * (MAX_QUAL + 2)) >> 1] {
70    static TABLE: OnceLock<[f64; ((MAX_QUAL + 1) * (MAX_QUAL + 2)) >> 1]> = OnceLock::new();
71    TABLE.get_or_init(|| {
72        let mut table = [0.0_f64; ((MAX_QUAL + 1) * (MAX_QUAL + 2)) >> 1];
73        let mut offset = 0usize;
74        for i in 0..=MAX_QUAL {
75            for j in 0..=i {
76                let log10_sum = approximate_log10_sum_log10(-0.1 * j as f64, -0.1 * i as f64);
77                let log10_m2m = (1.0 - 10f64.powf(log10_sum).min(1.0)).log10();
78                table[offset + j] = 10f64.powf(log10_m2m);
79            }
80            offset += i + 1;
81        }
82        table
83    })
84}

86pub(crate) fn logless_qual_to_trans_probs(ins_qual: u8, del_qual: u8, gcp: u8) -> [f64; 6] {
87    let gcp_err = qual_to_error_prob(gcp);
88    [
89        match_to_match_prob(ins_qual, del_qual),
90        qual_to_prob(gcp),
91        qual_to_error_prob(ins_qual),
92        gcp_err,
93        qual_to_error_prob(del_qual),
94        gcp_err,
95    ]
96}

98pub(crate) fn logless_build_transitions(
99    rn: usize,
100    insertion_gop: &[u8],
101    deletion_gop: &[u8],
102    overall_gcp: &[u8],
103) -> Vec<[f64; 6]> {
104    let mut transitions = vec![[0.0f64; 6]; rn + 1];
105    for i in 0..rn {
106        transitions[i + 1] =
107            logless_qual_to_trans_probs(insertion_gop[i], deletion_gop[i], overall_gcp[i]);
108    }
109    transitions
110}

112struct LoglessScratch {
113    transition: Vec<[f64; 6]>,
114    prior: Vec<f64>,
115    m: Vec<f64>,
116    ins: Vec<f64>,
117    del: Vec<f64>,
118}
119
120impl LoglessScratch {
121    fn new() -> Self {
122        Self {
123            transition: Vec::new(),
124            prior: Vec::new(),
125            m: Vec::new(),
126            ins: Vec::new(),
127            del: Vec::new(),
128        }
129    }
130
131    fn ensure(&mut self, rn: usize, hn: usize) {
132        let rows = rn + 1;
133        let cols = hn + 1;
134        let cells = rows * cols;
135        // Contig-scale dimensions previously ballooned Peak-RSS / footprint to tens of GiB.
136        debug_assert!(
137            rn <= 100_000 && hn <= 100_000 && cells <= 8_000_000,
138            "Logless PairHMM scratch ensure called with contig-scale dims rn={rn} hn={hn}"
139        );
140        if self.transition.len() < rows {
141            self.transition.resize(rows, [0.0; 6]);
142        }
143        if self.prior.len() < cells {
144            self.prior.resize(cells, 0.0);
145            self.m.resize(cells, 0.0);
146            self.ins.resize(cells, 0.0);
147            self.del.resize(cells, 0.0);
148        }
149    }
150
151    fn clear(&mut self) {
152        *self = Self::new();
153    }
154}

167/// Log10 P(read | haplotype) via GATK `LoglessPairHMM` (linear DP).
168pub fn logless_pairhmm_likelihood(
169    read_bases: &[u8],
170    read_quals: &[u8],
171    haplotype_bases: &[u8],
172    insertion_gop: &[u8],
173    deletion_gop: &[u8],
174    overall_gcp: &[u8],
175) -> GatkResult<f64> {
176    validate_inputs(
177        read_bases,
178        read_quals,
179        haplotype_bases,
180        insertion_gop,
181        deletion_gop,
182        overall_gcp,
183    )?;
184    let rn = read_bases.len();
185    let hn = haplotype_bases.len();
186    if rn == 0 {
187        return Ok(0.0);
188    }
189    const MAX_PAIRHMM_DIM: usize = 100_000;
190    const MAX_PAIRHMM_CELLS: usize = 8_000_000;
191    let cells = (rn + 1).saturating_mul(hn + 1);
192    if rn > MAX_PAIRHMM_DIM || hn > MAX_PAIRHMM_DIM || cells > MAX_PAIRHMM_CELLS {
193        return Err(GatkError::algorithm(format!(
194            "PairHMM Logless refused oversized DP (read_len={rn}, hap_len={hn}, cells={cells}); \
195             inputs must be assembly-region scale, not contig scale"
196        )));
197    }
198    LOGLESS_SCRATCH.with(|cell| {
199        let mut scratch = cell.borrow_mut();
200        scratch.ensure(rn, hn);
201        Ok(logless_pairhmm_likelihood_into(
202            read_bases,
203            read_quals,
204            haplotype_bases,
205            insertion_gop,
206            deletion_gop,
207            overall_gcp,
208            &mut scratch,
209        ))
210    })
211}

213/// Parity defaults (45/45/10) for Logless — same quals as Log10 parity path.
214pub fn logless_pairhmm_likelihood_parity_defaults(
215    read_bases: &[u8],
216    read_quals: &[u8],
217    haplotype_bases: &[u8],
218) -> GatkResult<f64> {
219    use crate::pairhmm_log10::{
220        GATK_PARITY_DEFAULT_DEL_QUAL, GATK_PARITY_DEFAULT_GCP, GATK_PARITY_DEFAULT_INS_QUAL,
221    };
222    let n = read_bases.len();
223    let ins = vec![GATK_PARITY_DEFAULT_INS_QUAL; n];
224    let del = vec![GATK_PARITY_DEFAULT_DEL_QUAL; n];
225    let gcp = vec![GATK_PARITY_DEFAULT_GCP; n];
226    logless_pairhmm_likelihood(read_bases, read_quals, haplotype_bases, &ins, &del, &gcp)
227}

229fn validate_inputs(
230    read_bases: &[u8],
231    read_quals: &[u8],
232    haplotype_bases: &[u8],
233    insertion_gop: &[u8],
234    deletion_gop: &[u8],
235    overall_gcp: &[u8],
236) -> GatkResult<()> {
237    if read_bases.len() != read_quals.len()
238        || read_bases.len() != insertion_gop.len()
239        || read_bases.len() != deletion_gop.len()
240        || read_bases.len() != overall_gcp.len()
241    {
242        return Err(GatkError::argument(
243            "PairHMM read arrays must have equal length",
244        ));
245    }
246    if haplotype_bases.is_empty() {
247        return Err(GatkError::argument("haplotype must be non-empty"));
248    }
249    Ok(())
250}

252fn logless_pairhmm_likelihood_into(
253    read_bases: &[u8],
254    read_quals: &[u8],
255    haplotype_bases: &[u8],
256    insertion_gop: &[u8],
257    deletion_gop: &[u8],
258    overall_gcp: &[u8],
259    scratch: &mut LoglessScratch,
260) -> f64 {
261    let rn = read_bases.len();
262    let hn = haplotype_bases.len();
263    let cols = hn + 1;
264
265    for i in 0..rn {
266        scratch.transition[i + 1] =
267            logless_qual_to_trans_probs(insertion_gop[i], deletion_gop[i], overall_gcp[i]);
268    }
269
270    let cells = (rn + 1) * cols;
271    for slot in &mut scratch.m[..cells] {
272        *slot = 0.0;
273    }
274    for slot in &mut scratch.ins[..cells] {
275        *slot = 0.0;
276    }
277    for slot in &mut scratch.del[..cells] {
278        *slot = 0.0;
279    }
280    for slot in &mut scratch.prior[..cells] {
281        *slot = 0.0;
282    }
283
284    for i in 0..rn {
285        let x = read_bases[i];
286        let qual = read_quals[i];
287        let match_p = qual_to_prob(qual);
288        let mismatch_p = qual_to_error_prob(qual) / 3.0;
289        let row = (i + 1) * cols;
290        for j in 0..hn {
291            let y = haplotype_bases[j];
292            scratch.prior[row + j + 1] = if x == y || x == b'N' || y == b'N' {
293                match_p
294            } else {
295                mismatch_p
296            };
297        }
298    }
299
300    let init_del = INITIAL_CONDITION / hn as f64;
301    for j in 0..=hn {
302        scratch.del[j] = init_del;
303    }
304
305    for i in 1..=rn {
306        let t = scratch.transition[i];
307        let row = i * cols;
308        let prev = (i - 1) * cols;
309        for j in 1..=hn {
310            let p = scratch.prior[row + j];
311            scratch.m[row + j] = p
312                * (scratch.m[prev + j - 1] * t[MATCH_TO_MATCH]
313                    + scratch.ins[prev + j - 1] * t[INDEL_TO_MATCH]
314                    + scratch.del[prev + j - 1] * t[INDEL_TO_MATCH]);
315            scratch.ins[row + j] = scratch.m[prev + j] * t[MATCH_TO_INSERTION]
316                + scratch.ins[prev + j] * t[INSERTION_TO_INSERTION];
317            scratch.del[row + j] = scratch.m[row + j - 1] * t[MATCH_TO_DELETION]
318                + scratch.del[row + j - 1] * t[DELETION_TO_DELETION];
319        }
320    }
321
322    let end_row = rn * cols;
323    let mut final_sum = 0.0;
324    for j in 1..=hn {
325        final_sum += scratch.m[end_row + j] + scratch.ins[end_row + j];
326    }
327    if final_sum <= 0.0 || !final_sum.is_finite() {
328        return f64::NEG_INFINITY;
329    }
330    final_sum.log10() - INITIAL_CONDITION_LOG10
331}

333/// Score one read against many haplotypes (scalar Logless; haplotypes sequential).
334pub fn logless_pairhmm_likelihoods(
335    read_bases: &[u8],
336    read_quals: &[u8],
337    haplotype_bases: &[&[u8]],
338    insertion_gop: &[u8],
339    deletion_gop: &[u8],
340    overall_gcp: &[u8],
341) -> GatkResult<Vec<f64>> {
342    haplotype_bases
343        .iter()
344        .map(|hap| {
345            logless_pairhmm_likelihood(
346                read_bases,
347                read_quals,
348                hap,
349                insertion_gop,
350                deletion_gop,
351                overall_gcp,
352            )
353        })
354        .collect()
355}

363    /// Looser than Logless↔Logless: Log10 vs Logless are different engines.
364    fn logless_vs_log10_policy() -> PairHmmFpPolicy {
365        PairHmmFpPolicy {
366            abs_epsilon: 1e-4,
367            rel_epsilon: 1e-3,
368        }

370
371    #[test]
372    fn initial_condition_constants_match_java() {
373        assert!((INITIAL_CONDITION - 2f64.powi(1020)).abs() / INITIAL_CONDITION < 1e-15);
374        assert!((INITIAL_CONDITION_LOG10 - 1020.0 * 2f64.log10()).abs() < 1e-12);
375    }
376
377    #[test]
378    fn perfect_match_finite_and_near_log10() {
379        let read = b"ACGTACGTAC";
380        let quals = vec![30u8; read.len()];
381        let hap = b"ACGTACGTAC";
382        let ll = logless_pairhmm_likelihood_parity_defaults(read, &quals, hap).unwrap();
383        let log10 = log10_pairhmm_likelihood_parity_defaults(read, &quals, hap).unwrap();
384        assert!(ll.is_finite(), "ll={ll}");
385        assert!(
386            pairhmm_fp_eq(ll, log10, logless_vs_log10_policy()),
387            "logless={ll} log10={log10}"
388        );
389    }
390
391    #[test]
392    fn mismatch_and_short_read() {
393        let read = b"A";
394        let quals = [20u8];
395        let hap = b"ACGT";
396        let ll = logless_pairhmm_likelihood_parity_defaults(read, &quals, hap).unwrap();
397        assert!(ll.is_finite() || ll.is_infinite());
398        let log10 = log10_pairhmm_likelihood_parity_defaults(read, &quals, hap).unwrap();
399        assert!(pairhmm_fp_eq(ll, log10, logless_vs_log10_policy()) || ll.is_infinite());
400    }
