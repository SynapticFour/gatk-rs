5/// GATK `MathUtils.scaleLogSpaceArrayForNumericalStability`.
6fn scale_log_space_array_for_numerical_stability(gl: &[f64]) -> Vec<f64> {
7    if gl.is_empty() {
8        return Vec::new();
9    }
10    let max_v = gl.iter().copied().fold(f64::NEG_INFINITY, f64::max);
11    gl.iter().map(|g| g - max_v).collect()
12}

14/// Diploid genotype pairs in GATK order: for each `j`, all `(i, j)` with `i <= j`.
15fn diploid_genotype_pairs(allele_count: usize) -> Vec<(usize, usize)> {
16    let mut pairs = Vec::new();
17    for j in 0..allele_count {
18        for i in 0..=j {
19            pairs.push((i, j));
20        }
21    }
22    pairs
23}

25fn num_genotype_likelihoods(allele_count: usize, ploidy: usize) -> usize {
26    debug_assert!(ploidy == 2, "parity v1 supports diploid only");
27    allele_count.saturating_mul(allele_count + 1) / 2
28}

30/// GATK `AlleleSubsettingUtils.subsettedPLIndices` for diploid genotypes.
31fn subsetted_pl_indices(
32    ploidy: usize,
33    original_allele_count: usize,
34    keep_allele_indices: &[usize],
35) -> Vec<usize> {
36    debug_assert_eq!(ploidy, 2);
37    let old_pairs = diploid_genotype_pairs(original_allele_count);
38    let new_pairs = diploid_genotype_pairs(keep_allele_indices.len());
39    let old_index: std::collections::HashMap<(usize, usize), usize> = old_pairs
40        .iter()
41        .enumerate()
42        .map(|(idx, &(i, j))| ((i, j), idx))
43        .collect();
44    new_pairs
45        .iter()
46        .map(|&(ni, nj)| {
47            let oi = keep_allele_indices[ni];
48            let oj = keep_allele_indices[nj];
49            let (oi, oj) = if oi <= oj { (oi, oj) } else { (oj, oi) };
50            *old_index.get(&(oi, oj)).expect("old genotype index")
51        })
52        .collect()
53}
54
55fn subset_ad(original_ad: &[i32], keep_allele_indices: &[usize]) -> Vec<i32> {
56    keep_allele_indices
57        .iter()
58        .map(|&i| original_ad[i])
59        .collect()
60}

62/// Subset log10 genotype likelihoods from `original_allele_count` to `keep_allele_indices`.
63pub fn subset_log10_genotype_likelihoods(
64    original_gl: &[f64],
65    original_allele_count: usize,
66    keep_allele_indices: &[usize],
67) -> GatkResult<Vec<f64>> {
68    let ploidy = 2;
69    let expected = num_genotype_likelihoods(original_allele_count, ploidy);
70    if original_gl.len() != expected {
71        return Err(GatkError::argument(format!(
72            "subset PL: expected {} log10 likelihoods for {original_allele_count} alleles, got {}",
73            expected,
74            original_gl.len()
75        )));
76    }
77    let map = subsetted_pl_indices(ploidy, original_allele_count, keep_allele_indices);
78    let picked: Vec<f64> = map.iter().map(|&idx| original_gl[idx]).collect();
79    Ok(scale_log_space_array_for_numerical_stability(&picked))
80}

82/// Log10 GL to integer PL (GATK `GenotypeBuilder` / `FastGenotype` round-trip).
83fn log10_gl_to_int_pl(gl: &[f64]) -> Vec<i32> {
84    if gl.is_empty() {
85        return Vec::new();
86    }
87    let max_v = gl.iter().copied().fold(f64::NEG_INFINITY, f64::max);
88    let mut pls: Vec<i32> = gl
89        .iter()
90        .map(|g| {
91            let shifted = (-10.0_f64) * (g - max_v);
92            shifted.min(i32::MAX as f64).round() as i32
93        })
94        .collect();
95    let min_pl = pls.iter().copied().min().unwrap_or(0);
96    for pl in &mut pls {
97        *pl -= min_pl;
98    }
99    pls
100}

102const SAC_STRANDS: usize = 2;
103
104fn subset_sac(original_sac: &[i32], keep_allele_indices: &[usize]) -> GatkResult<Vec<i32>> {
105    if original_sac.len() % SAC_STRANDS != 0 {
106        return Err(GatkError::argument(format!(
107            "SAC length {} not divisible by {SAC_STRANDS}",
108            original_sac.len()
109        )));
110    }
111    let mut out = Vec::with_capacity(keep_allele_indices.len() * SAC_STRANDS);
112    for &allele_idx in keep_allele_indices {
113        out.push(original_sac[SAC_STRANDS * allele_idx]);
114        out.push(original_sac[SAC_STRANDS * allele_idx + 1]);
115    }
116    Ok(out)
117}

119/// One row of `g-subset-pl` / `g-subset-vc` fixture output.
120/// # Invariants
121/// `pl` / `ad` lengths match diploid genotype count / allele count **after** subsetting.
122/// `allele_count_before` ≥ `allele_count_after`.
123/// # Ownership
124/// Owns PL/AD(/SAC) vectors for parity fixture comparison.
125/// # Mutation
126/// Immutable subset result.
127/// # Biological assumptions
128/// Allele subsetting remaps genotype likelihoods when dropping unused ALTs.
129/// # Java equivalence
130/// GATK allele-subsetting utilities (`AlleleSubsettingUtils` PL/AD/SAC remap).
131#[derive(Debug, Clone, PartialEq)]
132pub struct SubsetAllelesPlResult {
133    pub allele_count_before: usize,
134    pub allele_count_after: usize,
135    pub pl: Vec<i32>,
136    pub ad: Vec<i32>,
137    pub gq: Option<i32>,
138    pub sac: Option<Vec<i32>>,
139}

141/// Trim ACG to AC with het ref/C log10 PL fixture (GATK unit-test values).
142pub fn subset_trim_acg_to_ac(log10_pl: &[f64], ad: &[i32]) -> GatkResult<SubsetAllelesPlResult> {
143    let gl = subset_log10_genotype_likelihoods(log10_pl, 3, &[0, 1])?;
144    let pl = log10_gl_to_int_pl(&gl);
145    let ad_out = subset_ad(ad, &[0, 1]);
146    Ok(SubsetAllelesPlResult {
147        allele_count_before: 3,
148        allele_count_after: 2,
149        pl,
150        ad: ad_out,
151        gq: Some(200),
152        sac: None,
153    })
154}

171/// ACG to AC with PL + AD + SAC (GATK `AlleleSubsettingUtilsUnitTest` het AC row).
172pub fn subset_het_ac_sac(
173    log10_pl: &[f64],
174    ad: &[i32],
175    sac: &[i32],
176) -> GatkResult<SubsetAllelesPlResult> {
177    let gl = subset_log10_genotype_likelihoods(log10_pl, 3, &[0, 1])?;
178    let pl = log10_gl_to_int_pl(&gl);
179    let ad_out = subset_ad(ad, &[0, 1]);
180    let sac_out = subset_sac(sac, &[0, 1])?;
181    Ok(SubsetAllelesPlResult {
182        allele_count_before: 3,
183        allele_count_after: 2,
184        pl,
185        ad: ad_out,
186        gq: Some(200),
187        sac: Some(sac_out),
188    })
189}
