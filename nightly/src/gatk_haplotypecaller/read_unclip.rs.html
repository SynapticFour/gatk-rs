10/// HC default: `dontUseSoftClippedBases=false`, `overrideSoftclipFragmentCheck=false`.
11/// # Invariants
12/// [`Self::use_hard_clip_soft_bases`] mirrors GATK soft-clip vs hard-clip decision per read flags.
13/// Defaults match HC `AssemblyRegionArgumentCollection` soft-clip settings.
14/// # Ownership
15/// [`Copy`] policy passed into finalize/unclip helpers; records are mutably borrowed.
16/// # Mutation
17/// Policy immutable; read CIGAR/bases may be hard-clipped when policy demands.
18/// # Biological assumptions
19/// Soft-clipped bases are excluded when fragment size is ill-defined or `dontUseSoftClippedBases` is set.
20/// # Java equivalence
21/// GATK `AssemblyBasedCallerUtils.finalizeRegion` + `ReadClipper` / `dontUseSoftClippedBases`.
22#[derive(Debug, Clone, Copy)]
23pub struct HcSoftclipPolicy {
24    pub dont_use_soft_clipped_bases: bool,
25    pub override_softclip_fragment_check: bool,
26}
27
28impl HcSoftclipPolicy {
29    pub fn haplotype_caller_defaults() -> Self {
30        Self {
31            dont_use_soft_clipped_bases: false,
32            override_softclip_fragment_check: false,
33        }
34    }
35
36    pub fn use_hard_clip_soft_bases(&self, rec: &bam::Record) -> bool {
37        self.dont_use_soft_clipped_bases
38            || !(self.override_softclip_fragment_check || has_well_defined_fragment_size(rec))
39    }
40}

42/// `ReadUtils.hasWellDefinedFragmentSize`.
43pub fn has_well_defined_fragment_size(rec: &bam::Record) -> bool {
44    let flags = rec.flags();
45    const PAIRED: u16 = 0x1;
46    const UNMAPPED: u16 = 0x4;
47    const MATE_UNMAPPED: u16 = 0x8;
48    const REVERSE: u16 = 0x10;
49    const MATE_REVERSE: u16 = 0x20;
50    if rec.insert_size() == 0 {
51        return false;
52    }
53    if flags & PAIRED == 0 {
54        return false;
55    }
56    if flags & UNMAPPED != 0 || flags & MATE_UNMAPPED != 0 {
57        return false;
58    }
59    let read_rev = flags & REVERSE != 0;
60    let mate_rev = flags & MATE_REVERSE != 0;
61    read_rev != mate_rev
62}

64pub(crate) fn cigar_len(c: &Cigar) -> u32 {
65    match c {
66        Cigar::Match(n)
67        | Cigar::Ins(n)
68        | Cigar::Del(n)
69        | Cigar::SoftClip(n)
70        | Cigar::HardClip(n)
71        | Cigar::Equal(n)
72        | Cigar::Diff(n)
73        | Cigar::RefSkip(n)
74        | Cigar::Pad(n) => *n,
75    }
76}

78pub(crate) fn consumes_read_bases(c: &Cigar) -> bool {
79    matches!(
80        c,
81        Cigar::Match(_) | Cigar::Ins(_) | Cigar::SoftClip(_) | Cigar::Equal(_) | Cigar::Diff(_)
82    )

96fn revert_soft_clips_cigar(cigar: &[Cigar]) -> Vec<Cigar> {
97    cigar
98        .iter()
99        .map(|c| match c {
100            Cigar::SoftClip(n) => Cigar::Match(*n),
101            Cigar::Match(n) => Cigar::Match(*n),
102            Cigar::Ins(n) => Cigar::Ins(*n),
103            Cigar::Del(n) => Cigar::Del(*n),
104            Cigar::HardClip(n) => Cigar::HardClip(*n),
105            Cigar::Equal(n) => Cigar::Equal(*n),
106            Cigar::Diff(n) => Cigar::Diff(*n),
107            Cigar::RefSkip(n) => Cigar::RefSkip(*n),
108            Cigar::Pad(n) => Cigar::Pad(*n),
109        })
110        .collect()
111}

113fn cigar_op_key(c: Cigar) -> u8 {
114    match c {
115        Cigar::Match(_) => 0,
116        Cigar::Ins(_) => 1,
117        Cigar::Del(_) => 2,
118        Cigar::SoftClip(_) => 3,
119        Cigar::HardClip(_) => 4,
120        Cigar::Equal(_) => 5,
121        Cigar::Diff(_) => 6,
122        Cigar::RefSkip(_) => 7,
123        Cigar::Pad(_) => 8,
124    }
125}

127/// `CigarBuilder.make` — merge consecutive identical operators.
128fn normalize_cigar(elems: Vec<Cigar>) -> Vec<Cigar> {
129    let mut out = Vec::new();
130    for c in elems {
131        if c == Cigar::Match(0) {
132            continue;
133        }
134        if let Some(last) = out.last_mut() {
135            if cigar_op_key(*last) == cigar_op_key(c) {
136                let merged = cigar_len(last) + cigar_len(&c);
137                *last = op_with_len(c, merged);
138                continue;
139            }
140        }
141        out.push(c);
142    }
143    out
144}

146/// GATK `getSoftStart` / `getStart` (1-based ref coords; HTSJDK `POS` is 0-based).
147/// GATK `GATKRead.getSoftStart` (1-based).
148pub fn gatk_soft_start_1based(rec: &bam::Record) -> i64 {
149    let mut soft_start = rec.pos() + 1;
150    for c in rec.cigar().iter() {
151        if let Cigar::SoftClip(sc_len) = c {
152            soft_start -= i64::from(*sc_len);
153        } else if !matches!(c, Cigar::HardClip(_)) {
154            break;
155        }
156    }
157    soft_start
158}

161    template: &bam::Record,
162    cigar: &CigarString,
163    seq: &[u8],
164    qual: &[u8],

177/// `CigarUtils.clipCigar` (hard or soft clip operator).
178fn clip_cigar(cigar: &[Cigar], start: usize, stop: usize, clip_op: Cigar) -> Vec<Cigar> {
179    debug_assert!(matches!(clip_op, Cigar::SoftClip(_) | Cigar::HardClip(_)));
180    let clip_left = start == 0;
181    let mut new_cigar = Vec::new();
182    let mut element_start = 0usize;
183    for c in cigar {
184        if matches!(c, Cigar::HardClip(_)) {
185            new_cigar.push(*c);
186            continue;
187        }
188        let element_end = element_start
189            + if consumes_read_bases(c) {
190                cigar_len(c) as usize
191            } else {
192                0
193            };
194        if element_end <= start || element_start >= stop {
195            if consumes_read_bases(c) || (element_start != start && element_start != stop) {
196                new_cigar.push(*c);
197            }
198        } else {
199            let unclipped_len = if clip_left {
200                element_end.saturating_sub(stop)
201            } else {
202                start.saturating_sub(element_start)
203            };
204            let clipped_len = cigar_len(c) as usize - unclipped_len;
205            if unclipped_len == 0 {
206                if consumes_read_bases(c) {
207                    new_cigar.push(clip_op_with_len(clip_op, cigar_len(c)));
208                }
209            } else if clip_left {
210                new_cigar.push(clip_op_with_len(clip_op, clipped_len as u32));
211                new_cigar.push(op_with_len(*c, unclipped_len as u32));
212            } else {
213                new_cigar.push(op_with_len(*c, unclipped_len as u32));
214                new_cigar.push(clip_op_with_len(clip_op, clipped_len as u32));
215            }
216        }
217        element_start = element_end;
218    }
219    new_cigar
220}

222fn clip_op_with_len(op: Cigar, len: u32) -> Cigar {
223    match op {
224        Cigar::SoftClip(_) => Cigar::SoftClip(len),
225        Cigar::HardClip(_) => Cigar::HardClip(len),
226        _ => op,
227    }
228}

230fn op_with_len(c: Cigar, len: u32) -> Cigar {
231    match c {
232        Cigar::Match(_) => Cigar::Match(len),
233        Cigar::Ins(_) => Cigar::Ins(len),
234        Cigar::Del(_) => Cigar::Del(len),
235        Cigar::SoftClip(_) => Cigar::SoftClip(len),
236        Cigar::HardClip(_) => Cigar::HardClip(len),
237        Cigar::Equal(_) => Cigar::Equal(len),
238        Cigar::Diff(_) => Cigar::Diff(len),
239        Cigar::RefSkip(_) => Cigar::RefSkip(len),
240        Cigar::Pad(_) => Cigar::Pad(len),
241    }
242}

244/// `CigarUtils.alignmentStartShift` for left hard-clip (read-base index `num_clipped`).
245fn alignment_start_shift(cigar: &[Cigar], num_clipped: usize) -> i32 {
246    let mut ref_bases = 0i32;
247    let mut element_start = 0usize;
248    for c in cigar {
249        if matches!(c, Cigar::HardClip(_)) {
250            continue;
251        }
252        let element_end = element_start
253            + if consumes_read_bases(c) {
254                cigar_len(c) as usize
255            } else {
256                0
257            };
258        if element_end <= num_clipped {
259            if consumes_ref_bases(c) {
260                ref_bases += cigar_len(c) as i32;
261            }
262        } else if element_start < num_clipped {
263            let clipped_len = (num_clipped - element_start) as u32;
264            if consumes_ref_bases(c) {
265                ref_bases += clipped_len as i32;
266            }
267            break;
268        }
269        element_start = element_end;
270    }
271    ref_bases
272}

274/// `ClippingOp.applyHardClipBases`.
275fn apply_hard_clip_bases(rec: &bam::Record, start: usize, stop: usize) -> bam::Record {
276    let seq = rec.seq().as_bytes();
277    let qual = rec.qual().to_vec();
278    let read_len = seq.len();
279    if read_len == 0 {
280        return rec.clone();
281    }
282    let clip_len = stop.saturating_sub(start).saturating_add(1);
283    let new_length = read_len.saturating_sub(clip_len);
284    if new_length == 0 {
285        let mut out = bam::Record::new();
286        out.set_qname(rec.qname());
287        out.set_flags(rec.flags() | 0x4);
288        return out;
289    }
290    let copy_start = if start == 0 { stop + 1 } else { 0 };
291    let new_seq = &seq[copy_start..copy_start + new_length];
292    let new_qual = &qual[copy_start..copy_start + new_length.min(qual.len())];
293    const UNMAPPED: u16 = 0x4;
294    let old_cigar: Vec<Cigar> = rec.cigar().iter().copied().collect();
295    let cigar = if rec.flags() & UNMAPPED != 0 {
296        CigarString::from(vec![])
297    } else {
298        let elems = clip_cigar(&old_cigar, start, stop + 1, Cigar::HardClip(0));
299        CigarString::from(normalize_cigar(elems))
300    };
301    let mut out = replace_record_body(rec, &cigar, new_seq, new_qual);
302    if start == 0 && out.flags() & UNMAPPED == 0 {
303        let shift = alignment_start_shift(&old_cigar, stop + 1);
304        out.set_pos(rec.pos() + i64::from(shift));
305    }
306    normalize_record_cigar(&mut out);
307    out
308}

310/// Merge consecutive identical cigar ops (e.g. multiple hard clips from finalize steps).
311pub fn normalize_record_cigar(rec: &mut bam::Record) {
312    if rec.flags() & 0x4 != 0 {
313        return;
314    }
315    let elems: Vec<Cigar> = rec.cigar().iter().copied().collect();
316    let cigar = CigarString::from(normalize_cigar(elems));
317    let qname = rec.qname().to_vec();
318    let seq = rec.seq().as_bytes();
319    let qual = rec.qual().to_vec();
320    rec.set(&qname, Some(&cigar), &seq, &qual);
321}

323/// `ReadClipper.revertSoftClippedBases` + `ClippingOp.applyRevertSoftClippedBases`.
324pub fn revert_soft_clipped_bases(rec: &bam::Record) -> bam::Record {
325    let cigar_vec: Vec<Cigar> = rec.cigar().iter().copied().collect();
326    if cigar_vec.is_empty()
327        || (!cigar_vec.first().is_some_and(is_clipping_op)
328            && !cigar_vec.last().is_some_and(is_clipping_op))
329    {
330        return rec.clone();
331    }
332    // GATK uses `read.getSoftStart` on the original read (before S→M), not the reverted cigar.
333    let new_start = gatk_soft_start_1based(rec);
334    let new_cigar = CigarString::from(normalize_cigar(revert_soft_clips_cigar(&cigar_vec)));
335    let qual = rec.qual().to_vec();
336    let seq_bytes = rec.seq().as_bytes();
337    let mut out = replace_record_body(rec, &new_cigar, &seq_bytes, &qual);
338    if new_start <= 0 {
339        if out.flags() & 0x4 == 0 {
340            out.set_pos(0);
341        }
342        let stop = (-new_start) as usize;
343        out = apply_hard_clip_bases(&out, 0, stop);
344        if out.flags() & 0x4 == 0 {
345            out.set_pos(0);
346        }
347        return out;
348    }
349    out.set_pos(new_start - 1);
350    out
351}

353/// `ReadClipper.hardClipSoftClippedBases`.
354pub fn hard_clip_soft_clipped_bases(rec: &bam::Record) -> bam::Record {
355    let seq_len = rec.seq().as_bytes().len();
356    if seq_len == 0 {
357        return rec.clone();
358    }
359    let mut read_index = 0usize;
360    let mut cut_left = None::<usize>;
361    let mut cut_right = None::<usize>;
362    let mut right_tail = false;
363    for c in rec.cigar().iter() {
364        if matches!(c, Cigar::SoftClip(_)) {
365            if right_tail {
366                cut_right = Some(read_index);
367            } else {
368                cut_left = Some(read_index + cigar_len(c) as usize - 1);
369            }
370        } else if !matches!(c, Cigar::HardClip(_)) {
371            right_tail = true;
372        }
373        if consumes_read_bases(c) {
374            read_index += cigar_len(c) as usize;
375        }
376    }
377    let mut out = rec.clone();
378    if let Some(right) = cut_right {
379        out = apply_hard_clip_bases(&out, right, seq_len.saturating_sub(1));
380    }
381    if let Some(left) = cut_left {
382        out = apply_hard_clip_bases(&out, 0, left);
383    }
384    out
385}

400fn clip_low_qual_ends(rec: &bam::Record, low_qual: u8, clip_op: Cigar) -> bam::Record {
401    let hard = matches!(clip_op, Cigar::HardClip(_));
402    let seq_len = rec.seq().as_bytes().len();
403    if seq_len == 0 {
404        return rec.clone();
405    }
406    let quals = rec.qual();
407    let mut left = 0isize;
408    let mut right = seq_len.saturating_sub(1) as isize;
409    while right >= 0 && quals.get(right as usize).copied().unwrap_or(0) <= low_qual {
410        right -= 1;
411    }
412    while (left as usize) < seq_len && quals.get(left as usize).copied().unwrap_or(0) <= low_qual {
413        left += 1;
414    }
415    if left > right {
416        return empty_unmapped_read(rec);
417    }
418    let left = left as usize;
419    let right = right as usize;
420    if hard {
421        let mut out = rec.clone();
422        if right < seq_len.saturating_sub(1) {
423            out = apply_hard_clip_bases(&out, right.saturating_add(1), seq_len.saturating_sub(1));
424        }
425        if left > 0 {
426            out = apply_hard_clip_bases(&out, 0, left.saturating_sub(1));
427        }
428        return out;
429    }
430    let old_cigar: Vec<Cigar> = rec.cigar().iter().copied().collect();
431    let mut cigar = old_cigar;
432    if right < seq_len.saturating_sub(1) {
433        cigar = clip_cigar(&cigar, right.saturating_add(1), seq_len, clip_op);
434    }
435    if left > 0 {
436        cigar = clip_cigar(&cigar, 0, left, clip_op);
437    }
438    let cigar = CigarString::from(normalize_cigar(cigar));
439    let seq = rec.seq().as_bytes();
440    let qual = rec.qual();
441    let mut out = replace_record_body(rec, &cigar, &seq, qual);
442    normalize_record_cigar(&mut out);
443    out
444}

446/// First finalizeRegion soft-clip step (no tail-qual or adaptor clipping).
447pub fn apply_hc_softclip_pre_step(
448    rec: &bam::Record,
449    policy: &HcSoftclipPolicy,
450) -> (bam::Record, &'static str, Option<(i32, i32)>) {
451    if policy.use_hard_clip_soft_bases(rec) {
452        return (hard_clip_soft_clipped_bases(rec), "hard_clip", None);
453    }
454    let soft_start = (rec.pos() + 1) as i32;
455    let soft_end = alignment_end_1based(rec);
456    let out = revert_soft_clipped_bases(rec);
457    if out.flags() & 0x4 == 0 {
458        return (out, "revert", Some((soft_start, soft_end)));
459    }
460    (out, "revert", None)
461}

463pub fn alignment_end_1based(rec: &bam::Record) -> i32 {
464    let pos1 = (rec.pos() + 1) as i32;
465    let mut ref_len = 0i32;
466    for c in rec.cigar().iter() {
467        if consumes_ref_bases(c) {
468            ref_len += cigar_len(c) as i32;
469        }
470    }
471    if ref_len > 0 {
472        pos1 + ref_len - 1
473    } else {
474        pos1
475    }
476}

481/// GATK `ReadUtils.getAdaptorBoundary` (1-based reference coordinate).
482pub fn adaptor_boundary_1based(rec: &bam::Record) -> Option<i64> {
483    if !has_well_defined_fragment_size(rec) {
484        return None;
485    }
486    const REVERSE: u16 = 0x10;
487    if rec.flags() & REVERSE != 0 {
488        // GATK `getMateStart - 1` (0-based SAM `mpos` = 1-based mate start − 1).
489        Some(rec.mpos())
490    } else {
491        let insert = rec.insert_size().unsigned_abs() as i64;
492        Some((rec.pos() + 1) + insert)
493    }
494}

496fn is_inside_read_sam(rec: &bam::Record, ref_coord: i64) -> bool {
497    let start = rec.pos() + 1;
498    let end = i64::from(alignment_end_1based(rec));
499    ref_coord >= start && ref_coord <= end
500}

502/// Returns read index and whether the cigar op at that ref coord consumes read bases.
503fn read_index_and_op_for_ref_coord_1based(
504    rec: &bam::Record,
505    ref_coord_1based: i64,
506) -> Option<(usize, bool)> {
507    let alignment_start = gatk_soft_start_1based(rec);
508    if ref_coord_1based < alignment_start {
509        return None;
510    }
511    let mut last_read = 0usize;
512    let mut last_ref = alignment_start;
513    for c in rec.cigar().iter() {
514        let op_len = i64::from(cigar_len(c));
515        let op_consumes_read = consumes_read_bases(c);
516        let op_consumes_ref = consumes_ref_bases(c) || matches!(c, Cigar::SoftClip(_));
517        let first_read = last_read;
518        let first_ref = last_ref;
519        if op_consumes_read {
520            last_read += op_len as usize;
521        }
522        if op_consumes_ref {
523            last_ref += op_len;
524        }
525        if first_ref <= ref_coord_1based && ref_coord_1based < last_ref {
526            if matches!(c, Cigar::SoftClip(_) | Cigar::HardClip(_)) {
527                return None;
528            }
529            let read_pos = if op_consumes_read {
530                first_read + (ref_coord_1based - first_ref) as usize
531            } else {
532                first_read
533            };
534            return Some((read_pos, op_consumes_read));
535        }
536    }
537    None
538}

540fn read_index_at_ref_for_left_tail(rec: &bam::Record, ref_coord_1based: i64) -> Option<usize> {
541    // GATK left-tail hard clip uses `getReadIndexForReferenceCoordinate` then decrements stop only
542    // when the coordinate falls in a deletion (non-read-consuming op).
543    let (idx, op_consumes_read) = read_index_and_op_for_ref_coord_1based(rec, ref_coord_1based)?;
544    Some(if op_consumes_read {
545        idx
546    } else {
547        idx.saturating_sub(1)
548    })
549}
550
551fn read_index_at_ref_or_none(rec: &bam::Record, ref_coord_1based: i64) -> Option<usize> {
552    read_index_and_op_for_ref_coord_1based(rec, ref_coord_1based).map(|(i, _)| i)
553}

555/// GATK `ReadClipper.hardClipAdaptorSequence`.
556pub fn hard_clip_adaptor_sequence(rec: &bam::Record) -> bam::Record {
557    let Some(boundary) = adaptor_boundary_1based(rec) else {
558        return rec.clone();
559    };
560    if !is_inside_read_sam(rec, boundary) {
561        return rec.clone();
562    }
563    const REVERSE: u16 = 0x10;
564    if rec.flags() & REVERSE != 0 {
565        hard_clip_by_ref_left_tail(rec, boundary)
566    } else {
567        hard_clip_by_ref_right_tail(rec, boundary)
568    }
569}

571fn hard_clip_by_ref_left_tail(rec: &bam::Record, ref_stop_1based: i64) -> bam::Record {
572    let Some(stop_idx) = read_index_at_ref_for_left_tail(rec, ref_stop_1based) else {
573        return rec.clone();
574    };
575    apply_hard_clip_bases(rec, 0, stop_idx)
576}
577
578fn hard_clip_by_ref_right_tail(rec: &bam::Record, ref_start_1based: i64) -> bam::Record {
579    let Some(start_idx) = read_index_at_ref_or_none(rec, ref_start_1based) else {
580        return rec.clone();
581    };
582    let len = rec.seq().as_bytes().len();
583    if start_idx >= len {
584        return empty_unmapped_read(rec);
585    }
586    apply_hard_clip_bases(rec, start_idx, len.saturating_sub(1))
587}
588
589fn empty_unmapped_read(rec: &bam::Record) -> bam::Record {
590    let mut out = bam::Record::new();
591    out.set_qname(rec.qname());
592    out.set_flags(rec.flags() | 0x4);
593    out
594}
595
596/// GATK `ReadClipper.hardClipToRegion` on the assembly region padded span (1-based inclusive).
597pub fn hard_clip_to_region(
598    rec: &bam::Record,
599    ref_start_1based: u64,
600    ref_stop_1based: u64,
601) -> bam::Record {
602    let rs = ref_start_1based as i64;
603    let re = ref_stop_1based as i64;
604    let aln_start = rec.pos() + 1;
605    let aln_end = i64::from(alignment_end_1based(rec));
606    if aln_start > re || aln_end < rs {
607        return empty_unmapped_read(rec);
608    }
609    if aln_start < rs && aln_end > re {
610        let mut r = hard_clip_by_ref_left_tail(rec, rs - 1);
611        r = hard_clip_by_ref_right_tail(&r, re + 1);
612        return r;
613    }
614    if aln_start < rs {
615        return hard_clip_by_ref_left_tail(rec, rs - 1);
616    }
617    if aln_end > re {
618        return hard_clip_by_ref_right_tail(rec, re + 1);
619    }
620    rec.clone()
621}
622
623pub(crate) fn read_has_positive_cigar_length(rec: &bam::Record) -> bool {
624    rec.cigar()
625        .iter()
626        .any(|c| consumes_read_bases(c) && cigar_len(c) > 0)
627}
