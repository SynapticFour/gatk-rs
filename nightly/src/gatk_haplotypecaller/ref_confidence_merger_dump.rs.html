10/// Per-sample genotype row in H-D02 reference-confidence merge parity dumps.
11/// # Invariants
12/// Mirrors [`MergeGenotypeOut`](crate::ref_confidence_merger::MergeGenotypeOut) plus optional GQ for dump completeness.
13/// # Ownership
14/// Owns name and optional PL/AD/GQ vectors for TSV serialization.
15/// # Mutation
16/// Immutable dump snapshot.
17/// # Biological assumptions
18/// None — parity fixture/dump shape.
19/// # Java equivalence
20/// Rust-native dump of GATK `ReferenceConfidenceVariantContextMerger` unit-test outputs.
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub struct DumpGenotype {
23    pub name: String,
24    pub pl: Option<Vec<i32>>,
25    pub ad: Option<Vec<i32>>,
26    pub gq: Option<i32>,
27}

29/// Full merged-site dump for one H-D02 parity case.
30/// # Invariants
31/// When `merged_null` is true, allele/genotype vectors are empty (Java null merge).
32/// # Ownership
33/// Owns case metadata, alleles, and genotype dump rows.
34/// # Mutation
35/// Immutable after fixture merge or null-merge construction.
36/// # Biological assumptions
37/// None — test/dump artifact.
38/// # Java equivalence
39/// GATK `ReferenceConfidenceVariantContextMergerUnitTest` expected merge snapshots.
40#[derive(Debug, Clone, PartialEq, Eq)]
41pub struct RefConfidenceMergeDump {
42    pub case_id: String,
43    pub contig: String,
44    pub pos: u64,
45    pub alleles: Vec<String>,
46    pub genotypes: Vec<DumpGenotype>,
47    pub has_non_ref: bool,
48    pub merged_null: bool,
49}

330pub fn merge_reference_confidence_fixture(case_id: &str) -> GatkResult<RefConfidenceMergeDump> {
331    if case_id == "merge_spanning_ref_only" || case_id == "test08" {
332        let (contig, start, vcs) = unit_test_vcs(case_id)?;
333        if merge_reference_confidence(&contig, start, &vcs, None, true, false)?.is_none() {
334            return Ok(null_merge_dump(case_id));
335        }
336    }
337    let (contig, start, vcs) = unit_test_vcs(case_id)?;
338    let merged = merge_reference_confidence(&contig, start, &vcs, None, true, false)?
339        .ok_or_else(|| gatk_common::GatkError::argument("merge returned null"))?;
340    Ok(result_to_dump(case_id, merged))
341}

343pub fn dump_ref_confidence_merge_tsv(
344    merge: &RefConfidenceMergeDump,
345    out: &mut impl Write,
346) -> GatkResult<()> {
347    writeln!(out, "case_id\t{}", merge.case_id)?;
348    writeln!(out, "contig\t{}", merge.contig)?;
349    writeln!(out, "pos\t{}", merge.pos)?;
350    writeln!(out, "merged_null\t{}", merge.merged_null)?;
351    if merge.merged_null {
352        return Ok(());
353    }
354    writeln!(out, "allele_count\t{}", merge.alleles.len())?;
355    for (i, a) in merge.alleles.iter().enumerate() {
356        writeln!(out, "allele_{i}\t{a}")?;
357    }
358    writeln!(out, "sample_count\t{}", merge.genotypes.len())?;
359    for (gi, g) in merge.genotypes.iter().enumerate() {
360        writeln!(out, "genotype_{gi}_name\t{}", g.name)?;
361        if let Some(pl) = &g.pl {
362            writeln!(
363                out,
364                "genotype_{gi}_pl\t{}",
365                pl.iter()
366                    .map(|v| v.to_string())
367                    .collect::<Vec<_>>()
368                    .join(",")
369            )?;
370        }
371        if let Some(ad) = &g.ad {
372            writeln!(
373                out,
374                "genotype_{gi}_ad\t{}",
375                ad.iter()
376                    .map(|v| v.to_string())
377                    .collect::<Vec<_>>()
378                    .join(",")
379            )?;
380        }
381        if let Some(gq) = g.gq {
382            writeln!(out, "genotype_{gi}_gq\t{gq}")?;
383        }
384    }
385    writeln!(out, "has_non_ref\t{}", merge.has_non_ref)?;
386    Ok(())
387}
