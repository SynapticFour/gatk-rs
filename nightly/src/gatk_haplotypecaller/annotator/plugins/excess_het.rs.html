1//! GATK `ExcessHet` exact test (I-D03 production parity).
2
3const MIN_NEEDED_VALUE: f64 = 1.0e-16;
4/// GATK `ExcessHet.PHRED_SCALED_MIN_P_VALUE` (-10 * log10(1e-16)).
5const PHRED_SCALED_MIN_P_VALUE: f64 = 160.0;
6
7/// Right-sided exact test p-value (GATK `ExcessHet.exactTest`).
8pub fn exact_test(het_count: i32, ref_count: i32, hom_count: i32) -> f64 {
9    assert!(het_count >= 0 && ref_count >= 0 && hom_count >= 0);
10    let het_count = het_count as usize;
11    let ref_count = ref_count as usize;
12    let hom_count = hom_count as usize;
13
14    let (obs_hom_r, obs_hom_c) = if ref_count < hom_count {
15        (ref_count, hom_count)
16    } else {
17        (hom_count, ref_count)
18    };
19
20    let rare_copies = 2 * obs_hom_r + het_count;
21    let n = het_count + obs_hom_c + obs_hom_r;
22    if n == 0 {
23        return 1.0;
24    }
25
26    let mut probs = vec![0.0_f64; rare_copies + 1];
27    let mut mid = ((rare_copies as f64) * (2.0 * n as f64 - rare_copies as f64)
28        / (2.0 * n as f64 - 1.0))
29        .floor() as usize;
30    if (mid % 2) != (rare_copies % 2) {
31        mid += 1;
32    }
33
34    probs[mid] = 1.0;
35    let mut mysum = 1.0;
36
37    let mut curr_hets = mid;
38    let mut curr_hom_r = (rare_copies - mid) / 2;
39    let mut curr_hom_c = n - curr_hets - curr_hom_r;
40
41    while curr_hets >= 2 {
42        let potential_prob = probs[curr_hets] * curr_hets as f64 * (curr_hets - 1) as f64
43            / (4.0 * (curr_hom_r + 1) as f64 * (curr_hom_c + 1) as f64);
44        if potential_prob < MIN_NEEDED_VALUE {
45            break;
46        }
47        probs[curr_hets - 2] = potential_prob;
48        mysum += probs[curr_hets - 2];
49        curr_hets -= 2;
50        curr_hom_r += 1;
51        curr_hom_c += 1;
52    }
53
54    curr_hets = mid;
55    curr_hom_r = (rare_copies - mid) / 2;
56    curr_hom_c = n - curr_hets - curr_hom_r;
57
58    while curr_hets <= rare_copies.saturating_sub(2) {
59        let potential_prob = probs[curr_hets] * 4.0 * curr_hom_r as f64 * curr_hom_c as f64
60            / ((curr_hets + 2) as f64 * (curr_hets + 1) as f64);
61        if potential_prob < MIN_NEEDED_VALUE {
62            break;
63        }
64        probs[curr_hets + 2] = potential_prob;
65        mysum += probs[curr_hets + 2];
66        curr_hets += 2;
67        curr_hom_r = curr_hom_r.saturating_sub(1);
68        curr_hom_c = curr_hom_c.saturating_sub(1);
69    }
70
71    let right_pval = probs[het_count] / mysum;
72    if het_count == rare_copies {
73        return right_pval.clamp(0.0, 1.0);
74    }
75    let tail: f64 = probs[(het_count + 1)..].iter().sum();
76    (right_pval + tail / mysum).clamp(0.0, 1.0)
77}

79/// Phred-scaled excess het (GATK `ExcessHet.calculateEH(GenotypeCounts, sampleCount)`).
80pub fn excess_heterozygosity_phred(ref_count: u32, het_count: u32, hom_count: u32) -> f64 {
81    let sample_count = ref_count + het_count + hom_count;
82    if sample_count == 0 {
83        return 0.0;
84    }
85    let pval = exact_test(het_count as i32, ref_count as i32, hom_count as i32);
86    if pval < 1.0e-60 {
87        return PHRED_SCALED_MIN_P_VALUE;
88    }
89    let phred = -10.0 * pval.log10();
90    if phred == 0.0 {
91        0.0
92    } else {
93        phred
94    }
95}
