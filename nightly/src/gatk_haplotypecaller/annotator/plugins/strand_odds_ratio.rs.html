1//! GATK `StrandOddsRatio` (production).
2
3const PSEUDOCOUNT: f64 = 1.0;
4
5/// SOR from ref/alt × forward/reverse counts (GATK `StrandOddsRatio.calculateSOR`).
6pub fn strand_odds_ratio(ref_fw: u32, ref_rv: u32, alt_fw: u32, alt_rv: u32) -> f64 {
7    let table = [[ref_fw, ref_rv], [alt_fw, alt_rv]];
8    calculate_sor(&table)
9}
10
11fn calculate_sor(table: &[[u32; 2]; 2]) -> f64 {
12    let t00 = table[0][0] as f64 + PSEUDOCOUNT;
13    let t01 = table[0][1] as f64 + PSEUDOCOUNT;
14    let t10 = table[1][0] as f64 + PSEUDOCOUNT;
15    let t11 = table[1][1] as f64 + PSEUDOCOUNT;
16    let ratio = (t00 / t01) * (t11 / t10) + (t01 / t00) * (t10 / t11);
17    let ref_ratio = t00.min(t01) / t00.max(t01);
18    let alt_ratio = t10.min(t11) / t10.max(t11);
19    (ratio).ln() + ref_ratio.ln() - alt_ratio.ln()
20}
