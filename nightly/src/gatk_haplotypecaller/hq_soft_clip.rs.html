9/// `ReferenceConfidenceModel` HQ soft-clip path: `AlignmentUtils.countHighQualitySoftClips(..., threshold)`
10/// with [`RCM_HQ_SOFT_CLIP_QUAL_THRESHOLD`] — base must be **strictly greater** than this (Java `>`).
11pub const RCM_HQ_SOFT_CLIP_QUAL_THRESHOLD: u8 = 28;
12
13/// Legacy threshold used by read-level heuristics outside the RCM accumulator (kept for older tests/tools).
14pub const GATK_HQ_SOFT_CLIP_BASE_QUAL: u8 = 25;
15
16/// GATK `AlignmentUtils.countHighQualitySoftClips`: total count of soft-clip bases with qual
17/// **>** [`RCM_HQ_SOFT_CLIP_QUAL_THRESHOLD`], summed over every `S` CIGAR block.
18pub fn count_high_quality_soft_clip_bases_rcm(rec: &bam::Record) -> u32 {
19    let cigar = rec.cigar();
20    let qual = rec.qual();
21    if qual.is_empty() {
22        return 0;
23    }
24    let mut align_pos = 0usize;
25    let mut num_hq = 0u32;
26    for op in cigar.iter() {
27        let n = op.len() as usize;
28        match *op {
29            Cigar::SoftClip(_) => {
30                for i in 0..n {
31                    let q = qual.get(align_pos + i).copied().unwrap_or(0);
32                    if q > RCM_HQ_SOFT_CLIP_QUAL_THRESHOLD {
33                        num_hq += 1;
34                    }
35                }
36                align_pos += n;
37            }
38            Cigar::HardClip(_) | Cigar::Pad(_) | Cigar::Del(_) | Cigar::RefSkip(_) => {}
39            Cigar::Match(_) | Cigar::Ins(_) | Cigar::Diff(_) | Cigar::Equal(_) => {
40                align_pos += n;
41            }
42        }
43    }
44    num_hq
45}

47/// Longest contiguous HQ soft-clip run on either end of the read (bases with qual >= [`GATK_HQ_SOFT_CLIP_BASE_QUAL`]).
48pub fn max_hq_soft_clip_bases(rec: &bam::Record) -> u32 {
49    let qual = rec.qual();
50    let seq_len = rec.seq().len();
51    if qual.is_empty() || seq_len == 0 {
52        return 0;
53    }
54    let (lead, trail) = cigar_soft_clip_ends(&rec.cigar());
55    let lead_hq = hq_prefix_len(qual, lead as usize);
56    let trail_hq = hq_suffix_len(qual, trail as usize);
57    lead_hq.max(trail_hq) as u32
58}
59
60fn hq_prefix_len(qual: &[u8], clip_len: usize) -> usize {
61    let n = clip_len.min(qual.len());
62    let mut k = 0;
63    for &q in &qual[..n] {
64        if q >= GATK_HQ_SOFT_CLIP_BASE_QUAL {
65            k += 1;
66        } else {
67            break;
68        }
69    }
70    k
71}
72
73fn hq_suffix_len(qual: &[u8], clip_len: usize) -> usize {
74    let n = clip_len.min(qual.len());
75    if n == 0 {
76        return 0;
77    }
78    let start = qual.len() - n;
79    let mut k = 0;
80    for &q in qual[start..].iter().rev() {
81        if q >= GATK_HQ_SOFT_CLIP_BASE_QUAL {
82            k += 1;
83        } else {
84            break;
85        }
86    }
87    k
88}

90/// GATK-style running mean of per-read max HQ soft-clip lengths at a 1-based locus.
91pub fn hq_soft_clip_running_mean_at_locus(
92    records: &[bam::Record],
93    header: &bam::HeaderView,
94    contig: &str,
95    pos1: u64,
96    read_filters: &ReadFilterParams,
97) -> f64 {
98    let mut sum = 0.0;
99    let mut n = 0u32;
100    for rec in records {
101        if !passes_hc_read_filters_with_header(rec, header, read_filters) {
102            continue;
103        }
104        if !record_overlaps_closed_interval_1based(rec, header, contig, pos1, pos1, read_filters) {
105            continue;
106        }
107        let clip = max_hq_soft_clip_bases(rec);
108        if clip > 0 {
109            sum += clip as f64;
110            n += 1;
111        }
112    }
113    if n == 0 {
114        0.0
115    } else {
116        sum / f64::from(n)
117    }
118}

137    fn two_soft_clip_bases_counted_on_leading_clip() {

139        let hv = header();
140        let seq = b"ACGTACGTAC";
141        let qual = [30u8, 30, 10, 30, 30, 30, 30, 30, 30, 30];
142        let mut rec = Record::new();
143        rec.set(
144            b"sc",

146                Cigar::SoftClip(2),

148            ])),
149            seq,

151        );
152        rec.set_tid(hv.tid(b"chr1").unwrap() as i32);
