1//! Process-global optional NDJSON sink for semantic checkpoints.
2
3use super::schema::{RegionKey, SemanticStage, SemanticTraceEvent, SCHEMA_ID, TRACE_IMPL_RUST};
4use crate::runtime_config::RuntimeConfig;
5use serde_json::Value;
6use std::fs::File;
7use std::io::{BufWriter, Write};
8use std::path::PathBuf;
9use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
10use std::sync::Mutex;
11
12static ENABLED: AtomicBool = AtomicBool::new(false);
13static SEQ: AtomicU64 = AtomicU64::new(0);
14static SINK: Mutex<Option<NdjsonFileSink>> = Mutex::new(None);
15
16/// Opaque handle reserved for future non-global sinks.
17#[derive(Debug)]
18pub struct TraceSinkHandle;
19
20struct NdjsonFileSink {
21    writer: BufWriter<File>,
22}
23
24impl NdjsonFileSink {
25    fn open(path: PathBuf) -> std::io::Result<Self> {
26        let file = File::create(path)?;
27        Ok(Self {
28            writer: BufWriter::new(file),
29        })
30    }
31
32    fn write_event(&mut self, event: &SemanticTraceEvent) -> std::io::Result<()> {
33        serde_json::to_writer(&mut self.writer, event)?;
34        self.writer.write_all(b"\n")?;
35        Ok(())
36    }
37
38    fn flush(&mut self) -> std::io::Result<()> {
39        self.writer.flush()
40    }
41}
42
43/// Whether semantic tracing is active for this process.
44#[inline]
45pub fn is_enabled() -> bool {
46    ENABLED.load(Ordering::Relaxed)
47}
48
49/// Initialize from [`RuntimeConfig`]. No-op if already enabled or path unset.
50pub fn try_init_from_runtime(cfg: &RuntimeConfig) {
51    if is_enabled() {
52        return;
53    }
54    let Some(path) = cfg.debug.semantic_trace_path.as_ref() else {
55        return;
56    };
57    if path.is_empty() {
58        return;
59    }
60    let Ok(mut guard) = SINK.lock() else {
61        return;
62    };
63    if guard.is_some() {
64        ENABLED.store(true, Ordering::Relaxed);
65        return;
66    }
67    match NdjsonFileSink::open(PathBuf::from(path)) {
68        Ok(sink) => {
69            *guard = Some(sink);
70            SEQ.store(0, Ordering::Relaxed);
71            ENABLED.store(true, Ordering::Relaxed);
72        }
73        Err(e) => {
74            eprintln!("GATK_RS_SEMANTIC_TRACE: failed to open {path}: {e}");
75            ENABLED.store(false, Ordering::Relaxed);
76        }
77    }
78}

80/// Emit one event to the global sink (no-op when disabled).
81pub fn emit(stage: SemanticStage, region: Option<RegionKey>, payload: Value) {
82    if !is_enabled() {
83        return;
84    }
85    let Ok(mut guard) = SINK.lock() else {
86        return;
87    };
88    let Some(sink) = guard.as_mut() else {
89        return;
90    };
91    let seq = SEQ.fetch_add(1, Ordering::Relaxed) + 1;
92    let event = SemanticTraceEvent {
93        schema: SCHEMA_ID.to_string(),
94        seq,
95        impl_name: TRACE_IMPL_RUST.to_string(),
96        stage,
97        region,
98        payload,
99    };
100    if let Err(e) = sink.write_event(&event) {
101        eprintln!("GATK_RS_SEMANTIC_TRACE: write failed: {e}");
102    }
103}

105#[cfg(test)]
106pub(super) fn reset_for_tests() {
107    ENABLED.store(false, Ordering::Relaxed);
108    SEQ.store(0, Ordering::Relaxed);
109    if let Ok(mut guard) = SINK.lock() {
110        *guard = None;
111    }
112}
113
114#[cfg(test)]
115pub(super) fn flush_for_tests() {
116    if let Ok(mut guard) = SINK.lock() {
117        if let Some(sink) = guard.as_mut() {
118            let _ = sink.flush();
119        }
120    }
