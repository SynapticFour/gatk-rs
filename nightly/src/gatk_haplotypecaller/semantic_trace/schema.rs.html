1//! Semantic-trace event schema (`gatk_rs.hc.semantic_trace/v1`).

3use serde::{Deserialize, Serialize};
4use serde_json::Value;
5
6/// Schema identifier written on every NDJSON line.
7pub const SCHEMA_ID: &str = "gatk_rs.hc.semantic_trace/v1";
8
9/// Implementation label for Rust emitters.
10pub const TRACE_IMPL_RUST: &str = "rust";
11
12/// Implementation label for Java / projected emitters.
13pub const TRACE_IMPL_JAVA: &str = "java";

15/// Ordered production checkpoints (earlier stages sort before later for first-divergence).
16#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
17#[serde(rename_all = "snake_case")]
18pub enum SemanticStage {
19    /// Band-pass activity profile cut (pre-region materialization).
20    ActivityProfile,
21    /// Materialized active/inactive assembly-region boundary.
22    ActiveRegion,
23    /// Assembly graph / result-set metrics.
24    AssemblyGraph,
25    /// Reference haplotype path.
26    ReferencePath,
27    /// Candidate haplotypes after assembly/trim.
28    CandidateHaplotypes,
29    /// Read×haplotype PairHMM likelihoods.
30    ReadLikelihoods,
31    /// Site genotype likelihoods / PL.
32    GenotypeLikelihoods,
33    /// Inactive reference-confidence model path.
34    InactiveRcm,
35    /// VCF records emitted for the region.
36    VcfEmission,
37}
38
39impl SemanticStage {
40    /// Wire name matching serde `snake_case`.
41    pub fn as_str(self) -> &'static str {
42        match self {
43            Self::ActivityProfile => "activity_profile",
44            Self::ActiveRegion => "active_region",
45            Self::AssemblyGraph => "assembly_graph",
46            Self::ReferencePath => "reference_path",
47            Self::CandidateHaplotypes => "candidate_haplotypes",
48            Self::ReadLikelihoods => "read_likelihoods",
49            Self::GenotypeLikelihoods => "genotype_likelihoods",
50            Self::InactiveRcm => "inactive_rcm",
51            Self::VcfEmission => "vcf_emission",
52        }
53    }
54
55    /// Pipeline order index (lower = earlier).
56    pub fn order_index(self) -> u8 {
57        match self {
58            Self::ActivityProfile => 0,
59            Self::ActiveRegion => 1,
60            Self::AssemblyGraph => 2,
61            Self::ReferencePath => 3,
62            Self::CandidateHaplotypes => 4,
63            Self::ReadLikelihoods => 5,
64            Self::GenotypeLikelihoods => 6,
65            Self::InactiveRcm => 7,
66            Self::VcfEmission => 8,
67        }
68    }
69}

71/// Genomic region key shared across stages.
72#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
73pub struct RegionKey {
74    pub contig: String,
75    /// 1-based inclusive start (unpadded active/inactive span).
76    pub start: u64,
77    /// 1-based inclusive end.
78    pub end: u64,
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub is_active: Option<bool>,
81}
82
83/// One NDJSON semantic checkpoint.
84#[derive(Debug, Clone, Serialize, Deserialize)]
85pub struct SemanticTraceEvent {
86    pub schema: String,
87    /// Monotonic sequence number within the process (per sink).
88    pub seq: u64,
89    /// `"rust"` or `"java"`.
90    #[serde(rename = "impl")]
91    pub impl_name: String,
92    pub stage: SemanticStage,
93    #[serde(default, skip_serializing_if = "Option::is_none")]
94    pub region: Option<RegionKey>,
95    pub payload: Value,
96}
