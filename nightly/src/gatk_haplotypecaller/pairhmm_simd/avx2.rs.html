22const MATCH_TO_MATCH: usize = 0;
23const INDEL_TO_MATCH: usize = 1;
24const MATCH_TO_INSERTION: usize = 2;
25const INSERTION_TO_INSERTION: usize = 3;
26const MATCH_TO_DELETION: usize = 4;
27const DELETION_TO_DELETION: usize = 5;
28
29const LANES: usize = 4;
30
31struct Avx2Scratch {
32    m: Vec<f64>,
33    ins: Vec<f64>,
34    del: Vec<f64>,
35}
36
37impl Avx2Scratch {
38    fn new() -> Self {
39        Self {
40            m: Vec::new(),
41            ins: Vec::new(),
42            del: Vec::new(),
43        }
44    }
45
46    fn ensure(&mut self, cells: usize) {
47        let need = cells * LANES;
48        if self.m.len() < need {
49            self.m.resize(need, 0.0);
50            self.ins.resize(need, 0.0);
51            self.del.resize(need, 0.0);
52        }
53    }
54}

77/// Score haplotypes with AVX2 when available; otherwise portable packed f64.
78pub fn score_haps_avx2_f64(
79    read_bases: &[u8],
80    read_quals: &[u8],
81    haplotypes: &[&[u8]],
82    insertion_gop: &[u8],
83    deletion_gop: &[u8],
84    overall_gcp: &[u8],
85) -> GatkResult<Vec<f64>> {
86    if !is_x86_feature_detected!("avx2") {
87        return score_haps_logless_packed_f64(
88            read_bases,
89            read_quals,
90            haplotypes,
91            insertion_gop,
92            deletion_gop,
93            overall_gcp,
94        );
95    }
96    // SAFETY: `is_x86_feature_detected!("avx2")` is true above.
97    unsafe {
98        score_haps_avx2_f64_unchecked(
99            read_bases,
100            read_quals,
101            haplotypes,
102            insertion_gop,
103            deletion_gop,
104            overall_gcp,
105        )
106    }
107}

109#[target_feature(enable = "avx2")]
110unsafe fn score_haps_avx2_f64_unchecked(
111    read_bases: &[u8],
112    read_quals: &[u8],
113    haplotypes: &[&[u8]],
114    insertion_gop: &[u8],
115    deletion_gop: &[u8],
116    overall_gcp: &[u8],
117) -> GatkResult<Vec<f64>> {
118    if haplotypes.is_empty() {
119        return Ok(Vec::new());
120    }
121    let rn = read_bases.len();
122    if rn == 0 {
123        return Ok(vec![0.0; haplotypes.len()]);
124    }
125
126    let mut out = vec![0.0f64; haplotypes.len()];
127    let mut err = None;
128    AVX2_TRANSITIONS.with(|tcell| {
129        let mut transitions = tcell.borrow_mut();
130        logless_fill_transitions(
131            &mut transitions,
132            rn,
133            insertion_gop,
134            deletion_gop,
135            overall_gcp,
136        );
137        AVX2_BY_LEN.with(|by_len_cell| {
138            let mut by_len = by_len_cell.borrow_mut();
139            for v in by_len.values_mut() {
140                v.clear();
141            }
142            for (i, h) in haplotypes.iter().enumerate() {
143                by_len.entry(h.len()).or_default().push(i);
144            }
145            let lengths: Vec<usize> = by_len
146                .iter()
147                .filter(|(_, idxs)| !idxs.is_empty())
148                .map(|(len, _)| *len)
149                .collect();
150            AVX2_SCRATCH.with(|cell| {
151                let mut scratch = cell.borrow_mut();
152                for len in lengths {
153                    let Some(ordered) = by_len.get_mut(&len) else {
154                        continue;
155                    };
156                    ordered.sort_by(|&a, &b| haplotypes[a].cmp(haplotypes[b]));
157                    let mut subset = Vec::with_capacity(ordered.len());
158                    for &i in ordered.iter() {
159                        subset.push(haplotypes[i]);
160                    }
161                    let use_prefix = ordered.len() >= super::pack::PREFIX_REUSE_MIN_HAPS_AVX2
162                        && mean_consecutive_prefix_frac(&subset)
163                            >= super::pack::PREFIX_REUSE_OVER_SIMD_FRAC;
164                    if use_prefix {
165                        match score_haps_logless_packed_f64_with_transitions(
166                            read_bases,
167                            read_quals,
168                            &subset,
169                            &transitions,
170                        ) {
171                            Ok(scores) => {
172                                for (k, &i) in ordered.iter().enumerate() {
173                                    out[i] = scores[k];
174                                }
175                                AVX2_PREFIX_REUSE
176                                    .set(AVX2_PREFIX_REUSE.get() + ordered.len() as u64);
177                            }
178                            Err(e) => {
179                                err = Some(e);
180                                return;
181                            }
182                        }
183                        continue;
184                    }
185                    let mut chunks = ordered.chunks_exact(LANES);
186                    for pack_src in chunks.by_ref() {
187                        let pack = [
188                            haplotypes[pack_src[0]],
189                            haplotypes[pack_src[1]],
190                            haplotypes[pack_src[2]],
191                            haplotypes[pack_src[3]],
192                        ];
193                        let scores =
194                            score_pack4(read_bases, read_quals, &pack, &transitions, &mut scratch);
195                        for (k, &idx) in pack_src.iter().enumerate() {
196                            out[idx] = scores[k];
197                        }
198                        AVX2_PACK4.set(AVX2_PACK4.get() + 1);
199                    }
200                    for &i in chunks.remainder() {
201                        match score_one_hap_logless_f64_with_transitions(
202                            read_bases,
203                            read_quals,
204                            haplotypes[i],
205                            &transitions,
206                        ) {
207                            Ok(score) => out[i] = score,
208                            Err(e) => {
209                                err = Some(e);
210                                return;
211                            }
212                        }
213                        AVX2_LEFTOVER.set(AVX2_LEFTOVER.get() + 1);
214                    }
215                }
216            });
217        });
218    });
219    if let Some(e) = err {
220        return Err(e);
221    }
222    Ok(out)
223}

225/// SAFETY: caller must ensure AVX2 is available (`#[target_feature(enable = "avx2")]`).
226/// Equal-length packs only (`by_len` groups) — no per-j lane masks.
227#[target_feature(enable = "avx2")]
228unsafe fn score_pack4(
229    read_bases: &[u8],
230    read_quals: &[u8],
231    haps: &[&[u8]; 4],
232    transitions: &[[f64; 6]],
233    scratch: &mut Avx2Scratch,
234) -> [f64; 4] {
235    let rn = read_bases.len();
236    let hn = haps[0].len();
237    debug_assert!(haps.iter().all(|h| h.len() == hn));
238    let cols = hn + 1;
239    let cells = (rn + 1) * cols;
240    scratch.ensure(cells);
241    let m = &mut scratch.m;
242    let ins = &mut scratch.ins;
243    let del = &mut scratch.del;
244
245    m[..cols * LANES].fill(0.0);
246    ins[..cols * LANES].fill(0.0);
247    let init = INITIAL_CONDITION / hn as f64;
248    for j in 0..=hn {
249        let base = j * LANES;
250        for lane in 0..LANES {
251            del[base + lane] = init;
252        }
253    }
254
255    for i in 1..=rn {
256        let t = transitions[i];
257        let x = read_bases[i - 1];
258        let (match_p, mismatch_p) = logless_match_mismatch_prior(read_quals[i - 1]);
259        let tm2m = _mm256_set1_pd(t[MATCH_TO_MATCH]);
260        let ti2m = _mm256_set1_pd(t[INDEL_TO_MATCH]);
261        let tm2i = _mm256_set1_pd(t[MATCH_TO_INSERTION]);
262        let ti2i = _mm256_set1_pd(t[INSERTION_TO_INSERTION]);
263        let tm2d = _mm256_set1_pd(t[MATCH_TO_DELETION]);
264        let td2d = _mm256_set1_pd(t[DELETION_TO_DELETION]);
265        let row = i * cols;
266        let prev = (i - 1) * cols;
267        let col0 = row * LANES;
268        for lane in 0..LANES {
269            m[col0 + lane] = 0.0;
270            ins[col0 + lane] = 0.0;
271            del[col0 + lane] = 0.0;
272        }
273        for j in 1..=hn {
274            let idx = (row + j) * LANES;
275            let diag = ((prev + j - 1) * LANES) as isize;
276            let up = ((prev + j) * LANES) as isize;
277            let left = ((row + j - 1) * LANES) as isize;
278
279            let mut prior_arr = [0.0f64; 4];
280            for lane in 0..LANES {
281                let y = haps[lane][j - 1];
282                prior_arr[lane] = if x == y || x == b'N' || y == b'N' {
283                    match_p
284                } else {
285                    mismatch_p
286                };
287            }
288
289            // SAFETY: indices within SoA; AVX2 enabled; equal-length pack.
290            let m_diag = _mm256_loadu_pd(m.as_ptr().offset(diag));
291            let i_diag = _mm256_loadu_pd(ins.as_ptr().offset(diag));
292            let d_diag = _mm256_loadu_pd(del.as_ptr().offset(diag));
293            let m_up = _mm256_loadu_pd(m.as_ptr().offset(up));
294            let i_up = _mm256_loadu_pd(ins.as_ptr().offset(up));
295            let m_left = _mm256_loadu_pd(m.as_ptr().offset(left));
296            let d_left = _mm256_loadu_pd(del.as_ptr().offset(left));
297            let p = _mm256_loadu_pd(prior_arr.as_ptr());
298
299            let from_m = _mm256_mul_pd(m_diag, tm2m);
300            let from_i = _mm256_mul_pd(i_diag, ti2m);
301            let from_d = _mm256_mul_pd(d_diag, ti2m);
302            let sum = _mm256_add_pd(_mm256_add_pd(from_m, from_i), from_d);
303            let m_new = _mm256_mul_pd(p, sum);
304            let i_new = _mm256_add_pd(_mm256_mul_pd(m_up, tm2i), _mm256_mul_pd(i_up, ti2i));
305            let d_new = _mm256_add_pd(_mm256_mul_pd(m_left, tm2d), _mm256_mul_pd(d_left, td2d));
306
307            _mm256_storeu_pd(m.as_mut_ptr().add(idx), m_new);
308            _mm256_storeu_pd(ins.as_mut_ptr().add(idx), i_new);
309            _mm256_storeu_pd(del.as_mut_ptr().add(idx), d_new);
310        }
311    }
312
313    let end_row = rn * cols;
314    let mut sums = [0.0f64; 4];
315    for j in 1..=hn {
316        let idx = (end_row + j) * LANES;
317        let mv = _mm256_loadu_pd(m.as_ptr().add(idx));
318        let iv = _mm256_loadu_pd(ins.as_ptr().add(idx));
319        let s = _mm256_add_pd(mv, iv);
320        let mut tmp = [0.0f64; 4];
321        _mm256_storeu_pd(tmp.as_mut_ptr(), s);
322        for lane in 0..LANES {
323            sums[lane] += tmp[lane];
324        }
325    }
326
327    let mut out = [0.0f64; 4];
328    for lane in 0..LANES {
329        if sums[lane] <= 0.0 || !sums[lane].is_finite() {
330            out[lane] = f64::NEG_INFINITY;
331        } else {
332            out[lane] = sums[lane].log10() - INITIAL_CONDITION_LOG10;
333        }
334    }
335    out
