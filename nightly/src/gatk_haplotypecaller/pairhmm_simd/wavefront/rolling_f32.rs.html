1//! Rolling 2-row f32 Logless with f64 retry; dispatches to AVX2/NEON when available.
2
3use super::prep::ReadPrep;
4use super::rolling_f64::score_one_rolling_f64;
5use super::{select_wavefront_kernel, WavefrontKernel, WavefrontScratch};
6
7/// f32-safe seed (GKL-style `ldexpf(1, 120)`). Final log10 subtracts this scale so
8/// scores match f64 Logless up to rounding (`log10(c·I) − log10(I) = log10(c)`).
9pub const INITIAL_CONDITION_F32: f32 = f32::from_bits((127 + 120) << 23); // 2^120
10pub const INITIAL_CONDITION_F32_LOG10: f64 = 120.0 * std::f64::consts::LOG10_2;
11/// Reject f32 linear sums below this (relative to 2^120 scale) and retry f64.
12const MIN_ACCEPTED_F32_LINEAR_SUM: f32 = 1e-20;
13
14const MATCH_TO_MATCH: usize = 0;
15const INDEL_TO_MATCH: usize = 1;
16const MATCH_TO_INSERTION: usize = 2;
17const INSERTION_TO_INSERTION: usize = 3;
18const MATCH_TO_DELETION: usize = 4;
19const DELETION_TO_DELETION: usize = 5;
20
21/// Host-selected f32 kernel with f64 retry on underflow.
22pub fn score_one_rolling_f32_with_retry(
23    prep: &ReadPrep,
24    read_bases: &[u8],
25    hap: &[u8],
26    scratch: &mut WavefrontScratch,
27) -> f64 {
28    let sum = match select_wavefront_kernel() {
29        WavefrontKernel::Avx2F32 => {
30            #[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
31            {
32                // SAFETY: select_wavefront_kernel verified AVX2.
33                unsafe { super::avx2_f32::score_one_avx2_f32(prep, read_bases, hap, scratch) }
34            }
35            #[cfg(not(any(target_arch = "x86", target_arch = "x86_64")))]
36            {
37                score_one_portable_f32(prep, read_bases, hap, scratch)
38            }
39        }
40        WavefrontKernel::NeonF32 => {
41            #[cfg(target_arch = "aarch64")]
42            {
43                // SAFETY: select_wavefront_kernel verified NEON.
44                unsafe { super::neon_f32::score_one_neon_f32(prep, read_bases, hap, scratch) }
45            }
46            #[cfg(not(target_arch = "aarch64"))]
47            {
48                score_one_portable_f32(prep, read_bases, hap, scratch)
49            }
50        }
51        WavefrontKernel::PortableF32 => score_one_portable_f32(prep, read_bases, hap, scratch),
52    };
53
54    if !sum.is_finite() || sum < MIN_ACCEPTED_F32_LINEAR_SUM {
55        return score_one_rolling_f64(prep, read_bases, hap, scratch);
56    }
57    let s = sum as f64;
58    if s <= 0.0 || !s.is_finite() {
59        return f64::NEG_INFINITY;
60    }
61    s.log10() - INITIAL_CONDITION_F32_LOG10
62}

64/// Portable scalar f32 rolling kernel (returns linear sum, not log10).
65pub fn score_one_portable_f32(
66    prep: &ReadPrep,
67    read_bases: &[u8],
68    hap: &[u8],
69    scratch: &mut WavefrontScratch,
70) -> f32 {
71    let rn = read_bases.len();
72    let hn = hap.len();
73    if rn == 0 {
74        return 1.0;
75    }
76    let cols = hn + 1;
77    scratch.ensure_rolling_f32(cols);
78    let (m_a, m_b) = scratch.m32[..cols * 2].split_at_mut(cols);
79    let (ins_a, ins_b) = scratch.ins32[..cols * 2].split_at_mut(cols);
80    let (del_a, del_b) = scratch.del32[..cols * 2].split_at_mut(cols);
81
82    m_a.fill(0.0);
83    ins_a.fill(0.0);
84    let init_del = INITIAL_CONDITION_F32 / hn as f32;
85    for j in 0..=hn {
86        del_a[j] = init_del;
87    }
88
89    let mut prev_is_a = true;
90    for i in 1..=rn {
91        let t = prep.transitions_f32[i];
92        let x = read_bases[i - 1];
93        let (match_p, mismatch_p) = prep.match_mm_f32[i - 1];
94        let (m_prev, m_curr, ins_prev, ins_curr, del_prev, del_curr) = if prev_is_a {
95            (
96                &m_a[..],
97                &mut m_b[..],
98                &ins_a[..],
99                &mut ins_b[..],
100                &del_a[..],
101                &mut del_b[..],
102            )
103        } else {
104            (
105                &m_b[..],
106                &mut m_a[..],
107                &ins_b[..],
108                &mut ins_a[..],
109                &del_b[..],
110                &mut del_a[..],
111            )
112        };
113        m_curr[0] = 0.0;
114        ins_curr[0] = 0.0;
115        del_curr[0] = 0.0;
116        let tm2m = t[MATCH_TO_MATCH];
117        let ti2m = t[INDEL_TO_MATCH];
118        let tm2i = t[MATCH_TO_INSERTION];
119        let ti2i = t[INSERTION_TO_INSERTION];
120        let tm2d = t[MATCH_TO_DELETION];
121        let td2d = t[DELETION_TO_DELETION];
122        for j in 1..=hn {
123            let y = unsafe { *hap.get_unchecked(j - 1) };
124            let p = if x == y || x == b'N' || y == b'N' {
125                match_p
126            } else {
127                mismatch_p
128            };
129            // SAFETY: j in 1..=hn; buffers length cols = hn+1.
130            unsafe {
131                *m_curr.get_unchecked_mut(j) = p
132                    * (*m_prev.get_unchecked(j - 1) * tm2m
133                        + *ins_prev.get_unchecked(j - 1) * ti2m
134                        + *del_prev.get_unchecked(j - 1) * ti2m);
135                *ins_curr.get_unchecked_mut(j) =
136                    *m_prev.get_unchecked(j) * tm2i + *ins_prev.get_unchecked(j) * ti2i;
137                *del_curr.get_unchecked_mut(j) =
138                    *m_curr.get_unchecked(j - 1) * tm2d + *del_curr.get_unchecked(j - 1) * td2d;
139            }
140        }
141        prev_is_a = !prev_is_a;
142    }
143
144    let (m_end, ins_end) = if prev_is_a {
145        (&m_a[..], &ins_a[..])
146    } else {
147        (&m_b[..], &ins_b[..])
148    };
149    let mut final_sum = 0.0f32;
150    for j in 1..=hn {
151        final_sum += m_end[j] + ins_end[j];
152    }
153    final_sum
154}
