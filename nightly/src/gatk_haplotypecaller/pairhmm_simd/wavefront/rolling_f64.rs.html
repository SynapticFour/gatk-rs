1//! Rolling 2-row Logless PairHMM in f64 (oracle layout ≡ scalar full-matrix).
2
3use super::prep::ReadPrep;
4use super::WavefrontScratch;
5use crate::pairhmm_logless::{INITIAL_CONDITION, INITIAL_CONDITION_LOG10};
6
7const MATCH_TO_MATCH: usize = 0;
8const INDEL_TO_MATCH: usize = 1;
9const MATCH_TO_INSERTION: usize = 2;
10const INSERTION_TO_INSERTION: usize = 3;
11const MATCH_TO_DELETION: usize = 4;
12const DELETION_TO_DELETION: usize = 5;
13
14/// Score one (read, hap) with precomputed [`ReadPrep`] using rolling 2-row f64 DP.
15pub fn score_one_rolling_f64(
16    prep: &ReadPrep,
17    read_bases: &[u8],
18    hap: &[u8],
19    scratch: &mut WavefrontScratch,
20) -> f64 {
21    let rn = read_bases.len();
22    let hn = hap.len();
23    if rn == 0 {
24        return 0.0;
25    }
26    let cols = hn + 1;
27    scratch.ensure_rolling_f64(cols);
28    let (m_a, m_b) = scratch.m[..cols * 2].split_at_mut(cols);
29    let (ins_a, ins_b) = scratch.ins[..cols * 2].split_at_mut(cols);
30    let (del_a, del_b) = scratch.del[..cols * 2].split_at_mut(cols);
31
32    m_a.fill(0.0);
33    ins_a.fill(0.0);
34    let init_del = INITIAL_CONDITION / hn as f64;
35    for j in 0..=hn {
36        del_a[j] = init_del;
37    }
38
39    let mut prev_is_a = true;
40    for i in 1..=rn {
41        let t = prep.transitions_f64[i];
42        let x = read_bases[i - 1];
43        let (match_p, mismatch_p) = prep.match_mm_f64[i - 1];
44        let (m_prev, m_curr, ins_prev, ins_curr, del_prev, del_curr) = if prev_is_a {
45            (
46                &m_a[..],
47                &mut m_b[..],
48                &ins_a[..],
49                &mut ins_b[..],
50                &del_a[..],
51                &mut del_b[..],
52            )
53        } else {
54            (
55                &m_b[..],
56                &mut m_a[..],
57                &ins_b[..],
58                &mut ins_a[..],
59                &del_b[..],
60                &mut del_a[..],
61            )
62        };
63        m_curr[0] = 0.0;
64        ins_curr[0] = 0.0;
65        del_curr[0] = 0.0;
66        for j in 1..=hn {
67            let y = hap[j - 1];
68            let p = if x == y || x == b'N' || y == b'N' {
69                match_p
70            } else {
71                mismatch_p
72            };
73            m_curr[j] = p
74                * (m_prev[j - 1] * t[MATCH_TO_MATCH]
75                    + ins_prev[j - 1] * t[INDEL_TO_MATCH]
76                    + del_prev[j - 1] * t[INDEL_TO_MATCH]);
77            ins_curr[j] =
78                m_prev[j] * t[MATCH_TO_INSERTION] + ins_prev[j] * t[INSERTION_TO_INSERTION];
79            del_curr[j] =
80                m_curr[j - 1] * t[MATCH_TO_DELETION] + del_curr[j - 1] * t[DELETION_TO_DELETION];
81        }
82        prev_is_a = !prev_is_a;
83    }
84
85    let (m_end, ins_end) = if prev_is_a {
86        (&m_a[..], &ins_a[..])
87    } else {
88        (&m_b[..], &ins_b[..])
89    };
90    let mut final_sum = 0.0f64;
91    for j in 1..=hn {
92        final_sum += m_end[j] + ins_end[j];
93    }
94    if final_sum <= 0.0 || !final_sum.is_finite() {
95        return f64::NEG_INFINITY;
96    }
97    final_sum.log10() - INITIAL_CONDITION_LOG10
98}
