1//! Amortized read-side PairHMM preparation (transitions + emission priors).
2
3use crate::pairhmm_logless::{logless_match_mismatch_prior, logless_qual_to_trans_probs};
4
5/// Precomputed planes for one read, shared across all haplotypes.
6#[derive(Debug, Clone)]
7pub struct ReadPrep {
8    pub transitions_f64: Vec<[f64; 6]>,
9    pub transitions_f32: Vec<[f32; 6]>,
10    /// Per read base: (match_p, mismatch_p) as f32.
11    pub match_mm_f32: Vec<(f32, f32)>,
12    /// Per read base: (match_p, mismatch_p) as f64 (retry / rolling f64).
13    pub match_mm_f64: Vec<(f64, f64)>,
14}
15
16impl ReadPrep {
17    pub fn build(
18        read_bases: &[u8],
19        read_quals: &[u8],
20        insertion_gop: &[u8],
21        deletion_gop: &[u8],
22        overall_gcp: &[u8],
23    ) -> Self {
24        let rn = read_bases.len();
25        let mut transitions_f64 = vec![[0.0f64; 6]; rn + 1];
26        let mut transitions_f32 = vec![[0.0f32; 6]; rn + 1];
27        let mut match_mm_f64 = Vec::with_capacity(rn);
28        let mut match_mm_f32 = Vec::with_capacity(rn);
29        for i in 0..rn {
30            let t = logless_qual_to_trans_probs(insertion_gop[i], deletion_gop[i], overall_gcp[i]);
31            transitions_f64[i + 1] = t;
32            transitions_f32[i + 1] = [
33                t[0] as f32,
34                t[1] as f32,
35                t[2] as f32,
36                t[3] as f32,
37                t[4] as f32,
38                t[5] as f32,
39            ];
40            let (m, mm) = logless_match_mismatch_prior(read_quals[i]);
41            match_mm_f64.push((m, mm));
42            match_mm_f32.push((m as f32, mm as f32));
43        }
44        Self {
45            transitions_f64,
46            transitions_f32,
47            match_mm_f32,
48            match_mm_f64,
49        }
50    }
51
52    #[inline]
53    pub fn read_len(&self) -> usize {
54        self.match_mm_f64.len()
55    }
56}
