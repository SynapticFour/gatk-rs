1//! Assembly-path read length gate.
2//! Mirrors `HaplotypeCallerEngine.filterNonPassingReads` length check:
3//! `AlignmentUtils.unclippedReadLength(read) < READ_LENGTH_FILTER_THRESHOLD`.
4
5use rust_htslib::bam;
6use rust_htslib::bam::record::Cigar;
7
8/// `HaplotypeCallerEngine.READ_LENGTH_FILTER_THRESHOLD`.
9pub const GATK_READ_LENGTH_FILTER_THRESHOLD: usize = 10;
10
11/// `AlignmentUtils.unclippedReadLength` — read length minus soft-clip bases only.
12pub fn unclipped_read_length(rec: &bam::Record) -> usize {
13    let read_len = rec.seq().as_bytes().len();
14    let mut soft_clipped = 0usize;
15    for c in rec.cigar().iter() {
16        if let Cigar::SoftClip(n) = c {
17            soft_clipped += *n as usize;
18        }
19    }
20    read_len.saturating_sub(soft_clipped)
21}
22
23/// Length portion of `filterNonPassingReads` (other filters are separate PRE items).
24pub fn passes_read_length_filter(rec: &bam::Record) -> bool {
25    unclipped_read_length(rec) >= GATK_READ_LENGTH_FILTER_THRESHOLD
26}

28#[cfg(test)]
29mod tests {
30    use super::*;
31    use rust_htslib::bam::record::{Cigar, CigarString};
32
33    fn rec_with(cigar: &[Cigar], seq_len: usize) -> bam::Record {
34        let seq = vec![b'A'; seq_len];
35        let qual = vec![30u8; seq_len];
36        let mut r = bam::Record::new();
37        r.set(b"t", Some(&CigarString::from(cigar.to_vec())), &seq, &qual);
38        r
39    }
40
41    #[test]
42    fn unclipped_subtracts_soft_clips_only() {
43        let r = rec_with(&[Cigar::SoftClip(3), Cigar::Match(7)], 10);
44        assert_eq!(unclipped_read_length(&r), 7);
45        assert!(!passes_read_length_filter(&r));
46    }
47
48    #[test]
49    fn ten_match_bases_pass() {
50        let r = rec_with(&[Cigar::Match(10)], 10);
51        assert_eq!(unclipped_read_length(&r), 10);
52        assert!(passes_read_length_filter(&r));
53    }
54}
