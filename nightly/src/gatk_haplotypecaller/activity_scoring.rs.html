1//! Locus-by-locus activity scoring aligned with GATK `HaplotypeCallerEngine#isActive`.
2//! `calc_ref_vs_any_log10_genotype_likelihoods` mirrors `ReferenceConfidenceModel#calcGenotypeLikelihoodsOfRefVsAny`
3//! for the non–flow path (`readsWereRealigned = false`).
4//! This module ports the **pre-band-pass** probability path used for assembly-region discovery:
5//! 1. [`calc_ref_vs_any_log10_genotype_likelihoods`] — `ReferenceConfidenceModel#calcGenotypeLikelihoodsOfRefVsAny`
6//! for the non–flow-based reference model (`readsWereRealigned = false`).
7//! 2. [`calculate_single_sample_biallelic_non_ref_posterior`] — `AlleleFrequencyCalculator#calculateSingleSampleBiallelicNonRefPosterior`.
8//! Joint multisample `isActive` (≥2 samples with nonempty pileups) mirrors `AlleleFrequencyCalculator × GenotypingEngine`,
9//! feeding per-sample genotype likelihood vectors through **`htsjdk` PL quantization** (`GenotypeBuilder.PL(double[])`) exactly like HC;
10//! fallback `haplotype_caller_activity_profile_state_single_sample`-style pooling when only one nonzero pileup survives split.
11

38/// Parameters for HC activity scoring (defaults match a standard non–DRAGEN HC run).
39/// # Invariants
40/// `sample_ploidy` is typically 2; active-region confidence is capped at [`ACTIVE_REGION_STANDARD_CONFIDENCE_CAP`].
41/// Heterozygosity / SD feed Dirichlet pseudocounts for AF calculators.
42/// # Ownership
43/// Cloneable config; built from [`HaplotypeCallerConfig`] or defaults.
44/// # Mutation
45/// Snapshot per evaluator/profile; not mutated during locus scoring.
46/// # Biological assumptions
47/// Pre-band-pass ref-vs-any GL and non-ref posterior for assembly-region discovery.
48/// # Java equivalence
49/// GATK `HaplotypeCallerEngine#isActive` / `ReferenceConfidenceModel` / `AlleleFrequencyCalculator` knobs.
50#[derive(Debug, Clone)]
51pub struct HaplotypeCallerActivityScoringParams {
52    pub min_base_quality_score: u8,
53    /// Sample ploidy (`≥ 1`); type-enforced via [`crate::bio_ids::Ploidy`].
54    pub sample_ploidy: crate::bio_ids::Ploidy,
55    pub snp_heterozygosity: f64,
56    pub indel_heterozygosity: f64,
57    pub heterozygosity_standard_deviation: f64,
58    pub ref_model_deletion_qual: u8,
59    pub active_region_alt_multiplier: f64,
60    /// MinimalGenotypingEngine AF prior / active-region call confidence (Java `min(cap, user standardConfidenceForCalling)`).
61    pub active_region_standard_confidence_for_calling: f64,
62    /// `StandardCallerArgumentCollection.annotateAllSitesWithPLs` (affects `MinimalGenotypingEngine#forceKeepAllele` / QUAL branch).
63    pub annotate_all_sites_with_pls: bool,
64    /// `OutputMode.EMIT_ALL_CONFIDENT_SITES` when true; parity defaults to `EMIT_VARIANTS_ONLY`.
65    pub emit_all_confident_sites: bool,
66    /// When true, apply `(qual <= minBaseQual)` even for deletions (GATK flow-based path).
67    pub flow_based_reference_model: bool,
68    /// GATK `StandardCallerArgumentCollection.CONTAMINATION_FRACTION` applied to pileups before GL
69    /// (`AlleleBiasedDownsamplingUtils`). HC active-region engine forces `0.0`; parity gates may set `> 0`.
70    pub contamination_fraction_to_filter: f64,
71}

93    pub fn from_haplotype_caller_config(cfg: &HaplotypeCallerConfig) -> Self {
94        let mut s = Self::default();
95        s.min_base_quality_score = cfg.min_base_quality_score;
96        let user = cfg

98            .min(ACTIVE_REGION_STANDARD_CONFIDENCE_CAP);

106    }

109        self.snp_heterozygosity * self.ref_pseudocount()

113/// One read’s evidence at a locus after resolving deletion quality (GATK `ReadPileup` element).
114/// # Invariants
115/// `qual` is base quality or deletion surrogate (`REF_MODEL_DELETION_QUAL` path).
116/// `is_alt` reflects RCM before/after-assembly alt classification for the configured path.
117/// # Ownership
118/// [`Copy`] pileup cell; no ownership of the underlying BAM record.
119/// # Mutation
120/// Immutable per locus evaluation; contamination downsampling may drop observations wholesale.
121/// # Biological assumptions
122/// Encodes one read’s base/deletion evidence at a reference locus for activity GLs.
123/// # Java equivalence
124/// GATK `PileupElement` fields used by `ReferenceConfidenceModel` / `isActive`.

133    pub is_alt: bool,

135    pub is_next_to_soft_clip: bool,
136    /// `AlignmentUtils.countHighQualitySoftClips(read, RCM threshold)` for this read.
137    pub read_hq_soft_clip_base_count: u32,
138}

140/// GATK `ReferenceConfidenceModel.isAltBeforeAssembly` for `readsWereRealigned == false`.
141#[inline]
142pub fn is_alt_before_assembly(
143    read_base: u8,
144    ref_base: u8,
145    is_deletion: bool,
146    is_before_deletion_start: bool,
147    is_after_deletion_end: bool,
148    is_before_insertion: bool,
149    is_after_insertion: bool,
150    is_next_to_soft_clip: bool,
151) -> bool {
152    read_base != ref_base
153        || is_deletion
154        || is_before_deletion_start
155        || is_after_deletion_end
156        || is_before_insertion
157        || is_after_insertion
158        || is_next_to_soft_clip
159}

168fn qual_to_error_prob_log10_double(qual: f64) -> f64 {
169    qual * -0.1
170}

172fn qual_to_prob_log10_byte(qual: u8) -> f64 {
173    static CACHE: OnceLock<[f64; 256]> = OnceLock::new();
174    let cache = CACHE.get_or_init(|| {
175        let mut a = [0.0_f64; 256];
176        for i in 0..=254u16 {
177            let e = 10_f64.powf(-(i as f64) / 10.0);
178            a[i as usize] = (1.0 - e).log10();
179        }
180        a
181    });
182    cache[(qual as usize) & 0xff]
183}

185#[inline]
186fn qual_to_error_prob_log10_byte(qual: u8) -> f64 {
187    qual_to_error_prob_log10_double(qual as f64)
188}

190#[inline]
191fn fast_round_java(d: f64) -> usize {
192    if d > 0.0 {
193        (d + 0.5) as usize
194    } else {
195        (d - 0.5) as usize
196    }
197}

199fn jacobian_log_table_get(diff: f64) -> f64 {
200    const MAX_TOLERANCE: f64 = 8.0;
201    const TABLE_STEP: f64 = 0.0001;
202    static CACHE: OnceLock<Vec<f64>> = OnceLock::new();
203    let cache = CACHE.get_or_init(|| {
204        let n = (MAX_TOLERANCE / TABLE_STEP) as usize + 1;
205        (0..n)
206            .map(|k| (1.0 + 10_f64.powf(-(k as f64) * TABLE_STEP)).log10())
207            .collect::<Vec<_>>()
208    });
209    let inv_step = 1.0 / TABLE_STEP;
210    let mut idx = fast_round_java(diff * inv_step);
211    if idx >= cache.len() {
212        idx = cache.len() - 1;
213    }
214    cache[idx]
215}

217/// GATK `MathUtils.approximateLog10SumLog10(a, b)`.
218pub fn approximate_log10_sum_log10_pair(a: f64, b: f64) -> f64 {
219    const MAX_TOLERANCE: f64 = 8.0;
220    let (a, b) = if a > b { (b, a) } else { (a, b) };
221    if a == f64::NEG_INFINITY {
222        return b;
223    }
224    let diff = b - a;
225    if diff < MAX_TOLERANCE {
226        b + jacobian_log_table_get(diff)
227    } else {
228        b
229    }
230}

232/// Java `Math.round(double)` narrowed to PL-range values (`Math.floor(x + 0.5)` for finite `x`).
233#[inline]
234fn java_math_round_double_to_i32(x: f64) -> i32 {
235    debug_assert!(x.is_finite());
236    let rounded = (x + 0.5).floor();
237    if rounded < i32::MIN as f64 {
238        i32::MIN
239    } else if rounded > i32::MAX as f64 {
240        i32::MAX
241    } else {
242        rounded as i32
243    }
244}

246/// Genotype likelihoods as consumed by HC `AlleleFrequencyCalculator` after VC construction:
247/// [`GenotypeBuilder#PL(double[])`](https://samtools.github.io/htsjdk/javadoc/htsjdk/htsjdk/variant/variantcontext/GenotypeBuilder.html)
248/// converts log10 likelihoods → integer PLs then back (`GenotypeLikelihoods` / `FastGenotype#getLikelihoods`).
249pub(crate) fn genotype_log10_likelihoods_after_java_genotype_pl_roundtrip(gl: &[f64]) -> Vec<f64> {
250    if gl.is_empty() {
251        return Vec::new();
252    }
253    let adjust = gl.iter().copied().fold(f64::NEG_INFINITY, f64::max);
254    gl.iter()
255        .copied()
256        .map(|g| {
257            let shifted = (-10.0_f64) * (g - adjust);
258            let clamped = shifted.min(i32::MAX as f64);
259            let pl = java_math_round_double_to_i32(clamped);
260            (pl as f64) / (-10.0)
261        })
262        .collect()
263}

265/// GATK `MathUtils.log10SumLog10` over the full slice.
266pub fn log10_sum_log10(values: &[f64]) -> f64 {
267    if values.is_empty() {
268        return f64::NEG_INFINITY;
269    }
270    if values.len() == 1 {
271        return values[0];
272    }
273    let mut max_i = 0usize;
274    for i in 1..values.len() {
275        if values[i] > values[max_i] {
276            max_i = i;
277        }
278    }
279    let max_v = values[max_i];
280    if max_v == f64::NEG_INFINITY {
281        return max_v;
282    }
283    let mut sum = 1.0_f64;
284    for (i, &cur_val) in values.iter().enumerate() {
285        if i == max_i || cur_val == f64::NEG_INFINITY {
286            continue;
287        }
288        sum += 10_f64.powf(cur_val - max_v);
289    }
290    max_v + if sum != 1.0 { sum.log10() } else { 0.0 }
291}

293/// GATK `MathUtils.normalizeFromLog10ToLinearSpace` (returns probabilities summing to 1).
294/// Borrows the log10 vector so callers that still need it (e.g. genotype posteriors)
295/// do not clone before normalizing.
296pub fn normalize_from_log10_to_linear_space(log10_probs: &[f64]) -> Vec<f64> {
297    let s = log10_sum_log10(log10_probs);
298    log10_probs.iter().map(|x| 10_f64.powf(*x - s)).collect()
299}

301#[inline]
302fn max_element_index(values: &[f64]) -> usize {
303    debug_assert!(!values.is_empty());
304    let mut m = 0usize;
305    for i in 1..values.len() {
306        if values[i] > values[m] {
307            m = i;
308        }
309    }
310    m
311}

318/// Natural log of binomial coefficient C(n,k) — `CombinatoricsUtils.binomialCoefficientLog`.
319pub fn log_binomial_coefficient_natural(n: u32, k: u32) -> f64 {
320    debug_assert!(k <= n);
321    ln_gamma((n + 1) as f64) - ln_gamma((k + 1) as f64) - ln_gamma((n - k + 1) as f64)
322}

324/// GATK `AlleleFrequencyCalculator.calculateSingleSampleBiallelicNonRefPosterior` with priors from
325/// `AlleleFrequencyCalculator.makeCalculator(genotypeArgs)` (SNP branch).
326pub fn calculate_single_sample_biallelic_non_ref_posterior(
327    log10_genotype_likelihoods: &[f64],
328    return_zero_if_ref_is_max: bool,
329    params: &HaplotypeCallerActivityScoringParams,
330) -> f64 {
331    let ploidy = log10_genotype_likelihoods.len().saturating_sub(1);
332    if ploidy == 0 || log10_genotype_likelihoods.is_empty() {
333        return 0.0;
334    }
335
336    if return_zero_if_ref_is_max && max_element_index(log10_genotype_likelihoods) == 0 {
337        return 0.0;
338    }
339
340    let ref_pc = params.ref_pseudocount();
341    let snp_pc = params.snp_pseudocount();
342
343    let mut log10_unnorm = Vec::with_capacity(ploidy + 1);
344    for n in 0..=ploidy {
345        let ln_prior = log_binomial_coefficient_natural(ploidy as u32, n as u32)
346            + ln_gamma(n as f64 + snp_pc)
347            + ln_gamma(ploidy as f64 - n as f64 + ref_pc);
348        log10_unnorm.push(log10_genotype_likelihoods[n] + log_to_log10(ln_prior));
349    }
350
351    if return_zero_if_ref_is_max && max_element_index(&log10_unnorm) == 0 {
352        return 0.0;
353    }
354
355    let linear = normalize_from_log10_to_linear_space(&log10_unnorm);
356    1.0 - linear[0]
357}

359/// Multi-sample active probability from independent per-sample posteriors:
360/// `1 - Π_s (1 - P_s(non-ref))`.
361/// This gives `P(any sample is non-ref)` and provides a stable multi-sample
362/// activity signal until full joint genotyping scaffolding is wired.
363pub fn calculate_multi_sample_any_non_ref_posterior(
364    per_sample_log10_genotype_likelihoods: &[Vec<f64>],
365    return_zero_if_ref_is_max: bool,
366    params: &HaplotypeCallerActivityScoringParams,
367) -> f64 {
368    if per_sample_log10_genotype_likelihoods.is_empty() {
369        return 0.0;
370    }
371    let mut inactive_prob = 1.0_f64;
372    for gl in per_sample_log10_genotype_likelihoods {
373        let p = calculate_single_sample_biallelic_non_ref_posterior(
374            gl,
375            return_zero_if_ref_is_max,
376            params,
377        );
378        inactive_prob *= (1.0 - p).clamp(0.0, 1.0);
379    }
380    1.0 - inactive_prob
381}

391fn skip_observation(
392    obs: &PileupObservation,
393    qual: u8,
394    params: &HaplotypeCallerActivityScoringParams,
395) -> bool {
396    // Java includes bases at exactly minBaseQualityScore (Q10 when threshold is 10).
397    let qfilt = qual < params.min_base_quality_score;
398    qfilt && (params.flow_based_reference_model || !obs.is_deletion)

401/// Online mean matching GATK `MathUtils.RunningAverage` updates during RCM `calcGenotypeLikelihoodsOfRefVsAny`
402/// (each contributing pileup element that is not skipped; `isAlt && isNextToSoftClip` only).
403pub fn hq_soft_clip_running_mean_rcm_path(
404    pileup: &[PileupObservation],
405    params: &HaplotypeCallerActivityScoringParams,
406) -> f64 {
407    let mut mean = 0.0_f64;
408    let mut count = 0_u64;
409    for obs in pileup {
410        let qual = effective_qual(obs, params);
411        if skip_observation(obs, qual, params) {
412            continue;
413        }
414        if obs.is_alt && obs.is_next_to_soft_clip {
415            count += 1;
416            mean += (obs.read_hq_soft_clip_base_count as f64 - mean) / count as f64;
417        }
418    }
419    mean
420}

422/// Multisample ordering: one composite stream in encounter order of nonempty per-sample strata (matches
423/// `HaplotypeCallerEngine#isActive` iterating `splitContexts` values in `LinkedHashMap` order; Rust uses
424/// the same encounter order as `nonempty_stratified_sample_pileups_ordered`).
425pub fn hq_soft_clip_running_mean_rcm_stratified(
426    sample_piles: &[&[PileupObservation]],
427    params: &HaplotypeCallerActivityScoringParams,
428) -> f64 {
429    let mut mean = 0.0_f64;
430    let mut count = 0_u64;
431    for pile in sample_piles {
432        for obs in *pile {
433            let qual = effective_qual(obs, params);
434            if skip_observation(obs, qual, params) {
435                continue;
436            }
437            if obs.is_alt && obs.is_next_to_soft_clip {
438                count += 1;
439                mean += (obs.read_hq_soft_clip_base_count as f64 - mean) / count as f64;
440            }
441        }
442    }
443    mean
444}

446/// GATK `ReferenceConfidenceModel#calcGenotypeLikelihoodsOfRefVsAny` (log10 GL vector, length `ploidy + 1`).
447pub fn calc_ref_vs_any_log10_genotype_likelihoods(
448    ploidy: u32,
449    pileup: &[PileupObservation],
450    params: &HaplotypeCallerActivityScoringParams,
451) -> Vec<f64> {
452    let lc = (ploidy + 1) as usize;
453    let mut gl = vec![0.0_f64; lc];
454    let log10_ploidy = (ploidy as f64).log10();
455    let mut read_count: u32 = 0;
456
457    for obs in pileup {
458        let qual = effective_qual(obs, params);
459        if skip_observation(obs, qual, params) {
460            continue;
461        }
462        read_count += 1;
463        let is_alt = obs.is_alt;
464        let reference_likelihood;
465        let non_ref_likelihood;
466        if is_alt {
467            non_ref_likelihood = qual_to_prob_log10_byte(qual);
468            reference_likelihood = qual_to_error_prob_log10_byte(qual) + LOG10_ONE_THIRD;
469        } else {
470            reference_likelihood = qual_to_prob_log10_byte(qual);
471            non_ref_likelihood = qual_to_error_prob_log10_byte(qual) + LOG10_ONE_THIRD;
472        }
473        let read_weight = if is_alt {
474            params.active_region_alt_multiplier
475        } else {
476            1.0
477        };
478
479        gl[0] += read_weight * (reference_likelihood + log10_ploidy);
480        gl[lc - 1] += read_weight * (non_ref_likelihood + log10_ploidy);
481
482        let mut i = 1usize;
483        let mut j = lc.saturating_sub(2);
484        while i < lc - 1 {
485            let term = approximate_log10_sum_log10_pair(
486                reference_likelihood + (j as f64).log10(),
487                non_ref_likelihood + (i as f64).log10(),
488            );
489            gl[i] += read_weight * term;
490            i += 1;
491            if j == 0 {
492                break;
493            }
494            j -= 1;
495        }
496    }
497
498    let denom = (read_count as f64) * log10_ploidy;
499    for x in &mut gl {
500        *x -= denom;
501    }
502    gl
503}

517/// Multi-sample pileups (**nonempty strata only**) use the Java joint path; pooling when only one
518/// strata has reads.
519pub fn haplotype_caller_activity_profile_state_multi_sample(
520    contig: impl Into<String>,
521    pos: u64,
522    sample_pileups: &[Vec<PileupObservation>],
523    params: &HaplotypeCallerActivityScoringParams,
524) -> ActivityProfileState {
525    let contig = contig.into();
526    // Borrow nonempty strata — avoid cloning every observation at every locus.
527    let nonempty: Vec<&[PileupObservation]> = sample_pileups
528        .iter()
529        .filter(|pile| !pile.is_empty())
530        .map(|pile| pile.as_slice())
531        .collect();
532    match nonempty.len() {
533        0 => ActivityProfileState::new(contig.as_str(), pos, 0.0),
534        1 => haplotype_caller_activity_profile_state_single_sample(
535            contig.as_str(),
536            pos,
537            nonempty[0],
538            params,
539        ),
540        _ => {
541            crate::hc_joint_is_active::haplotype_caller_joint_multisample_is_active_activity_state(
542                contig, pos, &nonempty, params,
543            )
544        }
545    }
546}

588    #[test]
589    fn frozen_hom_ref_single_read_matches_python_reference() {
590        let params = HaplotypeCallerActivityScoringParams::default();
591        let obs = [PileupObservation {
592            read_base: b'A',
593            qual: 30,
594            is_deletion: false,
595            is_alt: false,
596            is_next_to_soft_clip: false,
597            read_hq_soft_clip_base_count: 0,
598        }];
599        let gl = calc_ref_vs_any_log10_genotype_likelihoods(2, &obs, &params);
600        assert_eq!(gl.len(), 3);
601        // Slight cross-platform float noise exists because the heterozygous term uses an
602        // approximate Jacobian log table. Keep a narrow bound while remaining platform-stable.
603        assert!((gl[0] + 0.00043451177401771).abs() < 5e-4, "gl={gl:?}");
604        // In the Java formula, heterozygous terms use log10(j)/log10(i) (no +log10(ploidy)),
605        // then the shared denominator subtracts log10(ploidy), so diploid het sits near -0.3010.
606        assert!((gl[1] + 0.30131962629328163).abs() < 5e-4, "gl={gl:?}");
607        assert!((gl[2] + 3.4771212547196626).abs() < 5e-4, "gl={gl:?}");
608        assert!(
609            gl[0] > gl[1],
610            "expected hom-ref > het for one high-Q ref read, gl={gl:?}"
611        );
612        let p = calculate_single_sample_biallelic_non_ref_posterior(&gl, true, &params);
613        assert!(p < 1e-6, "p={p}");
614    }

660    #[test]
661    fn multi_sample_probability_aggregates_any_non_ref() {
662        let params = HaplotypeCallerActivityScoringParams::default();
663        let s1 = vec![PileupObservation {
664            read_base: b'T',
665            qual: 30,
666            is_deletion: false,
667            is_alt: true,
668            is_next_to_soft_clip: false,
669            read_hq_soft_clip_base_count: 0,
670        }];
671        let s2 = vec![PileupObservation {
672            read_base: b'A',
673            qual: 30,
674            is_deletion: false,
675            is_alt: false,
676            is_next_to_soft_clip: false,
677            read_hq_soft_clip_base_count: 0,
678        }];
679        let gl1 = calc_ref_vs_any_log10_genotype_likelihoods(2, &s1, &params);
680        let gl2 = calc_ref_vs_any_log10_genotype_likelihoods(2, &s2, &params);
681        let p1 = calculate_single_sample_biallelic_non_ref_posterior(&gl1, true, &params);
682        let p2 = calculate_single_sample_biallelic_non_ref_posterior(&gl2, true, &params);
683        let pm = calculate_multi_sample_any_non_ref_posterior(&[gl1, gl2], true, &params);
684        let expected = 1.0 - (1.0 - p1) * (1.0 - p2);
685        assert!((pm - expected).abs() < 1e-12, "pm={pm} expected={expected}");
686    }
