1//! Mapped-read sanity checks for user BAM fields (lengths / coordinates).
2
3#![warn(clippy::unwrap_used, clippy::expect_used)]
4
5use gatk_common::{GatkError, GatkResult};
6
7pub fn validate_mapped_read_sanity(
8    read_len: usize,
9    qual_len: usize,
10    reference_start: i64,
11    reference_end: i64,
12) -> GatkResult<()> {
13    if read_len == 0 {
14        return Err(GatkError::read("Malformed read: empty read bases"));
15    }
16    if qual_len != 0 && qual_len != read_len {
17        return Err(GatkError::read(format!(
18            "Malformed read: base/quality length mismatch (bases={read_len}, quals={qual_len})"
19        )));
20    }
21    if reference_end < reference_start {
22        return Err(GatkError::read(format!(
23            "Malformed read: negative reference span (start={reference_start}, end={reference_end})"
24        )));
25    }
26    Ok(())
27}

40            other => panic!("expected Read error, got {other:?}"),
