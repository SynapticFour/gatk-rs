10/// `likelihood-engine-config` — HC default engine flags.
11pub fn dump_likelihood_engine_config_tsv(out: &mut impl Write) -> GatkResult<()> {
12    let cfg = HcLikelihoodEngineConfig::default();
13    writeln!(out, "key\tvalue").map_err(|e| GatkError::generic(format!("write: {e}")))?;
14    writeln!(out, "primary_engine\t{}", cfg.primary_engine_label())
15        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
16    writeln!(
17        out,
18        "stepwise_filtering\t{}",
19        cfg.filter_step_engine_active()
20    )
21    .map_err(|e| GatkError::generic(format!("write: {e}")))?;
22    writeln!(out, "dragstr_pair_hmm\t{}", cfg.uses_dragstr_pair_hmm())
23        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
24    writeln!(
25        out,
26        "flow_based\t{}",
27        cfg.implementation == HcLikelihoodImplementation::FlowBased
28    )
29    .map_err(|e| GatkError::generic(format!("write: {e}")))?;
30    Ok(())
31}

33/// `pcr-error-model` — adjusted ins qual at repeat lengths (CONSERVATIVE).
34pub fn dump_pcr_error_model_tsv(out: &mut impl Write) -> GatkResult<()> {
35    writeln!(out, "repeat_length\tadjusted_ins_qual")
36        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
37    let rate = PcrErrorModel::Conservative.rate_factor().unwrap();
38    for repeat in 0..=10 {
39        writeln!(out, "{repeat}\t{}", error_model_adjusted_qual(repeat, rate))
40            .map_err(|e| GatkError::generic(format!("write: {e}")))?;
41    }
42    Ok(())
43}

45/// `likelihood-pcr-read` — one read with PCR-adjusted likelihood vs unadjusted.
46pub fn dump_likelihood_pcr_read_tsv(cases_path: &Path, out: &mut impl Write) -> GatkResult<()> {
47    let text = std::fs::read_to_string(cases_path)
48        .map_err(|e| GatkError::generic(format!("read {}: {e}", cases_path.display())))?;
49    writeln!(out, "case_id\tmode\tlog10_likelihood")
50        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
51    for line in text.lines() {
52        let line = line.trim();
53        if line.is_empty() || line.starts_with('#') {
54            continue;
55        }
56        let cols: Vec<&str> = line.split('\t').collect();
57        if cols.len() < 4 {
58            return Err(GatkError::argument("pcr-read cases need 4 cols"));
59        }
60        let case_id = cols[0];
61        let read = cols[1].as_bytes();
62        let quals: Vec<u8> = cols[2]
63            .split(',')
64            .map(|p| p.parse::<u8>().map_err(|_| GatkError::argument("bad qual")))
65            .collect::<Result<_, _>>()?;
66        let hap = cols[3].as_bytes();
67        let ll_none = log10_pairhmm_likelihood_parity_defaults(read, &quals, hap)?;
68        let mut cfg = HcLikelihoodEngineConfig::default();
69        cfg.pcr_error_model = PcrErrorModel::None;
70        let mapq: u8 = cols.get(4).and_then(|s| s.parse().ok()).unwrap_or(60);
71        let ll_none2 = crate::likelihood_engine::log10_read_haplotype_likelihood(
72            &cfg, read, &quals, mapq, hap,
73        )?;
74        let ll_pcr = crate::likelihood_engine::log10_read_haplotype_likelihood(
75            &HcLikelihoodEngineConfig::default(),
76            read,
77            &quals,
78            mapq,
79            hap,
80        )?;
81        let _ = ll_none2;
82        writeln!(out, "{case_id}\tnone\t{ll_none:.17}")?;
83        writeln!(out, "{case_id}\tconservative\t{ll_pcr:.17}")?;
84    }
85    Ok(())
86}
