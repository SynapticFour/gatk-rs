1//! GATK `SeqGraph` simplification: mirrors `SeqGraph.simplifyGraphOnce` (GAP-E-03).
2//! Java reference: `.gatk-src/.../graphs/SeqGraph.java` — MergeDiamonds, MergeTails,
3//! SplitCommonSuffices (`CommonSuffixSplitter`), MergeCommonSuffices (`SharedSequenceMerger`), zip.
4
5use crate::seq_graph::SeqGraph;
6use std::collections::HashSet;

8const MIN_TAIL_SUFFIX_LEN: usize = 10;
9const MAX_SIMPLIFICATION_ITERATIONS: usize = 100;
10const MAX_REASONABLE_SIMPLIFICATION_CYCLES: usize = 100;
11
12/// Full Java `SeqGraph.simplifyGraph` loop.
13pub fn simplify_graph_full(graph: &mut SeqGraph) {
14    let _ = graph.zip_linear_chains();
15    let mut prev_signature: Option<Vec<u8>> = None;
16    for i in 0..MAX_SIMPLIFICATION_ITERATIONS {
17        if i > MAX_REASONABLE_SIMPLIFICATION_CYCLES {
18            break;
19        }
20        if !simplify_graph_once(graph) {
21            break;
22        }
23        if i > 5 {
24            let sig = graph_topology_signature(graph);
25            if prev_signature.as_ref() == Some(&sig) {
26                break;
27            }
28            prev_signature = Some(sig);
29        }
30    }
31}

33fn graph_topology_signature(graph: &SeqGraph) -> Vec<u8> {
34    let mut out = Vec::new();
35    out.extend_from_slice(&(graph.node_count() as u32).to_le_bytes());
36    out.extend_from_slice(&(graph.edge_count() as u32).to_le_bytes());
37    for v in 0..graph.node_count() {
38        out.extend_from_slice(&(graph.vertex_sequence(v).len() as u32).to_le_bytes());
39        out.extend(graph.vertex_sequence(v));
40    }
41    out
42}

44/// One Java `simplifyGraphOnce` iteration.
45fn simplify_graph_once(graph: &mut SeqGraph) -> bool {
46    let mut did = false;
47    did |= merge_diamonds_until_complete(graph);
48    did |= merge_tails_until_complete(graph);
49    did |= split_common_suffices_until_complete(graph);
50    did |= merge_common_suffices_until_complete(graph);
51    did |= graph.zip_linear_chains();
52    did
53}

55fn merge_diamonds_until_complete(graph: &mut SeqGraph) -> bool {
56    transform_until_complete(graph, try_merge_diamond)
57}
58
59fn merge_tails_until_complete(graph: &mut SeqGraph) -> bool {
60    transform_until_complete(graph, try_merge_tails)
61}
62
63fn split_common_suffices_until_complete(graph: &mut SeqGraph) -> bool {
64    let mut did = false;
65    let mut already_split: HashSet<usize> = HashSet::new();
66    loop {
67        let mut found = false;
68        for bottom in 0..graph.node_count() {
69            if already_split.contains(&bottom) {
70                continue;
71            }
72            if try_split_common_suffix(graph, bottom, &mut already_split) {
73                found = true;
74                did = true;
75                break;
76            }
77        }
78        if !found {
79            break;
80        }
81    }
82    did
83}

85fn merge_common_suffices_until_complete(graph: &mut SeqGraph) -> bool {
86    transform_until_complete(graph, try_merge_common_suffices)
87}

89fn transform_until_complete(
90    graph: &mut SeqGraph,
91    mut transform: impl FnMut(&mut SeqGraph, usize) -> bool,
92) -> bool {
93    let mut did_any = false;
94    loop {
95        let mut found = false;
96        for v in 0..graph.node_count() {
97            if transform(graph, v) {
98                found = true;
99                did_any = true;
100                break;
101            }
102        }
103        if !found {
104            break;
105        }
106    }
107    did_any
108}

110fn try_merge_diamond(graph: &mut SeqGraph, top: usize) -> bool {
111    let middles: Vec<usize> = graph.outgoing_nodes(top);
112    if middles.len() <= 1 {
113        return false;
114    }
115    let mut bottom: Option<usize> = None;
116    for &mi in &middles {
117        if graph.vertex_out_degree(mi) < 1 || graph.vertex_in_degree(mi) != 1 {
118            return false;
119        }
120        for mt in graph.outgoing_nodes(mi) {
121            match bottom {
122                None => bottom = Some(mt),
123                Some(b) if b == mt => {}
124                _ => return false,
125            }
126        }
127    }
128    let Some(bottom) = bottom else {
129        return false;
130    };
131    if graph.vertex_in_degree(bottom) != middles.len() {
132        return false;
133    }
134    let seqs: Vec<&[u8]> = middles.iter().map(|&m| graph.vertex_sequence(m)).collect();
135    let (prefix, suffix) = common_prefix_suffix(&seqs);
136    if prefix.is_empty() && suffix.is_empty() {
137        return false;
138    }
139    split_and_update(graph, top, Some(bottom), &middles, &prefix, &suffix)
140}

142fn try_merge_tails(graph: &mut SeqGraph, top: usize) -> bool {
143    let tails: Vec<usize> = graph.outgoing_nodes(top);
144    if tails.len() <= 1 {
145        return false;
146    }
147    for &t in &tails {
148        if !graph.is_sink_vertex(t) || graph.vertex_in_degree(t) > 1 {
149            return false;
150        }
151    }
152    let seqs: Vec<&[u8]> = tails.iter().map(|&t| graph.vertex_sequence(t)).collect();
153    let (prefix, suffix) = common_prefix_suffix(&seqs);
154    if suffix.len() < MIN_TAIL_SUFFIX_LEN {
155        return false;
156    }
157    split_and_update(graph, top, None, &tails, &prefix, &suffix)
158}

160/// GATK `CommonSuffixSplitter.split` — split incoming vertices of `bottom` on shared suffix.
161fn try_split_common_suffix(
162    graph: &mut SeqGraph,
163    bottom: usize,
164    already_split: &mut HashSet<usize>,
165) -> bool {
166    already_split.insert(bottom);
167    let prevs: Vec<usize> = graph.incoming_nodes(bottom);
168    if prevs.len() < 2 {
169        return false;
170    }
171    if !safe_to_split_suffix(graph, bottom, &prevs) {
172        return false;
173    }
174    let seqs: Vec<&[u8]> = prevs.iter().map(|&p| graph.vertex_sequence(p)).collect();
175    let suffix = common_suffix_only(&seqs);
176    if suffix.is_empty() {
177        return false;
178    }
179    if would_eliminate_ref_source(graph, &suffix, &prevs) {
180        return false;
181    }
182    if all_vertices_are_only_suffix(&seqs, &suffix) {
183        return false;
184    }
185
186    let suffix_v = graph.add_seq_vertex(suffix.clone());
187    let mut edges_to_remove: Vec<(usize, usize)> = Vec::new();
188    let mut mids_to_remove: HashSet<usize> = HashSet::new();
189
190    for &mid in &prevs {
191        mids_to_remove.insert(mid);
192        let mid_seq = graph.vertex_sequence(mid).to_vec();
193        let prefix = without_suffix(&mid_seq, &suffix);
194        let out_targets = graph.outgoing_nodes(mid);
195        if out_targets.len() != 1 {
196            return false;
197        }
198        let out_target = out_targets[0];
199        let (out_sup, out_ref) = graph.outgoing_edge_support_is_ref(mid);
200
201        let incoming_target = if let Some(prefix_bytes) = prefix {
202            let pv = graph.add_seq_vertex(prefix_bytes);
203            graph.add_or_update_edge(pv, suffix_v, out_sup, out_ref);
204            edges_to_remove.push((mid, out_target));
205            pv
206        } else {
207            edges_to_remove.push((mid, out_target));
208            suffix_v
209        };
210
211        // GATK: suffixV -> getEdgeTarget(outgoingEdgeOf(mid)) == bottom (CommonSuffixSplitter.java).
212        graph.add_or_update_edge(suffix_v, out_target, out_sup, out_ref);
213
214        for prev in graph.incoming_nodes(mid) {
215            let (in_sup, in_ref) = graph.incoming_edge_support_is_ref(mid);
216            graph.add_or_update_edge(prev, incoming_target, in_sup, in_ref);
217            edges_to_remove.push((prev, mid));
218        }
219    }
220
221    graph.remove_vertices_by_id(&mids_to_remove);
222    for (from, to) in edges_to_remove {
223        graph.remove_edge(from, to);
224    }
225    true
226}

228/// GATK `SharedSequenceMerger.merge` — merge identical incoming vertices of `bottom`.
229fn try_merge_common_suffices(graph: &mut SeqGraph, bottom: usize) -> bool {
230    let prevs: Vec<usize> = graph.incoming_nodes(bottom);
231    if prevs.is_empty() {
232        return false;
233    }
234    let first = graph.vertex_sequence(prevs[0]);
235    for &p in &prevs[1..] {
236        if graph.vertex_sequence(p) != first {
237            return false;
238        }
239    }
240    for &p in &prevs {
241        if graph.vertex_out_degree(p) != 1 || graph.outgoing_nodes(p)[0] != bottom {
242            return false;
243        }
244        if graph.vertex_in_degree(p) == 0 {
245            return false;
246        }
247    }
248
249    let mut merged_seq = first.to_vec();
250    merged_seq.extend_from_slice(graph.vertex_sequence(bottom));
251    let new_v = graph.add_seq_vertex(merged_seq);
252
253    let bottom_outs: Vec<usize> = graph.outgoing_nodes(bottom);
254    for &p in &prevs {
255        for prev in graph.incoming_nodes(p) {
256            let (sup, is_ref) = graph.incoming_edge_support_is_ref(p);
257            graph.add_or_update_edge(prev, new_v, sup, is_ref);
258        }
259    }
260    for &t in &bottom_outs {
261        let (sup, is_ref) = graph.outgoing_edge_support_is_ref(bottom);
262        graph.add_or_update_edge(new_v, t, sup, is_ref);
263    }
264
265    let mut remove: HashSet<usize> = prevs.iter().copied().collect();
266    remove.insert(bottom);
267    graph.remove_vertices_by_id(&remove);
268    true
269}

271fn safe_to_split_suffix(graph: &SeqGraph, bottom: usize, prevs: &[usize]) -> bool {
272    let bottom_outs: Vec<usize> = graph.outgoing_nodes(bottom);
273    if bottom_outs.len() != 1 {
274        return false;
275    }
276    let outgoing_of_bottom: HashSet<usize> = bottom_outs.into_iter().collect();
277    for &mid in prevs {
278        if mid == bottom {
279            return false;
280        }
281        let outs = graph.outgoing_nodes(mid);
282        if outs.len() != 1 || outs[0] != bottom {
283            return false;
284        }
285        if outgoing_of_bottom.contains(&mid) {
286            return false;
287        }
288    }
289    true
290}

292fn would_eliminate_ref_source(graph: &SeqGraph, suffix: &[u8], prevs: &[usize]) -> bool {
293    for &p in prevs {
294        if graph.is_ref_source_vertex(p) {
295            return graph.vertex_sequence(p).len() == suffix.len();
296        }
297    }
298    false
299}

301fn all_vertices_are_only_suffix(seqs: &[&[u8]], suffix: &[u8]) -> bool {
302    seqs.iter().all(|s| s.len() == suffix.len())
303}

305fn common_suffix_only(seqs: &[&[u8]]) -> Vec<u8> {
306    if seqs.is_empty() {
307        return Vec::new();
308    }
309    let min_len = seqs.iter().map(|s| s.len()).min().unwrap_or(0);
310    let mut suffix_len = 0usize;
311    'outer: for i in 0..min_len {
312        let b = seqs[0][seqs[0].len() - 1 - i];
313        for s in &seqs[1..] {
314            if s[s.len() - 1 - i] != b {
315                break 'outer;
316            }
317        }
318        suffix_len += 1;
319    }
320    if suffix_len == 0 {
321        Vec::new()
322    } else {
323        let start = seqs[0].len() - suffix_len;
324        seqs[0][start..].to_vec()
325    }
326}

328fn without_suffix(seq: &[u8], suffix: &[u8]) -> Option<Vec<u8>> {
329    if seq.len() < suffix.len() || !seq.ends_with(suffix) {
330        return None;
331    }
332    let prefix = &seq[..seq.len() - suffix.len()];
333    if prefix.is_empty() {
334        None
335    } else {
336        Some(prefix.to_vec())
337    }
338}

340fn split_and_update(
341    graph: &mut SeqGraph,
342    top: usize,
343    bottom: Option<usize>,
344    middles: &[usize],
345    prefix: &[u8],
346    suffix: &[u8],
347) -> bool {
348    let prefix_v = if prefix.is_empty() {
349        None
350    } else {
351        Some(graph.add_seq_vertex(prefix.to_vec()))
352    };
353    let suffix_v = if suffix.is_empty() {
354        None
355    } else {
356        Some(graph.add_seq_vertex(suffix.to_vec()))
357    };
358
359    let mut new_middles: Vec<usize> = Vec::new();
360    for &mid in middles {
361        let (in_sup, in_ref) = graph.incoming_edge_support_is_ref(mid);
362        let (out_sup, out_ref) = graph.outgoing_edge_support_is_ref(mid);
363        let mid_seq = graph.vertex_sequence(mid).to_vec();
364        if let Some(remaining) = without_prefix_suffix(&mid_seq, prefix, suffix) {
365            let nm = graph.add_seq_vertex(remaining);
366            new_middles.push(nm);
367            if let Some(pv) = prefix_v {
368                graph.add_or_update_edge(pv, nm, in_sup, in_ref);
369            }
370            if let Some(sv) = suffix_v {
371                graph.add_or_update_edge(nm, sv, out_sup, out_ref);
372            } else if let Some(b) = bottom {
373                graph.add_or_update_edge(nm, b, out_sup, out_ref);
374            }
375        } else if let (Some(pv), Some(sv)) = (prefix_v, suffix_v) {
376            graph.add_or_update_edge(pv, sv, in_sup.saturating_add(out_sup), in_ref || out_ref);
377        } else if let (Some(pv), None) = (prefix_v, suffix_v) {
378            if let Some(b) = bottom {
379                graph.add_or_update_edge(pv, b, in_sup.saturating_add(out_sup), in_ref || out_ref);
380            }
381        }
382    }
383
384    let mut top_to_mid_sup = 0u32;
385    let mut top_to_mid_ref = false;
386    for &mid in middles {
387        if let Some(idx) = graph.find_edge(top, mid) {
388            let e = &graph.edges_pub()[idx];
389            top_to_mid_sup = top_to_mid_sup.saturating_add(e.support);
390            top_to_mid_ref |= e.is_ref;
391        }
392    }
393
394    let remove: HashSet<usize> = middles.iter().copied().collect();
395    graph.remove_vertices_by_id(&remove);
396
397    if let Some(pv) = prefix_v {
398        graph.add_or_update_edge(top, pv, top_to_mid_sup, top_to_mid_ref);
399    } else {
400        for &nm in &new_middles {
401            let (sup, is_ref) = graph.incoming_edge_support_is_ref(nm);
402            graph.add_or_update_edge(top, nm, sup, is_ref);
403        }
404    }
405
406    if let (Some(sv), Some(b)) = (suffix_v, bottom) {
407        let (sup, is_ref) = graph.incoming_edge_support_is_ref(b);
408        graph.add_or_update_edge(sv, b, sup, is_ref);
409    }
410
411    !new_middles.is_empty() || prefix_v.is_some() || suffix_v.is_some()
412}

414fn common_prefix_suffix(seqs: &[&[u8]]) -> (Vec<u8>, Vec<u8>) {
415    if seqs.is_empty() {
416        return (Vec::new(), Vec::new());
417    }
418    let min_len = seqs.iter().map(|s| s.len()).min().unwrap_or(0);
419    let mut prefix_len = 0usize;
420    'prefix: for i in 0..min_len {
421        let b = seqs[0][i];
422        for s in &seqs[1..] {
423            if s[i] != b {
424                break 'prefix;
425            }
426        }
427        prefix_len += 1;
428    }
429    let remain = min_len.saturating_sub(prefix_len);
430    let mut suffix_len = 0usize;
431    'suffix: for i in 0..remain {
432        let b = seqs[0][seqs[0].len() - 1 - i];
433        for s in &seqs[1..] {
434            if s[s.len() - 1 - i] != b {
435                break 'suffix;
436            }
437        }
438        suffix_len += 1;
439    }
440    let prefix = seqs[0][..prefix_len].to_vec();
441    let suffix = if suffix_len == 0 {
442        Vec::new()
443    } else {
444        let start = seqs[0].len() - suffix_len;
445        seqs[0][start..].to_vec()
446    };
447    (prefix, suffix)
448}

450fn without_prefix_suffix(seq: &[u8], prefix: &[u8], suffix: &[u8]) -> Option<Vec<u8>> {
451    if seq.len() < prefix.len() + suffix.len() {
452        return None;
453    }
454    if !seq.starts_with(prefix) {
455        return None;
456    }
457    if !suffix.is_empty() && !seq.ends_with(suffix) {
458        return None;
459    }
460    let end = seq.len() - suffix.len();
461    let mid = &seq[prefix.len()..end];
462    if mid.is_empty() {
463        None
464    } else {
465        Some(mid.to_vec())
466    }
467}
