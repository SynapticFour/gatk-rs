22fn format_ll(v: f64) -> String {
23    if v.is_infinite() && v.is_sign_negative() {
24        "-inf".to_string()
25    } else {
26        format!("{v:.17}")
27    }
28}

30fn write_region_header(out: &mut impl Write, region: &AssemblyRegion) -> GatkResult<()> {
31    writeln!(out, "region_contig\t{}", region.contig)?;
32    writeln!(out, "region_start\t{}", region.start.get())?;
33    writeln!(out, "region_end\t{}", region.end.get())?;
34    writeln!(out, "is_active\t{}", region.is_active)?;
35    Ok(())
36}

38/// First active region: assemble haplotypes, score each read×haplotype with GATK `Log10PairHMM` defaults.
39pub fn dump_assembly_region_pairhmm_likelihoods_tsv(
40    ref_fasta: &Path,
41    bam_path: &Path,
42    interval_cli: &str,
43    padding: u64,
44    target: AssemblyRegionHaplotypeTarget,
45    out: &mut impl Write,
46) -> GatkResult<()> {
47    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
48    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
49    let filters = ReadFilterParams::gatk_standard_hc();
50    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
51    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
52    let regions = flatten_assembly_regions(&walk);
53    let mut ref_cache = ReferenceWindowCache::new(ref_fasta.to_path_buf(), 4);
54
55    let region = match target {
56        AssemblyRegionHaplotypeTarget::Active => regions
57            .iter()
58            .find(|r| {
59                matches!(
60                    call_disposition(r),
61                    AssemblyRegionCallDisposition::ActiveFull
62                )
63            })
64            .ok_or_else(|| GatkError::argument("no active assembly region in interval"))?,
65        AssemblyRegionHaplotypeTarget::Inactive => regions
66            .iter()
67            .find(|r| !r.is_active)
68            .ok_or_else(|| GatkError::argument("no inactive assembly region in interval"))?,
69    };
70
71    write_region_header(out, region)?;
72    if target == AssemblyRegionHaplotypeTarget::Inactive || !region.is_active {
73        writeln!(out, "read_count\t0")?;
74        writeln!(out, "haplotype_count\t0")?;
75        return Ok(());
76    }
77
78    let args = AssembleReadsArgs::default();
79    let finalized = finalize_region_reads_for_assembly(
80        &region.reads,
81        region,
82        args.correct_overlapping_base_qualities,
83        gatk_min_tail_quality_for_assembly(args.assembler.min_base_quality),
84        false,
85    );
86    let ars = assemble_reads(region, &dict, &mut ref_cache, &args)?;
87    let haplotypes: Vec<&[u8]> = ars.haplotypes.iter().map(|h| h.bases.as_slice()).collect();
88
89    writeln!(out, "read_count\t{}", finalized.len())?;
90    writeln!(out, "haplotype_count\t{}", haplotypes.len())?;
91    writeln!(out, "read_index\thaplotype_index\tlog10_likelihood")?;
92
93    const BQ_THRESHOLD: u8 = 18;
94    let mut read_records: Vec<_> = finalized.into_iter().collect();
95    read_records.sort_by(|a, b| a.qname().cmp(b.qname()).then_with(|| a.pos().cmp(&b.pos())));
96
97    for (ri, rec) in read_records.iter().enumerate() {
98        let bases = rec.seq().as_bytes();
99        let mut quals = rec.qual().to_vec();
100        let mapq = rec.mapq();
101        cap_read_base_qualities(&mut quals, mapq, BQ_THRESHOLD, false);
102        for (hi, hap) in haplotypes.iter().enumerate() {
103            let ll = log10_pairhmm_likelihood_parity_defaults(&bases, &quals, hap)?;
104            writeln!(out, "{ri}\t{hi}\t{}", format_ll(ll))?;
105        }
106    }
107    Ok(())
108}
