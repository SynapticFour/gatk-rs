1//! HC shard read-pipeline dump: IUPAC pre-transform, filter, post-transform.
2//! Downsample is **not** included (isolated in D.2). Matches `HcFullParityGateDump read-shard-pipeline`.
3
4use crate::dragen_mq::apply_dragen_mapping_quality_transform;
5use crate::read_model::{
6    passes_hc_read_filters_with_header, ReadFilterParams, GATK_HC_DEFAULT_MIN_MAPPING_QUALITY,
7};
8use crate::read_transformer::apply_iupac_strict_transform;
9use gatk_common::{GatkError, GatkResult};
10use rust_htslib::bam;
11use rust_htslib::bam::Read as _;
12use std::io::Write;
13use std::path::Path;
14
15fn seq_to_string(bytes: &[u8]) -> String {
16    String::from_utf8_lossy(bytes).into_owned()
17}
18
19/// TSV per read: `qname`, `flags`, `mapq`, `seq_raw`, `seq_after_pre`, `passes_hc_filter`, `seq_after_post`.
20/// Pipeline stages mirror `MultiIntervalLocalReadShard` without downsampler:
21/// `makeStandardHCReadTransformer` → `makeStandardHCReadFilters` → identity post-transform.
22pub fn dump_read_shard_pipeline_tsv(
23    alignment_path: &Path,
24    apply_dragen_mq: bool,
25    out: &mut impl Write,
26) -> GatkResult<()> {
27    writeln!(
28        out,
29        "qname\tflags\tmapq\tseq_raw\tseq_after_pre\tpasses_hc_filter\tseq_after_post"
30    )
31    .map_err(|e| GatkError::generic(format!("write: {e}")))?;
32
33    let filters = ReadFilterParams::gatk_standard_hc();
34    let mut reader = bam::Reader::from_path(alignment_path)
35        .map_err(|e| GatkError::generic(format!("open: {e}")))?;
36    let header = reader.header().clone();
37
38    for res in reader.records() {
39        let mut rec = res.map_err(|e| GatkError::generic(format!("read: {e}")))?;
40        let seq_raw = seq_to_string(&rec.seq().as_bytes());
41        apply_iupac_strict_transform(std::slice::from_mut(&mut rec));
42        let seq_after_pre = seq_to_string(&rec.seq().as_bytes());
43        let pass = passes_hc_read_filters_with_header(&rec, &header, &filters);
44        let mut post_rec = rec.clone();
45        if apply_dragen_mq {
46            apply_dragen_mapping_quality_transform(std::slice::from_mut(&mut post_rec));
47        }
48        let seq_after_post = seq_to_string(&post_rec.seq().as_bytes());
49        let mapq_after = post_rec.mapq();
50        writeln!(
51            out,
52            "{}\t{}\t{}\t{}\t{}\t{}\t{}",
53            String::from_utf8_lossy(rec.qname()),
54            rec.flags(),
55            if apply_dragen_mq {
56                mapq_after
57            } else {
58                rec.mapq()
59            },
60            seq_raw,
61            seq_after_pre,
62            if pass { "true" } else { "false" },
63            seq_after_post,
64        )
65        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
66    }
67
68    let _ = GATK_HC_DEFAULT_MIN_MAPPING_QUALITY;
69    Ok(())
70}

73mod tests {

76    #[test]
77    fn dump_header_present() {
78        let repo = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
79            .join("../parity/fixtures/hc-full-parity/d4/d4_iupac_mixed.sam");
80        if !repo.exists() {
81            return;
82        }
83        let mut buf = Vec::new();
84        dump_read_shard_pipeline_tsv(&repo, false, &mut buf).unwrap();
85        let s = String::from_utf8(buf).unwrap();
86        assert!(s.starts_with("qname\tflags\tmapq\tseq_raw"));
87        assert!(s.contains("r_mixed"));
88        assert!(s.contains("ANNN"));
89    }
