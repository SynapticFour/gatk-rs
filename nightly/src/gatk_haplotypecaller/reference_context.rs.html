1//! GATK `ReferenceContext` for assembly-region `apply` / `callRegion`.
2//! Wraps a 1-based inclusive reference interval and lazily-loaded bases from FASTA, matching
3//! `new ReferenceContext(reference, assemblyRegion.getPaddedSpan)` in Java.
4
5use gatk_common::{GatkError, GatkResult};
6use gatk_core::reference::{ReferenceWindowCache, SequenceDictionary};
7/// Reference bases for one locatable interval (typically the assembly region padded span).
8/// # Invariants
9/// `window_start`/`window_end` clip `[start, end]` to contig bounds; `bases` length matches inclusive window.
10/// Empty `bases` when the clipped window is inverted (`window_start > window_end`).
11/// # Ownership
12/// Owns contig name and reference byte vector loaded from FASTA cache.
13/// # Mutation
14/// Immutable after [`Self::from_interval`] construction.
15/// # Biological assumptions
16/// Uppercase reference bytes match padded assembly region span used in `apply` / `callRegion`.
17/// # Java equivalence
18/// GATK `ReferenceContext` over assembly region padded span.
19#[derive(Debug, Clone, PartialEq, Eq)]
20pub struct ReferenceContext {
21    pub contig: String,
22    /// Core interval (1-based inclusive).
23    pub start: u64,
24    pub end: u64,
25    /// Query window clipped to contig bounds (1-based inclusive).
26    pub window_start: u64,
27    pub window_end: u64,
28    /// Bases for `[window_start, window_end]` inclusive.
29    pub bases: Vec<u8>,
30}
31
32impl ReferenceContext {
33    pub fn empty() -> Self {
34        Self {
35            contig: String::new(),
36            start: 0,
37            end: 0,
38            window_start: 0,
39            window_end: 0,
40            bases: Vec::new(),
41        }
42    }
43
44    /// Load reference bases for a closed 1-based interval, clipped to `contig` length.
45    pub fn from_interval(
46        dictionary: &SequenceDictionary,
47        ref_cache: &mut ReferenceWindowCache,
48        contig: &str,
49        start1: u64,
50        end1: u64,
51    ) -> GatkResult<Self> {
52        let contig_len = dictionary
53            .contig(contig)
54            .map(|c| c.length)
55            .ok_or_else(|| GatkError::argument(format!("unknown contig {contig}")))?;
56        let window_start = start1.max(1);
57        let window_end = end1.min(contig_len);
58        let bases = if window_start > window_end {
59            Vec::new()
60        } else {
61            ref_cache
62                .get_interval_bytes(dictionary, contig, window_start, window_end)?
63                .to_vec()
64        };
65        Ok(Self {
66            contig: contig.to_string(),
67            start: start1,
68            end: end1,
69            window_start,
70            window_end,
71            bases,
72        })
73    }
74
75    pub fn len(&self) -> usize {
76        self.bases.len()
77    }
78
79    /// Uppercase ASCII reference string for parity dumps.
80    pub fn bases_ascii(&self) -> String {
81        self.bases
82            .iter()
83            .map(|&b| (b as char).to_ascii_uppercase())
84            .collect()
85    }
86
87    /// Expected length for a contiguous window (inclusive ends).
88    pub fn expected_window_len(&self) -> u64 {
89        self.window_end
90            .saturating_sub(self.window_start)
91            .saturating_add(1)
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn padded_span_length_matches_bases() {
101        let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../parity/fixtures");
102        let ref_fa = root.join("reference.fa");
103        let dict = SequenceDictionary::from_fasta_path(&ref_fa).unwrap();
104        let mut cache = ReferenceWindowCache::new(&ref_fa, 4);
105        let ctx = ReferenceContext::from_interval(&dict, &mut cache, "chr1", 1, 11).unwrap();
106        assert_eq!(ctx.len(), 11);
107        assert_eq!(ctx.expected_window_len(), 11);
108        assert_eq!(ctx.bases_ascii().len(), 11);
109    }
110
111    #[test]
112    fn extended_span_loads_full_window() {
113        let root = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../parity/fixtures");
114        let ref_fa = root.join("reference.fa");
115        let dict = SequenceDictionary::from_fasta_path(&ref_fa).unwrap();
116        let mut cache = ReferenceWindowCache::new(&ref_fa, 2);
117        let ctx = ReferenceContext::from_interval(&dict, &mut cache, "chr1", 1, 32).unwrap();
118        assert_eq!(ctx.window_start, 1);
119        assert_eq!(ctx.window_end, 32);
120        assert_eq!(ctx.len(), 32);
121    }
