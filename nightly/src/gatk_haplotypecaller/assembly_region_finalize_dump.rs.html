26#[derive(Debug, Clone)]
27struct FinalizeReadRow {
28    qname: String,
29    phase: &'static str,
30    flags: u16,
31    mapq: u8,
32    start: i64,
33    end: i64,
34    cigar: String,
35    seq_len: usize,
36    unclipped_len: usize,
37    fragment_length: i64,
38    unmapped: bool,
39}

41fn format_cigar(rec: &bam::Record) -> String {
42    rec.cigar()
43        .iter()
44        .map(|c| {
45            let op = match c {
46                Cigar::Match(_) => 'M',
47                Cigar::Ins(_) => 'I',
48                Cigar::Del(_) => 'D',
49                Cigar::SoftClip(_) => 'S',
50                Cigar::HardClip(_) => 'H',
51                Cigar::Equal(_) => '=',
52                Cigar::Diff(_) => 'X',
53                Cigar::RefSkip(_) => 'N',
54                Cigar::Pad(_) => 'P',
55            };
56            format!("{}{}", crate::read_unclip::cigar_len(c), op)
57        })
58        .collect()
59}

61fn finalize_read_row(rec: &bam::Record, phase: &'static str) -> FinalizeReadRow {
62    let unmapped = rec.tid() < 0 || rec.is_unmapped() || rec.seq().is_empty();
63    let start = if unmapped { 0 } else { rec.pos() + 1 };
64    let end = if unmapped {
65        0
66    } else {
67        i64::from(alignment_end_1based(rec))
68    };
69    FinalizeReadRow {
70        qname: String::from_utf8_lossy(rec.qname()).into_owned(),
71        phase,
72        flags: rec.flags(),
73        mapq: rec.mapq(),
74        start,
75        end,
76        cigar: if unmapped {
77            String::new()
78        } else {
79            format_cigar(rec)
80        },
81        seq_len: rec.seq().as_bytes().len(),
82        unclipped_len: unclipped_read_length(rec),
83        fragment_length: rec.insert_size(),
84        unmapped,
85    }
86}

93/// First active region: raw iterator reads vs production `finalizeRegion` rows.
94pub fn dump_assembly_region_finalize_reads_tsv(
95    ref_fasta: &Path,
96    bam_path: &Path,
97    interval_cli: &str,
98    padding: u64,
99    out: &mut impl Write,
100) -> GatkResult<()> {
101    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
102    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
103    let filters = ReadFilterParams::gatk_standard_hc();
104    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
105    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
106    let regions = flatten_assembly_regions(&walk);
107    let region = select_first_active_region(&regions)?;
108
109    let min_base_quality = 10u8;
110    let min_tail = gatk_min_tail_quality_for_assembly(min_base_quality);
111    let finalized = assembly_reads_for_production(&region.reads, region, min_tail, true, false);
112
113    let header = bam::Reader::from_path(bam_path)
114        .map_err(|e| GatkError::generic(format!("open bam: {e}")))?
115        .header()
116        .clone();
117    fn sort_java<R: std::borrow::Borrow<bam::Record>>(
118        reads: &[R],
119        phase: &'static str,
120        header: &bam::HeaderView,
121    ) -> Vec<FinalizeReadRow> {
122        let mut sorted: Vec<&bam::Record> = reads.iter().map(|r| r.borrow()).collect();
123        sorted.sort_by(|a, b| {
124            compare_read_coordinates_java(a, b, header)
125                .cmp(&0)
126                .then_with(|| a.qname().cmp(b.qname()))
127        });
128        sorted
129            .into_iter()
130            .map(|r| finalize_read_row(r, phase))
131            .collect()
132    }
133    let raw_rows = sort_java(&region.reads, "raw", &header);
134    let fin_rows = sort_java(&finalized, "finalize", &header);
135
136    writeln!(out, "region_contig\t{}", region.contig)
137        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
138    writeln!(out, "region_start\t{}", region.start.get())
139        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
140    writeln!(out, "region_end\t{}", region.end.get())
141        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
142    writeln!(out, "extended_start\t{}", region.extended_start.get())
143        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
144    writeln!(out, "extended_end\t{}", region.extended_end.get())
145        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
146    writeln!(out, "read_path\tfinalize").map_err(|e| GatkError::generic(format!("write: {e}")))?;
147    writeln!(out, "raw_read_count\t{}", raw_rows.len())
148        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
149    writeln!(out, "finalize_read_count\t{}", fin_rows.len())
150        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
151    if !matches!(
152        call_disposition(region),
153        AssemblyRegionCallDisposition::ActiveFull
154    ) {
155        writeln!(out, "warn\tregion_inactive_using_reads_for_asm_diagnostic")
156            .map_err(|e| GatkError::generic(format!("write: {e}")))?;
157    }
158    writeln!(
159        out,
160        "read\tqname\tphase\tflags\tmapq\tstart\tend\tcigar\tseq_len\tunclipped_len\tfragment_length\tunmapped"
161    )
162    .map_err(|e| GatkError::generic(format!("write: {e}")))?;
163    for row in raw_rows.iter().chain(fin_rows.iter()) {
164        writeln!(
165            out,
166            "read\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}",
167            row.qname,
168            row.phase,
169            row.flags,
170            row.mapq,
171            row.start,
172            row.end,
173            row.cigar,
174            row.seq_len,
175            row.unclipped_len,
176            row.fragment_length,
177            row.unmapped
178        )
179        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
180    }
181    Ok(())
182}
