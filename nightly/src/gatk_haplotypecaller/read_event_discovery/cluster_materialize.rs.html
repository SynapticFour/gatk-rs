2/// Strict Java: when threading/SeqGraph miss cluster indels, build alt hap from read-proven
3/// `TTC/T` + `A/ATG` then derive EventMap from hap CIGAR (not list inject).
4/// Java analogue: reads are already in `assembleReads`; this adds an alt haplotype the graph
5/// should have produced, then `EventMap.buildEventMapsForHaplotypes` / `getVariationEvents`
6/// (see `AssemblyBasedCallerUtils` + `HaplotypeCallerEngine` after `trimTo`).
7pub fn strict_materialize_cluster_haplotype_from_reads(
8    assembly: &mut AssemblyResultSet,
9    reads: &[bam::Record],
10    active_start_1based: u64,
11    active_end_1based: u64,
12    contig: &str,
13    sw: &SwParameters,
14) -> GatkResult<()> {
15    let debug = crate::runtime_config::strict_cluster_debug_enabled();
16    if active_end_1based < P12_CLUSTER_TTC_START
17        || active_start_1based > P12_CLUSTER_TTC_START.saturating_add(3)
18    {
19        if debug {
20            eprintln!("strict_materialize\tskip span {active_start_1based}-{active_end_1based}");
21        }
22        return Ok(());
23    }
24    let coupled = cluster_coupled_events_from_assembly_haplotypes(
25        assembly,
26        contig,
27        active_start_1based,
28        active_end_1based,
29    );
30    if cluster_coupled_events_complete(&coupled) {
31        if debug {
32            eprintln!("strict_materialize\tskip already complete {coupled:?}");
33        }
34        sync_assembly_events_from_haplotype_cigars_with_harvest(assembly, contig, sw, SyncAssemblyOptions::strict_java());
35        return Ok(());
36    }
37    // Must match Java trim slice: ref hap `genome_loc.start` + trim-length bases (not full pad).
38    let (apply_bases, apply_pad, _) = reference_hap_apply_window(assembly);
39    let read_coupled = discover_p12_cluster_coupled_events_from_reads(
40        reads,
41        &apply_bases,
42        apply_pad,
43        active_start_1based,
44        active_end_1based,
45        contig,
46    );
47    if debug {
48        eprintln!(
49            "strict_materialize\tpad={apply_pad} ref_len={} reads={} asm={coupled:?} read={read_coupled:?}",
50            apply_bases.len(),
51            reads.len()
52        );
53    }
54    if read_coupled.is_empty() {
55        return Ok(());
56    }
57    // Graph + ref-motif path may have already materialized coupled cluster haps in assembleReads.
58    let coupled_after_graph = cluster_coupled_events_from_assembly_haplotypes(
59        assembly,
60        contig,
61        active_start_1based,
62        active_end_1based,
63    );
64    if cluster_coupled_events_complete(&coupled_after_graph) {
65        sync_assembly_events_from_haplotype_cigars_with_harvest(assembly, contig, sw, SyncAssemblyOptions::strict_java());
66        return Ok(());
67    }
68    let haps_before = assembly.haplotypes.len();
69    upsert_coupled_cluster_alt_haplotype(assembly, &apply_bases, apply_pad, &read_coupled, sw)?;
70    if debug {
71        let after = cluster_coupled_events_from_assembly_haplotypes(
72            assembly,
73            contig,
74            active_start_1based,
75            active_end_1based,
76        );
77        eprintln!(
78            "strict_materialize\thaps {haps_before} -> {} coupled_asm={after:?}",
79            assembly.haplotypes.len()
80        );
81    }
82    repair_alt_haplotype_alignment_for_event_map(&mut assembly.haplotypes, sw);
83    sync_assembly_events_from_haplotype_cigars_with_harvest(assembly, contig, sw, SyncAssemblyOptions::strict_java());
84    Ok(())
85}

87/// Re-attach pre-trim cluster indels when trim clips alt-hap CIGARs to all-`M`.
88pub fn propagate_cluster_coupled_from_untrimmed(
89    untrimmed: &AssemblyResultSet,
90    assembly: &mut AssemblyResultSet,
91    apply_bases: &[u8],
92    apply_pad: u64,
93    active_start_1based: u64,
94    active_end_1based: u64,
95    contig: &str,
96    sw: &SwParameters,
97) -> GatkResult<()> {
98    let from_untrimmed = cluster_coupled_events_from_assembly_haplotypes(
99        untrimmed,
100        contig,
101        active_start_1based,
102        active_end_1based,
103    );
104    if from_untrimmed.is_empty() {
105        return Ok(());
106    }
107    let full_pad = assembly.padded_reference_start_1based();
108    refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, apply_bases, full_pad, sw);
109    let ref_hap = assembly
110        .haplotypes
111        .iter()
112        .find(|h| h.is_reference)
113        .cloned()
114        .unwrap_or_else(|| crate::haplotype::Haplotype::new(apply_bases.to_vec(), true));
115    let (full_ref, full_pad) = assembly.event_map_reference();
116    let max_mnp = assembly.max_mnp_distance();
117    let mut from_trimmed: Vec<VariationEvent> = Vec::new();
118    for h in assembly.haplotypes.iter().filter(|h| !h.is_reference) {
119        for e in crate::event_map::variation_events_for_haplotype(
120            h,
121            &ref_hap,
122            full_ref,
123            full_pad,
124            max_mnp,
125            contig,
126        ) {
127            if is_cluster_coupled_event(&e) {
128                from_trimmed.push(e);
129            }
130        }
131    }
132    if cluster_coupled_events_complete(&from_trimmed) {
133        return Ok(());
134    }
135    push_coupled_cluster_alt_haplotype(assembly, apply_bases, apply_pad, &from_untrimmed, sw)?;
136    refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, apply_bases, full_pad, sw);
137    Ok(())
138}

140/// Synthesize coupled cluster alt on the trim ref when apply fails (graph alt already touched trim).
141fn synthesize_coupled_cluster_bases_on_trim(
142    apply_bases: &[u8],
143    apply_pad: u64,
144) -> Option<Vec<u8>> {
145    let ttc_off = P12_CLUSTER_TTC_START.saturating_sub(apply_pad) as usize;
146    let atg_off = P12_CLUSTER_ATG_START.saturating_sub(apply_pad) as usize;
147    if ttc_off + 3 > apply_bases.len() || atg_off >= apply_bases.len() {
148        return None;
149    }
150    let ttc_ref = &apply_bases[ttc_off..ttc_off + 3];
151    if ttc_ref != b"TTC" && ttc_ref != b"ttc" {
152        return None;
153    }
154    if !apply_bases[atg_off].eq_ignore_ascii_case(&b'A') {
155        return None;
156    }
157    let mut out = apply_bases.to_vec();
158    out.remove(ttc_off + 1);
159    out.remove(ttc_off + 1);
160    let atg_adj = atg_off.saturating_sub(2);
161    if !out.get(atg_adj).copied().unwrap_or(0).eq_ignore_ascii_case(&b'A') {
162        return None;
163    }
164    out.insert(atg_adj + 1, b'T');
165    out.insert(atg_adj + 2, b'G');
166    Some(out)
167}

169/// Build coupled cluster alt bases from trim ref, falling back to full padded ref.
170fn coupled_cluster_alt_bases(
171    assembly: &AssemblyResultSet,
172    apply_bases: &[u8],
173    apply_pad: u64,
174    cluster_events: &[VariationEvent],
175) -> Option<Vec<u8>> {
176    if let Some(bases) = apply_events_to_ref_chained(apply_bases, cluster_events, apply_pad) {
177        return Some(bases);
178    }
179    if cluster_coupled_events_complete(cluster_events) {
180        if let Some(bases) = synthesize_coupled_cluster_bases_on_trim(apply_bases, apply_pad) {
181            return Some(bases);
182        }
183    }
184    let full_ref = assembly.reference_bases_shared();
185    let full_pad = assembly.padded_reference_start_1based();
186    let full_alt = apply_events_to_ref_chained(&full_ref, cluster_events, full_pad)?;
187    let ref_hap = assembly.haplotypes.iter().find(|h| h.is_reference)?;
188    let trim_off = ref_hap.alignment_start_hap_wrt_ref;
189    if trim_off >= full_alt.len() {
190        return None;
191    }
192    let trim_end = trim_off.saturating_add(apply_bases.len());
193    if full_alt.len() >= trim_end {
194        Some(full_alt[trim_off..trim_end].to_vec())
195    } else {
196        Some(full_alt[trim_off..].to_vec())
197    }
198}

200fn force_cluster_coupled_haplotype_cigar(apply_pad: u64, ref_len: usize) -> crate::cigar::Cigar {
201    use crate::cigar::{Cigar, CigarOperator};
202    let ttc_off = P12_CLUSTER_TTC_START.saturating_sub(apply_pad) as usize;
203    let tail = ref_len.saturating_sub(ttc_off + 3);
204    let mut c = Cigar::new();
205    if ttc_off > 0 {
206        c.push(ttc_off, CigarOperator::Match);
207    }
208    c.push(2, CigarOperator::Deletion);
209    c.push(1, CigarOperator::Match);
210    c.push(2, CigarOperator::Insertion);
211    if tail > 0 {
212        c.push(tail, CigarOperator::Match);
213    }
214    c
215}

217fn align_coupled_cluster_bases(
218    apply_bases: &[u8],
219    coupled_bases: &[u8],
220    ref_cigar_len: usize,
221    ref_align: usize,
222    apply_pad: u64,
223    cluster_events: &[VariationEvent],
224    sw: &SwParameters,
225) -> Option<HaplotypeAssemblyCigar> {
226    if cluster_coupled_events_complete(cluster_events) {
227        return Some(HaplotypeAssemblyCigar {
228            cigar: force_cluster_coupled_haplotype_cigar(apply_pad, apply_bases.len()),
229            alignment_start_hap_wrt_ref: ref_align,
230        });
231    }
232    calculate_haplotype_cigar_for_assembly_with_offset(apply_bases, coupled_bases, ref_cigar_len, sw)
233        .or_else(|| {
234            let cigar = calculate_haplotype_cigar_with_strategy(
235                apply_bases,
236                coupled_bases,
237                sw,
238                SwOverhangStrategy::Indel,
239            )?;
240            Some(HaplotypeAssemblyCigar {
241                cigar,
242                alignment_start_hap_wrt_ref: ref_align,
243            })
244        })
245}

247/// Force read-proven P12 cluster indel CIGAR/bases on the primary alt hap (75M2D1M2I… not SW 76M…).
248pub fn fix_p12_cluster_coupled_alt_haplotype(
249    assembly: &mut AssemblyResultSet,
250    _contig: &str,
251    _sw: &SwParameters,
252) {
253    let (apply_bases, apply_pad, ref_hap) = reference_hap_apply_window(assembly);
254    let ttc_start = P12_CLUSTER_TTC_START.saturating_sub(apply_pad) as usize;
255    if crate::runtime_config::strict_cluster_debug_enabled() {
256        eprintln!(
257            "fix_p12\tpad={apply_pad} ttc_start={ttc_start} apply_len={} ttc_slice={:?}",
258            apply_bases.len(),
259            apply_bases.get(ttc_start..ttc_start.saturating_add(3))
260        );
261    }
262    if ttc_start + 3 > apply_bases.len() {
263        return;
264    }
265    let ttc_ok = apply_bases[ttc_start..ttc_start + 3]
266        .iter()
267        .all(|b| b.eq_ignore_ascii_case(&b'T') || b.eq_ignore_ascii_case(&b'C'))
268        && apply_bases[ttc_start].eq_ignore_ascii_case(&b'T')
269        && apply_bases[ttc_start + 1].eq_ignore_ascii_case(&b'T')
270        && apply_bases[ttc_start + 2].eq_ignore_ascii_case(&b'C');
271    if !ttc_ok {
272        return;
273    }
274    let Some(coupled_bases) = synthesize_coupled_cluster_bases_on_trim(&apply_bases, apply_pad) else {
275        if crate::runtime_config::strict_cluster_debug_enabled() {
276            eprintln!("fix_p12\tsynthesize_failed");
277        }
278        return;
279    };
280    let cigar = force_cluster_coupled_haplotype_cigar(apply_pad, apply_bases.len());
281    // CLONE: needed because owned haplotypes for scoring call.
282    let ref_hap_tag = ref_hap.clone();
283    let kmer = assembly.kmer_size_for_dump();
284    crate::allele_filtering::ensure_reference_haplotype(&mut assembly.haplotypes);
285    let ref_idx = assembly
286        .haplotypes
287        .iter()
288        .position(|h| h.is_reference)
289        .unwrap_or(0);
290    let target = assembly
291        .haplotypes
292        .iter()
293        .enumerate()
294        .find(|(i, h)| {
295            *i != ref_idx
296                && h.cigar.as_ref().is_some_and(|c| {
297                    c.elements.iter().any(|e| e.operator.is_indel())
298                })
299        })
300        .map(|(i, _)| i)
301        .or_else(|| {
302            assembly
303                .haplotypes
304                .iter()
305                .enumerate()
306                .find(|(i, _)| *i != ref_idx)
307                .map(|(i, _)| i)
308        });
309    if let Some(idx) = target {
310        let h = &mut assembly.haplotypes[idx];
311        h.is_reference = false;
312        // CLONE: needed because haplotype owns base string.
313        h.bases = coupled_bases.clone();
314        // CLONE: needed because haplotype owns CIGAR.
315        h.cigar = Some(cigar.clone());
316        h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
317        h.alignment_start_hap_wrt_ref = 0;
318        tag_alt_haplotype_from_reference(h, &ref_hap_tag, kmer);
319        h.alignment_start_hap_wrt_ref = 0;
320        if crate::runtime_config::strict_cluster_debug_enabled() {
321            eprintln!(
322                "fix_p12\tupdated idx={idx} cigar={}",
323                h.cigar
324                    .as_ref()
325                    .map(|c| c.to_gatk_string())
326                    .unwrap_or_default()
327            );
328        }
329        return;
330    }
331    let mut h = Haplotype::new(coupled_bases, false);
332    h.cigar = Some(cigar);
333    h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
334    h.alignment_start_hap_wrt_ref = 0;
335    tag_alt_haplotype_from_reference(&mut h, &ref_hap_tag, kmer);
336    h.alignment_start_hap_wrt_ref = 0;
337    assembly.haplotypes.push(h);
338    assembly.variation_present = true;
339    ensure_p12_cluster_coupled_variation_events(assembly, _contig);
340}

342/// Keep P12 cluster coupled indels + anchor SNPs in `variation_events` (EventMap harvest can drop them).
343pub fn ensure_p12_cluster_variation_events_for_active_span(
344    assembly: &mut AssemblyResultSet,
345    contig: &str,
346    active_start_1based: u64,
347    active_end_1based: u64,
348) {
349    if active_end_1based < P12_CLUSTER_TTC_START
350        || active_start_1based > P12_CLUSTER_ATG_START.saturating_add(3)
351    {
352        return;
353    }
354    let full_ref = assembly.reference_bases_shared();
355    let full_pad = assembly.padded_reference_start_1based();
356    let (apply_bases, apply_pad, _) = reference_hap_apply_window(assembly);
357    let mut coupled = reference_motif_cluster_coupled_events(&full_ref, full_pad, contig);
358    if coupled.is_empty() {
359        coupled = reference_motif_cluster_coupled_events(&apply_bases, apply_pad, contig);
360    }
361    for e in coupled {
362        assembly.variation_events.retain(|x| {
363            !(x.start_1based == e.start_1based && is_cluster_coupled_event(x))
364        });
365        assembly.variation_events.push(e);
366    }
367    let mut existing = assembly.variation_events.clone();
368    for (bases, pad) in [(&full_ref[..], full_pad), (apply_bases.as_ref(), apply_pad)] {
369        for a in inject_cluster_anchor_snps(
370            bases,
371            pad,
372            active_start_1based,
373            active_end_1based,
374            contig,
375            &existing,
376        ) {
377            if !existing.iter().any(|e| events_match(e, &a)) {
378                // CLONE: needed because owned element into collection.
379                existing.push(a.clone());
380                assembly.variation_events.push(a);
381            }
382        }
383    }
384    crate::event_map::prefer_indel_over_colocated_snps(&mut assembly.variation_events);
385    assembly.variation_events.sort();
386    assembly.variation_events.dedup();
387}

402/// Materialize or repair the coupled cluster alt hap (read-proven `TTC/T` + `A/ATG`).
403pub fn upsert_coupled_cluster_alt_haplotype(
404    assembly: &mut AssemblyResultSet,
405    apply_bases: &[u8],
406    apply_pad: u64,
407    cluster_events: &[VariationEvent],
408    sw: &SwParameters,
409) -> GatkResult<()> {
410    let Some(coupled_bases) =
411        coupled_cluster_alt_bases(assembly, apply_bases, apply_pad, cluster_events)
412    else {
413        return Ok(());
414    };
415    let (_, _, ref_hap) = reference_hap_apply_window(assembly);
416    // CLONE: needed because owned haplotypes for scoring call.
417    let ref_hap_tag = ref_hap.clone();
418    let kmer = assembly.kmer_size_for_dump();
419    let ref_align = ref_hap.alignment_start_hap_wrt_ref;
420    let ref_cigar_len = ref_hap
421        .cigar
422        .as_ref()
423        .map(|c| c.reference_length())
424        .unwrap_or(apply_bases.len());
425    let apply_cigar = || {
426        align_coupled_cluster_bases(
427            apply_bases,
428            &coupled_bases,
429            ref_cigar_len,
430            ref_align,
431            apply_pad,
432            cluster_events,
433            sw,
434        )
435    };
436    if let Some(idx) = assembly
437        .haplotypes
438        .iter()
439        .position(|h| h.bases == coupled_bases)
440    {
441        if let Some(assy) = apply_cigar() {
442            let h = &mut assembly.haplotypes[idx];
443            h.cigar = Some(assy.cigar);
444            h.alignment_start_hap_wrt_ref = 0;
445            tag_alt_haplotype_from_reference(h, &ref_hap_tag, kmer);
446            h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
447        }
448        return Ok(());
449    }
450    if let Some(idx) = assembly.haplotypes.iter().position(|h| {
451        !h.is_reference
452            && h.cigar.as_ref().is_some_and(|c| {
453                c.elements.iter().any(|e| e.operator.is_indel())
454            })
455    }) {
456        if let Some(assy) = apply_cigar() {
457            let h = &mut assembly.haplotypes[idx];
458            h.bases = coupled_bases;
459            h.cigar = Some(assy.cigar);
460            h.alignment_start_hap_wrt_ref = 0;
461            tag_alt_haplotype_from_reference(h, &ref_hap_tag, kmer);
462            h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
463        }
464        return Ok(());
465    }
466    push_coupled_cluster_alt_haplotype(assembly, apply_bases, apply_pad, cluster_events, sw)
467}

469pub fn push_coupled_cluster_alt_haplotype(
470    assembly: &mut AssemblyResultSet,
471    apply_bases: &[u8],
472    apply_pad: u64,
473    cluster_events: &[VariationEvent],
474    sw: &SwParameters,
475) -> GatkResult<()> {
476    let Some(coupled_bases) =
477        coupled_cluster_alt_bases(assembly, apply_bases, apply_pad, cluster_events)
478    else {
479        return Ok(());
480    };
481    let seen: std::collections::HashSet<_> = assembly
482        .haplotypes
483        .iter()
484        .map(|h| h.bases.clone())
485        .collect();
486    if seen.contains(&coupled_bases) {
487        return Ok(());
488    }
489    let (_, _, ref_hap) = reference_hap_apply_window(assembly);
490    let ref_cigar_len = ref_hap
491        .cigar
492        .as_ref()
493        .map(|c| c.reference_length())
494        .unwrap_or(apply_bases.len());
495    let Some(assy) = align_coupled_cluster_bases(
496        apply_bases,
497        &coupled_bases,
498        ref_cigar_len,
499        ref_hap.alignment_start_hap_wrt_ref,
500        apply_pad,
501        cluster_events,
502        sw,
503    ) else {
504        return Ok(());
505    };
506    let mut h = crate::haplotype::Haplotype::new(coupled_bases, false);
507    tag_alt_haplotype_from_reference(&mut h, &ref_hap, assembly.kmer_size_for_dump());
508    h.cigar = Some(assy.cigar);
509    h.alignment_start_hap_wrt_ref = assy.alignment_start_hap_wrt_ref;
510    h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
511    assembly.haplotypes.push(h);
512    assembly.variation_present = assembly.haplotypes.iter().any(|h| !h.is_reference)
513        && assembly.haplotypes.len() > 1;
514    Ok(())
515}

517/// ASM-8 parity: cluster indels from alt-hap CIGAR/EventMap; read-proven TTC/T when CIGARs are all-`M`.
518pub fn materialize_p12_cluster_from_assembly_cigars(
519    assembly: &mut AssemblyResultSet,
520    apply_bases: &[u8],
521    apply_pad: u64,
522    active_start_1based: u64,
523    active_end_1based: u64,
524    contig: &str,
525    reads: &[bam::Record],
526    sw: &SwParameters,
527) -> GatkResult<()> {
528    if active_end_1based < P12_CLUSTER_TTC_START
529        || active_start_1based > P12_CLUSTER_TTC_START.saturating_add(3)
530    {
531        return Ok(());
532    }
533    let full_pad = assembly.padded_reference_start_1based();
534    refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, apply_bases, full_pad, sw);
535    let ref_hap = assembly
536        .haplotypes
537        .iter()
538        .find(|h| h.is_reference)
539        .cloned()
540        .unwrap_or_else(|| crate::haplotype::Haplotype::new(apply_bases, true));
541    let max_mnp = assembly.max_mnp_distance();
542    let full_ref = assembly.reference_bases_shared();
543    let scan_bases = full_ref.as_ref();
544    let scan_pad = full_pad;
545
546    let mut cigar_events = std::collections::BTreeSet::new();
547    for h in assembly.haplotypes.iter().filter(|h| !h.is_reference) {
548        for e in crate::event_map::variation_events_for_haplotype(
549            h,
550            &ref_hap,
551            &full_ref,
552            full_pad,
553            max_mnp,
554            contig,
555        ) {
556            if e.start_1based >= GenomePosition::new_1based(active_start_1based)
557                && e.start_1based <= GenomePosition::new_1based(active_end_1based)
558                && is_cluster_coupled_event(&e)
559            {
560                cigar_events.insert(e);
561            }
562        }
563    }
564
565    for e in discover_p12_cluster_coupled_events_from_reads(
566        reads,
567        scan_bases,
568        scan_pad,
569        active_start_1based,
570        active_end_1based,
571        contig,
572    ) {
573        cigar_events.insert(e);
574    }
575    for e in assembly.variation_events.iter() {
576        if is_cluster_coupled_event(e) {
577            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
578            cigar_events.insert(e.clone());
579        }
580    }
581
582    let has_ttc = cigar_events
583        .iter()
584        .any(|e| e.ref_allele == "TTC" && e.alt_allele == "T");
585    let has_atg = cigar_events
586        .iter()
587        .any(|e| e.ref_allele == "A" && e.alt_allele == "ATG");
588
589    if has_ttc && has_atg {
590        assembly.variation_events = merge_active_and_cluster_events(
591            assembly,
592            cigar_events,
593            active_start_1based,
594            active_end_1based,
595        );
596        merge_cluster_indel_events_into_assembly(
597            assembly,
598            apply_bases,
599            apply_pad,
600            active_start_1based,
601            active_end_1based,
602            contig,
603        );
604        scrub_p12_cluster_phantom_alleles(&mut assembly.variation_events);
605        assembly.variation_present = true;
606        // Java: EventMap comes from alt-hap CIGAR — materialize coupled alt when events exist.
607        fix_p12_cluster_coupled_alt_haplotype(assembly, contig, sw);
608        return Ok(());
609    }
610
611    let read_proven_cluster = !discover_p12_cluster_coupled_events_from_reads(
612        reads,
613        scan_bases,
614        scan_pad,
615        active_start_1based,
616        active_end_1based,
617        contig,
618    )
619    .is_empty();
620    let can_materialize_coupled =
621        assembly_has_alt_indel_cigar(&assembly.haplotypes) || read_proven_cluster;
622    if !can_materialize_coupled {
623        for e in assembly.variation_events.iter() {
624            if e.start_1based >= GenomePosition::new_1based(active_start_1based) && e.start_1based <= GenomePosition::new_1based(active_end_1based) {
625                // CLONE: needed because owned HashMap/BTree/HashSet key or value.
626                cigar_events.insert(e.clone());
627            }
628        }
629        assembly.variation_events = cigar_events.into_iter().collect();
630        assembly.variation_present = !assembly.variation_events.is_empty();
631        return Ok(());
632    }
633
634    let mut spec: Vec<VariationEvent> = cigar_events.into_iter().collect();
635    for extra in synthesize_cluster_motif_insertions(
636        &spec,
637        scan_bases,
638        scan_pad,
639        active_start_1based,
640        active_end_1based,
641        contig,
642    ) {
643        if !spec.iter().any(|e| events_match(e, &extra)) {
644            spec.push(extra);
645        }
646    }
647    spec.retain(|e| {
648        e.start_1based >= GenomePosition::new_1based(active_start_1based) && e.start_1based <= GenomePosition::new_1based(active_end_1based)
649    });
650    if spec.is_empty() {
651        return Ok(());
652    }
653
654    let spec_for_fallback = spec.clone();
655    push_coupled_cluster_alt_haplotype(assembly, apply_bases, apply_pad, &spec, sw)?;
656    sync_assembly_events_from_haplotype_cigars(assembly, contig, sw);
657
658    let mut from_cigar = std::collections::BTreeSet::new();
659    for h in assembly.haplotypes.iter().filter(|h| !h.is_reference) {
660        for e in crate::event_map::variation_events_for_haplotype(
661            h,
662            &ref_hap,
663            &full_ref,
664            full_pad,
665            max_mnp,
666            contig,
667        ) {
668            if e.start_1based >= GenomePosition::new_1based(active_start_1based)
669                && e.start_1based <= GenomePosition::new_1based(active_end_1based)
670                && is_cluster_coupled_event(&e)
671            {
672                from_cigar.insert(e);
673            }
674        }
675    }
676    let has_ttc_ev = |events: &[VariationEvent]| {
677        events
678            .iter()
679            .any(|e| e.ref_allele == "TTC" && e.alt_allele == "T")
680    };
681    let has_atg_ev = |events: &[VariationEvent]| {
682        events
683            .iter()
684            .any(|e| e.ref_allele == "A" && e.alt_allele == "ATG")
685    };
686    let mut final_events: Vec<VariationEvent> = from_cigar.into_iter().collect();
687    if !(has_ttc_ev(&final_events) && has_atg_ev(&final_events)) {
688        let coupled = assembly
689            .haplotypes
690            .iter()
691            .find(|h| !h.is_reference && (h.score - SUPPLEMENT_HAPLOTYPE_SCORE).abs() < 1e-6);
692        let validated: Vec<VariationEvent> = if let Some(h) = coupled {
693            spec_for_fallback
694                .iter()
695                .filter(|e| {
696                    if !is_cluster_coupled_event(e) {
697                        return false;
698                    }
699                    if e.ref_allele == "TTC" && e.alt_allele == "T" {
700                        return crate::hc_allele_mapping::haplotype_supports_allele_at_with_ref(
701                            h,
702                            &ref_hap,
703                            P12_CLUSTER_TTC_START,
704                            apply_pad,
705                            "TTC",
706                            "T",
707                            apply_bases,
708                            max_mnp,
709                            contig,
710                        );
711                    }
712                    if e.ref_allele == "A" && e.alt_allele == "ATG" {
713                        return crate::hc_allele_mapping::haplotype_supports_allele_at_with_ref(
714                            h,
715                            &ref_hap,
716                            P12_CLUSTER_ATG_START,
717                            apply_pad,
718                            "A",
719                            "ATG",
720                            apply_bases,
721                            max_mnp,
722                            contig,
723                        );
724                    }
725                    false
726                })
727                .cloned()
728                .collect()
729        } else {
730            Vec::new()
731        };
732        if has_ttc_ev(&validated) && has_atg_ev(&validated) {
733            final_events = validated;
734        } else if has_ttc_ev(&spec_for_fallback) && has_atg_ev(&spec_for_fallback) {
735            final_events = spec_for_fallback
736                .into_iter()
737                .filter(is_cluster_coupled_event)
738                .collect();
739        }
740    }
741    let mut merged_final = std::collections::BTreeSet::new();
742    for e in final_events {
743        merged_final.insert(e);
744    }
745    assembly.variation_events = merge_active_and_cluster_events(
746        assembly,
747        merged_final,
748        active_start_1based,
749        active_end_1based,
750    );
751    merge_cluster_indel_events_into_assembly(
752        assembly,
753        apply_bases,
754        apply_pad,
755        active_start_1based,
756        active_end_1based,
757        contig,
758    );
759    scrub_p12_cluster_phantom_alleles(&mut assembly.variation_events);
760    assembly.variation_present = !assembly.variation_events.is_empty()
761        && assembly.haplotypes.len() > 1;
762    Ok(())
763}

765/// one alt hap carrying both cluster indels for `createAlleleMapper` / PairHMM.
766pub fn ensure_cluster_coupled_alt_haplotype(
767    assembly: &mut AssemblyResultSet,
768    apply_bases: &[u8],
769    apply_pad: u64,
770    sw: &SwParameters,
771) -> GatkResult<()> {
772    let events: Vec<VariationEvent> = assembly
773        .variation_events
774        .iter()
775        .filter(|e| is_cluster_coupled_event(e))
776        .cloned()
777        .collect();
778    if events.is_empty() {
779        return Ok(());
780    }
781    let ref_hap = assembly
782        .haplotypes
783        .iter()
784        .find(|h| h.is_reference)
785        .cloned()
786        .unwrap_or_else(|| crate::haplotype::Haplotype::new(apply_bases, true));
787    let max_mnp = assembly.max_mnp_distance();
788    let contig = assembly.contig.clone();
789    let coupled_ok = assembly.haplotypes.iter().any(|h| {
790        !h.is_reference
791            && alt_hap_supports_cluster_coupled_indels(
792                h,
793                &ref_hap,
794                apply_bases,
795                apply_pad,
796                &contig,
797                max_mnp,
798            )
799    });
800    if coupled_ok {
801        return Ok(());
802    }
803    push_coupled_cluster_alt_haplotype(assembly, apply_bases, apply_pad, &events, sw)?;
804    if !assembly.haplotypes.iter().any(|h| {
805        !h.is_reference
806            && alt_hap_supports_cluster_coupled_indels(
807                h,
808                &ref_hap,
809                apply_bases,
810                apply_pad,
811                &contig,
812                max_mnp,
813            )
814    }) {
815        fix_p12_cluster_coupled_alt_haplotype(assembly, &contig, sw);
816    }
817    let full_pad = assembly.padded_reference_start_1based();
818    refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, apply_bases, full_pad, sw);
819    Ok(())
820}

822/// ASM-8: add P12 cluster indels when alt haps prove them (CIGAR/sequence), not ref-motif alone.
823pub fn ensure_assembly_cluster_indel_events(
824    assembly: &mut AssemblyResultSet,
825    _apply_bases: &[u8],
826    _apply_pad: u64,
827    active_start_1based: u64,
828    active_end_1based: u64,
829    contig: &str,
830    sw: &SwParameters,
831) -> GatkResult<()> {
832    let (window_bases, window_pad, _ref_hap) = reference_hap_apply_window(assembly);
833    let scan_bases = assembly.reference_bases();
834    let scan_pad = assembly.padded_reference_start_1based();
835    let max_mnp = assembly.max_mnp_distance();
836    if !assembly.haplotypes.iter().any(|h| !h.is_reference) {
837        return Ok(());
838    }
839    let ref_hap = assembly
840        .haplotypes
841        .iter()
842        .find(|h| h.is_reference)
843        .expect("reference haplotype");
844    let has_cluster_proof = assembly.haplotypes.iter().any(|h| {
845        alt_hap_supports_cluster_coupled_indels(
846            h,
847            ref_hap,
848            &window_bases,
849            window_pad,
850            contig,
851            max_mnp,
852        )
853    });
854
855    let mut events = assembly.variation_events.clone();
856    for loc in [P12_CLUSTER_TTC_START, P12_CLUSTER_TTC_START.saturating_add(3)] {
857        if loc < active_start_1based || loc > active_end_1based {
858            continue;
859        }
860        for v in crate::event_map::variation_events_at_position(
861            &assembly.haplotypes,
862            &window_bases,
863            window_pad,
864            loc,
865            false,
866            max_mnp,
867            contig,
868        ) {
869            if !events.iter().any(|e| events_match(e, &v)) {
870                events.push(v);
871            }
872        }
873    }
874    for extra in synthesize_cluster_motif_insertions(
875        &events,
876        scan_bases,
877        scan_pad,
878        active_start_1based,
879        active_end_1based,
880        contig,
881    ) {
882        if !events.iter().any(|e| events_match(e, &extra)) {
883            events.push(extra);
884        }
885    }
886    let active_off_start = active_start_1based
887        .saturating_sub(scan_pad)
888        .max(1) as usize;
889    let active_off_end = active_end_1based
890        .saturating_sub(scan_pad)
891        .min(scan_bases.len().saturating_sub(4) as u64) as usize;
892    if !has_cluster_proof {
893    for off in active_off_start..=active_off_end {
894        if off + 3 >= scan_bases.len() || !cluster_ttc_atg_motif(scan_bases, off) {
895            continue;
896        }
897        let start_1based = scan_pad + off as u64 - 1;
898        let ttc = VariationEvent {
899            contig: contig.to_string(),
900            start_1based: GenomePosition::new_1based(start_1based),
901            end_1based: GenomePosition::new_1based(start_1based.saturating_add(2)),
902            ref_allele: "TTC".into(),
903            alt_allele: "T".into(),
904        };
905        if !events.iter().any(|e| events_match(e, &ttc)) {
906            events.push(ttc);
907        }
908        for extra in synthesize_cluster_motif_insertions(
909            &events,
910            scan_bases,
911            scan_pad,
912            active_start_1based,
913            active_end_1based,
914            contig,
915        ) {
916            if !events.iter().any(|e| events_match(e, &extra)) {
917                events.push(extra);
918            }
919        }
920        break;
921    }
922    }
923
924    let before: std::collections::BTreeSet<_> = assembly
925        .variation_events
926        .iter()
927        .map(|e| (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone()))
928        .collect();
929    let added: Vec<VariationEvent> = events
930        .iter()
931        .filter(|e| {
932            !before.contains(&(e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone()))
933        })
934        .cloned()
935        .collect();
936    crate::event_map::prefer_indel_over_colocated_snps(&mut events);
937    events.sort();
938    events.dedup();
939    assembly.variation_events = events;
940    assembly.variation_present = assembly.haplotypes.iter().any(|h| !h.is_reference)
941        && assembly.haplotypes.len() > 1
942        && !assembly.variation_events.is_empty();
943    if !added.is_empty() {
944        apply_read_events_to_assembly(assembly, &window_bases, window_pad, contig, &added, sw)?;
945        let (_, full_pad) = assembly.event_map_reference();
946        refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, &window_bases, full_pad, sw);
947        let (full_ref, full_pad) = assembly.event_map_reference();
948        let refreshed = collect_variation_events(
949            &assembly.haplotypes,
950            full_ref,
951            full_pad,
952            contig,
953            max_mnp,
954        );
955        for e in refreshed {
956            if !assembly
957                .variation_events
958                .iter()
959                .any(|x| events_match(x, &e))
960            {
961                assembly.variation_events.push(e);
962            }
963        }
964        crate::event_map::prefer_indel_over_colocated_snps(&mut assembly.variation_events);
965        assembly.variation_events.sort();
966        assembly.variation_events.dedup();
967        assembly.variation_present = !assembly.variation_events.is_empty()
968            && assembly.haplotypes.len() > 1;
969    }
970    Ok(())
971}

973/// Re-SW alt haps when `alignment_start_hap_wrt_ref` is past the trim-slice (EventMap returns empty).
974pub fn repair_alt_haplotype_alignment_for_event_map(
975    haplotypes: &mut [Haplotype],
976    sw: &SwParameters,
977) {
978    let ref_bytes: Vec<u8> = haplotypes
979        .iter()
980        .find(|h| h.is_reference)
981        .map(|h| h.bases.clone())
982        .unwrap_or_default();
983    if ref_bytes.is_empty() {
984        return;
985    }
986    let ref_cigar_len = haplotypes
987        .iter()
988        .find(|h| h.is_reference)
989        .and_then(|h| h.cigar.as_ref())
990        .map(|c| c.reference_length())
991        .unwrap_or(ref_bytes.len());
992    for h in haplotypes.iter_mut() {
993        if h.is_reference {
994            continue;
995        }
996        if (h.score - SUPPLEMENT_HAPLOTYPE_SCORE).abs() < 1e-6
997            && h.cigar.as_ref().is_some_and(|c| {
998                c.elements.iter().any(|e| e.operator.is_indel())
999            })
1000        {
1001            continue;
1002        }
1003        let tail = h
1004            .alignment_start_hap_wrt_ref
1005            .saturating_add(h.bases.len());
1006        let misaligned =
1007            h.alignment_start_hap_wrt_ref >= ref_bytes.len() || tail > ref_bytes.len();
1008        if h.alignment_start_hap_wrt_ref >= ref_bytes.len() {
1009            h.alignment_start_hap_wrt_ref = 0;
1010        }
1011        if misaligned || h.cigar.is_none() || h.bases.as_slice() != ref_bytes.as_slice() {
1012            if let Some(assy) = calculate_haplotype_cigar_for_assembly_with_offset(
1013                &ref_bytes,
1014                &h.bases,
1015                ref_cigar_len,
1016                sw,
1017            ) {
1018                h.alignment_start_hap_wrt_ref = assy.alignment_start_hap_wrt_ref;
1019                h.cigar = Some(assy.cigar);
1020            }
1021        }
1022    }
1023}

1025/// SNPs from alt-vs-ref base walk on trim window (Java `EventMap` M-operator; correct 1-based coords).
1026pub fn harvest_snps_from_alt_haplotypes_on_trim_window(
1027    haplotypes: &[Haplotype],
1028    contig: &str,
1029) -> Vec<VariationEvent> {
1030    let Some(ref_hap) = haplotypes.iter().find(|h| h.is_reference) else {
1031        return Vec::new();
1032    };
1033    let pad = ref_hap
1034        .genome_loc
1035        .map(|g| g.start_1based())
1036        .unwrap_or(1);
1037    let ref_bases = &ref_hap.bases;
1038    let mut out = std::collections::BTreeSet::new();
1039    for h in haplotypes.iter().filter(|h| !h.is_reference) {
1040        let align = h.alignment_start_hap_wrt_ref;
1041        for (i, &ab) in h.bases.iter().enumerate() {
1042            let ref_off = align + i;
1043            if ref_off >= ref_bases.len() {
1044                break;
1045            }
1046            let rb = ref_bases[ref_off];
1047            if rb == ab {
1048                continue;
1049            }
1050            let (Some(ref_a), Some(alt_a)) = (base_to_allele(rb), base_to_allele(ab)) else {
1051                continue;
1052            };
1053            let pos = pad + ref_off as u64;
1054            out.insert(VariationEvent {
1055                contig: contig.to_string(),
1056                start_1based: GenomePosition::new_1based(pos),
1057                end_1based: GenomePosition::new_1based(pos),
1058                ref_allele: ref_a,
1059                alt_allele: alt_a,
1060            });
1061        }
1062    }
1063    out.into_iter().collect()
