1/// True when one ACGT base is ≥70% of the insertion (poly-A/T plugs in homopolymer runs).
2fn insertion_is_low_complexity(inserted: &[u8]) -> bool {
3    if inserted.is_empty() {
4        return true;
5    }
6    let mut counts = [0u32; 4];
7    for &b in inserted {
8        match b.to_ascii_uppercase() {
9            b'A' => counts[0] += 1,
10            b'C' => counts[1] += 1,
11            b'G' => counts[2] += 1,
12            b'T' => counts[3] += 1,
13            _ => {}
14        }
15    }
16    let n = inserted.len() as u32;
17    let max = counts.into_iter().max().unwrap_or(0);
18    max.saturating_mul(10) >= n.saturating_mul(7)
19}

21/// Long insertions present in read sequence between ref anchors but not as CIGAR `I`
22/// (L11 holdout `20:15001894`: Java +36 INS; BAM shows 2D/4D while query carries the motif).
23/// Narrower than [`discover_motif_insertion_events_from_reads`] (which is off on `strict`):
24/// only `ins_len ∈ [10, MAX)`, support ≥2 — cheaper on M4 and fewer short-motif FPs.
25/// STR self-similarity makes naive plug matching explode (same read matches many offsets).
26/// Production filters (phenotype, not locus pins):
27/// 1. inserted ≠ reference continuation (true alt vs ref-matching read-ahead);
28/// 2. `LONG_INS_REMATCH_BP` bases after the plug rematch reference (back on haplotype);
29/// 3. left-normalized: last inserted base ≠ anchor (reject tandem-shiftable forms);
30/// 4. reject low-complexity (homopolymer-dominated) insertions.
31fn discover_long_insertion_events_from_reads(
32    reads: &[bam::Record],
33    ref_bases: &[u8],
34    pad_start_1based: u64,
35    active_start_1based: u64,
36    active_end_1based: u64,
37    contig: &str,
38) -> Vec<(u32, VariationEvent)> {
39    use crate::read_projection::query_index_at_reference_position;
40
41    const MIN_LONG_INS_LEN: usize = IndelSpan::LONG_INSERTION_MIN.get();
42    const MIN_LONG_INS_READ_SUPPORT: u32 = 2;
43    const LONG_INS_REMATCH_BP: usize = 10;
44    let pad_start0 = pad_start_1based.saturating_sub(1) as i64;
45    let active_off_start = active_start_1based
46        .saturating_sub(pad_start_1based) as usize;
47    let active_off_end = active_end_1based
48        .saturating_sub(pad_start_1based)
49        .min(ref_bases.len().saturating_sub(3) as u64) as usize;
50    let mut support: BTreeMap<(u64, String, String), u32> = BTreeMap::new();
51
52    let max_ins = MAX_VARIATION_EVENT_ALLELE_LENGTH.saturating_sub(1);
53    if max_ins < MIN_LONG_INS_LEN {
54        return Vec::new();
55    }
56    for ins_len in MIN_LONG_INS_LEN..=max_ins {
57        for off in active_off_start..=active_off_end {
58            // Need anchor, after, and rematch bases on reference.
59            if off + 2 + LONG_INS_REMATCH_BP > ref_bases.len() {
60                continue;
61            }
62            if off + 1 + ins_len > ref_bases.len() {
63                continue;
64            }
65            let anchor = ref_bases[off];
66            let after = ref_bases[off + 1];
67            if !is_regular_base(anchor) || !is_regular_base(after) {
68                continue;
69            }
70            let pos = pad_start_1based + off as u64;
71            if !in_active_span(pos, active_start_1based, active_end_1based) {
72                continue;
73            }
74            let left0 = pad_start0 + off as i64;
75            let ref_cont = &ref_bases[off + 1..off + 1 + ins_len];
76            let ref_rematch = &ref_bases[off + 2..off + 2 + LONG_INS_REMATCH_BP];
77
78            for rec in reads {
79                if rec.is_unmapped() || rec.tid() < 0 {
80                    continue;
81                }
82                let cigar = CigarString(rec.cigar().iter().copied().collect());
83                let start0 = rec.pos();
84                let seq = rec.seq().as_bytes();
85                let Some(ql) = query_index_at_reference_position(start0, &cigar, left0) else {
86                    continue;
87                };
88                if !seq
89                    .get(ql)
90                    .copied()
91                    .unwrap_or(0).eq_ignore_ascii_case(&anchor)
92                {
93                    continue;
94                }
95                let Some(inserted) = query_subseq(&seq, ql + 1, ins_len) else {
96                    continue;
97                };
98                if !seq
99                    .get(ql + 1 + ins_len)
100                    .copied()
101                    .unwrap_or(0).eq_ignore_ascii_case(&after)
102                {
103                    continue;
104                }
105                if !inserted.iter().all(|&b| is_regular_base(b)) {
106                    continue;
107                }
108                // Reject low-complexity plugs (poly-A/T etc.) — holdout 15040030 false +20
109                // over true CA>C deletion in an A-run.
110                if insertion_is_low_complexity(inserted) {
111                    continue;
112                }
113                // Reject ref-identical read-ahead (common in M-aligned STR).
114                if inserted.eq_ignore_ascii_case(ref_cont) {
115                    continue;
116                }
117                // Require rematch onto reference after the insertion plug.
118                let rematch_start = ql + 1 + ins_len + 1;
119                let Some(q_rematch) = query_subseq(&seq, rematch_start, LONG_INS_REMATCH_BP) else {
120                    continue;
121                };
122                if !q_rematch.eq_ignore_ascii_case(ref_rematch) {
123                    continue;
124                }
125                // Left-normalized: last inserted base ≠ anchor (tandem-shiftable otherwise).
126                if inserted
127                    .last()
128                    .copied()
129                    .unwrap_or(0).eq_ignore_ascii_case(&anchor)
130                {
131                    continue;
132                }
133                let mut alt_allele = vec![anchor];
134                alt_allele.extend_from_slice(inserted);
135                let ref_allele_s = allele_bytes_to_string(vec![anchor]);
136                let alt_allele_s = allele_bytes_to_string(alt_allele);
137                *support
138                    // CLONE: needed because owned HashMap entry key.
139                    .entry((pos, ref_allele_s.clone(), alt_allele_s.clone()))
140                    .or_default() += 1;
141            }
142        }
143    }
144
145    let mut scored = Vec::new();
146    for ((pos, ref_allele, alt_allele), count) in support {
147        if count < MIN_LONG_INS_READ_SUPPORT {
148            continue;
149        }
150        let event = VariationEvent {
151            contig: contig.to_string(),
152            start_1based: GenomePosition::new_1based(pos),
153            end_1based: GenomePosition::new_1based(pos),
154            ref_allele,
155            alt_allele,
156        };
157        if allele_len_ok(&event) && event.ref_allele != event.alt_allele {
158            scored.push((count, event));
159        }
160    }
161    // Prefer longest ALT at the same POS (nested STR lengths collapse to Java-scale allele).
162    scored.sort_by(|a, b| {
163        a.1.start_1based
164            .cmp(&b.1.start_1based)
165            .then_with(|| b.1.alt_allele.len().cmp(&a.1.alt_allele.len()))
166            .then_with(|| b.0.cmp(&a.0))
167    });
168    let mut best_at_pos: BTreeMap<u64, (u32, VariationEvent)> = BTreeMap::new();
169    for (count, event) in scored {
170        let pos = event.start_1based.get();
171        best_at_pos.entry(pos).or_insert((count, event));
172    }
173    // Drop satellite long-INS that start inside a longer upstream insertion's span
174    // (same misaligned STR haplotype; holdout 15001894 +36 vs 15001902/924 plugs).
175    let mut ordered: Vec<(u32, VariationEvent)> = best_at_pos.into_values().collect();
176    ordered.sort_by(|a, b| {
177        b.1.alt_allele
178            .len()
179            .cmp(&a.1.alt_allele.len())
180            .then_with(|| b.0.cmp(&a.0))
181            .then_with(|| a.1.start_1based.cmp(&b.1.start_1based))
182    });
183    let mut kept: Vec<(u32, VariationEvent)> = Vec::new();
184    for (count, event) in ordered {
185        let pos = event.start_1based.get();
186        let span = event.alt_allele.len().saturating_sub(event.ref_allele.len());
187        let satellite = kept.iter().any(|(_, k)| {
188            let kpos = k.start_1based.get();
189            let kspan = k.alt_allele.len().saturating_sub(k.ref_allele.len());
190            kspan >= span
191                && kpos < pos
192                && pos <= kpos.saturating_add(kspan as u64)
193        });
194        if !satellite {
195            kept.push((count, event));
196        }
197    }
198    kept
199}
