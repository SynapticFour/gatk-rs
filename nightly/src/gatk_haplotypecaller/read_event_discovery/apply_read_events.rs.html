1pub fn prune_spillover_supplement_haplotypes(assembly: &mut AssemblyResultSet) {
2    let ref_len = assembly
3        .haplotypes
4        .iter()
5        .find(|h| h.is_reference)
6        .map(|h| h.bases.len())
7        .unwrap_or(0);
8    if ref_len == 0 {
9        return;
10    }
11    // L11: allow long insertions up to EventMap allele cap (was ref+8, which pruned
12    // +36 INS haps → empty-mapper rescue + prefer_dominant dropped nearby true indels).
13    let max_alt_len = ref_len.saturating_add(MAX_VARIATION_EVENT_ALLELE_LENGTH);
14    let full_len = assembly.reference_bases().len();
15    assembly.haplotypes.retain(|h| {
16        if h.is_reference || h.bases.len() <= max_alt_len {
17            return true;
18        }
19        // Spine SNPs outside the trimmed apply window are built on the full padded ref;
20        // do not treat those as spillover (otherwise CIGAR sync never sees the SNP).
21        h.bases.len() == full_len && (h.score - SUPPLEMENT_HAPLOTYPE_SCORE).abs() < 1e-6
22    });
23}

25fn apply_read_events_to_assembly(
26    assembly: &mut AssemblyResultSet,
27    _ref_bases: &[u8],
28    _pad_start: u64,
29    contig: &str,
30    read_events: &[VariationEvent],
31    sw: &SwParameters,
32) -> GatkResult<()> {
33    if read_events.is_empty() {
34        return Ok(());
35    }
36    let (apply_bases, apply_pad, ref_hap) = reference_hap_apply_window(assembly);
37    let ref_cigar_len = ref_hap
38        .cigar
39        .as_ref()
40        .map(|c| c.reference_length())
41        .unwrap_or(apply_bases.len());
42
43    let (full_ref, full_pad) = assembly.event_map_reference();
44    let full_ref_vec = assembly.reference_bases_shared();
45    let full_pad_genomic = assembly.padded_reference_start_1based();
46    let asm_events = collect_variation_events(
47        &assembly.haplotypes,
48        full_ref,
49        full_pad,
50        contig,
51        assembly.max_mnp_distance(),
52    );
53
54    let mut seen: std::collections::HashSet<Vec<u8>> = assembly
55        .haplotypes
56        .iter()
57        .map(|h| h.bases.clone())
58        .collect();
59
60    let kmer = assembly.kmer_size_for_dump();
61    let mut push_alt =
62        |ref_for_cigar: &[u8],
63         pad_for_cigar: u64,
64         event: Option<&VariationEvent>,
65         alt_bases: Vec<u8>|
66         -> GatkResult<()> {
67            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
68            if !seen.insert(alt_bases.clone()) {
69                return Ok(());
70            }
71            let cigar_len = if ref_for_cigar.len() == apply_bases.len() {
72                ref_cigar_len
73            } else {
74                ref_for_cigar.len()
75            };
76            // L11: long insertions in STR must use the deterministic single-event CIGAR.
77            // SW often emits a frankenstein CIGAR → EventMap floods (~189 events) and
78            // prefer_dominant_spanning_indels drops nearby true indels (15002023 C>CAT).
79            let force_single = event.is_some_and(|ev| {
80                ev.alt_allele.len() > ev.ref_allele.len()
81                    && ev.indel_span().is_long_insertion_span()
82            });
83            let mut cigar = if force_single {
84                event.and_then(|ev| {
85                    cigar_for_single_indel_event(ref_for_cigar, pad_for_cigar, ev)
86                })
87            } else {
88                None
89            };
90            if cigar.is_none() {
91                cigar = calculate_haplotype_cigar_with_strategy(
92                    ref_for_cigar,
93                    &alt_bases,
94                    sw,
95                    SwOverhangStrategy::Indel,
96                )
97                .or_else(|| {
98                    calculate_haplotype_cigar_for_assembly(
99                        ref_for_cigar,
100                        &alt_bases,
101                        cigar_len,
102                        sw,
103                    )
104                });
105            }
106            let cigar_has_indel = cigar
107                .as_ref()
108                .is_some_and(|c| c.elements.iter().any(|e| e.operator.is_indel()));
109            if !cigar_has_indel {
110                if let Some(ev) = event {
111                    cigar = cigar_for_single_indel_event(ref_for_cigar, pad_for_cigar, ev);
112                }
113            }
114            let Some(cigar) = cigar else {
115                return Ok(());
116            };
117            let mut h = Haplotype::new(alt_bases, false);
118            tag_alt_haplotype_from_reference(&mut h, &ref_hap, kmer);
119            h.cigar = Some(cigar);
120            h.score = SUPPLEMENT_HAPLOTYPE_SCORE;
121            assembly.haplotypes.push(h);
122            Ok(())
123        };
124
125    // L11: never chain a long insertion with other spine events — SW on the
126    // frankenstein haplotype explodes EventMap and drops nearby true indels.
127    let has_long_ins = read_events.iter().any(|e| {
128        e.alt_allele.len() > e.ref_allele.len() && e.indel_span().is_long_insertion_span()
129    });
130    let chained = if has_long_ins {
131        None
132    } else {
133        apply_events_to_ref_chained(&apply_bases, read_events, apply_pad)
134    };
135    if let Some(cluster_hap) = chained {
136        push_alt(&apply_bases, apply_pad, None, cluster_hap)?;
137    } else {
138        for event in read_events {
139            if let Some(alt_bases) = apply_event_to_ref(&apply_bases, event, apply_pad) {
140                push_alt(&apply_bases, apply_pad, Some(event), alt_bases)?;
141            } else if let Some(alt_bases) =
142                apply_event_to_ref(&full_ref_vec, event, full_pad_genomic)
143            {
144                push_alt(&full_ref_vec, full_pad_genomic, Some(event), alt_bases)?;
145            }
146        }
147    }
148
149    let asm_indels_for_restore: Vec<VariationEvent> = asm_events
150        .iter()
151        .filter(|e| e.is_indel() && !homopolymer_motif_phantom(e))
152        .cloned()
153        .collect();
154    let mut events: Vec<VariationEvent> = asm_events
155        .into_iter()
156        .filter(|e| !homopolymer_motif_phantom(e))
157        .collect();
158    for event in read_events {
159        if !events.iter().any(|e| events_match(e, event)) {
160            // CLONE: needed because owned element into collection.
161            events.push(event.clone());
162        }
163    }
164    crate::event_map::prefer_indel_over_colocated_snps(&mut events);
165    crate::event_map::prefer_dominant_spanning_indels(&mut events);
166    // L11: prefer_dominant can drop distant true indels when the EventMap flood around a
167    // newly injected long-INS includes long DELs. Restore asm indels far from new long-INS.
168    const RESTORE_FAR_FROM_LONG_INS_BP: u64 = IndelSpan::FRAGMENT_WINDOW_BP;
169    let new_long_ins: Vec<(u64, IndelSpan)> = read_events
170        .iter()
171        .filter(|e| e.alt_allele.len() > e.ref_allele.len() && e.indel_span().is_long_insertion_span())
172        .map(|e| (e.start_1based.get(), e.indel_span()))
173        .collect();
174    if !new_long_ins.is_empty() {
175        for e in &asm_indels_for_restore {
176            if events.iter().any(|x| events_match(x, e)) {
177                continue;
178            }
179            let p = e.start_1based.get();
180            let near = new_long_ins.iter().any(|(lp, lspan)| {
181                p.abs_diff(*lp) <= RESTORE_FAR_FROM_LONG_INS_BP.max(lspan.get() as u64)
182            });
183            if !near {
184                // CLONE: needed because owned element into collection.
185                events.push(e.clone());
186            }
187        }
188    }
189    events.sort();
190    events.dedup();
191    assembly.variation_events = events;
192    assembly.variation_present = assembly.haplotypes.iter().any(|h| !h.is_reference)
193        && assembly.haplotypes.len() > 1;
194    prune_spillover_supplement_haplotypes(assembly);
195    Ok(())
196}
