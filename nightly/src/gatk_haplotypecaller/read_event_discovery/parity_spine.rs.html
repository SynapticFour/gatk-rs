1/// Drop phantom SNPs at P12 cluster indel loci when coupled `TTC/T` + `A/ATG` are present.
2pub fn scrub_p12_cluster_phantom_alleles(events: &mut Vec<VariationEvent>) {
3    let has_atg = events.iter().any(|e| {
4        e.start_1based == GenomePosition::new_1based(P12_CLUSTER_ATG_START) && e.ref_allele == "A" && e.alt_allele == "ATG"
5    });
6    let has_ttc = events.iter().any(|e| {
7        e.start_1based == GenomePosition::new_1based(P12_CLUSTER_TTC_START) && e.ref_allele == "TTC" && e.alt_allele == "T"
8    });
9    events.retain(|e| {
10        if has_atg
11            && e.start_1based == GenomePosition::new_1based(P12_CLUSTER_ATG_START)
12            && !(e.ref_allele == "A" && e.alt_allele == "ATG")
13        {
14            return false;
15        }
16        if has_ttc
17            && e.start_1based == GenomePosition::new_1based(P12_CLUSTER_TTC_START)
18            && !(e.ref_allele == "TTC" && e.alt_allele == "T")
19        {
20            return false;
21        }
22        true
23    });
24    crate::event_map::prefer_indel_over_colocated_snps(events);
25}

27fn merge_active_and_cluster_events(
28    assembly: &AssemblyResultSet,
29    cluster_events: std::collections::BTreeSet<VariationEvent>,
30    active_start_1based: u64,
31    active_end_1based: u64,
32) -> Vec<VariationEvent> {
33    let has_atg = cluster_events
34        .iter()
35        .any(|e| e.ref_allele == "A" && e.alt_allele == "ATG");
36    let has_ttc = cluster_events
37        .iter()
38        .any(|e| e.ref_allele == "TTC" && e.alt_allele == "T");
39    let mut merged = cluster_events;
40    for e in assembly.variation_events.iter() {
41        if e.start_1based < GenomePosition::new_1based(active_start_1based) || e.start_1based > GenomePosition::new_1based(active_end_1based) {
42            continue;
43        }
44        if has_atg
45            && e.start_1based == GenomePosition::new_1based(P12_CLUSTER_ATG_START)
46            && !(e.ref_allele == "A" && e.alt_allele == "ATG")
47        {
48            continue;
49        }
50        if has_ttc
51            && e.start_1based == GenomePosition::new_1based(P12_CLUSTER_TTC_START)
52            && !(e.ref_allele == "TTC" && e.alt_allele == "T")
53        {
54            continue;
55        }
56        // CLONE: needed because owned HashMap/BTree/HashSet key or value.
57        merged.insert(e.clone());
58    }
59    let mut out: Vec<VariationEvent> = merged.into_iter().collect();
60    scrub_p12_cluster_phantom_alleles(&mut out);
61    out
62}

64/// Parity spine: read-proven indels missing from assembly EventMap (genome-wide + harness).
65///
66/// `materialize_alt_haps`: when true, SW-rematerialize alt haplotypes so EventMap sync
67/// retains the alleles. List-only mode (`false`) is only safe when a later path merges
68/// prior events without a strict CIGAR-only rebuild — post-HMM StrictJava sync does not.
69pub fn parity_spine_read_proven_indels(
70    assembly: &mut AssemblyResultSet,
71    reads: &[SharedBamRecord],
72    active_start_1based: u64,
73    active_end_1based: u64,
74    sw: &SwParameters,
75    materialize_alt_haps: bool,
76) -> GatkResult<()> {
77    if reads.is_empty() || ref_bases_empty(assembly) {
78        return Ok(());
79    }
80    const MIN_GENERIC_INDEL_READS: u32 = 2;
81    // Dense GIAB windows can carry many true indels per active region.
82    const MAX_SPINE_INDELS: usize = 64;
83    let (apply_bases, apply_pad, _) = reference_hap_apply_window(assembly);
84    // Scan on full padded ref — trim/event-map windows can exclude indel anchors that
85    // still lie inside the active genotyping span.
86    let scan_bases = assembly.reference_bases();
87    let scan_pad = assembly.padded_reference_start_1based();
88    let contig = assembly.contig.clone();
89    let max_mnp = assembly.max_mnp_distance();
90    let graph = collect_variation_events(
91        &assembly.haplotypes,
92        scan_bases,
93        scan_pad,
94        &contig,
95        max_mnp,
96    );
97    // Dense RT-first hit path: EventMap indels already on haplotype CIGARs — skip the
98    // multi-pass read rediscovery (TTCT / indel / SNP-collapse / long-INS). L9 listed
99    // materialize is empty when every listed indel is graph-encoded. SNP spine still
100    // runs separately for Java SNP FNs.
101    let has_alt = assembly.haplotypes.iter().any(|h| !h.is_reference);
102    let indel_cigar_complete = has_alt
103        && assembly
104            .variation_events
105            .iter()
106            .filter(|e| e.is_indel())
107            .all(|e| graph.iter().any(|g| events_match(g, e)));
108    if indel_cigar_complete {
109        crate::runtime_config::rss_trace_checkpoint(
110            "parity_spine_indel_skip_cigar_complete",
111            &format!(
112                "events={} graph={}",
113                assembly.variation_events.len(),
114                graph.len()
115            ),
116        );
117        return Ok(());
118    }
119    let existing: std::collections::BTreeSet<(u64, String, String)> = assembly
120        .variation_events
121        .iter()
122        .chain(graph.iter())
123        .map(|e| (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone()))
124        .collect();
125    let mut candidates: Vec<(u32, VariationEvent)> = Vec::new();
126    for (support, e) in discover_ttct_deletions_from_reads(
127        reads,
128        scan_bases,
129        scan_pad,
130        active_start_1based,
131        active_end_1based,
132        &contig,
133    ) {
134        // CLONE: needed because owned composite key for dedup/lookup.
135        let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
136        if !existing.contains(&key) && !candidates.iter().any(|(_, c)| events_match(c, &e)) {
137            candidates.push((support.max(MIN_GENERIC_INDEL_READS), e));
138        }
139    }
140    let mut read_indels = discover_indel_events_from_reads(
141        reads,
142        scan_bases,
143        scan_pad,
144        active_start_1based,
145        active_end_1based,
146        &contig,
147    );
148    // L10: admit support==1 indels that properly extend a same-start indel with
149    // support≥2 (nested STR / longest-allele; holdout 20:15031984 18D beside 8D).
150    let strong_starts: BTreeSet<u64> = read_indels
151        .iter()
152        .filter(|(s, e)| *s >= MIN_GENERIC_INDEL_READS && e.is_indel())
153        .map(|(_, e)| e.start_1based.get())
154        .collect();
155    for (support, e) in read_indels.drain(..) {
156        if !e.is_indel() {
157            continue;
158        }
159        let admit = support >= MIN_GENERIC_INDEL_READS
160            || (support >= 1
161                && strong_starts.contains(&e.start_1based.get())
162                && e.ref_allele.len() >= 5);
163        if !admit {
164            continue;
165        }
166        // CLONE: needed because owned composite key for dedup/lookup.
167        let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
168        if !existing.contains(&key) && !candidates.iter().any(|(_, c)| events_match(c, &e)) {
169            candidates.push((support.max(1), e));
170        }
171    }
172    let snps = discover_snp_events_from_reads(
173        reads,
174        scan_bases,
175        scan_pad,
176        active_start_1based,
177        active_end_1based,
178        &contig,
179        false,
180        ReadEventDiscoveryOptions::strict(),
181    );
182    let mut snp_buf = snps;
183    for (_, e) in collapse_snps_to_deletions(
184        &mut snp_buf,
185        scan_bases,
186        scan_pad,
187        active_start_1based,
188        active_end_1based,
189        &contig,
190    ) {
191        if e.is_indel() {
192            // CLONE: needed because owned composite key for dedup/lookup.
193            let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
194            if !existing.contains(&key) && !candidates.iter().any(|(_, c)| events_match(c, &e)) {
195                candidates.push((MIN_GENERIC_INDEL_READS, e));
196            }
197        }
198    }
199    // L9: events already on the genotyping list but missing from hap CIGARs still need
200    // alt-hap materialization (spine previously skipped them via `existing`).
201    // Same span floor as ensure_alt_haplotypes — avoid short fragment inflation.
202    const MIN_LISTED_MATERIALIZE_SPAN: usize = 5;
203    for e in &assembly.variation_events {
204        if !e.is_indel() {
205            continue;
206        }
207        if e.ref_allele.len().abs_diff(e.alt_allele.len()) < MIN_LISTED_MATERIALIZE_SPAN {
208            continue;
209        }
210        if graph.iter().any(|g| events_match(g, e)) {
211            continue;
212        }
213        if candidates.iter().any(|(_, c)| events_match(c, e)) {
214            continue;
215        }
216        let (rr, ra) = read_allele_depths_at_locus(reads, e, apply_pad);
217        let (rr2, ra2) = read_allele_depths_at_locus(reads, e, scan_pad);
218        let (rref, ralt) = if ra2 >= ra { (rr2, ra2) } else { (rr, ra) };
219        if genome_wide_genotype_read_support(e, rref, ralt) {
220            // CLONE: needed — candidate list owns VariationEvent for later materialization.
221            candidates.push((ralt.max(0) as u32, e.clone()));
222        }
223    }
224    // L11: long insertions in query sequence without CIGAR I (holdout 20:15001894).
225    let mut long_ins_pos: Vec<(u64, IndelSpan)> = Vec::new();
226    for (support, e) in discover_long_insertion_events_from_reads(
227        reads,
228        scan_bases,
229        scan_pad,
230        active_start_1based,
231        active_end_1based,
232        &contig,
233    ) {
234        let span = e.indel_span();
235        // Track every discovered long-INS locus (even if already on EventMap) so fragment
236        // spray drop still fires.
237        long_ins_pos.push((e.start_1based.get(), span));
238        // CLONE: needed because owned composite key for dedup/lookup.
239        let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
240        if !existing.contains(&key) && !candidates.iter().any(|(_, c)| events_match(c, &e)) {
241            candidates.push((support, e));
242        }
243    }
244    // Also treat long insertions already on candidates / assembly EventMap.
245    for (_, e) in &candidates {
246        if e.alt_allele.len() > e.ref_allele.len() && e.indel_span().is_long_insertion_span() {
247            let pos = e.start_1based.get();
248            let span = e.indel_span();
249            if !long_ins_pos.iter().any(|(p, _)| *p == pos) {
250                long_ins_pos.push((pos, span));
251            }
252        }
253    }
254    for e in &assembly.variation_events {
255        if e.is_indel()
256            && e.alt_allele.len() > e.ref_allele.len()
257            && e.indel_span().is_long_insertion_span()
258        {
259            let pos = e.start_1based.get();
260            let span = e.indel_span();
261            if !long_ins_pos.iter().any(|(p, _)| *p == pos) {
262                long_ins_pos.push((pos, span));
263            }
264        }
265    }
266    // A3: when a long insertion is present, drop short fragment sprays in-window so they
267    // are not chained onto the long-allele haplotype (frankenstein EventMap wiped nearby
268    // true indels e.g. holdout 15002023 C>CAT).
269    if !long_ins_pos.is_empty() {
270        candidates.retain(|(_, e)| {
271            let span = e.indel_span();
272            if !span.is_short_fragment() {
273                return true;
274            }
275            let p = e.start_1based.get();
276            !long_ins_pos
277                .iter()
278                .any(|(lp, lspan)| IndelSpan::nests_beside_long(p, *lp, *lspan))
279        });
280    }
281        candidates.sort_by(|a, b| b.0.cmp(&a.0).then_with(|| a.1.start_1based.cmp(&b.1.start_1based)));
282    candidates.truncate(MAX_SPINE_INDELS);
283    if candidates.is_empty() {
284        return Ok(());
285    }
286    if crate::parity_harness::env_flag_set("GATK_RS_INDEL_HAP_TRACE") {
287        for (sup, e) in candidates
288            .iter()
289            .filter(|(_, e)| e.ref_allele.len().abs_diff(e.alt_allele.len()) >= 10)
290            .take(8)
291        {
292            eprintln!(
293                "L10-spine-indel cand support={} {}:{} {}>{} span={}",
294                sup,
295                e.contig,
296                e.start_1based.get(),
297                e.ref_allele,
298                e.alt_allele,
299                e.ref_allele.len().abs_diff(e.alt_allele.len())
300            );
301        }
302    }
303    let events: Vec<VariationEvent> = candidates.into_iter().map(|(_, e)| e).collect();
304    if materialize_alt_haps {
305        apply_read_events_to_assembly(
306            assembly,
307            &apply_bases,
308            apply_pad,
309            &contig,
310            &events,
311            sw,
312        )?;
313        sync_assembly_events_from_haplotype_cigars(assembly, &contig, sw);
314    } else {
315        for e in events {
316            if !assembly.variation_events.iter().any(|x| events_match(x, &e)) {
317                assembly.variation_events.push(e);
318            }
319        }
320        sort_dedup_variation_events(assembly);
321    }
322    scrub_p12_cluster_phantom_alleles(&mut assembly.variation_events);
323    Ok(())
324}

326/// Parity spine: biallelic SNPs from reads when EventMap has no event at that locus (Java `getVariationEvents` gap).
327/// See [`parity_spine_read_proven_indels`] for `materialize_alt_haps`.
328pub fn parity_spine_read_proven_snps(
329    assembly: &mut AssemblyResultSet,
330    reads: &[SharedBamRecord],
331    active_start_1based: u64,
332    active_end_1based: u64,
333    sw: &SwParameters,
334    materialize_alt_haps: bool,
335) -> GatkResult<()> {
336    if reads.is_empty() || ref_bases_empty(assembly) {
337        return Ok(());
338    }
339    // Dense GIAB active regions can exceed the old spine-era cap of 20; 64 keeps
340    // throughput bounded (128 made dense HC ~6× slower without clearing remaining FNs).
341    const MAX_SPINE_SNPS: usize = 64;
342    let (apply_bases, apply_pad, _) = reference_hap_apply_window(assembly);
343    let (scan_bases, scan_pad) = assembly.event_map_reference();
344    let contig = assembly.contig.clone();
345    let existing: std::collections::BTreeSet<(u64, String, String)> = assembly
346        .variation_events
347        .iter()
348        .map(|e| (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone()))
349        .collect();
350    let snps = discover_snp_events_from_reads(
351        reads,
352        scan_bases,
353        scan_pad,
354        active_start_1based,
355        active_end_1based,
356        &contig,
357        false,
358        ReadEventDiscoveryOptions::strict(),
359    );
360    let mut candidates: Vec<VariationEvent> = Vec::new();
361    for (_, e) in snps {
362        if candidates.len() >= MAX_SPINE_SNPS {
363            break;
364        }
365        if e.ref_allele.len() != 1 || e.alt_allele.len() != 1 {
366            continue;
367        }
368        if e.start_1based >= GenomePosition::new_1based(P12_CLUSTER_CORE_START) && e.start_1based <= GenomePosition::new_1based(P12_CLUSTER_CORE_END) {
369            continue;
370        }
371        // L8 pin retirement: do not skip merely because *some* graph event shares the locus
372        // require the exact REF/ALT already present (wrong colocated alleles blocked FN SNPs).
373        // CLONE: needed because owned composite key for dedup/lookup.
374        let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
375        if existing.contains(&key) || candidates.iter().any(|c| events_match(c, &e)) {
376            continue;
377        }
378        candidates.push(e);
379    }
380    if candidates.is_empty() {
381        return Ok(());
382    }
383    if materialize_alt_haps {
384        apply_read_events_to_assembly(
385            assembly,
386            &apply_bases,
387            apply_pad,
388            &contig,
389            &candidates,
390            sw,
391        )?;
392        sync_assembly_events_from_haplotype_cigars(assembly, &contig, sw);
393    } else {
394        for e in candidates {
395            if !assembly.variation_events.iter().any(|x| events_match(x, &e)) {
396                assembly.variation_events.push(e);
397            }
398        }
399        sort_dedup_variation_events(assembly);
400    }
401    Ok(())
402}
