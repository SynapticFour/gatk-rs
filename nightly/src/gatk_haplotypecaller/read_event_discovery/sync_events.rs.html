4/// Options for [`sync_assembly_events_from_haplotype_cigars_with_harvest`] (Sprint I-3).
5/// # Invariants
6/// [`Self::strict_java`] sets `strict_event_map_only == true` and disables trim SNP harvest.
7/// Harvest and strict modes are mutually exclusive for production `CallRegionArgs::strict_java`.
8/// # Ownership
9/// [`Copy`] config passed into sync helpers; assembly result sets are mutably borrowed.
10/// # Mutation
11/// Immutable options tag; sync functions mutate `AssemblyResultSet` variation events.
12/// # Biological assumptions
13/// Post-trim SNP harvest is a parity supplement, not GATK production EventMap behavior.
14/// # Java equivalence
15/// Mirrors GATK `AssemblyResultSet.regenerateVariationEvents` / `EventMap.buildEventMapsForHaplotypes` when strict.
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17pub struct SyncAssemblyOptions {
18    /// Post-`trim_to` SNP harvest from alt haplotypes (off under production `strict_java`).
19    pub harvest_trim_snps: bool,
20    /// When true, events come only from EventMap / CIGAR regen (production `strict_java`).
21    pub strict_event_map_only: bool,
22}
23
24impl SyncAssemblyOptions {
25    /// Production `CallRegionArgs::strict_java` — EventMap only, no trim SNP harvest.
26    pub const fn strict_java() -> Self {
27        Self {
28            harvest_trim_snps: false,
29            strict_event_map_only: true,
30        }
31    }
32
33    /// Non-strict / parity: optional harvest + side-list merge behavior.
34    pub const fn with_harvest(harvest_trim_snps: bool) -> Self {
35        Self {
36            harvest_trim_snps,
37            strict_event_map_only: false,
38        }
39    }
40
41    pub const fn from_strict_java(is_strict_java: bool) -> Self {
42        if is_strict_java {
43            Self::strict_java()
44        } else {
45            Self::with_harvest(true)
46        }
47    }
48}

50/// ASM-8 + Java `EventMap`: refresh alt CIGARs on trim slice; events from full padded ref (GATK parity).
51pub fn sync_assembly_events_from_haplotype_cigars(

53    contig: &str,
54    sw: &SwParameters,
55) {
56    sync_assembly_events_from_haplotype_cigars_with_harvest(
57        assembly,
58        contig,
59        sw,
60        SyncAssemblyOptions::with_harvest(false),
61    );
62}

64/// Like [`sync_assembly_events_from_haplotype_cigars`] with optional trim-window SNP harvest (post-`trim_to` only).
65/// When `options.strict_event_map_only` is true (production `strict_java`), events come only from
66/// [`collect_variation_events`] after [`refresh_alt_haplotype_indel_cigars`] — matching GATK
67/// `AssemblyResultSet.regenerateVariationEvents` / `EventMap.buildEventMapsForHaplotypes`.
68pub fn sync_assembly_events_from_haplotype_cigars_with_harvest(
69    assembly: &mut AssemblyResultSet,
70    contig: &str,
71    sw: &SwParameters,
72    options: SyncAssemblyOptions,
73) {
74    let harvest_trim_snps = options.harvest_trim_snps;
75    let strict_event_map_only = options.strict_event_map_only;
76    let trimmed_ref = assembly.apply_bases_shared();
77    let (_, full_pad) = assembly.event_map_reference();
78    if harvest_trim_snps && !strict_event_map_only {
79        repair_alt_haplotype_alignment_for_event_map(&mut assembly.haplotypes, sw);
80    }
81    refresh_alt_haplotype_indel_cigars(&mut assembly.haplotypes, trimmed_ref.as_ref(), full_pad, sw);
82    let prior_events = std::mem::take(&mut assembly.variation_events);
83    let (full_ref, full_pad) = assembly.event_map_reference();
84    let mut events = collect_variation_events(
85        &assembly.haplotypes,
86        full_ref,
87        full_pad,
88        contig,
89        assembly.max_mnp_distance(),
90    );
91    // R4-2: outside contig 2, keep prior read-proven indels across CIGAR regen (assembly often
92    // fails to encode genome-wide indels on alt haplotypes; list + pileup AD carry them).
93    // Same retain for biallelic SNPs: trim-window materialize + spillover prune can leave
94    // strong hets list-only (call-rate dig 21:9411785), and CIGAR-only regen would drop them.
95    let genome_wide_contig = contig != "2" && contig != "chr2";
96    let mut event_keys: std::collections::HashSet<(u64, String, String)> = events
97        .iter()
98        .map(|e| (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone()))
99        .collect();
100    let merge_preserved = |events: &mut Vec<VariationEvent>,
101                           keys: &mut std::collections::HashSet<(u64, String, String)>,
102                           source: &[VariationEvent]| {
103        for e in source {
104            let keep = is_cluster_coupled_event(e)
105                || is_cluster_ctc_del(e)
106                || is_cluster_anchor_snp(e)
107                // ASM-8: read-proven gap SNPs survive strict CIGAR regen (post-backfill).
108                || (strict_java_asm8_only_enabled() && is_p12_phase_e_gap_event(e))
109                || (!strict_java_asm8_only_enabled()
110                    && (is_p12_phase_e_gap_event(e) || is_java_diff_oracle_allele(e)))
111                || (genome_wide_contig && e.is_indel())
112                || (genome_wide_contig
113                    && e.ref_allele.len() == 1
114                    && e.alt_allele.len() == 1);
115            if !keep {
116                continue;
117            }
118            let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
119            if keys.insert(key) {
120                // CLONE: needed because owned element into collection.
121                events.push(e.clone());
122            }
123        }
124    };
125    merge_preserved(&mut events, &mut event_keys, &prior_events);
126    if !strict_event_map_only
127        && harvest_trim_snps {
128            for e in harvest_snps_from_alt_haplotypes_on_trim_window(&assembly.haplotypes, contig) {
129                let key = (e.start_1based.get(), e.ref_allele.clone(), e.alt_allele.clone());
130                if event_keys.insert(key) {
131                    events.push(e);
132                }
133            }
134        }
135    crate::event_map::prefer_indel_over_colocated_snps(&mut events);
136    crate::event_map::prefer_dominant_spanning_indels(&mut events);
137    events.sort_by_key(|e| e.start_1based);
138    events.dedup_by(|a, b| {
139        a.start_1based == b.start_1based
140            && a.ref_allele == b.ref_allele
141            && a.alt_allele == b.alt_allele
142    });
143    if !strict_event_map_only {
144        scrub_p12_cluster_phantom_alleles(&mut events);
145    }
146    assembly.variation_events = events;
147    assembly.variation_present = assembly.haplotypes.iter().any(|h| !h.is_reference)
148        && assembly.haplotypes.len() > 1
149        && !assembly.variation_events.is_empty();
150}
