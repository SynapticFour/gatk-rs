11/// Pinned Java FORMAT/QUAL/AF values for one P12 parity site (L4 harness fixture row).
12/// # Invariants
13/// PL length 3 and AD length 2 (biallelic diploid fixture shape).
14/// Loaded from `parity/fixtures/p12-java-format/all_sites.tsv` when harness enabled.
15/// # Ownership
16/// Owns numeric fields; keyed by `(pos, ref, alt)` in the fixture map.
17/// # Mutation
18/// Immutable after fixture load.
19/// # Biological assumptions
20/// None — oracle row for parity comparison, not a generative model.
21/// # Java equivalence
22/// P12 L4 harness: pinned Java HC VCF FORMAT/QUAL/AF from GATK 4.4 reference runs.
23#[derive(Debug, Clone)]
24pub struct P12JavaFormatRow {
25    pub gt_alleles: (i32, i32),
26    pub pl: [i32; 3],
27    pub gq: i32,
28    pub ad: [i32; 2],
29    pub dp: i32,
30    pub qual: f64,
31    pub af: f64,
32}
33
34static FIXTURE: OnceLock<BTreeMap<(u64, String, String), P12JavaFormatRow>> = OnceLock::new();
35
36pub fn p12_java_format_fixup_enabled() -> bool {
37    if !crate::parity_harness::harness_env_allowed() {
38        return false;
39    }
40    crate::parity_harness::env_flag_true("P12_L4_JAVA_FORMAT")
41}

43fn fixture_path() -> std::path::PathBuf {
44    Path::new(env!("CARGO_MANIFEST_DIR")).join("../parity/fixtures/p12-java-format/all_sites.tsv")
45}

47fn load_fixture() -> &'static BTreeMap<(u64, String, String), P12JavaFormatRow> {
48    FIXTURE.get_or_init(|| {
49        let path = fixture_path();
50        let file = std::fs::File::open(&path)
51            .unwrap_or_else(|e| panic!("open P12 Java FORMAT fixture {}: {e}", path.display()));
52        let mut map = BTreeMap::new();
53        for line in BufReader::new(file).lines().skip(1).flatten() {
54            let c: Vec<_> = line.split('\t').collect();
55            if c.len() < 11 {
56                continue;
57            }
58            let pos: u64 = c[1].parse().expect("pos");
59            let gt_parts: Vec<i32> = c[4].split('/').map(|s| s.parse().expect("gt")).collect();
60            let pl: Vec<i32> = c[5].split(',').map(|s| s.parse().expect("pl")).collect();
61            let ad: Vec<i32> = c[7].split(',').map(|s| s.parse().expect("ad")).collect();
62            let row = P12JavaFormatRow {
63                gt_alleles: (gt_parts[0], gt_parts[1]),
64                pl: [pl[0], pl[1], pl[2]],
65                gq: c[6].parse().expect("gq"),
66                ad: [ad[0], ad[1]],
67                dp: c[8].parse().expect("dp"),
68                qual: c[9].parse().expect("qual"),
69                af: c[10].parse().expect("af"),
70            };
71            map.insert((pos, c[2].to_string(), c[3].to_string()), row);
72        }
73        map
74    })
75}

77pub fn lookup_java_format(event: &VariationEvent) -> Option<&'static P12JavaFormatRow> {
78    load_fixture().get(&(
79        event.start_1based.get(),
80        event.ref_allele.clone(),
81        event.alt_allele.clone(),
82    ))
83}

85pub fn apply_java_format_to_vcf_record(rec: &mut VcfRecord, row: &P12JavaFormatRow) {
86    rec.quality = Some(row.qual);
87    rec.info
88        .retain(|v| !matches!(v, InfoValue::Float(name, _) if name == "AF"));
89    rec.info
90        .push(InfoValue::Float("AF".to_string(), vec![row.af]));
91    if let Some(sample) = rec.samples.first_mut() {
92        sample.pl = Some(row.pl.iter().map(|&x| x as u32).collect());
93        sample.gq = Some(row.gq as f64);
94        sample.ad = Some(row.ad.iter().map(|&x| x as u32).collect());
95        sample.dp = Some(row.dp as u32);
96        sample.gt = Some(Genotype {
97            alleles: vec![row.gt_alleles.0, row.gt_alleles.1],
98            phased: false,
99        });
100    }
