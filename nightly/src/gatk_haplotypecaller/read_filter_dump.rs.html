1//! HC read-filter decision dump.
2
3use crate::read_model::{
4    standard_hc_read_filter_failure_index, GATK_HC_DEFAULT_MIN_MAPPING_QUALITY,
5    STANDARD_HC_READ_FILTER_JAVA_NAMES,
6};
7use gatk_common::{GatkError, GatkResult};
8use rust_htslib::bam;
9use rust_htslib::bam::Read as _;
10use std::io::Write;
11use std::path::Path;
12
13/// L2 delimiter between per-read rows and the `CountingReadFilter`-style summary (matches Java gate dump).
14pub const HC_READ_FILTER_COUNT_SECTION: &str = "---HC_READ_FILTER_COUNTS---";
15
16/// TSV: `qname`, `flags`, `mapq`, `passes_hc_filter` (`true`/`false`).
17/// Matches `HcFullParityGateDump read-filters`: `HaplotypeCallerEngine.makeStandardHCReadFilters`
18/// at default MQ threshold ([`GATK_HC_DEFAULT_MIN_MAPPING_QUALITY`]).
19/// Appends [`HC_READ_FILTER_COUNT_SECTION`] plus `filter\tfiltered_count` rows mirroring Java
20/// `CountingReadFilter` leaf counts (AND short-circuit order).
21pub fn dump_hc_read_filter_tsv(alignment_path: &Path, out: &mut impl Write) -> GatkResult<()> {
22    writeln!(out, "qname\tflags\tmapq\tpasses_hc_filter")
23        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
24    let mut reader = bam::Reader::from_path(alignment_path)
25        .map_err(|e| GatkError::generic(format!("open: {e}")))?;
26    let header = reader.header().clone();
27    let mut counts = [0u64; STANDARD_HC_READ_FILTER_JAVA_NAMES.len()];
28    for res in reader.records() {
29        let rec = res.map_err(|e| GatkError::generic(format!("read: {e}")))?;
30        let fail = standard_hc_read_filter_failure_index(
31            &rec,
32            &header,
33            GATK_HC_DEFAULT_MIN_MAPPING_QUALITY,
34        );
35        let pass = fail.is_none();
36        if let Some(i) = fail {
37            counts[i] += 1;
38        }
39        writeln!(
40            out,
41            "{}\t{}\t{}\t{}",
42            String::from_utf8_lossy(rec.qname()),
43            rec.flags(),
44            rec.mapq(),
45            if pass { "true" } else { "false" }
46        )
47        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
48    }
49    writeln!(out, "{HC_READ_FILTER_COUNT_SECTION}")
50        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
51    writeln!(out, "filter\tfiltered_count")
52        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
53    for (i, name) in STANDARD_HC_READ_FILTER_JAVA_NAMES.iter().enumerate() {
54        writeln!(out, "{name}\t{}", counts[i])
55            .map_err(|e| GatkError::generic(format!("write: {e}")))?;
56    }
57    Ok(())
58}
