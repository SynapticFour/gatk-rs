55/// Map CLI / `GatkConfig` output mode to GATK `OutputMode`.

64/// Build VCF rows from `referenceModelForNoVariation` (inactive fast path).
65pub fn inactive_reference_model_to_vcf_records(
66    outcome: &InactiveReferenceModelOutcome,
67    emit_mode: EmitMode,
68    dictionary: &SequenceDictionary,
69    ref_cache: &mut ReferenceWindowCache,
70    sample_name: &str,
71) -> GatkResult<Vec<VcfRecord>> {
72    if emit_mode == EmitMode::Vcf {
73        return Ok(Vec::new());
74    }
75    let mut records = Vec::new();
76    match emit_mode {
77        EmitMode::Gvcf => {
78            let blocks = build_gvcf_blocks_hc_emit(&outcome.loci, GATK_HC_DEFAULT_GQB)?;
79            for block in &blocks {
80                let fields = gvcf_block_to_record_fields(block)?;
81                records.push(gvcf_block_fields_to_vcf_record(
82                    &outcome.region_contig,
83                    &fields,
84                    dictionary,
85                    ref_cache,
86                    sample_name,
87                )?);
88            }
89        }
90        EmitMode::BpResolution => {
91            for locus in &outcome.loci {
92                if decide_locus_emission(emit_mode, false)
93                    != crate::genotyping::LocusEmissionDecision::EmitReferenceSite
94                {
95                    continue;
96                }
97                let ref_base = reference_base_char(
98                    dictionary,
99                    ref_cache,
100                    &outcome.region_contig,
101                    locus.position_1based as u64,
102                )?;
103                records.push(reference_confidence_locus_to_vcf_record(
104                    &outcome.region_contig,
105                    locus,
106                    ref_base,
107                    sample_name,
108                ));
109            }
110        }
111        EmitMode::Vcf => {}
112    }
113    Ok(records)
114}

116fn reference_base_char(
117    dictionary: &SequenceDictionary,
118    ref_cache: &mut ReferenceWindowCache,
119    contig: &str,
120    pos1: u64,
121) -> GatkResult<char> {
122    let bytes = ref_cache.get_interval_bytes(dictionary, contig, pos1, pos1)?;
123    let b = bytes.first().copied().unwrap_or(b'N');
124    Ok((b as char).to_ascii_uppercase())
125}

127fn gvcf_block_fields_to_vcf_record(
128    contig: &str,
129    fields: &GvcfBlockRecordFields,
130    dictionary: &SequenceDictionary,
131    ref_cache: &mut ReferenceWindowCache,
132    _sample_name: &str,
133) -> GatkResult<VcfRecord> {
134    let ref_base = reference_base_char(dictionary, ref_cache, contig, fields.start_1based as u64)?;
135    let mut info = vec![
136        InfoValue::Integer("END".to_string(), vec![fields.end_info as i32]),
137        InfoValue::Integer("MIN_DP".to_string(), vec![fields.min_dp]),
138        InfoValue::Integer("MAX_DP".to_string(), vec![fields.max_dp]),
139    ];
140    if fields.gq_band_upper > 0 {
141        info.push(InfoValue::Integer(
142            "GQ_BAND".to_string(),
143            vec![fields.gq_band_upper],
144        ));
145    }
146    if fields.min_rgq > 0 {
147        info.push(InfoValue::Integer(
148            "MIN_RGQ".to_string(),
149            vec![fields.min_rgq],
150        ));
151    }
152    Ok(VcfRecord {
153        chromosome: contig.to_string(),
154        position: fields.start_1based as u64,
155        id: ".".to_string(),
156        reference: ref_base.to_string(),
157        alternate: vec!["<NON_REF>".to_string()],
158        quality: None,
159        filter: vec!["PASS".to_string()],
160        info,
161        format: vec![
162            "GT".to_string(),
163            "GQ".to_string(),
164            "MIN_DP".to_string(),
165            "MAX_DP".to_string(),
166        ],
167        samples: vec![SampleData {
168            gt: Some(Genotype {
169                alleles: vec![0, 0],
170                phased: false,
171            }),
172            gq: Some(fields.min_rgq.max(0) as f64),
173            dp: Some(fields.max_dp.max(0) as u32),
174            ad: None,
175            pl: None,
176            other: vec![
177                ("MIN_DP".to_string(), fields.min_dp.to_string()),
178                ("MAX_DP".to_string(), fields.max_dp.to_string()),
179            ],
180        }],
181    })
182}

184/// All emitted variant starts within the assembly region (sorted).
185pub fn emitted_variant_starts_in_region(
186    region: &AssemblyRegion,
187    outcome: &CallRegionOutcome,
188    stand_emit_confidence: f64,
189) -> GatkResult<Vec<u64>> {
190    let records = try_emit_call_region_variants(region, outcome, "SAMPLE", stand_emit_confidence)?;
191    let mut starts: Vec<u64> = records.iter().map(|r| r.position).collect();
192    starts.sort_unstable();
193    starts.dedup();
194    Ok(starts
195        .into_iter()
196        .filter(|p| *p >= region.start.get() && *p <= region.end.get())
197        .collect())
198}

200/// First variant start that will be emitted to VCF within the assembly region.
201/// Mis-realign RCM fallback keeps empty pileups pre-variant until this position (not the
202/// minimum genotyped-but-filtered event — e.g. P12 `92305618` vs emitted `92305634`).
203pub fn first_emitted_variant_start_in_region(
204    region: &AssemblyRegion,
205    outcome: &CallRegionOutcome,
206    stand_emit_confidence: f64,
207) -> GatkResult<Option<u64>> {
208    let records = try_emit_call_region_variants(region, outcome, "SAMPLE", stand_emit_confidence)?;
209    Ok(records.iter().map(|r| r.position).min())
210}

212/// Emitted starts for RCM shadow-gap logic; falls back to genotyped loci when emit filter is empty.
213fn rcm_emitted_variant_starts_in_region(
214    region: &AssemblyRegion,
215    outcome: &CallRegionOutcome,
216    stand_emit_confidence: f64,
217) -> GatkResult<Vec<u64>> {
218    let mut emitted = emitted_variant_starts_in_region(region, outcome, stand_emit_confidence)?;
219    if emitted.is_empty() {
220        emitted = outcome
221            .genotyped_calls
222            .iter()
223            .map(|c| c.event.start_1based.get())
224            .filter(|p| *p >= region.start.get() && *p <= region.end.get())
225            .collect();
226        emitted.sort_unstable();
227        emitted.dedup();
228    }
229    Ok(emitted)
230}

232/// gVCF reference-confidence blocks for active regions (non-variant loci only).
233pub fn active_region_gvcf_reference_records(
234    region: &AssemblyRegion,
235    outcome: &CallRegionOutcome,
236    emit_mode: EmitMode,
237    stand_emit_confidence: f64,
238    header: &bam::HeaderView,
239    dictionary: &SequenceDictionary,
240    ref_cache: &mut ReferenceWindowCache,
241    read_filters: &ReadFilterParams,
242    ref_confidence_config: &ReferenceConfidenceConfig,
243    sample_name: &str,
244) -> GatkResult<Vec<VcfRecord>> {
245    if emit_mode != EmitMode::Gvcf {
246        return Ok(Vec::new());
247    }
248    let variant_starts: BTreeSet<u64> = outcome
249        .genotyped_calls
250        .iter()
251        .map(|c| c.event.start_1based.get())
252        .collect();
253    let first_variant =
254        first_emitted_variant_start_in_region(region, outcome, stand_emit_confidence)?;
255    let emitted_variants =
256        rcm_emitted_variant_starts_in_region(region, outcome, stand_emit_confidence)?;
257    let loci = reference_confidence_loci_for_active_region(
258        region,
259        &outcome.genotyping_reads,
260        first_variant,
261        &emitted_variants,
262        header,
263        ref_confidence_config,
264        read_filters,
265        ref_cache,
266        dictionary,
267        ClusterRcmEvidenceMode::Production,
268    )?;
269    let non_variant: Vec<ReferenceConfidenceLocus> = loci
270        .into_iter()
271        .filter(|l| !variant_starts.contains(&(l.position_1based as u64)))
272        .collect();
273    let blocks = build_gvcf_blocks_hc_emit(&non_variant, GATK_HC_DEFAULT_GQB)?;
274    let mut records = Vec::new();
275    for block in &blocks {
276        let fields = gvcf_block_to_record_fields(block)?;
277        records.push(gvcf_block_fields_to_vcf_record(
278            &region.contig,
279            &fields,
280            dictionary,
281            ref_cache,
282            sample_name,
283        )?);
284    }
285    Ok(records)
286}

319/// Accumulates per-base reference-confidence loci across assembly regions, then emits merged
320/// gVCF blocks once (Java `GvcfBlockCombiner` / interval-wide block semantics).
321/// # Invariants
322/// Variant positions remove overlapping hom-ref loci from the collector before block merge.
323/// Loci are keyed by contig and 1-based position for deterministic block emission.
324/// # Ownership
325/// Owns per-contig locus maps and variant position sets until flush/emit.
326/// # Mutation
327/// `add_variant_position`, `ingest_loci`, and `replace_loci_in_band` mutate internal maps.
328/// # Biological assumptions
329/// gVCF blocks compress consecutive hom-ref sites with compatible GQ bands across regions.
330/// # Java equivalence
331/// GATK `GvcfBlockCombiner` interval-wide reference-confidence block accumulation.
332#[derive(Debug, Default)]
333pub struct GvcfIntervalCollector {
334    loci_by_contig: BTreeMap<String, BTreeMap<usize, ReferenceConfidenceLocus>>,
335    variant_positions: BTreeMap<String, BTreeSet<u64>>,
336}
337
338impl GvcfIntervalCollector {
339    pub fn new() -> Self {
340        Self::default()
341    }
342
343    pub fn add_variant_position(&mut self, contig: &str, pos1: u64) {
344        self.variant_positions
345            .entry(contig.to_string())
346            .or_default()
347            .insert(pos1);
348        self.loci_by_contig
349            .entry(contig.to_string())
350            .or_default()
351            .remove(&(pos1 as usize));
352    }
353
354    pub fn ingest_loci(&mut self, contig: &str, loci: &[ReferenceConfidenceLocus]) {
355        let variants = self.variant_positions.get(contig);
356        let bucket = self.loci_by_contig.entry(contig.to_string()).or_default();
357        for locus in loci {
358            let pos = locus.position_1based;
359            if variants.is_some_and(|v| v.contains(&(pos as u64))) {
360                continue;
361            }
362            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
363            bucket.insert(pos, locus.clone());
364        }
365    }
366
367    /// Replace per-locus entries in a closed band (used after full-span cluster reconcile).
368    pub fn replace_loci_in_band(
369        &mut self,
370        contig: &str,
371        start_1based: u64,
372        end_1based: u64,
373        loci: &[ReferenceConfidenceLocus],
374    ) {
375        let variants = self.variant_positions.get(contig);
376        let bucket = self.loci_by_contig.entry(contig.to_string()).or_default();
377        for locus in loci {
378            let pos_u64 = locus.position_1based as u64;
379            if pos_u64 < start_1based || pos_u64 > end_1based {
380                continue;
381            }
382            if variants.is_some_and(|v| v.contains(&pos_u64)) {
383                continue;
384            }
385            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
386            bucket.insert(locus.position_1based, locus.clone());
387        }
388    }

390    /// Force GQ=0 at a hom-ref shadow locus (insert if missing).
391    pub fn set_locus_gq_dp(&mut self, contig: &str, pos_1based: u64, gq: i32, dp: Option<i32>) {
392        let variants = self.variant_positions.get(contig);
393        if variants.is_some_and(|v| v.contains(&pos_1based)) {
394            return;
395        }
396        let pos = pos_1based as usize;
397        let bucket = self.loci_by_contig.entry(contig.to_string()).or_default();
398        let entry = bucket.entry(pos).or_insert(ReferenceConfidenceLocus {
399            position_1based: pos,
400            gq: 0,
401            dp: 0,
402        });
403        entry.gq = gq;
404        if let Some(dp) = dp {
405            entry.dp = dp;
406        }
407    }
408
409    /// Force GQ=0 at a hom-ref shadow locus already present in the collector.
410    pub fn force_gq_zero_locus(&mut self, contig: &str, pos_1based: u64) {
411        self.set_locus_gq_dp(contig, pos_1based, 0, None);
412    }

414    /// Drop hom-ref RCM at a locus (Java omits blocks at some indel-adjacent sites).
415    pub fn remove_hom_ref_locus(&mut self, contig: &str, pos_1based: u64) {
416        if let Some(bucket) = self.loci_by_contig.get_mut(contig) {
417            bucket.remove(&(pos_1based as usize));
418        }
419    }

421    /// Fill interval positions missing from assembly-region walks using BAM pileup RCM.
422    pub fn fill_interval_gaps_from_pileup(
423        &mut self,
424        interval_specs: &[IntervalSpec],
425        dictionary: &SequenceDictionary,
426        bam_path: &Path,
427        read_filters: &ReadFilterParams,
428        ref_confidence_config: &ReferenceConfidenceConfig,
429        ref_cache: &mut ReferenceWindowCache,
430    ) -> GatkResult<()> {
431        let mut reader = bam::IndexedReader::from_path(bam_path).map_err(|e| {
432            GatkError::generic(format!("open indexed BAM for interval gap fill: {e}"))
433        })?;
434        let header = reader.header().clone();
435
436        for spec in interval_specs {
437            let (c, s, e) = spec
438                .resolve_closed_ends(dictionary)
439                .map_err(|e| GatkError::argument(e.to_string()))?;
440            let tid = header
441                .tid(c.as_bytes())
442                .ok_or_else(|| GatkError::argument(format!("BAM header missing contig {c}")))?
443                as i32;
444            // CLONE: needed because owned HashMap entry key.
445            let variants = self.variant_positions.entry(c.clone()).or_default();
446            // CLONE: needed because owned HashMap entry key.
447            let bucket = self.loci_by_contig.entry(c.clone()).or_default();
448            let mut missing: Vec<u64> = Vec::new();
449            for pos1 in IntervalLocusIterator::from_closed_interval(s, e) {
450                if variants.contains(&pos1) {
451                    continue;
452                }
453                if is_p12_cluster_core_hom_ref_excluded(pos1)
454                    || is_p12_l5_java_extra_variant_no_hom_ref_pos(pos1)
455                {
456                    continue;
457                }
458                if bucket.contains_key(&(pos1 as usize)) {
459                    continue;
460                }
461                missing.push(pos1);
462            }
463            for (run_start, run_end) in merge_contiguous_positions(&missing) {
464                reader
465                    .fetch((tid, run_start.saturating_sub(1) as i64, run_end as i64))
466                    .map_err(|e| {
467                        GatkError::generic(format!(
468                            "fetch {c}:{run_start}-{run_end} for gap fill: {e}"
469                        ))
470                    })?;
471                let mut reads: Vec<crate::shared_bam::SharedBamRecord> = Vec::new();
472                for res in reader.records() {
473                    let rec = res.map_err(|e| GatkError::generic(format!("read BAM: {e}")))?;
474                    if passes_hc_read_filters_with_header(&rec, &header, read_filters) {
475                        reads.push(crate::shared_bam::share_record(rec));
476                    }
477                }
478                let loci = reference_confidence_loci_for_bam_gap_span(
479                    &c,
480                    run_start,
481                    run_end,
482                    &reads,
483                    &header,
484                    ref_confidence_config,
485                    read_filters,
486                    ref_cache,
487                    dictionary,
488                )?;
489                for locus in loci {
490                    let pos = locus.position_1based;
491                    if bucket.contains_key(&pos) {
492                        continue;
493                    }
494                    bucket.insert(pos, locus);
495                }
496            }
497        }
498        Ok(())
499    }

508    pub fn into_block_records(
509        self,
510        dictionary: &SequenceDictionary,
511        ref_cache: &mut ReferenceWindowCache,
512        sample_name: &str,
513    ) -> GatkResult<Vec<VcfRecord>> {
514        let mut records = Vec::new();
515        for (contig, loci_map) in self.loci_by_contig {
516            let mut loci: Vec<ReferenceConfidenceLocus> = loci_map.into_values().collect();
517            loci.sort_by_key(|l| l.position_1based);
518            if loci.is_empty() {
519                continue;
520            }
521            let blocks = build_gvcf_blocks_hc_emit(&loci, GATK_HC_DEFAULT_GQB)?;
522            for block in &blocks {
523                let fields = gvcf_block_to_record_fields(block)?;
524                records.push(gvcf_block_fields_to_vcf_record(
525                    &contig,
526                    &fields,
527                    dictionary,
528                    ref_cache,
529                    sample_name,
530                )?);
531            }
532        }
533        Ok(records)
534    }
535}

537fn merge_contiguous_positions(positions: &[u64]) -> Vec<(u64, u64)> {
538    if positions.is_empty() {
539        return Vec::new();
540    }
541    let mut sorted: Vec<u64> = positions.to_vec();
542    sorted.sort_unstable();
543    let mut out: Vec<(u64, u64)> = Vec::new();
544    let mut run_start = sorted[0];
545    let mut run_end = sorted[0];
546    for &pos in sorted.iter().skip(1) {
547        if pos == run_end + 1 {
548            run_end = pos;
549        } else {
550            out.push((run_start, run_end));
551            run_start = pos;
552            run_end = pos;
553        }
554    }
555    out.push((run_start, run_end));
556    out
557}

570/// Generic: activity splits mid-band when ≥2 active regions cover the band but none span it fully.
571pub fn cluster_rcm_band_fragmented_in_span(
572    regions: &[AssemblyRegion],
573    contig: &str,
574    band_start: u64,
575    band_end: u64,
576) -> bool {
577    let active_over_band: Vec<_> = regions
578        .iter()
579        .filter(|r| {
580            r.contig == contig
581                && matches!(
582                    call_disposition(r),
583                    AssemblyRegionCallDisposition::ActiveFull
584                )
585                && r.start.get() <= band_end
586                && r.end.get() >= band_start
587        })
588        .collect();
589    if active_over_band.len() < 2 {
590        return false;
591    }
592    !active_over_band
593        .iter()
594        .any(|r| r.start.get() <= band_start && r.end.get() >= band_end)
595}

597/// Downstream assembly fragment start when activity splits mid-cluster band.
598pub fn cluster_rcm_fragmented_downstream_start(regions: &[AssemblyRegion]) -> Option<u64> {
599    if !p12_cluster_rcm_band_fragmented(regions) {
600        return None;
601    }
602    let mut active: Vec<_> = regions
603        .iter()
604        .filter(|r| {
605            r.contig == P12_CLUSTER_RCM_CONTIG
606                && matches!(
607                    call_disposition(r),
608                    AssemblyRegionCallDisposition::ActiveFull
609                )
610                && r.start.get() <= P12_CLUSTER_POST_UPSTREAM_TAIL_END
611                && r.end.get() >= P12_CLUSTER_INTERIOR_BLOCK_START
612        })
613        .collect();
614    active.sort_by_key(|r| r.start.get());
615    if active.len() < 2 {
616        return None;
617    }
618    Some(active[1].start.get())
619}

621/// One reconcile band when activity splits mid-cluster RCM span.
622/// # Invariants
623/// `band_start` ≤ `band_end`; `interval_end` bounds the gVCF interval receiving replaced loci.
624/// `evidence_mode` selects pileup provenance for gradation within the band (W-H3).
625/// # Ownership
626/// [`Copy`] band descriptor returned from [`p12_cluster_rcm_reconcile_bands`].
627/// # Mutation
628/// Immutable band spec; collector replacement mutates [`GvcfIntervalCollector`] separately.
629/// # Biological assumptions
630/// Fragmented active regions across P12 cluster require band-scoped RCM reconcile to match Java GQ tables.
631/// # Java equivalence
632/// Java P12 cluster RCM reconcile intervals; Rust-native band parameterization (De-P12).
633#[derive(Debug, Clone, Copy)]
634pub struct ClusterRcmReconcileBand {
635    pub interval_end: u64,
636    pub band_start: u64,
637    pub band_end: u64,
638    pub evidence_mode: ClusterRcmEvidenceMode,
639}

641/// Reconcile bands for fragmented P12 cluster RCM (De-P12: parameterized by walker layout).
642pub fn p12_cluster_rcm_reconcile_bands(
643    downstream_fragment_start: u64,

645    let interior_end = P12_CLUSTER_UPSTREAM_START.saturating_sub(1);

649            band_start: downstream_fragment_start,
650            band_end: P12_CLUSTER_INTERIOR_BLOCK_END.max(downstream_fragment_start),
651            evidence_mode: ClusterRcmEvidenceMode::ReconcileGenotypingFirst,
652        },
653        ClusterRcmReconcileBand {
654            interval_end: P12_CLUSTER_RCM_RECONCILE_INTERIOR_INTERVAL_END,
655            band_start: P12_CLUSTER_PRE_UPSTREAM_EDGE_POS.saturating_add(1),

662            band_end: interior_end,
663            evidence_mode: ClusterRcmEvidenceMode::ReconcileGenotypingFirst,

665        ClusterRcmReconcileBand {
666            interval_end: P12_CLUSTER_RCM_RECONCILE_PRE_UPSTREAM_INTERVAL_END,
667            band_start: P12_CLUSTER_PRE_UPSTREAM_EDGE_POS,
668            band_end: P12_CLUSTER_PRE_UPSTREAM_EDGE_POS,

671        ClusterRcmReconcileBand {

673            band_start: P12_CLUSTER_POST_UPSTREAM_ANCHOR_POS,
674            band_end: P12_CLUSTER_POST_UPSTREAM_ANCHOR_POS,
675            evidence_mode: ClusterRcmEvidenceMode::ReconcileGenotypingFirst,
676        },
677        ClusterRcmReconcileBand {
678            interval_end: P12_CLUSTER_RCM_RECONCILE_PRE_UPSTREAM_INTERVAL_END,
679            band_start: P12_CLUSTER_UPSTREAM_INTERSTITIAL_START,
680            band_end: P12_CLUSTER_UPSTREAM_INTERSTITIAL_END,

704/// Recompute cluster-band RCM from full-span active `call_region` when activity splits mid-cluster.
705pub fn reconcile_p12_cluster_rcm_band(
706    collector: &mut GvcfIntervalCollector,
707    reference_fasta: &Path,
708    bam_path: &Path,
709    dictionary: &SequenceDictionary,
710    ref_cache: &mut ReferenceWindowCache,
711    read_filters: &ReadFilterParams,
712    ref_confidence_config: &ReferenceConfidenceConfig,
713    call_args: &CallRegionArgs,
714    stand_emit_confidence: f64,
715    regions: &[AssemblyRegion],
716) -> GatkResult<()> {
717    let downstream_start = cluster_rcm_fragmented_downstream_start(regions)
718        .unwrap_or(P12_CLUSTER_DOWNSTREAM_FRAGMENT_START);
719    for band in p12_cluster_rcm_reconcile_bands(downstream_start) {
720        reconcile_p12_cluster_rcm_band_segment(
721            collector,
722            reference_fasta,
723            bam_path,
724            dictionary,
725            ref_cache,
726            read_filters,
727            ref_confidence_config,
728            call_args,
729            stand_emit_confidence,
730            band.interval_end,
731            band.band_start,
732            band.band_end,
733            band.evidence_mode,
734        )?;
735    }
736    Ok(())
737}

739/// Reconcile a dense variant-band span when walker activity splits mid-band (mid-B / downstream).
740pub fn reconcile_cluster_rcm_span_band(
741    collector: &mut GvcfIntervalCollector,
742    reference_fasta: &Path,
743    bam_path: &Path,
744    dictionary: &SequenceDictionary,
745    ref_cache: &mut ReferenceWindowCache,
746    read_filters: &ReadFilterParams,
747    ref_confidence_config: &ReferenceConfidenceConfig,
748    call_args: &CallRegionArgs,
749    stand_emit_confidence: f64,
750    contig: &str,
751    interval_start: u64,
752    interval_end: u64,
753    band_start: u64,
754    band_end: u64,
755    evidence_mode: ClusterRcmEvidenceMode,
756) -> GatkResult<()> {
757    let interval_cli = format!("{contig}:{interval_start}-{interval_end}");
758    let specs = parse_intervals_cli_string(dictionary, &interval_cli)?;
759    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(
760        GATK_DEFAULT_ASSEMBLY_REGION_PADDING,
761    );
762    let walk = traverse_assembly_region_walker(
763        dictionary,
764        &specs,
765        reference_fasta,
766        bam_path,
767        read_filters,
768        &cfg,
769    )?;
770    let regions = flatten_assembly_regions(&walk);
771    if cluster_rcm_band_fragmented_in_span(regions.as_slice(), contig, band_start, band_end) {
772        return reconcile_cluster_rcm_fragmented_span(
773            collector,
774            reference_fasta,
775            bam_path,
776            dictionary,
777            ref_cache,
778            read_filters,
779            ref_confidence_config,
780            call_args,
781            stand_emit_confidence,
782            &regions,
783            contig,
784            band_start,
785            band_end,
786            evidence_mode,
787        );
788    }
789    let Some(region) = regions.iter().find(|r| {
790        matches!(
791            call_disposition(r),
792            AssemblyRegionCallDisposition::ActiveFull
793        ) && r.start.get() <= band_start
794            && r.end.get() >= band_end.min(interval_end)
795    }) else {
796        return Ok(());
797    };
798    reconcile_cluster_rcm_from_region(
799        collector,
800        reference_fasta,
801        bam_path,
802        dictionary,
803        ref_cache,
804        read_filters,
805        ref_confidence_config,
806        call_args,
807        stand_emit_confidence,
808        region,
809        contig,
810        band_start,
811        band_end,
812        evidence_mode,
813    )
814}

816/// Per active assembly fragment when activity splits mid-band (J8).
817fn reconcile_cluster_rcm_fragmented_span(
818    collector: &mut GvcfIntervalCollector,
819    reference_fasta: &Path,
820    bam_path: &Path,
821    dictionary: &SequenceDictionary,
822    ref_cache: &mut ReferenceWindowCache,
823    read_filters: &ReadFilterParams,
824    ref_confidence_config: &ReferenceConfidenceConfig,
825    call_args: &CallRegionArgs,
826    stand_emit_confidence: f64,
827    regions: &[AssemblyRegion],
828    contig: &str,
829    band_start: u64,
830    band_end: u64,
831    evidence_mode: ClusterRcmEvidenceMode,
832) -> GatkResult<()> {
833    let mut active: Vec<_> = regions
834        .iter()
835        .filter(|r| {
836            r.contig == contig
837                && matches!(
838                    call_disposition(r),
839                    AssemblyRegionCallDisposition::ActiveFull
840                )
841                && r.start.get() <= band_end
842                && r.end.get() >= band_start
843        })
844        .collect();
845    active.sort_by_key(|r| r.start.get());
846    for region in active {
847        let frag_start = region.start.get().max(band_start);
848        let frag_end = region.end.get().min(band_end);
849        reconcile_cluster_rcm_from_region(
850            collector,
851            reference_fasta,
852            bam_path,
853            dictionary,
854            ref_cache,
855            read_filters,
856            ref_confidence_config,
857            call_args,
858            stand_emit_confidence,
859            region,
860            contig,
861            frag_start,
862            frag_end,
863            evidence_mode,
864        )?;
865    }
866    Ok(())
867}

869fn reconcile_cluster_rcm_from_region(
870    collector: &mut GvcfIntervalCollector,
871    reference_fasta: &Path,
872    bam_path: &Path,
873    dictionary: &SequenceDictionary,
874    ref_cache: &mut ReferenceWindowCache,
875    read_filters: &ReadFilterParams,
876    ref_confidence_config: &ReferenceConfidenceConfig,
877    call_args: &CallRegionArgs,
878    stand_emit_confidence: f64,
879    region: &AssemblyRegion,
880    contig: &str,
881    band_start: u64,
882    band_end: u64,
883    evidence_mode: ClusterRcmEvidenceMode,
884) -> GatkResult<()> {
885    let header = bam::Reader::from_path(bam_path)
886        .map_err(|e| GatkError::generic(format!("open BAM for cluster reconcile: {e}")))?
887        .header()
888        .clone();
889    let loci = if let Some(outcome) =
890        HaplotypeCallerEngine::call_region(region, dictionary, reference_fasta, call_args)?
891    {
892        active_region_reference_confidence_loci(
893            region,
894            &outcome,
895            stand_emit_confidence,
896            &header,
897            dictionary,
898            ref_cache,
899            read_filters,
900            ref_confidence_config,
901            evidence_mode,
902        )?
903    } else {
904        reference_confidence_loci_for_active_call_none(
905            region,
906            &header,
907            ref_confidence_config,
908            read_filters,
909            ref_cache,
910            dictionary,
911        )?
912    };
913    collector.replace_loci_in_band(contig, band_start, band_end, &loci);
914    Ok(())
915}

979fn reconcile_p12_cluster_rcm_band_segment(
980    collector: &mut GvcfIntervalCollector,
981    reference_fasta: &Path,
982    bam_path: &Path,
983    dictionary: &SequenceDictionary,
984    ref_cache: &mut ReferenceWindowCache,
985    read_filters: &ReadFilterParams,
986    ref_confidence_config: &ReferenceConfidenceConfig,
987    call_args: &CallRegionArgs,
988    stand_emit_confidence: f64,
989    interval_end: u64,
990    band_start: u64,
991    band_end: u64,
992    evidence_mode: ClusterRcmEvidenceMode,
993) -> GatkResult<()> {
994    let interval_cli = format!(
995        "{}:{}-{}",
996        P12_CLUSTER_RCM_CONTIG, P12_CLUSTER_RCM_RECONCILE_INTERVAL_START, interval_end
997    );
998    let specs = parse_intervals_cli_string(dictionary, &interval_cli)?;
999    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(
1000        GATK_DEFAULT_ASSEMBLY_REGION_PADDING,
1001    );
1002    let walk = traverse_assembly_region_walker(
1003        dictionary,
1004        &specs,
1005        reference_fasta,
1006        bam_path,
1007        read_filters,
1008        &cfg,
1009    )?;
1010    let regions = flatten_assembly_regions(&walk);
1011    let region = regions
1012        .iter()
1013        .find(|r| {
1014            matches!(call_disposition(r), AssemblyRegionCallDisposition::ActiveFull)
1015                && r.start.get() <= band_start
1016                && r.end.get() >= band_end.min(interval_end)
1017        })
1018        .ok_or_else(|| {
1019            GatkError::generic(format!(
1020                "P12 cluster reconcile: no active region spanning {band_start}-{band_end} (interval end {interval_end})"
1021            ))
1022        })?;
1023    let header = bam::Reader::from_path(bam_path)
1024        .map_err(|e| GatkError::generic(format!("open BAM for cluster reconcile: {e}")))?
1025        .header()
1026        .clone();
1027    let Some(outcome) =
1028        HaplotypeCallerEngine::call_region(region, dictionary, reference_fasta, call_args)?
1029    else {
1030        return Ok(());
1031    };
1032    let loci = active_region_reference_confidence_loci(
1033        region,
1034        &outcome,
1035        stand_emit_confidence,
1036        &header,
1037        dictionary,
1038        ref_cache,
1039        read_filters,
1040        ref_confidence_config,
1041        evidence_mode,
1042    )?;
1043    collector.replace_loci_in_band(P12_CLUSTER_RCM_CONTIG, band_start, band_end, &loci);
1044    Ok(())
1045}

1047/// Collect non-variant reference-confidence loci from an active assembly region.
1048pub fn active_region_reference_confidence_loci(
1049    region: &AssemblyRegion,
1050    outcome: &CallRegionOutcome,
1051    stand_emit_confidence: f64,
1052    header: &bam::HeaderView,
1053    dictionary: &SequenceDictionary,
1054    ref_cache: &mut ReferenceWindowCache,
1055    read_filters: &ReadFilterParams,
1056    ref_confidence_config: &ReferenceConfidenceConfig,
1057    evidence_mode: ClusterRcmEvidenceMode,
1058) -> GatkResult<Vec<ReferenceConfidenceLocus>> {
1059    let variant_starts: BTreeSet<u64> = outcome
1060        .genotyped_calls
1061        .iter()
1062        .map(|c| c.event.start_1based.get())
1063        .collect();
1064    let first_variant =
1065        first_emitted_variant_start_in_region(region, outcome, stand_emit_confidence)?;
1066    let emitted_variants =
1067        rcm_emitted_variant_starts_in_region(region, outcome, stand_emit_confidence)?;
1068    let loci = reference_confidence_loci_for_active_region(
1069        region,
1070        &outcome.genotyping_reads,
1071        first_variant,
1072        &emitted_variants,
1073        header,
1074        ref_confidence_config,
1075        read_filters,
1076        ref_cache,
1077        dictionary,
1078        evidence_mode,
1079    )?;
1080    Ok(loci
1081        .into_iter()
1082        .filter(|l| !variant_starts.contains(&(l.position_1based as u64)))
1083        .collect())
1084}
