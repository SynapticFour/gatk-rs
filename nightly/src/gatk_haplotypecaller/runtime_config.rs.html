6/// Execution / CLI surface flags (always readable in production builds).
7/// # Invariants
8/// `legacy_provisional` is rejected at runtime when set (Sprint B removal).
9/// `scaffold_output` and `activate_output_opt_out` are independent emit toggles.
10/// # Ownership
11/// [`Clone`] sub-config owned by [`RuntimeConfig`]; parsed once at process edges.
12/// # Mutation
13/// Immutable after [`RuntimeConfig::from_env`]; must not be mutated mid-run for parity.
14/// # Biological assumptions
15/// None documented (process control flags only).
16/// # Java equivalence
17/// Rust-native env surface; no direct Java argument collection (see `ENV_INVENTORY.md`).
18#[derive(Debug, Clone, Default)]
19pub struct ExecutionConfig {
20    /// `GATK_RS_HC_SCAFFOLD_OUTPUT=1` — header-only VCF.
21    pub scaffold_output: bool,
22    /// `GATK_RS_HC_ACTIVATE_OUTPUT=0` — explicit opt-out of variant emit.
23    pub activate_output_opt_out: bool,
24    /// `GATK_RS_HC_LEGACY_PROVISIONAL=1` — rejected (Sprint B removal).
25    pub legacy_provisional: bool,
26}
27
28/// Diagnostics that must never change emit/genotype results.
29/// # Invariants
30/// Debug flags affect logging/tracing only; documented as non-behavior-changing.
31/// `debug_tile_overlaps` may be O(tiles×BAM) and is for scaffold diagnostics only.
32/// # Ownership
33/// [`Clone`] sub-config owned by [`RuntimeConfig`].
34/// # Mutation
35/// Immutable after env parse; helpers may re-read env on each call for dump-only paths.
36/// # Biological assumptions
37/// None documented.
38/// # Java equivalence
39/// Rust-native diagnostics env vars; not mirrored in GATK CLI.
40#[derive(Debug, Clone, Default)]
41pub struct DebugConfig {
42    /// `GATK_RS_STRICT_CLUSTER_DEBUG=1` — stderr tracing for cluster materialize.
43    pub strict_cluster_debug: bool,
44    /// `GATK_RS_HC_DEBUG_TILE_OVERLAPS=1` — O(tiles×BAM) scaffold overlap count (R4-1).
45    pub debug_tile_overlaps: bool,
46    /// `GATK_RS_SEMANTIC_TRACE=<path>` — observe-only NDJSON semantic checkpoints.
47    /// Never changes genotype/emit results; see `semantic_trace` module.
48    pub semantic_trace_path: Option<String>,
49}

51/// Full runtime config. Prefer [`RuntimeConfig::from_env`] at process edges.
52/// # Invariants
53/// Production algorithm modules must not call `std::env::var` directly (use this module).
54/// Harness-gated flags remain in [`crate::parity_harness`], not here.
55/// # Ownership
56/// Owns [`ExecutionConfig`] and [`DebugConfig`]; cheap to clone at startup.
57/// # Mutation
58/// Immutable after construction except dump helpers that re-parse env for debug booleans.
59/// # Biological assumptions
60/// None documented.
61/// # Java equivalence
62/// Rust-native centralized env config (Sprint I-4); GATK uses CLI args instead.
63#[derive(Debug, Clone, Default)]
64pub struct RuntimeConfig {
65    pub execution: ExecutionConfig,
66    pub debug: DebugConfig,
67}
68
69impl RuntimeConfig {
70    /// Parse production-facing env vars once. Harness flags stay in [`crate::parity_harness`].
71    pub fn from_env() -> Self {
72        Self {
73            execution: ExecutionConfig {
74                scaffold_output: env_truthy("GATK_RS_HC_SCAFFOLD_OUTPUT"),
75                activate_output_opt_out: env_is("GATK_RS_HC_ACTIVATE_OUTPUT", &["0", "false"]),
76                legacy_provisional: env_truthy("GATK_RS_HC_LEGACY_PROVISIONAL"),
77            },
78            debug: DebugConfig {
79                strict_cluster_debug: env_truthy("GATK_RS_STRICT_CLUSTER_DEBUG"),
80                debug_tile_overlaps: env_truthy("GATK_RS_HC_DEBUG_TILE_OVERLAPS"),
81                semantic_trace_path: std::env::var("GATK_RS_SEMANTIC_TRACE")
82                    .ok()
83                    .filter(|p| !p.is_empty() && p != "0" && !p.eq_ignore_ascii_case("off")),
84            },
85        }
86    }
87}
88
89fn env_truthy(name: &str) -> bool {
90    std::env::var(name)
91        .map(|v| matches!(v.as_str(), "1" | "true" | "TRUE" | "yes" | "YES"))
92        .unwrap_or(false)
93}
94
95fn env_is(name: &str, values: &[&str]) -> bool {
96    std::env::var(name).ok().is_some_and(|v| {
97        values
98            .iter()
99            .any(|want| v == *want || v.eq_ignore_ascii_case(want))
100    })
101}
102
103/// Process-wide debug helpers (cheap; reads env each call — dump/debug only).
104pub fn strict_cluster_debug_enabled() -> bool {
105    RuntimeConfig::from_env().debug.strict_cluster_debug
106}
