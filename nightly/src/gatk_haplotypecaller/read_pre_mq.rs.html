1//! Assembly-path mapping-quality gate.
2//! Mirrors `HaplotypeCallerEngine.filterNonPassingReads`:
3//! `rec.getMappingQuality < hcArgs.mappingQualityThreshold`.
4
5use gatk_core::MappingQuality;
6use rust_htslib::bam;
7
8/// `HaplotypeCallerEngine.DEFAULT_READ_QUALITY_FILTER_THRESHOLD` / `hcArgs.mappingQualityThreshold` default.
9pub const GATK_ASSEMBLY_MQ_FILTER_THRESHOLD: u8 = 20;
10
11/// MQ portion of `filterNonPassingReads` (strict `<` in Java → keep when `mapq >= threshold`).
12/// SAM MAPQ `255` ([`MappingQuality::Unavailable`]) is treated as passing, matching Java's
13/// unsigned comparison (`255 >= threshold` for normal HC thresholds).
14#[inline]
15pub fn passes_assembly_mq_filter(rec: &bam::Record, threshold: u8) -> bool {
16    match MappingQuality::from_sam_mapq(rec.mapq()) {
17        MappingQuality::Unavailable => true,
18        MappingQuality::Score(q) => q >= threshold,
19    }
20}
21
22/// Default HC assembly-path threshold (20).
23#[allow(dead_code)] // convenience wrapper; call sites pass an explicit threshold today
24#[inline]
25pub fn passes_assembly_mq_filter_default(rec: &bam::Record) -> bool {
26    passes_assembly_mq_filter(rec, GATK_ASSEMBLY_MQ_FILTER_THRESHOLD)
27}

29#[cfg(test)]
30mod tests {
31    use super::*;
32    use rust_htslib::bam::record::{Cigar, CigarString};
33
34    fn rec_with_mapq(mapq: u8) -> bam::Record {
35        let mut r = bam::Record::new();
36        r.set(
37            b"t",
38            Some(&CigarString::from(vec![Cigar::Match(10)])),
39            b"AAAAAAAAAA",
40            &vec![30u8; 10],
41        );
42        r.set_mapq(mapq);
43        r
44    }
45
46    #[test]
47    fn threshold_twenty_is_inclusive() {
48        assert!(passes_assembly_mq_filter_default(&rec_with_mapq(20)));
49        assert!(!passes_assembly_mq_filter_default(&rec_with_mapq(19)));
50    }
51
52    #[test]
53    fn mq_unavailable_passes_assembly_path() {
54        assert!(passes_assembly_mq_filter_default(&rec_with_mapq(255)));
55        assert_eq!(
56            MappingQuality::from_sam_mapq(255),
57            MappingQuality::Unavailable
58        );
59    }
