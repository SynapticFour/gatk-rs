5fn log10_factorial(n: u32) -> f64 {
6    if n <= 1 {
7        return 0.0;
8    }
9    (2..=n).map(|i| (i as f64).log10()).sum()
10}
11
12fn log10_binomial(n: u32, k: u32) -> f64 {
13    log10_factorial(n) - log10_factorial(k) - log10_factorial(n - k)
14}

16/// GATK `computeMaxAcceptableAlleleCount(ploidy, maxGenotypeCount)`.
17pub fn compute_max_acceptable_allele_count(
18    ploidy: u32,
19    max_genotype_count: u32,
20) -> GatkResult<u32> {
21    if ploidy == 0 {
22        return Err(GatkError::argument("ploidy must be >= 1"));
23    }
24    if max_genotype_count == 0 {
25        return Err(GatkError::argument("max_genotype_count must be >= 1"));
26    }
27    if ploidy == 1 {
28        return Ok(max_genotype_count);
29    }
30    let log10_max = (max_genotype_count as f64).log10();
31    let x = 10_f64.powf((log10_factorial(ploidy) + log10_max) / (ploidy as f64));
32    let lower = ((x.floor() as i32) - (ploidy as i32) - 1).max(2) as u32;
33    let upper = x.ceil().max(2.0) as u32;
34    for a in (lower..=upper).rev() {
35        let log10_gt = log10_binomial(ploidy + a - 1, a - 1);
36        if log10_max >= log10_gt {
37            return Ok(a);
38        }
39    }
40    Ok(2)
41}

43/// Haplotype score precedence for allele trimming (GATK `whichAllelesToKeepBasedonHapScores` lite).
44/// # Invariants
45/// `best_score` ≥ `second_best_score` for the allele’s haplotype score list.
46/// `allele_index` indexes the allele list being trimmed.
47/// # Ownership
48/// Owned score triple used while ranking alleles to keep.
49/// # Mutation
50/// Ephemeral ranking record; discarded after allele selection.
51/// # Biological assumptions
52/// Alleles with stronger supporting haplotype scores are preferred when capping allele count.
53/// # Java equivalence
54/// GATK `HaplotypeCallerGenotypingEngine.whichAllelesToKeepBasedonHapScores` ranking cell.
55#[derive(Debug, Clone, PartialEq)]
56pub struct AlleleHaplotypeScore {
57    pub allele_index: usize,
58    pub best_score: f64,
59    pub second_best_score: f64,
60}
61
62/// Best and second-best scores without allocating a sorted clone (ascending-sort last two).
63#[inline]
64fn top_two_scores(scores: &[f64]) -> (f64, f64) {
65    let mut best = f64::NEG_INFINITY;
66    let mut second = f64::NEG_INFINITY;
67    for &s in scores {
68        if s > best {
69            second = best;
70            best = s;
71        } else if s > second {
72            second = s;
73        }
74    }
75    (best, second)
76}

78/// GATK `HaplotypeCallerGenotypingEngine.whichAllelesToKeepBasedonHapScores`.
79#[allow(dead_code)] // thin wrapper over `_with_ref`; kept for GATK-shaped call sites
80pub fn which_alleles_to_keep_by_haplotype_scores<S: AsRef<[f64]>>(
81    scores_per_allele: &[S],
82    desired_allele_count: usize,
83) -> Vec<usize> {
84    which_alleles_to_keep_by_haplotype_scores_with_ref(
85        scores_per_allele,
86        desired_allele_count,
87        None,
88    )
89}

91/// Same as [`which_alleles_to_keep_by_haplotype_scores`] with optional per-allele reference flags (GATK ref-priority tie-break).
92pub fn which_alleles_to_keep_by_haplotype_scores_with_ref<S: AsRef<[f64]>>(
93    scores_per_allele: &[S],
94    desired_allele_count: usize,
95    is_reference: Option<&[bool]>,
96) -> Vec<usize> {
97    if scores_per_allele.len() <= desired_allele_count {
98        return (0..scores_per_allele.len()).collect();
99    }
100    let mut ranked: Vec<AlleleHaplotypeScore> = scores_per_allele
101        .iter()
102        .enumerate()
103        .map(|(i, scores)| {
104            // Top-2 without sorting a cloned score vector (same as sort-then-take-last-two).
105            let (best, second) = top_two_scores(scores.as_ref());
106            AlleleHaplotypeScore {
107                allele_index: i,
108                best_score: best,
109                second_best_score: second,
110            }
111        })
112        .collect();
113    ranked.sort_by(|a, b| compare_haplotype_allele_scores(a, b, is_reference));
114    let keep = desired_allele_count.min(ranked.len());
115    let mut out: Vec<usize> = ranked.iter().take(keep).map(|a| a.allele_index).collect();
116    out.sort_unstable();
117    out
118}

120fn compare_haplotype_allele_scores(
121    a: &AlleleHaplotypeScore,
122    b: &AlleleHaplotypeScore,
123    is_reference: Option<&[bool]>,
124) -> std::cmp::Ordering {
125    if let Some(flags) = is_reference {
126        let a_ref = flags.get(a.allele_index).copied().unwrap_or(false);
127        let b_ref = flags.get(b.allele_index).copied().unwrap_or(false);
128        if a_ref && !b_ref {
129            return std::cmp::Ordering::Less;
130        }
131        if !a_ref && b_ref {
132            return std::cmp::Ordering::Greater;
133        }
134    }
135    b.best_score
136        .total_cmp(&a.best_score)
137        .then(b.second_best_score.total_cmp(&a.second_best_score))
138        .then(a.allele_index.cmp(&b.allele_index))
139}
