1//! GATK `GraphBasedKBestHaplotypeFinder` on [`SeqGraph`] (`Path.getBases` stitching).
2
3use crate::kbest_haplotype::{log_penalty, KBestPath};
4use crate::seq_graph::SeqGraph;
5use gatk_common::{GatkError, GatkResult};
6use std::cmp::Ordering;
7use std::collections::{BinaryHeap, HashSet};
8
9/// Same bound as the read-threading k-best finder — see `kbest_haplotype`.
10const MAX_KBEST_HEAP_PATHS: usize = 1_024;
11const MAX_KBEST_PATH_EDGES: usize = 4_096;
12
13#[derive(Debug, Clone)]
14struct PathState {
15    start: usize,
16    edges: Vec<(usize, usize)>,
17    last: usize,
18    score: f64,
19    is_reference: bool,
20    /// Edge count as a cheap heap tie-break (full bases sort happens at the end).
21    edge_count: usize,
22}
23
24impl PathState {
25    fn new(start: usize) -> Self {
26        Self {
27            start,
28            edges: Vec::new(),
29            last: start,
30            score: 0.0,
31            is_reference: false,
32            edge_count: 0,
33        }
34    }
35
36    fn extend(&self, graph: &SeqGraph, to: usize, edge_support: u32, total_outgoing: u32) -> Self {
37        // Same as read-threading k-best: no `Vec::clone` on the frontier path.
38        let mut edges = Vec::with_capacity(self.edges.len() + 1);
39        edges.extend_from_slice(&self.edges);
40        edges.push((self.last, to));
41        let penalty = log_penalty(edge_support, total_outgoing);
42        Self {
43            start: self.start,
44            edges,
45            last: to,
46            score: self.score + penalty,
47            is_reference: self.is_reference && graph.edge_is_ref(self.last, to),
48            edge_count: self.edge_count + 1,
49        }
50    }
51}
52
53struct HeapItem {
54    score_bits: u64,
55    tie: usize,
56    path: PathState,
57}
58
59impl Ord for HeapItem {
60    fn cmp(&self, other: &Self) -> Ordering {
61        self.score_bits
62            .cmp(&other.score_bits)
63            .then_with(|| other.tie.cmp(&self.tie))
64    }
65}
66
67impl PartialOrd for HeapItem {
68    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
69        Some(self.cmp(other))
70    }
71}
72
73impl PartialEq for HeapItem {
74    fn eq(&self, other: &Self) -> bool {
75        self.score_bits == other.score_bits && self.tie == other.tie
76    }
77}
78
79impl Eq for HeapItem {}

81/// K-best paths on a cleaned sequence graph (GATK `GraphBasedKBestHaplotypeFinder`).
82pub fn find_best_haplotypes_seq_graph(
83    graph: &SeqGraph,
84    max_number_of_haplotypes: usize,
85) -> GatkResult<Vec<KBestPath>> {
86    if max_number_of_haplotypes == 0 {
87        return Ok(Vec::new());
88    }
89    let source = graph
90        .reference_source_vertex()
91        .ok_or_else(|| GatkError::algorithm("seq kbest: no reference source vertex"))?;
92    let sink = graph
93        .reference_sink_vertex()
94        .ok_or_else(|| GatkError::algorithm("seq kbest: no reference sink vertex"))?;
95    let sinks: HashSet<usize> = HashSet::from([sink]);
96
97    let mut result = Vec::new();
98    let mut heap: BinaryHeap<HeapItem> = BinaryHeap::new();
99    heap.push(HeapItem {
100        score_bits: 0.0_f64.to_bits(),
101        tie: 0,
102        path: PathState::new(source),
103    });
104
105    let mut vertex_counts = vec![0usize; graph.node_count()];
106    // Same Peak bound as read-threading k-best (50k → 12k for dense GIAB shards).
107    const MAX_KBEST_EXPANSIONS: usize = 12_000;
108    let mut expansions = 0usize;
109
110    while !heap.is_empty() && result.len() < max_number_of_haplotypes {
111        if crate::runtime_config::hc_rss_abort_triggered() {
112            crate::runtime_config::rss_trace_checkpoint(
113                "seq_kbest_rss_abort",
114                &format!("expansions={expansions} results={}", result.len()),
115            );
116            break;
117        }
118        let item = heap.pop().expect("non-empty");
119        let path = item.path;
120        if sinks.contains(&path.last) {
121            result.push(KBestPath {
122                start: path.start,
123                edges: path.edges,
124                score: path.score,
125                is_reference: path.is_reference,
126            });
127            continue;
128        }
129        if path.edges.len() >= MAX_KBEST_PATH_EDGES
130            || expansions >= MAX_KBEST_EXPANSIONS
131            || heap.len() >= MAX_KBEST_HEAP_PATHS
132        {
133            continue;
134        }
135        if vertex_counts[path.last] < max_number_of_haplotypes {
136            vertex_counts[path.last] += 1;
137            expansions += 1;
138            let outs = graph.outgoing_nodes(path.last);
139            let total: u32 = outs
140                .iter()
141                .filter_map(|&t| graph.edge_support(path.last, t))
142                .sum();
143            for to in outs {
144                if let Some(support) = graph.edge_support(path.last, to) {
145                    if heap.len() >= MAX_KBEST_HEAP_PATHS {
146                        break;
147                    }
148                    let extended = path.extend(graph, to, support, total);
149                    heap.push(HeapItem {
150                        score_bits: extended.score.to_bits(),
151                        tie: extended.edge_count,
152                        path: extended,
153                    });
154                }
155            }
156        }
157    }
158
159    result.sort_by(|a, b| {
160        b.score
161            .partial_cmp(&a.score)
162            .unwrap_or(Ordering::Equal)
163            .then_with(|| {
164                let ab = graph.path_bases_bytes(a.start, &a.edges);
165                let bb = graph.path_bases_bytes(b.start, &b.edges);
166                bb.cmp(&ab)
167            })
168    });
169    Ok(result)
170}
