17/// HC INFO + QUAL slice aligned with GATK `HaplotypeCaller` default annotators on biallelic sites.
18/// # Invariants
19/// Fields mirror GATK default INFO keys for emitted biallelic variant sites (AC/AF/AN/DP/FS/SOR/etc.).
20/// `qual` uses AF-calculator path with Java-observed posterior convention for emitted variants.
21/// # Ownership
22/// Owns scalar annotation values; genotype and region data are consumed at construction time.
23/// # Mutation
24/// Immutable annotation bundle produced by [`annotate_hc_variant_site`].
25/// # Biological assumptions
26/// Single-sample diploid biallelic SNP/indels with standard short-read INFO semantics.
27/// # Java equivalence
28/// GATK `HaplotypeCaller` default variant annotators (J-D01 INFO/QUAL slice).
29#[derive(Debug, Clone, PartialEq)]
30pub struct HcVariantSiteAnnotations {
31    pub qual: f64,
32    pub ac: i32,
33    pub af: f64,
34    pub an: i32,
35    pub dp: i32,
36    pub excess_het: f64,
37    pub fs: f64,
38    pub mleac: i32,
39    pub mleaf: f64,
40    pub mq: f64,
41    pub qd: f64,
42    pub sor: f64,
43    pub read_pos_rank_sum: f64,
44    pub inbreeding_coeff: f64,
45}

47/// Phred-scaled site QUAL from GATK `GenotypingEngine` + `AlleleFrequencyCalculator`
48/// (`use-posteriors-to-calculate-qual` false, default HC).
49pub fn qual_from_af_calculation(genotype_log10_likelihoods: &[f64]) -> GatkResult<f64> {
50    let af = calculate_biallelic_af_em(
51        &[genotype_log10_likelihoods],
52        &AfCalculatorConfig::default(),
53    )?;
54    // Observed on p11: Java uses `log10ProbOnlyRefAlleleExists` for emitted variant QUAL.
55    let log10_conf = af.log10_posterior_no_variant + 0.0;
56    Ok((-10.0 * log10_conf) + 0.0)
57}

59/// GATK `QualByDepth.getDepth` for a single variant genotype.
60pub fn qd_depth_for_variant(gt: &Genotype, fields: &GenotypeFormatFields) -> i32 {
61    if !is_het_or_hom_var(gt) {
62        return 0;
63    }
64    if !fields.ad.is_empty() {
65        let total_ad: i32 = fields.ad.iter().map(|d| d.as_i32()).sum();
66        if total_ad != 0 {
67            let alt_ad = total_ad.saturating_sub(fields.ad[0].as_i32().max(0));
68            let ad_restricted = if alt_ad > 1 { total_ad } else { 0 };
69            if ad_restricted > 0 {
70                return ad_restricted;
71            }
72            return total_ad;
73        }
74    }
75    fields.dp.as_i32().max(0)
76}
77
78fn is_het_or_hom_var(gt: &Genotype) -> bool {
79    matches!(gt.alleles.as_slice(), [0, 1] | [1, 0] | [1, 1])
80}

82pub fn annotate_hc_variant_site(
83    region: Option<&AssemblyRegion>,
84    position_1based: u64,
85    ref_allele: &str,
86    alt_allele: &str,
87    genotype: &RegionGenotypeResult,
88    _config: &HcGenotypingConfig,
89) -> GatkResult<HcVariantSiteAnnotations> {
90    let gl_for_qual = genotype_log10_likelihoods_after_java_genotype_pl_roundtrip(
91        &genotype.genotype_log10_likelihoods,
92    );
93    let qual = qual_from_af_calculation(&gl_for_qual)?;
94    let best_idx =
95        crate::genotyping::biallelic_genotype_index_from_pl(&genotype.format.pl).as_usize();
96    let gt = genotype_from_index(best_idx);
97    let dp = genotype.format.dp.as_i32().max(0);
98    let qd_depth = qd_depth_for_variant(&gt, &genotype.format);
99    let (ac, af, an, mleac, mleaf) = mle_alleles_from_genotype_index(best_idx);
100    let (ref_fw, ref_rv, alt_fw, alt_rv, mq_sum, mq_n) =
101        read_strand_evidence_at_site(region, position_1based, ref_allele, alt_allele);
102    let (ref_positions, alt_positions) =
103        read_offset_evidence_at_site(region, position_1based, ref_allele, alt_allele);
104    let fs = fisher_strand::fisher_strand_statistic(ref_fw, ref_rv, alt_fw, alt_rv);
105    let sor = strand_odds_ratio::strand_odds_ratio(ref_fw, ref_rv, alt_fw, alt_rv);
106    let rp = read_pos_rank_sum::read_pos_rank_sum(&ref_positions, &alt_positions);
107    let qd = qual_by_depth::qual_by_depth(qual, qd_depth);
108    let mq = if mq_n > 0 {
109        mq_sum as f64 / mq_n as f64
110    } else {
111        0.0
112    };
113    let (ref_n, het_n, hom_alt_n) = genotype_counts_from_index(best_idx); // usize diploid index
114    let excess_het_phred = excess_het::excess_heterozygosity_phred(ref_n, het_n, hom_alt_n);
115    let inbreeding_coeff = if ref_n + het_n + hom_alt_n > 0 {
116        1.0 - (het_n as f64) / (ref_n + het_n + hom_alt_n) as f64
117    } else {
118        0.0
119    };
120    Ok(HcVariantSiteAnnotations {
121        qual,
122        ac,
123        af,
124        an,
125        dp,
126        excess_het: excess_het_phred,
127        fs,
128        mleac,
129        mleaf,
130        mq,
131        qd,
132        sor,
133        read_pos_rank_sum: rp,
134        inbreeding_coeff,
135    })
136}

138fn read_offset_evidence_at_site(
139    region: Option<&AssemblyRegion>,
140    position_1based: u64,
141    ref_allele: &str,
142    alt_allele: &str,
143) -> (Vec<f64>, Vec<f64>) {
144    let mut ref_pos = Vec::new();
145    let mut alt_pos = Vec::new();
146    let ref_b = ref_allele.as_bytes().first().copied().unwrap_or(b'N');
147    let alt_b = alt_allele.as_bytes().first().copied().unwrap_or(b'N');
148    let pos = position_1based as i32;
149    let Some(region) = region else {
150        return (ref_pos, alt_pos);
151    };
152    for rec in &region.reads {
153        let read_start = rec.pos() + 1;
154        let Some(base) = read_base_at_ref_coord_1based(rec, pos) else {
155            continue;
156        };
157        let offset = (position_1based as i64 - read_start) as f64;
158        if base.eq_ignore_ascii_case(&alt_b) {
159            alt_pos.push(offset);
160        } else if base.eq_ignore_ascii_case(&ref_b) {
161            ref_pos.push(offset);
162        }
163    }
164    (ref_pos, alt_pos)
165}

167fn mle_alleles_from_genotype_index(best: usize) -> (i32, f64, i32, i32, f64) {
168    let an = 2;
169    match best {
170        0 => (0, 0.0, an, 0, 0.0),
171        1 => (1, 0.5, an, 1, 0.5),
172        _ => (2, 1.0, an, 2, 1.0),
173    }
174}
175
176fn genotype_counts_from_index(best: usize) -> (u32, u32, u32) {
177    match best {
178        0 => (1, 0, 0),
179        1 => (0, 1, 0),
180        _ => (0, 0, 1),
181    }
182}

184fn read_strand_evidence_at_site(
185    region: Option<&AssemblyRegion>,
186    position_1based: u64,
187    ref_allele: &str,
188    alt_allele: &str,
189) -> (u32, u32, u32, u32, u64, u32) {
190    let mut ref_fw = 0u32;
191    let mut ref_rv = 0u32;
192    let mut alt_fw = 0u32;
193    let mut alt_rv = 0u32;
194    let mut mq_sum = 0u64;
195    let mut mq_n = 0u32;
196    let ref_b = ref_allele.as_bytes().first().copied().unwrap_or(b'N');
197    let alt_b = alt_allele.as_bytes().first().copied().unwrap_or(b'N');
198    let pos = position_1based as i32;
199    let Some(region) = region else {
200        return (ref_fw, ref_rv, alt_fw, alt_rv, mq_sum, mq_n);
201    };
202    for rec in &region.reads {
203        let Some(base) = read_base_at_ref_coord_1based(rec, pos) else {
204            continue;
205        };
206        let reverse = rec.flags() & FLAG_REVERSE != 0;
207        let supports_alt = base.eq_ignore_ascii_case(&alt_b);
208        let supports_ref = base.eq_ignore_ascii_case(&ref_b);
209        if supports_alt {
210            if reverse {
211                alt_rv += 1;
212            } else {
213                alt_fw += 1;
214            }
215            mq_sum += u64::from(rec.mapq());
216            mq_n += 1;
217        } else if supports_ref {
218            if reverse {
219                ref_rv += 1;
220            } else {
221                ref_fw += 1;
222            }
223        }
224    }
225    (ref_fw, ref_rv, alt_fw, alt_rv, mq_sum, mq_n)
226}

228fn genotype_from_index(best: usize) -> Genotype {
229    let alleles = match best {
230        0 => vec![0, 0],
231        1 => vec![0, 1],
232        _ => vec![1, 1],
233    };
234    Genotype {
235        alleles,
236        phased: false,
237    }
238}
