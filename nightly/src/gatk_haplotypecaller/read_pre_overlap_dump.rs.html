1//! PRE.4 gate dump: `read-pre-overlap`.
2
3use crate::fragment_overlap::{
4    clean_overlapping_read_pairs, format_quals, overlapping_pairs_indices,
5};
6use gatk_common::{GatkError, GatkResult};
7use rust_htslib::bam;
8use rust_htslib::bam::Read as _;
9use std::collections::HashMap;
10use std::io::Write;
11use std::path::Path;
12
13pub fn dump_read_pre_overlap_tsv(alignment_path: &Path, out: &mut impl Write) -> GatkResult<()> {
14    writeln!(out, "qname\tqual_in\tqual_out\toverlap_pair")
15        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
16
17    let mut reader = bam::Reader::from_path(alignment_path)
18        .map_err(|e| GatkError::generic(format!("open: {e}")))?;
19    let mut records = Vec::new();
20    let mut qual_in = HashMap::new();
21    for res in reader.records() {
22        let rec = res.map_err(|e| GatkError::generic(format!("read: {e}")))?;
23        qual_in.insert(rec.qname().to_vec(), format_quals(&rec));
24        records.push(rec);
25    }
26    records.sort_by_key(|r| (r.tid(), r.pos()));
27    let pair_indices = overlapping_pairs_indices(&records)?;
28    let mut in_pair = std::collections::HashSet::new();
29    for (a, b) in &pair_indices {
30        in_pair.insert(records[*a].qname().to_vec());
31        in_pair.insert(records[*b].qname().to_vec());
32    }
33    clean_overlapping_read_pairs(&mut records, true)?;
34    for rec in &records {
35        let qname = rec.qname();
36        let qual_in_s = qual_in.get(qname).map(String::as_str).unwrap_or("");
37        writeln!(
38            out,
39            "{}\t{}\t{}\t{}",
40            String::from_utf8_lossy(qname),
41            qual_in_s,
42            format_quals(rec),
43            in_pair.contains(qname)
44        )
45        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
46    }
47    Ok(())
48}
