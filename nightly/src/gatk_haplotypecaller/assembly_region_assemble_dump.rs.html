24/// Which assembly region to dump (first match in interval stream).
25/// # Invariants
26/// `Active` selects first `ActiveFull` disposition; `Inactive` dumps metadata only (no assembly).
27/// # Ownership
28/// [`Copy`] CLI target enum.
29/// # Mutation
30/// Immutable dump selector.
31/// # Biological assumptions
32/// None — parity dump routing.
33/// # Java equivalence
34/// Rust-native E2E dump selector over GATK `callRegion` active vs inactive branches.
35#[derive(Debug, Clone, Copy, PartialEq, Eq)]
36pub enum AssemblyRegionHaplotypeTarget {
37    /// First region with `ActiveFull` disposition (GATK `assembleReads` path).
38    Active,
39    /// First inactive region — metadata only, no assembly (`E2E.5`).
40    Inactive,
41}
42
43impl AssemblyRegionHaplotypeTarget {
44    pub fn parse_cli(s: &str) -> Option<Self> {
45        match s {
46            "active" => Some(Self::Active),
47            "inactive" => Some(Self::Inactive),
48            _ => None,
49        }
50    }
51}

53fn write_region_meta(
54    out: &mut impl Write,
55    region: &AssemblyRegion,
56    status: &str,
57    kmer_size: usize,
58) -> GatkResult<()> {
59    writeln!(out, "region_contig\t{}", region.contig)
60        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
61    writeln!(out, "region_start\t{}", region.start.get())
62        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
63    writeln!(out, "region_end\t{}", region.end.get())
64        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
65    writeln!(out, "is_active\t{}", region.is_active)
66        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
67    writeln!(out, "status\t{status}").map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
68    writeln!(out, "kmer_size\t{kmer_size}")
69        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
70    Ok(())
71}

73/// First matching region in interval stream → haplotype TSV (E2E / `callRegion` assembly slice).
74pub fn dump_assembly_region_haplotypes_tsv(
75    ref_fasta: &Path,
76    bam_path: &Path,
77    interval_cli: &str,
78    padding: u64,
79    target: AssemblyRegionHaplotypeTarget,
80    out: &mut impl Write,
81) -> GatkResult<()> {
82    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
83    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
84    let filters = ReadFilterParams::gatk_standard_hc();
85    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
86    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
87    let regions = flatten_assembly_regions(&walk);
88    let mut ref_cache = ReferenceWindowCache::new(ref_fasta.to_path_buf(), 4);
89
90    let region = match target {
91        AssemblyRegionHaplotypeTarget::Active => select_region_for_asm_dump(&regions)
92            .ok_or_else(|| GatkError::argument("no active assembly region in interval"))?,
93        AssemblyRegionHaplotypeTarget::Inactive => regions
94            .iter()
95            .find(|r| !r.is_active)
96            .ok_or_else(|| GatkError::argument("no inactive assembly region in interval"))?,
97    };
98
99    if target == AssemblyRegionHaplotypeTarget::Inactive || !region.is_active {
100        write_region_meta(out, region, "inactive_skip", 0)?;
101        return Ok(());
102    }
103
104    let reference = assembly_reference_read(&dict, &mut ref_cache, region)?;
105    let assemble_args = AssembleReadsArgs::default();
106    let assembly_set = crate::assembly_based_caller::assemble_reads(
107        region,
108        &dict,
109        &mut ref_cache,
110        &assemble_args,
111    )?;
112    // Java `emitAssemblyRegionHaplotypesFromMaterial`: `isVariationPresent` is still false
113    // after `runLocalAssembly` (SeqGraph `findBestPaths` uses `add(h)` only). Kmer metadata is 0.
114    write_region_meta(out, region, "just_assembled_reference", 0)?;
115    let ref_bases = reference.bases.as_bytes();
116    let haplotypes = if assembly_set.haplotypes.len() == 1
117        && assembly_set.haplotypes[0].is_reference
118        && assembly_set.haplotypes[0].bases.as_slice() != ref_bases
119    {
120        // Cyclic k=85 graphs can mark a stitched ref path `is_reference` with non-ref-length bases.
121        vec![reference_haplotype_for_assembly_region(
122            &reference, region, &dict,
123        )]
124    } else {
125        assembly_set.haplotypes
126    };
127    write_haplotype_rows_with_ref_recovery(&haplotypes, ref_bases, &reference.bases, out)
128}

130/// Per-kmer SeqGraph probe for the first active region (assembly divergence diagnostics).
131pub fn dump_assembly_region_kmer_probe_tsv(
132    ref_fasta: &Path,
133    bam_path: &Path,
134    interval_cli: &str,
135    padding: u64,
136    out: &mut impl Write,
137) -> GatkResult<()> {
138    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
139    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
140    let filters = ReadFilterParams::gatk_standard_hc();
141    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
142    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
143    let regions = flatten_assembly_regions(&walk);
144    let mut ref_cache = ReferenceWindowCache::new(ref_fasta.to_path_buf(), 4);
145
146    let region = select_region_for_asm_dump(&regions)
147        .ok_or_else(|| GatkError::argument("no assembly region with reads in interval"))?;
148    if !matches!(
149        call_disposition(region),
150        AssemblyRegionCallDisposition::ActiveFull
151    ) {
152        writeln!(out, "warn\tregion_inactive_kmer_probe_on_inactive_reads")
153            .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
154    }
155
156    let reference = assembly_reference_read(&dict, &mut ref_cache, region)?;
157    let assemble_args = AssembleReadsArgs::default();
158    let reads = assembly_reads_for_java_materialize_dump(&region.reads);
159
160    writeln!(out, "region_contig\t{}", region.contig)
161        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
162    writeln!(out, "region_start\t{}", region.start.get())
163        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
164    writeln!(out, "region_end\t{}", region.end.get())
165        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
166    writeln!(out, "padded_ref_len\t{}", reference.bases.len())
167        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
168    writeln!(out, "read_count\t{}", reads.len())
169        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
170
171    let rows = probe_seq_graph_kmer_attempts(&reference, &reads, &assemble_args.assembler)?;
172    writeln!(
173        out,
174        "phase\tkmer\tallow_low_complexity\tallow_non_unique\toutcome\tthread_nodes\tthread_edges\tcleanup_status\thas_ref_source\thas_ref_sink\tref_path_matches\tkbest_paths\textracted_haps\tnon_ref_haps\tpath_bases_len\tpath_eq_ref"
175    )
176    .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
177    for r in &rows {
178        writeln!(
179            out,
180            "{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}",
181            r.phase,
182            r.kmer_size,
183            r.allow_low_complexity,
184            r.allow_non_unique_ref,
185            r.outcome,
186            r.thread_nodes,
187            r.thread_edges,
188            r.cleanup_status,
189            r.has_ref_source,
190            r.has_ref_sink,
191            r.ref_path_matches,
192            r.kbest_paths,
193            r.extracted_haps,
194            r.non_ref_haps,
195            r.path_bases_len,
196            r.path_eq_ref_bases,
197        )
198        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
199    }
200    Ok(())
