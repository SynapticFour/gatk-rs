1//! CombineGVCFs — merge per-sample gVCFs into a multi-sample gVCF (no joint genotyping).
2//! Observable Java contract (GATK 4.4 `CombineGVCFs`):
3//! Union sample columns from input gVCFs.
4//! At each genomic breakpoint induced by record starts / `END`+1, merge overlapping
5//! reference-confidence VCs via `ReferenceConfidenceVariantContextMerger` semantics
6//! ([`crate::ref_confidence_merger::merge_reference_confidence`]).
7//! Remap PL/AD onto the unified allele list; missing samples emit no-call genotypes.
8//! Hom-ref `<NON_REF>` blocks carry `INFO/END`; variant sites emit discrete rows.
9//! This module is a Rust-native walker over that merger kernel — not a Java class clone.
10

63/// Run CombineGVCFs from filesystem paths (CLI entry).
64pub fn run_combine_gvcfs(args: &CombineGvcfsArgs) -> GatkResult<()> {
65    if args.variant_paths.is_empty() {
66        return Err(GatkError::argument(
67            "CombineGVCFs requires at least one -V input",
68        ));
69    }
70    let dictionary = SequenceDictionary::from_fasta_path(&args.reference)?;
71    let clip = resolve_clip_spans(&dictionary, args.intervals.as_deref())?;
72    let mut loaded = Vec::with_capacity(args.variant_paths.len());
73    for path in &args.variant_paths {
74        loaded.push(load_gvcf(path)?);
75    }
76    let records = combine_loaded_gvcfs(&loaded, Some(args.reference.as_path()), clip.as_ref())?;
77    let header = build_combined_header(&loaded, Some(args.reference.as_path()))?;
78    let mut writer = VcfWriter::new(&args.output, header)?;
79    writer.write_header()?;
80    writer.write_records(&records)?;
81    Ok(())
82}

84/// Merge already-loaded gVCFs (unit-test / harness entry).
85/// `reference` supplies bases when no input VC starts at a locus (spanning-only merges).
86/// `clip` restricts emission to inclusive spans per contig when present.
87pub fn combine_loaded_gvcfs(
88    inputs: &[LoadedGvcf],
89    reference: Option<&Path>,
90    clip: Option<&HashMap<String, Vec<Span>>>,
91) -> GatkResult<Vec<VcfRecord>> {
92    if inputs.is_empty() {
93        return Err(GatkError::argument("CombineGVCFs: no inputs"));
94    }
95    let sample_order: Vec<String> = inputs.iter().map(|g| g.sample_name.clone()).collect();
96    let mut by_contig: BTreeMap<String, Vec<(usize, &GvcfSite)>> = BTreeMap::new();
97    for (si, g) in inputs.iter().enumerate() {
98        for site in &g.sites {
99            by_contig
100                // CLONE: needed because owned HashMap entry key.
101                .entry(site.contig.clone())
102                .or_default()
103                .push((si, site));
104        }
105    }
106
107    let mut out = Vec::new();
108    for (contig, sites) in by_contig {
109        let clip_spans = clip.and_then(|m| m.get(&contig)).map(|v| v.as_slice());
110        if clip.is_some() && clip_spans.is_none() {
111            continue;
112        }
113        let breakpoints = collect_breakpoints(&sites, clip_spans);
114        if breakpoints.len() < 2 {
115            continue;
116        }
117        let mut contig_recs = Vec::new();
118        for w in breakpoints.windows(2) {
119            let loc_start = w[0];
120            let loc_end = w[1].saturating_sub(1);
121            if loc_end < loc_start {
122                continue;
123            }
124            if let Some(spans) = clip_spans {
125                if !spans
126                    .iter()
127                    .any(|s| overlaps(s.start, s.end, loc_start, loc_end))
128                {
129                    continue;
130                }
131            }
132
133            let covering = covering_sites(&sites, loc_start);
134            if covering.is_empty() {
135                continue;
136            }
137
138            let mut merge_inputs = Vec::with_capacity(covering.len());
139            let mut format_carry: HashMap<String, SampleData> = HashMap::new();
140            for &(si, site) in &covering {
141                let sample = &inputs[si].sample_name;
142                // CLONE: needed because owned HashMap/BTree/HashSet key or value.
143                format_carry.insert(sample.clone(), site.sample.clone());
144                merge_inputs.push(site_to_merge_input(site, sample));
145            }
146            // Java `CombineGVCFs.endPreviousStates` walks overlapping VCs from last→first
147            // before `ReferenceConfidenceVariantContextMerger.merge`, which fixes ALT order.
148            merge_inputs.reverse();
149
150            let ref_base = if merge_inputs.iter().any(|v| v.start == loc_start) {
151                None
152            } else {
153                match reference {
154                    Some(path) => Some(reference_base_at_1based(path, &contig, loc_start)?),
155                    None => {
156                        // Tests without FASTA: use first covering site's REF first base.
157                        Some(covering[0].1.ref_allele.as_bytes()[0].to_ascii_uppercase())
158                    }
159                }
160            };
161
162            let merged = match merge_reference_confidence(
163                &contig,
164                loc_start,
165                &merge_inputs,
166                ref_base,
167                false,
168                false,
169            )? {
170                Some(m) => m,
171                None => continue,
172            };
173
174            let rec = merge_result_to_record(&merged, loc_end, &sample_order, &format_carry)?;
175            contig_recs.push(rec);
176        }
177        coalesce_homref_blocks(&mut contig_recs);
178        out.extend(contig_recs);
179    }
180    Ok(out)
181}

183/// Load a single-sample (or first-sample) gVCF from disk.
184pub fn load_gvcf(path: &Path) -> GatkResult<LoadedGvcf> {
185    let mut reader = VcfReader::from_file(path)?;
186    let header = reader.header().clone();
187    let sample_name =
188        header.samples.first().cloned().ok_or_else(|| {
189            GatkError::argument(format!("gVCF has no samples: {}", path.display()))
190        })?;
191    let mut sites = Vec::new();
192    while let Some(rec) = reader.read_next_record()? {
193        sites.push(record_to_site(rec, &sample_name, path)?);
194    }
195    Ok(LoadedGvcf {
196        path: path.to_path_buf(),
197        sample_name,
198        header,
199        sites,
200    })
201}

214fn resolve_clip_spans(
215    dictionary: &SequenceDictionary,
216    intervals: Option<&str>,
217) -> GatkResult<Option<HashMap<String, Vec<Span>>>> {
218    let Some(iv) = intervals else {
219        return Ok(None);
220    };
221    let specs = gatk_core::reference::parse_intervals_cli_string(dictionary, iv)?;
222    let mut map: HashMap<String, Vec<Span>> = HashMap::new();
223    for spec in specs {
224        let start = spec.start.unwrap_or(1);
225        let end = match spec.end {
226            Some(e) => e,
227            None => {
228                let c = dictionary.contig(&spec.contig).ok_or_else(|| {
229                    GatkError::argument(format!(
230                        "whole-contig -L {} needs contig length in dictionary",
231                        spec.contig
232                    ))
233                })?;
234                c.length
235            }
236        };
237        map.entry(spec.contig)
238            .or_default()
239            .push(Span { start, end });
240    }
241    Ok(Some(map))
242}

248fn record_end(rec: &VcfRecord) -> u64 {
249    for info in &rec.info {
250        if let InfoValue::Integer(id, vals) = info {
251            if id == "END" {
252                if let Some(&e) = vals.first() {
253                    return e as u64;
254                }
255            }
256        }
257    }
258    rec.position + rec.reference.len().saturating_sub(1) as u64
259}

261fn record_to_site(rec: VcfRecord, _sample_name: &str, path: &Path) -> GatkResult<GvcfSite> {
262    let end = record_end(&rec);
263    let sample = rec.samples.first().cloned().unwrap_or_else(|| SampleData {
264        gt: None,
265        gq: None,
266        dp: None,
267        ad: None,
268        pl: None,
269        other: Vec::new(),
270    });
271    Ok(GvcfSite {
272        contig: rec.chromosome,
273        start: rec.position,
274        end,
275        ref_allele: rec.reference,
276        alts: rec.alternate,
277        sample,
278        source: path.display().to_string(),
279    })

282fn site_to_merge_input(site: &GvcfSite, sample: &str) -> MergeVcInput {
283    let mut alleles = Vec::with_capacity(1 + site.alts.len());
284    alleles.push(MergeAllele {
285        bases: site.ref_allele.clone(),
286        is_reference: true,
287    });
288    for alt in &site.alts {
289        alleles.push(MergeAllele {
290            // CLONE: needed because owned haplotypes for scoring call.
291            bases: alt.clone(),
292            is_reference: false,
293        });
294    }
295    let pl = site.sample.pl.as_ref().map(|v| {
296        v.iter()
297            .map(|&x| i32::try_from(x).unwrap_or(i32::MAX))
298            .collect()
299    });
300    let ad = site.sample.ad.as_ref().map(|v| {
301        v.iter()
302            .map(|&x| i32::try_from(x).unwrap_or(i32::MAX))
303            .collect()
304    });
305    MergeVcInput {
306        source: site.source.clone(),
307        start: site.start,
308        alleles,
309        genotypes: vec![MergeGenotype {
310            sample: sample.to_string(),
311            pl,
312            ad,
313        }],
314    }
315}

317fn collect_breakpoints(sites: &[(usize, &GvcfSite)], clip_spans: Option<&[Span]>) -> Vec<u64> {
318    let mut bp = BTreeSet::new();
319    for (_, s) in sites {
320        bp.insert(s.start);
321        bp.insert(s.end.saturating_add(1));
322        // Spanning variant bodies need per-base merges (`*` remapping).
323        let has_variant_alt = s.alts.iter().any(|a| a != NON_REF_ALLELE);
324        if has_variant_alt && s.end > s.start {
325            for p in (s.start + 1)..=s.end {
326                bp.insert(p);
327            }
328        }
329    }
330    if let Some(spans) = clip_spans {
331        for s in spans {
332            bp.insert(s.start);
333            bp.insert(s.end.saturating_add(1));
334        }
335    }
336    bp.into_iter().collect()
337}

339fn covering_sites<'a>(sites: &'a [(usize, &GvcfSite)], loc: u64) -> Vec<(usize, &'a GvcfSite)> {
340    let mut out = Vec::new();
341    let mut seen_sample = BTreeSet::new();
342    for &(si, site) in sites {
343        if site.start <= loc && loc <= site.end && seen_sample.insert(si) {
344            out.push((si, site));
345        }
346    }
347    out
348}

354fn merge_result_to_record(
355    merged: &crate::ref_confidence_merger::RefConfidenceMergeResult,
356    interval_end: u64,
357    sample_order: &[String],
358    format_carry: &HashMap<String, SampleData>,
359) -> GatkResult<VcfRecord> {
360    let ref_allele = merged
361        .alleles
362        .first()
363        .cloned()
364        .ok_or_else(|| GatkError::argument("merged alleles empty"))?;
365    let alts: Vec<String> = merged.alleles.iter().skip(1).cloned().collect();
366    let n_alleles = merged.alleles.len();
367
368    let by_name: HashMap<&str, &crate::ref_confidence_merger::MergeGenotypeOut> = merged
369        .genotypes
370        .iter()
371        .map(|g| (g.name.as_str(), g))
372        .collect();
373
374    let mut samples = Vec::with_capacity(sample_order.len());
375    for name in sample_order {
376        if let Some(g) = by_name.get(name.as_str()) {
377            let carry = format_carry.get(name);
378            let pl_u32: Option<Vec<u32>> = g.pl.as_ref().map(|pl| {
379                pl.iter()
380                    .map(|&x| u32::try_from(x.max(0)).unwrap_or(u32::MAX))
381                    .collect()
382            });
383            let ad_u32: Option<Vec<u32>> = g.ad.as_ref().map(|ad| {
384                ad.iter()
385                    .map(|&x| u32::try_from(x.max(0)).unwrap_or(0))
386                    .collect()
387            });
388            let gt = pl_u32
389                .as_ref()
390                .and_then(|pl| best_diploid_gt_from_pl_u32(pl.as_slice(), n_alleles));
391            // CLONE: needed because multi-owner or ownership transfer into new structure.
392            let other = carry.map(|c| c.other.clone()).unwrap_or_default();
393            // Preserve MIN_DP and any non-standard FORMAT keys from the covering record.
394            samples.push(SampleData {
395                gt,
396                gq: carry.and_then(|c| c.gq),
397                dp: carry.and_then(|c| c.dp),
398                ad: ad_u32,
399                pl: pl_u32,
400                other,
401            });
402        } else {
403            samples.push(SampleData {
404                gt: None,
405                gq: None,
406                dp: None,
407                ad: None,
408                pl: None,
409                other: Vec::new(),
410            });
411        }
412    }
413
414    let mut info = Vec::new();
415    let block = is_homref_nonref_only(&alts) && ref_allele.len() == 1;
416    if block && interval_end > merged.pos {
417        info.push(InfoValue::Integer(
418            "END".to_string(),
419            vec![interval_end as i32],
420        ));
421    }
422
423    Ok(VcfRecord {
424        chromosome: merged.contig.clone(),
425        position: merged.pos,
426        id: ".".to_string(),
427        reference: ref_allele,
428        alternate: alts,
429        quality: None,
430        filter: vec![".".to_string()],
431        info,
432        format: vec![
433            "GT".to_string(),
434            "GQ".to_string(),
435            "DP".to_string(),
436            "AD".to_string(),
437            "PL".to_string(),
438            "MIN_DP".to_string(),
439        ],
440        samples,
441    })
442}

444fn best_diploid_gt_from_pl_u32(pl: &[u32], n_alleles: usize) -> Option<Genotype> {
445    if n_alleles == 0 || pl.is_empty() {
446        return None;
447    }
448    let mut pairs = Vec::new();
449    for j in 0..n_alleles {
450        for i in 0..=j {
451            pairs.push((i, j));
452        }
453    }
454    if pl.len() != pairs.len() {
455        return None;
456    }
457    let (best_idx, _) = pl.iter().enumerate().min_by_key(|(_, &v)| v)?;
458    let (a, b) = pairs[best_idx];
459    Some(Genotype {
460        alleles: vec![a as i32, b as i32],
461        phased: false,
462    })
463}

465/// Merge adjacent hom-ref `<NON_REF>` blocks with identical per-sample FORMAT payloads.
466fn coalesce_homref_blocks(recs: &mut Vec<VcfRecord>) {
467    if recs.len() < 2 {
468        return;
469    }
470    let mut out: Vec<VcfRecord> = Vec::with_capacity(recs.len());
471    for rec in recs.drain(..) {
472        let can_merge = out.last().is_some_and(|prev| {
473            is_homref_nonref_only(&prev.alternate)
474                && is_homref_nonref_only(&rec.alternate)
475                && prev.chromosome == rec.chromosome
476                && prev.reference == rec.reference
477                && samples_pl_ad_eq(&prev.samples, &rec.samples)
478                && record_end(prev) + 1 == rec.position
479        });
480        if can_merge {
481            let new_end = record_end(&rec);
482            if let Some(prev) = out.last_mut() {
483                set_end(prev, new_end);
484            }
485        } else {
486            out.push(rec);
487        }
488    }
489    *recs = out;
490}
491
492fn samples_pl_ad_eq(a: &[SampleData], b: &[SampleData]) -> bool {
493    if a.len() != b.len() {
494        return false;
495    }
496    a.iter().zip(b.iter()).all(|(x, y)| {
497        x.pl == y.pl && x.ad == y.ad && x.dp == y.dp && x.gq == y.gq && x.other == y.other
498    })
499}

501fn set_end(rec: &mut VcfRecord, end: u64) {
502    rec.info
503        .retain(|v| !matches!(v, InfoValue::Integer(id, _) if id == "END"));
504    if end > rec.position {
505        rec.info
506            .push(InfoValue::Integer("END".to_string(), vec![end as i32]));
507    }
508}

510fn build_combined_header(inputs: &[LoadedGvcf], reference: Option<&Path>) -> GatkResult<VcfHeader> {
511    let mut header = VcfHeader::default();
512    header.file_format = "VCFv4.2".to_string();
513    header.source = Some("gatk-rs CombineGVCFs".to_string());
514    if let Some(r) = reference {
515        header.reference = Some(r.display().to_string());
516    }
517
518    let mut contig_ids = BTreeSet::new();
519    for g in inputs {
520        for c in &g.header.contigs {
521            if c.id.is_empty() {
522                continue;
523            }
524            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
525            if contig_ids.insert(c.id.clone()) {
526                // CLONE: needed because owned element into collection.
527                header.contigs.push(c.clone());
528            }
529        }
530    }
531    if header.contigs.is_empty() {
532        let mut seen = BTreeSet::new();
533        for g in inputs {
534            for s in &g.sites {
535                // CLONE: needed because owned HashMap/BTree/HashSet key or value.
536                if seen.insert(s.contig.clone()) {
537                    header.contigs.push(Contig {
538                        // CLONE: needed because VCF header contig id is owned separately from `seen`.
539                        id: s.contig.clone(),
540                        length: None,
541                        md5: None,
542                        assembly: None,
543                        species: None,
544                        uri: None,
545                    });
546                }
547            }
548        }
549    }
550
551    header.info_fields = vec![
552        InfoField {
553            id: "END".to_string(),
554            number: "1".to_string(),
555            type_field: "Integer".to_string(),
556            description: "Stop position of the interval".to_string(),
557            source: None,
558            version: None,
559        },
560        InfoField {
561            id: "DP".to_string(),
562            number: "1".to_string(),
563            type_field: "Integer".to_string(),
564            description: "Approximate read depth; some reads may have been filtered".to_string(),
565            source: None,
566            version: None,
567        },
568    ];
569    header.format_fields = vec![
570        FormatField {
571            id: "GT".to_string(),
572            number: "1".to_string(),
573            type_field: "String".to_string(),
574            description: "Genotype".to_string(),
575        },
576        FormatField {
577            id: "GQ".to_string(),
578            number: "1".to_string(),
579            type_field: "Integer".to_string(),
580            description: "Genotype Quality".to_string(),
581        },
582        FormatField {
583            id: "DP".to_string(),
584            number: "1".to_string(),
585            type_field: "Integer".to_string(),
586            description:
587                "Approximate read depth (reads with MQ=255 or with bad mates are filtered)"
588                    .to_string(),
589        },
590        FormatField {
591            id: "AD".to_string(),
592            number: "R".to_string(),
593            type_field: "Integer".to_string(),
594            description: "Allelic depths for the ref and alt alleles in the order listed"
595                .to_string(),
596        },
597        FormatField {
598            id: "PL".to_string(),
599            number: "G".to_string(),
600            type_field: "Integer".to_string(),
601            description: "Normalized, Phred-scaled likelihoods for genotypes".to_string(),
602        },
603        FormatField {
604            id: "MIN_DP".to_string(),
605            number: "1".to_string(),
606            type_field: "Integer".to_string(),
607            description: "Minimum DP observed within the GVCF block".to_string(),
608        },
609    ];
610    header.samples = inputs.iter().map(|g| g.sample_name.clone()).collect();
611    Ok(header)
612}
