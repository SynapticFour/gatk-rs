12/// Inputs for region genotyping (active-window EventMap walk).
13/// # Invariants
14/// Likelihood indices align with `likelihood_reads` / `haplotypes`.
15/// Active window and pad starts are 1-based; `ref_bytes` covers the padded span.
16/// # Ownership
17/// Lifetime-bound borrows of likelihoods, BAM records, haplotypes, and events.
18/// # Mutation
19/// View-only bundle; engines must not mutate borrowed slices.
20/// # Biological assumptions
21/// One assembly region ready for `assignGenotypeLikelihoods`-style genotyping.
22/// # Java equivalence
23/// Rust-native input bundle for GATK genotyping engine region walk (L7-B3).
24pub struct GenotypeSiteRegion<'a> {
25    pub likelihoods: &'a [RegionReadLikelihood],
26    pub likelihood_reads: &'a [SharedBamRecord],
27    pub pileup_reads: &'a [SharedBamRecord],
28    pub supplemental_pileup_reads: Option<&'a [SharedBamRecord]>,
29    pub haplotypes: &'a [Haplotype],
30    pub ref_bytes: &'a [u8],
31    pub pad_start_1based: GenomePosition,
32    pub full_reference_bases: &'a [u8],
33    pub full_reference_pad_1based: GenomePosition,
34    pub active_start_1based: GenomePosition,
35    pub active_end_1based: GenomePosition,
36    pub contig: &'a str,
37    pub max_mnp_distance: usize,
38    pub stored_events: &'a [VariationEvent],
39    pub graph_events: &'a [VariationEvent],
40}
