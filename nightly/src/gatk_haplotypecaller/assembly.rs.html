335    /// GATK `BaseGraph.removePathsNotConnectedToRef`.
336    pub fn remove_paths_not_connected_to_ref(&mut self) -> GatkResult<()> {
337        let source = self
338            .reference_source_vertex()
339            .ok_or_else(|| GatkError::algorithm("removePathsNotConnectedToRef: no ref source"))?;
340        let sink = self
341            .reference_sink_vertex()
342            .ok_or_else(|| GatkError::algorithm("removePathsNotConnectedToRef: no ref sink"))?;
343
344        let mut from_source = HashSet::new();
345        let mut stack = vec![source];
346        from_source.insert(source);
347        while let Some(v) = stack.pop() {
348            for to in self.outgoing_nodes(v) {
349                if from_source.insert(to) {
350                    stack.push(to);
351                }
352            }
353        }
354
355        let mut from_sink = HashSet::new();
356        let mut stack = vec![sink];
357        from_sink.insert(sink);
358        while let Some(v) = stack.pop() {
359            for from in self.incoming_nodes(v) {
360                if from_sink.insert(from) {
361                    stack.push(from);
362                }
363            }
364        }
365
366        from_source.retain(|v| from_sink.contains(v));
367        let remove: HashSet<usize> = (0..self.node_count())
368            .filter(|v| !from_source.contains(v))
369            .collect();
370        self.remove_nodes(&remove);
371        self.cleanup_isolated_nodes();
372        Ok(())
373    }

486    /// Remove short dead-end chains with only single-path continuation.
487    pub fn remove_dangling_paths(&mut self, max_nodes: usize) {
488        if max_nodes == 0 {
489            return;
490        }
491        let mut remove_nodes = HashSet::new();
492        for node_id in 0..self.nodes.len() {
493            let out_degree = self.outgoing.get(&node_id).map(|x| x.len()).unwrap_or(0);
494            if out_degree != 0 {
495                continue;
496            }
497            let mut path = vec![node_id];
498            let mut cur = node_id;
499            while path.len() < max_nodes {
500                // Lifetime: only need to inspect predecessor ids; borrow the set.
501                let Some(preds) = self.incoming.get(&cur) else {
502                    break;
503                };
504                if preds.len() != 1 {
505                    break;
506                }
507                let Some(&pred) = preds.iter().next() else {
508                    break;
509                };
510                let pred_out = self.outgoing.get(&pred).map(|x| x.len()).unwrap_or(0);
511                if pred_out != 1 {
512                    break;
513                }
514                path.push(pred);
515                cur = pred;
516            }
517            if path.len() <= max_nodes {
518                for n in path {
519                    remove_nodes.insert(n);
520                }
521            }
522        }
523        self.remove_nodes(&remove_nodes);
524    }

526    pub fn cleanup_isolated_nodes(&mut self) {
527        let mut keep = HashSet::new();
528        for (from, to) in self.edges.keys() {
529            keep.insert(*from);
530            keep.insert(*to);
531        }

533            .filter(|n| !keep.contains(n))
534            .collect::<HashSet<_>>();
535        self.remove_nodes(&remove);
536    }

538    pub(crate) fn remove_nodes(&mut self, remove: &HashSet<usize>) {
539        if remove.is_empty() {
540            return;
541        }
542        let old_nodes = std::mem::take(&mut self.nodes);
543        let mut new_nodes = Vec::new();
544        let mut remap = HashMap::new();
545        for n in old_nodes {
546            if !remove.contains(&n.id) {
547                let new_id = new_nodes.len();
548                remap.insert(n.id, new_id);
549                new_nodes.push(KmerNode {
550                    id: new_id,
551                    kmer: n.kmer,
552                    support: n.support,
553                });
554            }
555        }
556        let mut new_edges = HashMap::new();
557        let mut new_ref_edges = HashSet::new();
558        for ((from, to), support) in &self.edges {
559            if let (Some(&nf), Some(&nt)) = (remap.get(from), remap.get(to)) {
560                new_edges.insert((nf, nt), *support);
561                if self.ref_edges.contains(&(*from, *to)) {
562                    new_ref_edges.insert((nf, nt));
563                }
564            }
565        }
566        let new_ref_nodes: HashSet<usize> = self
567            .ref_nodes
568            .iter()
569            .filter_map(|n| remap.get(n).copied())
570            .collect();
571        self.nodes = new_nodes;
572        self.edges = new_edges;
573        self.ref_edges = new_ref_edges;
574        self.ref_nodes = new_ref_nodes;
575        self.kmer_to_id.clear();
576        self.outgoing.clear();
577        self.incoming.clear();
578        for n in &self.nodes {
579            // Arc clone — shared bytes with `nodes` (no second kmer allocation).
580            self.kmer_to_id.insert(std::sync::Arc::clone(&n.kmer), n.id);
581        }
582        for (from, to) in self.edges.keys() {
583            self.outgoing.entry(*from).or_default().insert(*to);
584            self.incoming.entry(*to).or_default().insert(*from);
585        }
586    }

588    pub fn extract_candidate_haplotypes(
589        &self,
590        max_haplotypes: usize,
591        max_bases: usize,
592    ) -> Vec<CandidateHaplotype> {
593        if self.nodes.is_empty() || max_haplotypes == 0 || max_bases == 0 {
594            return Vec::new();
595        }
596        let mut starts = (0..self.nodes.len())
597            .filter(|id| self.incoming.get(id).map(|x| x.is_empty()).unwrap_or(true))
598            .collect::<Vec<_>>();
599        if starts.is_empty() {
600            starts = (0..self.nodes.len()).collect();
601        }
602        starts.sort_unstable();
603
604        let mut heap: BinaryHeap<(u32, Reverse<Vec<u8>>)> = BinaryHeap::new();
605        for s in starts {
606            let mut visited = HashSet::new();
607            let mut current = self.nodes[s].kmer.to_vec(); // owned scratch for DFS path growth
608            self.dfs_haplotypes(s, &mut current, 0, max_bases, &mut visited, &mut heap);
609        }
610        let mut out = heap
611            .into_iter()
612            .map(|(support, Reverse(sequence))| CandidateHaplotype { sequence, support })
613            .collect::<Vec<_>>();
614        out.sort_by(|a, b| {
615            b.support
616                .cmp(&a.support)
617                .then_with(|| a.sequence.cmp(&b.sequence))
618        });
619        out.truncate(max_haplotypes);
620        out
621    }

623    /// Basic graph invariants used by phase-5 contracts.
624    pub fn validate_basic_invariants(&self) -> GatkResult<()> {
625        for n in &self.nodes {
626            if n.kmer.len() != self.kmer_size {
627                return Err(GatkError::algorithm(format!(
628                    "node {} has kmer length {}, expected {}",
629                    n.id,
630                    n.kmer.len(),
631                    self.kmer_size
632                )));
633            }
634        }
635        for ((from, to), support) in &self.edges {
636            if *support == 0 {
637                return Err(GatkError::algorithm(format!(
638                    "edge ({from}->{to}) has zero support"
639                )));
640            }
641            if *from >= self.nodes.len() || *to >= self.nodes.len() {
642                return Err(GatkError::algorithm(format!(
643                    "edge ({from}->{to}) points outside node set"
644                )));
645            }
646            let from_k = &self.nodes[*from].kmer;
647            let to_k = &self.nodes[*to].kmer;
648            if !from_k[1..].eq(&to_k[..self.kmer_size - 1]) {
649                return Err(GatkError::algorithm(format!(
650                    "edge ({from}->{to}) violates k-1 overlap invariant"
651                )));
652            }
653        }
654        if self.has_cycle_internal() {
655            return Err(GatkError::algorithm(
656                "graph contains a cycle; expected DAG after cleanup for current wave-B fixtures",
657            ));
658        }
659        Ok(())
660    }
661
662    fn has_cycle_internal(&self) -> bool {
663        let mut indeg = vec![0usize; self.nodes.len()];
664        for (_, to) in self.edges.keys() {
665            indeg[*to] += 1;
666        }
667        let mut stack = indeg
668            .iter()
669            .enumerate()
670            .filter_map(|(i, d)| if *d == 0 { Some(i) } else { None })
671            .collect::<Vec<_>>();
672        let mut seen = 0usize;
673        while let Some(n) = stack.pop() {
674            seen += 1;
675            if let Some(outs) = self.outgoing.get(&n) {
676                for to in outs {
677                    indeg[*to] -= 1;
678                    if indeg[*to] == 0 {
679                        stack.push(*to);
680                    }
681                }
682            }
683        }
684        seen != self.nodes.len()
685    }
686
687    fn dfs_haplotypes(
688        &self,
689        node: usize,
690        current: &mut Vec<u8>,
691        support_acc: u32,
692        max_bases: usize,
693        visited: &mut HashSet<usize>,
694        out: &mut BinaryHeap<(u32, Reverse<Vec<u8>>)>,
695    ) {
696        if current.len() > max_bases || visited.contains(&node) {
697            return;
698        }
699        visited.insert(node);
700        // Lifetime: DFS only reads adjacency; iterate borrowed successors.
701        let outs = self.outgoing.get(&node);
702        if outs.map(|s| s.is_empty()).unwrap_or(true) {
703            // CLONE: needed — heap owns completed haplotype sequences.
704            out.push((support_acc.max(1), Reverse(current.clone())));
705            visited.remove(&node);
706            return;
707        }
708        let Some(outs) = outs else {
709            visited.remove(&node);
710            return;
711        };
712        for &to in outs {
713            let edge_w = *self.edges.get(&(node, to)).unwrap_or(&0);
714            let next_base = self.nodes[to].kmer.last().copied().unwrap_or(b'N');
715            current.push(next_base);
716            self.dfs_haplotypes(
717                to,
718                current,
719                support_acc.saturating_add(edge_w),
720                max_bases,
721                visited,
722                out,
723            );
724            current.pop();
725        }
726        visited.remove(&node);
727    }
728}

730/// Wave-A graph-first assembly API (not on production `call_region` path).
731/// Prefer [`crate::read_threading_assembler`] / [`crate::assembly_based_caller`].
732/// Exercised by `#[cfg(test)]` unit tests; kept for that contract outside test cfg.
733#[allow(dead_code)]
734#[derive(Debug, Clone)]
735pub(crate) struct AssemblyEngine {
736    params: AssemblyGraphParams,
737}
738
739#[allow(dead_code)] // see struct note — unit-tested under cfg(test)
740impl AssemblyEngine {
741    /// Create a new assembly engine.
742    pub fn new(params: AssemblyGraphParams) -> GatkResult<Self> {
743        Ok(Self { params })
744    }
745
746    /// Assemble candidate haplotypes from reads (delegates to [`crate::read_threading_assembler`] when reference provided).
747    pub fn assemble(&self, reads: &[AssemblyRead]) -> GatkResult<Vec<CandidateHaplotype>> {
748        self.assemble_with_optional_ref(None, reads)
749    }
750
751    /// Assemble with optional reference sequence (GATK ref-threaded path).
752    pub fn assemble_with_optional_ref(
753        &self,
754        reference: Option<&AssemblyRead>,
755        reads: &[AssemblyRead],
756    ) -> GatkResult<Vec<CandidateHaplotype>> {
757        if let Some(reference) = reference {
758            let mut args = crate::read_threading_assembler::ReadThreadingAssemblerArgs::default();
759            args.kmer_sizes = vec![self.params.kmer_size.as_usize()];
760            args.min_base_quality = self.params.min_base_quality;
761            args.min_prune_factor = self.params.min_edge_weight;
762            args.num_best_haplotypes_per_graph = self.params.max_haplotypes;
763            let result = crate::read_threading_assembler::assemble_from_ref_and_reads(
764                reference, reads, &args,
765            )?;
766            return Ok(result
767                .haplotypes
768                .into_iter()
769                .map(|h| CandidateHaplotype {
770                    sequence: h.bases,
771                    support: h.score.max(0.0) as u32,
772                })
773                .collect());
774        }
775        let mut graph = AssemblyGraph::from_reads(reads, &self.params)?;
776        let mut pruning = AssemblyGraphPruningParams::gatk_haplotype_caller_defaults();
777        pruning.min_prune_factor = self.params.min_edge_weight;
778        graph.apply_pruning(&pruning);
779        graph.remove_dangling_paths(self.params.dangling_path_max_nodes);
780        graph.cleanup_isolated_nodes();
781        Ok(graph.extract_candidate_haplotypes(
782            self.params.max_haplotypes,
783            self.params.max_haplotype_bases,
784        ))
785    }
786}
