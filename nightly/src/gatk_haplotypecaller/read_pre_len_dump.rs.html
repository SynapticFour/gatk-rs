1//! PRE.2 gate dump: `read-pre-len`.
2
3use crate::read_pre_len::{passes_read_length_filter, unclipped_read_length};
4use gatk_common::{GatkError, GatkResult};
5use rust_htslib::bam;
6use rust_htslib::bam::record::Cigar;
7use rust_htslib::bam::Read as _;
8use std::io::Write;
9use std::path::Path;
10
11pub fn dump_read_pre_len_tsv(alignment_path: &Path, out: &mut impl Write) -> GatkResult<()> {
12    writeln!(
13        out,
14        "qname\tcigar\tread_length\tunclipped_length\tpasses_len_filter"
15    )
16    .map_err(|e| GatkError::generic(format!("write: {e}")))?;
17
18    let mut reader = bam::Reader::from_path(alignment_path)
19        .map_err(|e| GatkError::generic(format!("open: {e}")))?;
20    for res in reader.records() {
21        let rec = res.map_err(|e| GatkError::generic(format!("read: {e}")))?;
22        let unclipped = unclipped_read_length(&rec);
23        writeln!(
24            out,
25            "{}\t{}\t{}\t{}\t{}",
26            String::from_utf8_lossy(rec.qname()),
27            format_cigar(&rec),
28            rec.seq().as_bytes().len(),
29            unclipped,
30            passes_read_length_filter(&rec)
31        )
32        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
33    }
34    Ok(())
35}
36
37fn format_cigar(rec: &bam::Record) -> String {
38    rec.cigar()
39        .iter()
40        .map(|c| {
41            let op = match c {
42                Cigar::Match(_) => 'M',
43                Cigar::Ins(_) => 'I',
44                Cigar::Del(_) => 'D',
45                Cigar::SoftClip(_) => 'S',
46                Cigar::HardClip(_) => 'H',
47                Cigar::Equal(_) => '=',
48                Cigar::Diff(_) => 'X',
49                Cigar::RefSkip(_) => 'N',
50                Cigar::Pad(_) => 'P',
51            };
52            format!("{}{}", crate::read_unclip::cigar_len(c), op)
53        })
54        .collect()
55}
