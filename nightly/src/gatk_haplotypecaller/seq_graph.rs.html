7/// Sequence-labeled vertex after k-mer graph → sequence graph conversion.
8/// # Invariants
9/// `id` is dense index into owning [`SeqGraph::vertices`].
10/// `sequence` holds vertex bases (full k-mer at source, suffix byte at non-source joins).
11/// # Ownership
12/// Owns base sequence bytes; graph owns vertex vector.
13/// # Mutation
14/// Graph cleanup may merge/replace vertex sequences in place.
15/// # Biological assumptions
16/// Vertex sequence is literal assembly graph bases on the reference/read path.
17/// # Java equivalence
18/// GATK `SeqVertex` in `SeqGraph` (`BaseGraph.toSequenceGraph`).
19#[derive(Debug, Clone)]
20pub struct SeqVertex {
21    pub id: usize,
22    pub sequence: Vec<u8>,
23}
24
25/// Directed edge in a sequence graph with support and reference-path flag.
26/// # Invariants
27/// `from` / `to` index valid vertices; `is_ref` marks reference-spine edges.
28/// # Ownership
29/// [`Copy`] edge record in [`SeqGraph::edges`].
30/// # Mutation
31/// Support may increase on merge; edges removed during cleanup passes.
32/// # Biological assumptions
33/// Edge support reflects read evidence multiplicity at the junction.
34/// # Java equivalence
35/// GATK `SeqEdge` / reference edge marking in `SeqGraph`.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub struct SeqEdge {
38    pub from: usize,
39    pub to: usize,
40    pub support: u32,
41    pub is_ref: bool,
42}
43
44/// Outcome of [`SeqGraph::cleanup_seq_graph`] — variation present or reference-only.
45/// # Invariants
46/// Mutually exclusive assembly outcomes for one cleanup pass.
47/// # Ownership
48/// [`Copy`] enum returned from cleanup.
49/// # Mutation
50/// Immutable status tag.
51/// # Biological assumptions
52/// `AssembledSomeVariation` means non-ref paths survived cleanup connected to ref endpoints.
53/// # Java equivalence
54/// GATK `SeqGraph` cleanup status (`ASSEMBLED_SOME_VARIATION`, `JUST_ASSEMBLED_REFERENCE`).
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum SeqGraphCleanupStatus {
57    AssembledSomeVariation,
58    JustAssembledReference,
59}

61/// Base sequence graph for k-best haplotype discovery after threading conversion.
62/// # Invariants
63/// Reference source/sink vertices exist after successful cleanup when variation is assembled.
64/// Vertex `id` values remain dense after prune/zip operations.
65/// # Ownership
66/// Owns vertices, edges, and adjacency indexes; built from [`AssemblyGraph`] without retaining reads.
67/// # Mutation
68/// Cleanup methods (`clean_non_ref_paths`, `zip_linear_chains`, `simplify_graph`, etc.) mutate in place.
69/// # Biological assumptions
70/// Graph encodes colinear sequence alternatives over the padded reference window.
71/// # Java equivalence
72/// GATK `SeqGraph` (`BaseGraph.toSequenceGraph`, `ReadThreadingAssembler.cleanupSeqGraph`).
73#[derive(Debug, Clone)]
74pub struct SeqGraph {
75    pub kmer_size: usize,
76    vertices: Vec<SeqVertex>,
77    edges: Vec<SeqEdge>,
78    outgoing: HashMap<usize, Vec<usize>>,
79    incoming: HashMap<usize, Vec<usize>>,
80}

83    pub fn from_assembly_graph(graph: &AssemblyGraph) -> Self {
84        let kmer_size = graph.kmer_size;
85        let mut vertices = Vec::new();
86        let mut id_map = HashMap::new();
87        for (i, node) in graph.nodes().iter().enumerate() {
88            let is_source = graph.incoming_count(i) == 0;
89            let seq = additional_sequence(&node.kmer, is_source);
90            let id = vertices.len();
91            vertices.push(SeqVertex { id, sequence: seq });
92            id_map.insert(i, id);
93        }
94        let mut edges = Vec::new();
95        let mut outgoing: HashMap<usize, Vec<usize>> = HashMap::new();
96        let mut incoming: HashMap<usize, Vec<usize>> = HashMap::new();
97        for e in graph.edges_sorted() {
98            let from = *id_map.get(&e.from).expect("edge from");
99            let to = *id_map.get(&e.to).expect("edge to");
100            let is_ref = graph.edge_is_ref(e.from, e.to);
101            edges.push(SeqEdge {
102                from,
103                to,
104                support: e.support,
105                is_ref,
106            });
107            outgoing.entry(from).or_default().push(to);
108            incoming.entry(to).or_default().push(from);
109        }
110        Self {
111            kmer_size,
112            vertices,
113            edges,
114            outgoing,
115            incoming,
116        }
117    }

139    fn is_ref_source(&self, v: usize) -> bool {
140        if self.vertices.len() == 1 {
141            return true;
142        }
143        if self
144            .incoming
145            .get(&v)
146            .into_iter()
147            .flatten()
148            .any(|&p| self.edge_is_ref(p, v))
149        {
150            return false;
151        }
152        self.outgoing
153            .get(&v)
154            .into_iter()
155            .flatten()
156            .any(|&t| self.edge_is_ref(v, t))
157    }

166            .into_iter()
167            .flatten()
168            .any(|&t| self.edge_is_ref(v, t))
169        {
170            return false;

228    pub fn reference_path_bytes(&self) -> Option<Vec<u8>> {
229        let source = self.reference_source_vertex()?;
230        let sink = self.reference_sink_vertex()?;
231        let mut path = self.vertices[source].sequence.clone();
232        let mut cur = source;
233        while cur != sink {
234            let outs = self.outgoing_of(cur);
235            let next = *outs
236                .iter()
237                .find(|&&t| self.edge_is_ref(cur, t))
238                .or_else(|| outs.first())?;
239            path.extend_from_slice(&self.vertices[next].sequence);
240            cur = next;
241        }
242        Some(path)
243    }

245    pub fn clean_non_ref_paths(&mut self) {
246        let Some(source) = self.reference_source_vertex() else {
247            return;
248        };
249        let Some(sink) = self.reference_sink_vertex() else {
250            return;
251        };
252        let mut to_remove: HashSet<(usize, usize)> = HashSet::new();
253        let mut check: Vec<(usize, usize)> = self
254            .edges
255            .iter()
256            .filter(|e| e.to == source && !e.is_ref)
257            .map(|e| (e.from, e.to))
258            .collect();
259        while let Some((from, to)) = check.pop() {
260            if !to_remove.insert((from, to)) {
261                continue;
262            }
263            for e in &self.edges {
264                if e.to == from && !e.is_ref {
265                    check.push((e.from, e.to));
266                }
267            }
268        }
269        check = self
270            .edges
271            .iter()
272            .filter(|e| e.from == sink && !e.is_ref)
273            .map(|e| (e.from, e.to))
274            .collect();
275        while let Some((from, to)) = check.pop() {
276            if !to_remove.insert((from, to)) {
277                continue;
278            }
279            for e in &self.edges {
280                if e.from == to && !e.is_ref {
281                    check.push((e.from, e.to));
282                }
283            }
284        }
285        self.edges.retain(|e| !to_remove.contains(&(e.from, e.to)));
286        self.rebuild_index();
287        self.remove_singleton_orphan_vertices();
288    }

290    pub fn zip_linear_chains(&mut self) -> bool {
291        let starts: Vec<usize> = (0..self.vertices.len())
292            .filter(|&s| self.is_linear_chain_start(s))
293            .collect();
294        if starts.is_empty() {
295            return false;
296        }
297        let mut merged = false;
298        for start in starts {
299            let chain = self.trace_linear_chain(start);
300            if chain.len() > 1 {
301                merged |= self.merge_linear_chain(&chain);
302            }
303        }
304        merged
305    }
306
307    fn is_linear_chain_start(&self, source: usize) -> bool {
308        if self.out_degree(source) != 1 {
309            return false;
310        }
311        let indeg = self.in_degree(source);
312        indeg != 1
313            || self
314                .incoming_of(source)
315                .first()
316                .map(|&p| self.out_degree(p) > 1)
317                .unwrap_or(false)
318    }
319
320    fn trace_linear_chain(&self, zip_start: usize) -> Vec<usize> {
321        let mut chain = vec![zip_start];
322        let mut last = zip_start;
323        loop {
324            if self.out_degree(last) != 1 {
325                break;
326            }
327            let target = self.outgoing_of(last)[0];
328            if self.in_degree(target) != 1 || last == target {
329                break;
330            }
331            let last_ref = self.is_reference_node(last);
332            let target_ref = self.is_reference_node(target);
333            if last_ref != target_ref {
334                break;
335            }
336            chain.push(target);
337            last = target;
338        }
339        chain
340    }

347    fn merge_linear_chain(&mut self, chain: &[usize]) -> bool {
348        if chain.len() < 2 {
349            return false;
350        }
351        let keep = chain[0];
352        let remove: HashSet<usize> = chain[1..].iter().copied().collect();
353        let mut merged_seq = self.vertices[keep].sequence.clone();
354        for &v in &chain[1..] {
355            merged_seq.extend_from_slice(&self.vertices[v].sequence);
356        }
357
358        let kept: Vec<usize> = (0..self.vertices.len())
359            .filter(|i| !remove.contains(i))
360            .collect();
361        let old_to_new: HashMap<usize, usize> = kept
362            .iter()
363            .enumerate()
364            .map(|(new_id, &old_id)| (old_id, new_id))
365            .collect();
366
367        let mut new_vertices = Vec::with_capacity(kept.len());
368        for &old_id in &kept {
369            let mut v = self.vertices[old_id].clone();
370            if old_id == keep {
371                v.sequence = merged_seq.clone();
372            }
373            v.id = old_to_new[&old_id];
374            new_vertices.push(v);
375        }
376
377        let mut new_edges = Vec::new();
378        let mut seen: HashSet<(usize, usize, bool)> = HashSet::new();
379        for e in &self.edges {
380            let Some(&from) = old_to_new.get(&e.from) else {
381                continue;
382            };
383            let Some(&to) = old_to_new.get(&e.to) else {
384                continue;
385            };
386            if from == to {
387                continue;
388            }
389            if seen.insert((from, to, e.is_ref)) {
390                new_edges.push(SeqEdge {
391                    from,
392                    to,
393                    support: e.support,
394                    is_ref: e.is_ref,
395                });
396            }
397        }
398        self.vertices = new_vertices;
399        self.edges = new_edges;
400        self.rebuild_index();
401        true
402    }

404    fn rebuild_index(&mut self) {
405        let mut outgoing: HashMap<usize, Vec<usize>> = HashMap::new();
406        let mut incoming: HashMap<usize, Vec<usize>> = HashMap::new();
407        for (i, e) in self.edges.iter().enumerate() {
408            let _ = i;
409            outgoing.entry(e.from).or_default().push(e.to);
410            incoming.entry(e.to).or_default().push(e.from);
411        }
412        self.outgoing = outgoing;
413        self.incoming = incoming;
414    }

416    pub fn remove_singleton_orphan_vertices(&mut self) {
417        loop {
418            let orphans: HashSet<usize> = (0..self.vertices.len())
419                .filter(|&v| {
420                    self.in_degree(v) == 0 && self.out_degree(v) == 0 && !self.is_ref_source(v)
421                })
422                .collect();
423            if orphans.is_empty() {
424                break;
425            }
426            self.vertices.retain(|v| !orphans.contains(&v.id));
427            for (i, v) in self.vertices.iter_mut().enumerate() {
428                v.id = i;
429            }
430            self.edges
431                .retain(|e| !orphans.contains(&e.from) && !orphans.contains(&e.to));
432            self.rebuild_index();
433        }
434    }

436    /// GATK `BaseGraph.removeVerticesNotConnectedToRefRegardlessOfEdgeDirection`.
437    pub fn remove_vertices_not_connected_to_ref_regardless_of_direction(&mut self) {
438        let Some(source) = self.reference_source_vertex() else {
439            return;
440        };
441        let mut keep = HashSet::new();
442        let mut stack = vec![source];
443        keep.insert(source);
444        while let Some(v) = stack.pop() {
445            for &t in self.outgoing_of(v) {
446                if keep.insert(t) {
447                    stack.push(t);
448                }
449            }
450            for &p in self.incoming_of(v) {
451                if keep.insert(p) {
452                    stack.push(p);
453                }
454            }
455        }
456        let remove: HashSet<usize> = (0..self.vertices.len())
457            .filter(|v| !keep.contains(v))
458            .collect();
459        if remove.is_empty() {
460            return;
461        }
462        self.vertices.retain(|v| !remove.contains(&v.id));
463        for (i, v) in self.vertices.iter_mut().enumerate() {
464            v.id = i;
465        }
466        self.edges
467            .retain(|e| !remove.contains(&e.from) && !remove.contains(&e.to));
468        self.rebuild_index();
469    }

471    pub fn remove_paths_not_connected_to_ref(&mut self) -> GatkResult<()> {
472        let source = self
473            .reference_source_vertex()
474            .ok_or_else(|| GatkError::algorithm("seq graph: no ref source"))?;
475        let sink = self
476            .reference_sink_vertex()
477            .ok_or_else(|| GatkError::algorithm("seq graph: no ref sink"))?;
478        let mut from_source = HashSet::new();
479        let mut stack = vec![source];
480        from_source.insert(source);
481        while let Some(v) = stack.pop() {
482            for &t in self.outgoing_of(v) {
483                if from_source.insert(t) {
484                    stack.push(t);
485                }
486            }
487        }
488        let mut from_sink = HashSet::new();
489        let mut stack = vec![sink];
490        from_sink.insert(sink);
491        while let Some(v) = stack.pop() {
492            for &p in self.incoming_of(v) {
493                if from_sink.insert(p) {
494                    stack.push(p);
495                }
496            }
497        }
498        from_source.retain(|v| from_sink.contains(v));
499        let remove: HashSet<usize> = (0..self.vertices.len())
500            .filter(|v| !from_source.contains(v))
501            .collect();
502        self.vertices.retain(|v| !remove.contains(&v.id));
503        for (i, v) in self.vertices.iter_mut().enumerate() {
504            v.id = i;
505        }
506        self.edges
507            .retain(|e| !remove.contains(&e.from) && !remove.contains(&e.to));
508        self.rebuild_index();
509        Ok(())
510    }

614    pub fn cleanup_seq_graph(&mut self) -> SeqGraphCleanupStatus {
615        self.zip_linear_chains();
616        self.remove_singleton_orphan_vertices();
617        self.remove_vertices_not_connected_to_ref_regardless_of_direction();
618        self.simplify_graph();
619        if self.reference_source_vertex().is_none() || self.reference_sink_vertex().is_none() {
620            return SeqGraphCleanupStatus::JustAssembledReference;
621        }
622        // GATK `ReadThreadingAssembler.cleanupSeqGraph`: after removePaths + simplify, proceed to
623        // ASSEMBLED_SOME_VARIATION (dummy vertex if needed) — no second ref source/sink abort.
624        let _ = self.remove_paths_not_connected_to_ref();
625        self.simplify_graph();
626        if self.vertices.len() == 1 {
627            let complete = 0usize;
628            let dummy_id = self.vertices.len();
629            self.vertices.push(SeqVertex {
630                id: dummy_id,
631                sequence: Vec::new(),
632            });
633            self.edges.push(SeqEdge {
634                from: complete,
635                to: dummy_id,
636                support: 0,
637                is_ref: true,
638            });
639            self.rebuild_index();
640        }
641        SeqGraphCleanupStatus::AssembledSomeVariation
642    }

645fn additional_sequence(kmer: &str, is_source: bool) -> Vec<u8> {

649fn additional_sequence_bytes(seq: &[u8], is_source: bool) -> Vec<u8> {
650    if is_source {
651        seq.to_vec()
652    } else {
653        seq.last().map(|&c| vec![c]).unwrap_or_default()

675    #[test]
676    fn prepared_p5_seqgraph_dump_path_has_nodes() {
677        let repo = Path::new(env!("CARGO_MANIFEST_DIR")).join("..");
678        let reference =
679            load_assembly_ref_tsv(&repo.join("parity/fixtures/hc-full-parity/e4/p5_case1_ref.tsv"))
680                .unwrap();
681        let reads = load_assembly_reads_tsv(
682            &repo.join("parity/fixtures/hc-full-parity/e4/p5_case1_reads.tsv"),
683        )
684        .unwrap();
685        let args = ReadThreadingAssemblerArgs {
686            kmer_sizes: vec![3],
687            min_base_quality: 10,
688            min_prune_factor: 2,
689            min_dangling_branch_length: 4,
690            recover_dangling_heads: true,
691            ..Default::default()
692        };
693        let graph =
694            build_threading_graph_for_haplotype_dump(&reference, &reads, 3, &args, true, false)
695                .unwrap()
696                .expect("graph");
697        assert!(graph.node_count() > 0, "rt graph");
698        let mut seq = SeqGraph::from_assembly_graph(&graph);
699        assert!(seq.node_count() > 0, "seq after from");
700        seq.clean_non_ref_paths();
701        let status = seq.cleanup_seq_graph();
702        assert!(seq.node_count() > 0, "after cleanup status={status:?}");
703        let path = seq.reference_path_bytes().unwrap();
704        assert_eq!(std::str::from_utf8(&path).unwrap(), reference.bases);
705    }
