7/// Sequence-labeled vertex after k-mer graph → sequence graph conversion.
8/// # Invariants
9/// `id` is dense index into the owning sequence graph vertex list.
10/// `sequence` holds vertex bases (full k-mer at source, suffix byte at non-source joins).
11/// # Ownership
12/// Owns base sequence bytes; graph owns vertex vector.
13/// # Mutation
14/// Graph cleanup may merge/replace vertex sequences in place.
15/// # Biological assumptions
16/// Vertex sequence is literal assembly graph bases on the reference/read path.
17/// # Java equivalence
18/// GATK `SeqVertex` in `SeqGraph` (`BaseGraph.toSequenceGraph`).
19#[derive(Debug, Clone)]
20pub struct SeqVertex {
21    pub id: usize,
22    pub sequence: Vec<u8>,
23}
24
25/// Directed edge in a sequence graph with support and reference-path flag.
26/// # Invariants
27/// `from` / `to` index valid vertices; `is_ref` marks reference-spine edges.
28/// # Ownership
29/// [`Copy`] edge record in the owning sequence graph edge list.
30/// # Mutation
31/// Support may increase on merge; edges removed during cleanup passes.
32/// # Biological assumptions
33/// Edge support reflects read evidence multiplicity at the junction.
34/// # Java equivalence
35/// GATK `SeqEdge` / reference edge marking in `SeqGraph`.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub struct SeqEdge {
38    pub from: usize,
39    pub to: usize,
40    pub support: u32,
41    pub is_ref: bool,
42}
43
44/// Outcome of [`SeqGraph::cleanup_seq_graph`] — variation present or reference-only.
45/// # Invariants
46/// Mutually exclusive assembly outcomes for one cleanup pass.
47/// # Ownership
48/// [`Copy`] enum returned from cleanup.
49/// # Mutation
50/// Immutable status tag.
51/// # Biological assumptions
52/// `AssembledSomeVariation` means non-ref paths survived cleanup connected to ref endpoints.
53/// # Java equivalence
54/// GATK `SeqGraph` cleanup status (`ASSEMBLED_SOME_VARIATION`, `JUST_ASSEMBLED_REFERENCE`).
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum SeqGraphCleanupStatus {
57    AssembledSomeVariation,
58    JustAssembledReference,
59}

61/// Base sequence graph for k-best haplotype discovery after threading conversion.
62/// # Invariants
63/// Reference source/sink vertices exist after successful cleanup when variation is assembled.
64/// Vertex `id` values remain dense after prune/zip operations.
65/// # Ownership
66/// Owns vertices, edges, and adjacency indexes; built from [`AssemblyGraph`] without retaining reads.
67/// # Mutation
68/// Cleanup methods (`clean_non_ref_paths`, `zip_linear_chains`, `simplify_graph`, etc.) mutate in place.
69/// # Biological assumptions
70/// Graph encodes colinear sequence alternatives over the padded reference window.
71/// # Java equivalence
72/// GATK `SeqGraph` (`BaseGraph.toSequenceGraph`, `ReadThreadingAssembler.cleanupSeqGraph`).
73#[derive(Debug, Clone)]
74pub struct SeqGraph {
75    pub kmer_size: usize,
76    vertices: Vec<SeqVertex>,
77    edges: Vec<SeqEdge>,
78    outgoing: HashMap<usize, Vec<usize>>,
79    incoming: HashMap<usize, Vec<usize>>,
80}

83    pub fn from_assembly_graph(graph: &AssemblyGraph) -> Self {
84        let kmer_size = graph.kmer_size;
85        let mut vertices = Vec::new();
86        let mut id_map = HashMap::new();
87        for (i, node) in graph.nodes().iter().enumerate() {
88            let is_source = graph.incoming_count(i) == 0;
89            let seq = additional_sequence_bytes(&node.kmer, is_source);
90            let id = vertices.len();
91            vertices.push(SeqVertex { id, sequence: seq });
92            id_map.insert(i, id);
93        }
94        let mut edges = Vec::new();
95        let mut outgoing: HashMap<usize, Vec<usize>> = HashMap::new();
96        let mut incoming: HashMap<usize, Vec<usize>> = HashMap::new();
97        for e in graph.edges_sorted() {
98            let from = *id_map.get(&e.from).expect("edge from");
99            let to = *id_map.get(&e.to).expect("edge to");
100            let is_ref = graph.edge_is_ref(e.from, e.to);
101            edges.push(SeqEdge {
102                from,
103                to,
104                support: e.support,
105                is_ref,
106            });
107            outgoing.entry(from).or_default().push(to);
108            incoming.entry(to).or_default().push(from);
109        }
110        Self {
111            kmer_size,
112            vertices,
113            edges,
114            outgoing,
115            incoming,
116        }
117    }

144    fn is_ref_source(&self, v: usize) -> bool {
145        if self.vertices.len() == 1 {
146            return true;
147        }
148        if self
149            .incoming
150            .get(&v)
151            .into_iter()
152            .flatten()
153            .any(|&p| self.edge_is_ref(p, v))
154        {
155            return false;
156        }
157        self.outgoing
158            .get(&v)
159            .into_iter()
160            .flatten()
161            .any(|&t| self.edge_is_ref(v, t))
162    }

165        if self.vertices.len() == 1 {
166            return true;
167        }
168        if self
169            .outgoing
170            .get(&v)
171            .into_iter()
172            .flatten()
173            .any(|&t| self.edge_is_ref(v, t))
174        {
175            return false;

203        let first = if edges.is_empty() { start } else { edges[0].0 };
204        let mut bases = additional_sequence_bytes(&self.vertices[first].sequence, true);

207                &self.vertices[to].sequence,

233    pub fn reference_path_bytes(&self) -> Option<Vec<u8>> {
234        let source = self.reference_source_vertex()?;
235        let sink = self.reference_sink_vertex()?;
236        let mut path = self.vertices[source].sequence.clone();
237        let mut cur = source;
238        while cur != sink {
239            let outs = self.outgoing_of(cur);
240            let next = *outs
241                .iter()
242                .find(|&&t| self.edge_is_ref(cur, t))
243                .or_else(|| outs.first())?;
244            path.extend_from_slice(&self.vertices[next].sequence);
245            cur = next;
246        }
247        Some(path)
248    }

250    pub fn clean_non_ref_paths(&mut self) {
251        let Some(source) = self.reference_source_vertex() else {
252            return;
253        };
254        let Some(sink) = self.reference_sink_vertex() else {
255            return;
256        };
257        let mut to_remove: HashSet<(usize, usize)> = HashSet::new();
258        let mut check: Vec<(usize, usize)> = self
259            .edges
260            .iter()
261            .filter(|e| e.to == source && !e.is_ref)
262            .map(|e| (e.from, e.to))
263            .collect();
264        while let Some((from, to)) = check.pop() {
265            if !to_remove.insert((from, to)) {
266                continue;
267            }
268            for e in &self.edges {
269                if e.to == from && !e.is_ref {
270                    check.push((e.from, e.to));
271                }
272            }
273        }
274        check = self
275            .edges
276            .iter()
277            .filter(|e| e.from == sink && !e.is_ref)
278            .map(|e| (e.from, e.to))
279            .collect();
280        while let Some((from, to)) = check.pop() {
281            if !to_remove.insert((from, to)) {
282                continue;
283            }
284            for e in &self.edges {
285                if e.from == to && !e.is_ref {
286                    check.push((e.from, e.to));
287                }
288            }
289        }
290        self.edges.retain(|e| !to_remove.contains(&(e.from, e.to)));
291        self.rebuild_index();
292        self.remove_singleton_orphan_vertices();
293    }

295    pub fn zip_linear_chains(&mut self) -> bool {
296        let starts: Vec<usize> = (0..self.vertices.len())
297            .filter(|&s| self.is_linear_chain_start(s))
298            .collect();
299        if starts.is_empty() {
300            return false;
301        }
302        let mut merged = false;
303        for start in starts {
304            let chain = self.trace_linear_chain(start);
305            if chain.len() > 1 {
306                merged |= self.merge_linear_chain(&chain);
307            }
308        }
309        merged
310    }
311
312    fn is_linear_chain_start(&self, source: usize) -> bool {
313        if self.out_degree(source) != 1 {
314            return false;
315        }
316        let indeg = self.in_degree(source);
317        indeg != 1
318            || self
319                .incoming_of(source)
320                .first()
321                .map(|&p| self.out_degree(p) > 1)
322                .unwrap_or(false)
323    }
324
325    fn trace_linear_chain(&self, zip_start: usize) -> Vec<usize> {
326        let mut chain = vec![zip_start];
327        let mut last = zip_start;
328        loop {
329            if self.out_degree(last) != 1 {
330                break;
331            }
332            let target = self.outgoing_of(last)[0];
333            if self.in_degree(target) != 1 || last == target {
334                break;
335            }
336            let last_ref = self.is_reference_node(last);
337            let target_ref = self.is_reference_node(target);
338            if last_ref != target_ref {
339                break;
340            }
341            chain.push(target);
342            last = target;
343        }
344        chain
345    }

352    fn merge_linear_chain(&mut self, chain: &[usize]) -> bool {
353        if chain.len() < 2 {
354            return false;
355        }
356        let keep = chain[0];
357        let remove: HashSet<usize> = chain[1..].iter().copied().collect();
358        // Take keep sequence then extend — avoid cloning every vertex seq on the chain.
359        let mut merged_seq = std::mem::take(&mut self.vertices[keep].sequence);
360        for &v in &chain[1..] {
361            merged_seq.extend_from_slice(&self.vertices[v].sequence);
362        }
363
364        let kept: Vec<usize> = (0..self.vertices.len())
365            .filter(|i| !remove.contains(i))
366            .collect();
367        let old_to_new: HashMap<usize, usize> = kept
368            .iter()
369            .enumerate()
370            .map(|(new_id, &old_id)| (old_id, new_id))
371            .collect();
372
373        let mut new_vertices = Vec::with_capacity(kept.len());
374        for &old_id in &kept {
375            let sequence = if old_id == keep {
376                std::mem::take(&mut merged_seq)
377            } else {
378                std::mem::take(&mut self.vertices[old_id].sequence)
379            };
380            new_vertices.push(SeqVertex {
381                id: old_to_new[&old_id],
382                sequence,
383            });
384        }
385
386        let mut new_edges = Vec::new();
387        let mut seen: HashSet<(usize, usize, bool)> = HashSet::new();
388        for e in &self.edges {
389            let Some(&from) = old_to_new.get(&e.from) else {
390                continue;
391            };
392            let Some(&to) = old_to_new.get(&e.to) else {
393                continue;
394            };
395            if from == to {
396                continue;
397            }
398            if seen.insert((from, to, e.is_ref)) {
399                new_edges.push(SeqEdge {
400                    from,
401                    to,
402                    support: e.support,
403                    is_ref: e.is_ref,
404                });
405            }
406        }
407        self.vertices = new_vertices;
408        self.edges = new_edges;
409        self.rebuild_index();
410        true
411    }

413    fn rebuild_index(&mut self) {
414        self.outgoing.clear();
415        self.incoming.clear();
416        for e in &self.edges {
417            self.outgoing.entry(e.from).or_default().push(e.to);
418            self.incoming.entry(e.to).or_default().push(e.from);
419        }
420    }

422    pub fn remove_singleton_orphan_vertices(&mut self) {
423        loop {
424            let orphans: HashSet<usize> = (0..self.vertices.len())
425                .filter(|&v| {
426                    self.in_degree(v) == 0 && self.out_degree(v) == 0 && !self.is_ref_source(v)
427                })
428                .collect();
429            if orphans.is_empty() {
430                break;
431            }
432            self.vertices.retain(|v| !orphans.contains(&v.id));
433            for (i, v) in self.vertices.iter_mut().enumerate() {
434                v.id = i;
435            }
436            self.edges
437                .retain(|e| !orphans.contains(&e.from) && !orphans.contains(&e.to));
438            self.rebuild_index();
439        }
440    }

442    /// GATK `BaseGraph.removeVerticesNotConnectedToRefRegardlessOfEdgeDirection`.
443    pub fn remove_vertices_not_connected_to_ref_regardless_of_direction(&mut self) {
444        let Some(source) = self.reference_source_vertex() else {
445            return;
446        };
447        let mut keep = HashSet::new();
448        let mut stack = vec![source];
449        keep.insert(source);
450        while let Some(v) = stack.pop() {
451            for &t in self.outgoing_of(v) {
452                if keep.insert(t) {
453                    stack.push(t);
454                }
455            }
456            for &p in self.incoming_of(v) {
457                if keep.insert(p) {
458                    stack.push(p);
459                }
460            }
461        }
462        let remove: HashSet<usize> = (0..self.vertices.len())
463            .filter(|v| !keep.contains(v))
464            .collect();
465        if remove.is_empty() {
466            return;
467        }
468        self.vertices.retain(|v| !remove.contains(&v.id));
469        for (i, v) in self.vertices.iter_mut().enumerate() {
470            v.id = i;
471        }
472        self.edges
473            .retain(|e| !remove.contains(&e.from) && !remove.contains(&e.to));
474        self.rebuild_index();
475    }

477    pub fn remove_paths_not_connected_to_ref(&mut self) -> GatkResult<()> {
478        let source = self
479            .reference_source_vertex()
480            .ok_or_else(|| GatkError::algorithm("seq graph: no ref source"))?;
481        let sink = self
482            .reference_sink_vertex()
483            .ok_or_else(|| GatkError::algorithm("seq graph: no ref sink"))?;
484        let mut from_source = HashSet::new();
485        let mut stack = vec![source];
486        from_source.insert(source);
487        while let Some(v) = stack.pop() {
488            for &t in self.outgoing_of(v) {
489                if from_source.insert(t) {
490                    stack.push(t);
491                }
492            }
493        }
494        let mut from_sink = HashSet::new();
495        let mut stack = vec![sink];
496        from_sink.insert(sink);
497        while let Some(v) = stack.pop() {
498            for &p in self.incoming_of(v) {
499                if from_sink.insert(p) {
500                    stack.push(p);
501                }
502            }
503        }
504        from_source.retain(|v| from_sink.contains(v));
505        let remove: HashSet<usize> = (0..self.vertices.len())
506            .filter(|v| !from_source.contains(v))
507            .collect();
508        self.vertices.retain(|v| !remove.contains(&v.id));
509        for (i, v) in self.vertices.iter_mut().enumerate() {
510            v.id = i;
511        }
512        self.edges
513            .retain(|e| !remove.contains(&e.from) && !remove.contains(&e.to));
514        self.rebuild_index();
515        Ok(())
516    }

570        is_ref: bool,
571    ) {
572        if let Some(idx) = self.find_edge(from, to) {
573            let e = &mut self.edges[idx];
574            e.support = e.support.saturating_add(support);
575            e.is_ref |= is_ref;
576        } else {

579                to,
580                support,
581                is_ref,

620    pub fn cleanup_seq_graph(&mut self) -> SeqGraphCleanupStatus {
621        self.zip_linear_chains();
622        self.remove_singleton_orphan_vertices();
623        self.remove_vertices_not_connected_to_ref_regardless_of_direction();
624        self.simplify_graph();
625        if self.reference_source_vertex().is_none() || self.reference_sink_vertex().is_none() {
626            return SeqGraphCleanupStatus::JustAssembledReference;
627        }
628        // GATK `ReadThreadingAssembler.cleanupSeqGraph`: after removePaths + simplify, proceed to
629        // ASSEMBLED_SOME_VARIATION (dummy vertex if needed) — no second ref source/sink abort.
630        let _ = self.remove_paths_not_connected_to_ref();
631        self.simplify_graph();
632        if self.vertices.len() == 1 {
633            let complete = 0usize;
634            let dummy_id = self.vertices.len();
635            self.vertices.push(SeqVertex {
636                id: dummy_id,
637                sequence: Vec::new(),
638            });
639            self.edges.push(SeqEdge {
640                from: complete,
641                to: dummy_id,
642                support: 0,
643                is_ref: true,
644            });
645            self.rebuild_index();
646        }
647        SeqGraphCleanupStatus::AssembledSomeVariation
648    }

651fn additional_sequence_bytes(seq: &[u8], is_source: bool) -> Vec<u8> {
652    if is_source {

654    } else {

677    #[test]
678    fn prepared_p5_seqgraph_dump_path_has_nodes() {
679        let repo = Path::new(env!("CARGO_MANIFEST_DIR")).join("..");
680        let reference =
681            load_assembly_ref_tsv(&repo.join("parity/fixtures/hc-full-parity/e4/p5_case1_ref.tsv"))
682                .unwrap();
683        let reads = load_assembly_reads_tsv(
684            &repo.join("parity/fixtures/hc-full-parity/e4/p5_case1_reads.tsv"),
685        )
686        .unwrap();
687        let args = ReadThreadingAssemblerArgs {
688            kmer_sizes: vec![3],
689            min_base_quality: 10,
690            min_prune_factor: 2,
691            min_dangling_branch_length: 4,
692            recover_dangling_heads: true,
693            ..Default::default()
694        };
695        let graph =
696            build_threading_graph_for_haplotype_dump(&reference, &reads, 3, &args, true, false)
697                .unwrap()
698                .expect("graph");
699        assert!(graph.node_count() > 0, "rt graph");
700        let mut seq = SeqGraph::from_assembly_graph(&graph);
701        assert!(seq.node_count() > 0, "seq after from");
702        seq.clean_non_ref_paths();
703        let status = seq.cleanup_seq_graph();
704        assert!(seq.node_count() > 0, "after cleanup status={status:?}");
705        let path = seq.reference_path_bytes().unwrap();
706        assert_eq!(path.as_slice(), reference.bases.as_slice());
707    }
