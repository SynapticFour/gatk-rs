1//! Multisample HaplotypeCaller `isActive` (`HaplotypeCallerEngine`: joint `MinimalGenotypingEngine#calculateGenotypes`,
2//! AlleleFrequencyCalculator EM, QUAL → [`QualityUtils#qualToProb`](https://github.com/broadinstitute/gatk)).
3//! Per-sample genotype likelihoods are quantized like `htsjdk` `GenotypeBuilder#PL(double[])` before AF/QUAL, matching VC round-tripping in GATK.
4//! Assumes stratified nonempty per-sample piles (Java `AlignmentContext.splitContextBySampleName`).

15const JAVA_EM_THRESHOLD: f64 = 0.1;
16/// `AFCalculationResult.EPSILON`.
17const AFC_RESULT_EPSILON: f64 = 1.0e-10;

19/// Branch threshold for `NaturalLogUtils.log1mexp` (= `Math.log(0.5)` = `-LN_2`).
20const LOG1MEXP_THRESHOLD: f64 = -std::f64::consts::LN_2;
21
22const LOG10_TWO: f64 = std::f64::consts::LOG10_2;
23
24#[inline]
25fn log1mexp(a: f64) -> f64 {
26    if a > 0.0 {
27        f64::NAN
28    } else if a == 0.0 {
29        f64::NEG_INFINITY
30    } else if a < LOG1MEXP_THRESHOLD {
31        (1.0 - a.exp()).ln()
32    } else {
33        (-a.exp_m1()).ln()
34    }
35}

37/// GATK `MathUtils.log10OneMinusPow10`.
38pub fn log10_one_minus_pow10(a: f64) -> f64 {
39    if a > 0.0 {
40        f64::NAN
41    } else if a == 0.0 {
42        f64::NEG_INFINITY
43    } else {
44        let b = a * std::f64::consts::LN_10;
45        log1mexp(b) * std::f64::consts::LOG10_E
46    }
47}

49#[inline]
50fn qual_to_error_prob_log10(phred_threshold: f64) -> f64 {
51    phred_threshold * -0.1
52}

54#[inline]
55fn qual_to_prob_from_nonnegative_qual(qual: f64) -> f64 {
56    debug_assert!(qual >= 0.0, "qual={qual}");
57    1.0 - 10_f64.powf(-qual / 10.0)
58}

60fn dirichlet_log10_mean_weights(alpha: &[f64]) -> Vec<f64> {
61    let sum: f64 = alpha.iter().sum();
62    alpha.iter().map(|a| (a / sum).log10()).collect()
63}

65/// Heterozygous genotype log₁₀ combination multiplier (`MathUtils.log10Factorial(ploidy) − ∑log10(count!)`).
66#[inline]
67fn log10_combo_diploid_heterozygous() -> f64 {
68    LOG10_TWO
69}

71fn normalize_log10_in_place(vals: &mut [f64]) {
72    let s = log10_sum_log10(vals);
73    for x in vals.iter_mut() {
74        *x -= s;
75    }
76}

78fn genotype_log10_posteriors_diploid_biallelic_sample(
79    log10_gl: &[f64; 3],
80    log10_af: &[f64; 2],
81) -> [f64; 3] {
82    let combos = [0.0_f64, log10_combo_diploid_heterozygous(), 0.0];
83    let mut v = [
84        combos[0] + log10_gl[0] + 2.0 * log10_af[0],
85        combos[1] + log10_gl[1] + log10_af[0] + log10_af[1],
86        combos[2] + log10_gl[2] + 2.0 * log10_af[1],
87    ];
88    normalize_log10_in_place(&mut v);
89    v
90}

92fn effective_diploid_biallelic_allele_counts(
93    per_sample_gl: &[Vec<f64>],
94    log10_af: &[f64; 2],
95) -> Option<[f64; 2]> {
96    let mut acc = [f64::NEG_INFINITY, f64::NEG_INFINITY];
97    for gl in per_sample_gl {
98        let arr: &[f64; 3] = gl.as_slice().try_into().ok()?;
99        let post = genotype_log10_posteriors_diploid_biallelic_sample(arr, log10_af);
100        // `post[g]` is log₁₀ P(g|θ); allele copy mass is P(g)×count ⇒ log₁₀(P×c)=post[g]+log₁₀(c).
101        // `log10_sum_log10(post[g], log10(c))` would mean log₁₀(10^post+10^log c) and does **not** match Java `AF`.
102        acc[0] = log10_sum_log10(&[acc[0], post[0] + LOG10_TWO]);
103        acc[1] = log10_sum_log10(&[acc[1], post[2] + LOG10_TWO]);
104        acc[0] = log10_sum_log10(&[acc[0], post[1]]);
105        acc[1] = log10_sum_log10(&[acc[1], post[1]]);
106    }
107    Some([10_f64.powf(acc[0]), 10_f64.powf(acc[1])])
108}

110/// Allele-frequency EM + `calculateGenotypes` QUAL path (biallelic diploid, no spanning deletion).
111pub fn haplotype_caller_joint_multisample_is_active_activity_state(
112    contig: impl Into<String>,
113    pos: u64,
114    nonempty_sample_piles: &[&[PileupObservation]],
115    scoring: &HaplotypeCallerActivityScoringParams,
116) -> ActivityProfileState {
117    let contig = contig.into();
118    if nonempty_sample_piles.len() < 2 {
119        let merged: Vec<_> = nonempty_sample_piles
120            .iter()
121            .flat_map(|p| p.iter().copied())
122            .collect();
123        return haplotype_caller_activity_profile_state_minimal_genotyping(
124            contig, pos, &merged, scoring,
125        );
126    }
127
128    let hq_soft_clip_running_mean =
129        crate::activity_scoring::hq_soft_clip_running_mean_rcm_stratified(
130            nonempty_sample_piles,
131            scoring,
132        );
133
134    let mut per_sample_gl = Vec::with_capacity(nonempty_sample_piles.len());
135    for p in nonempty_sample_piles {
136        per_sample_gl.push(calc_ref_vs_any_log10_genotype_likelihoods(
137            scoring.sample_ploidy.as_u32(),
138            p,
139            scoring,
140        ));
141    }
142
143    let per_sample_java_pl: Vec<Vec<f64>> = per_sample_gl
144        .iter()
145        .map(|gl| genotype_log10_likelihoods_after_java_genotype_pl_roundtrip(gl))
146        .collect();
147
148    let active_prob =
149        if scoring.sample_ploidy.get() == 2 && per_sample_java_pl.iter().all(|gl| gl.len() == 3) {
150            joint_active_prob_via_af_em_and_qual(&per_sample_java_pl, scoring).unwrap_or(0.0)
151        } else {
152            crate::activity_scoring::calculate_multi_sample_any_non_ref_posterior(
153                &per_sample_java_pl,
154                true,
155                scoring,
156            )
157        };
158
159    let original = per_sample_gl
160        .iter()
161        .map(|gl| {
162            let max_i = max_element_index(gl);
163            gl[max_i] - gl[0]
164        })
165        .fold(f64::NEG_INFINITY, f64::max);
166
167    activity_state_meta(
168        contig,
169        pos,
170        active_prob,
171        original,
172        hq_soft_clip_running_mean,
173    )
174}

176fn activity_state_meta(
177    contig: String,
178    pos: u64,
179    active_prob: f64,
180    original: f64,
181    hq_soft_clip_running_mean: f64,
182) -> ActivityProfileState {
183    let evidence = if hq_soft_clip_running_mean
184        > crate::activity_scoring::AVERAGE_HQ_SOFTCLIPS_HQ_BASES_THRESHOLD
185    {
186        crate::activity_profile::ActivityEvidence::HighQualitySoftClips {
187            clip_bases: hq_soft_clip_running_mean as u32,
188        }
189    } else {
190        crate::activity_profile::ActivityEvidence::Plain
191    };
192
193    ActivityProfileState {
194        contig: std::sync::Arc::from(contig),
195        pos,
196        active_prob,
197        original_active_prob: original,
198        evidence,
199    }
200}

202/// `per_sample_gl` must match `htsjdk GenotypeLikelihoods` after VC construction (`PL(double[])` quantization).
203fn joint_active_prob_via_af_em_and_qual(
204    per_sample_gl: &[Vec<f64>],
205    scoring: &HaplotypeCallerActivityScoringParams,
206) -> Option<f64> {
207    // `HaplotypeCallerEngine` uses `FAKE_ALT = <FAKE_ALT>` (symbolic, `Allele#length` 0) vs `FAKE_REF = N`
208    // (length 1). `AlleleFrequencyCalculator` therefore applies `indelPseudocount` to the alt, not `snpPseudocount`.
209    let ref_pc = scoring.ref_pseudocount();
210    let alt_pc_prior = scoring.indel_heterozygosity * ref_pc;
211    let prior_counts = [ref_pc, alt_pc_prior];
212
213    let n_allele = 2_usize;
214    let flat_freq = -(n_allele as f64).log10();
215
216    let mut log10_af = [flat_freq, flat_freq];
217    let mut allele_counts = [0.0_f64; 2];
218
219    for _ in 0..256 {
220        let new_counts = effective_diploid_biallelic_allele_counts(per_sample_gl, &log10_af)?;
221        let max_diff = (new_counts[0] - allele_counts[0])
222            .abs()
223            .max((new_counts[1] - allele_counts[1]).abs());
224        allele_counts = new_counts;
225        let posterior_pc = [
226            prior_counts[0] + allele_counts[0],
227            prior_counts[1] + allele_counts[1],
228        ];
229        log10_af = {
230            let w = dirichlet_log10_mean_weights(&posterior_pc);
231            [w[0], w[1]]
232        };
233        if max_diff <= JAVA_EM_THRESHOLD {
234            break;
235        }
236    }
237
238    let mut log10_p_no_variant = 0.0_f64;
239    for gl in per_sample_gl {
240        let arr: &[f64; 3] = gl.as_slice().try_into().ok()?;
241        let post = genotype_log10_posteriors_diploid_biallelic_sample(arr, &log10_af);
242        // GATK uses `+=` over per-sample log10 P(hom-ref | θ), i.e. log10 ∏_s p_s.
243        log10_p_no_variant += post[0];
244    }
245
246    let call_conf = scoring.active_region_standard_confidence_for_calling;
247    let alt_plausible =
248        log10_p_no_variant + AFC_RESULT_EPSILON < qual_to_error_prob_log10(call_conf);
249    let site_is_monomorphic = !alt_plausible;
250
251    let log10_vc_confidence = vc_log10_p_error_assignment(
252        site_is_monomorphic,
253        scoring.annotate_all_sites_with_pls,
254        log10_p_no_variant,
255    );
256    let phred_scaled_qual = (-10.0_f64) * log10_vc_confidence;
257
258    let emits = gc_passes_emit_threshold(
259        phred_scaled_qual,
260        site_is_monomorphic,
261        scoring.emit_all_confident_sites,
262        scoring.active_region_standard_confidence_for_calling,
263    ) && supplemental_emit_filters();
264
265    if !emits {
266        return Some(0.0);
267    }
268
269    Some(qual_to_prob_from_nonnegative_qual(phred_scaled_qual))
270}

272/// Remaining unconditional `calculateGenotypes` veto paths are no-ops for HC IsActive probes (no spanning del, allele cap).
273#[inline]
274fn supplemental_emit_filters() -> bool {
275    true
276}
277
278/// `GenotypingEngine#calculateGenotypes` veto that requires `calculateGenotypes` continuation (false here).
279/// Mirrors `emitAllActiveSites` / GVCF quirks absent in active-region evaluator.
280#[inline]
281fn vc_log10_p_error_assignment(
282    site_is_monomorphic: bool,
283    annotate_all_pls: bool,
284    log10_p_posterior_no_variant: f64,
285) -> f64 {
286    if !site_is_monomorphic || annotate_all_pls {
287        log10_p_posterior_no_variant + 0.0
288    } else {
289        log10_one_minus_pow10(log10_p_posterior_no_variant) + 0.0
290    }
291}
292
293fn gc_passes_emit_threshold(
294    phred_scaled_confidence: f64,
295    site_is_monomorphic: bool,
296    emit_all_confident_sites: bool,
297    standard_confidence_for_calling: f64,
298) -> bool {
299    let mode_all_confident = emit_all_confident_sites;
300    (mode_all_confident || !site_is_monomorphic)
301        && phred_scaled_confidence >= standard_confidence_for_calling
302}
