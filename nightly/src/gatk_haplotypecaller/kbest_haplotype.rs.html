8/// One haplotype path with GATK log-penalty score.
9/// # Invariants
10/// `edges` form an ordered walk from `start` through the [`AssemblyGraph`].
11/// `score` is the GATK log-penalty path score used for k-best ranking.
12/// # Ownership
13/// Owns edge list; borrows graph in [`KBestPath::bases`].
14/// # Mutation
15/// Immutable path after k-best search.
16/// # Biological assumptions
17/// Candidate haplotype sequence over the local assembly graph.
18/// # Java equivalence
19/// GATK `KBestHaplotype` / `GraphBasedKBestHaplotypeFinder` path.
20#[derive(Debug, Clone)]
21pub struct KBestPath {
22    pub start: usize,
23    pub edges: Vec<(usize, usize)>,
24    pub score: f64,
25    pub is_reference: bool,
26}
27
28impl KBestPath {
29    pub fn bases(&self, graph: &AssemblyGraph) -> Vec<u8> {
30        graph.path_bases(self.start, &self.edges)
31    }
32}

34/// Cap on in-flight k-best frontier paths. Bushy/cyclic local graphs (NA12878
35/// `20:10098169-10098441`) otherwise fill this heap with long `PathState`s and reach
36/// multi-GiB Peak-RSS — the realistic-window failure mode.
37///
38/// Keep well below `haplotype_budget × max_path_edges × 16B` ≈ tens of MiB.
39const MAX_KBEST_HEAP_PATHS: usize = 1_024;
40
41/// Cap on edges in a single in-flight path. Cyclic graphs otherwise let `edges` grow
42/// without bound → GiB-scale PathStates.
43///
44/// Must clear padded HC spans (active + ~500 bp each side → often 1–2 kb of k-mers,
45/// i.e. ≫256 edges). Cap 256 zeroed k-best on normal GIAB windows (chr21 recall
46/// collapse: `paths=0` everywhere, dangling fragments only). Cyclic wander remains
47/// bounded by heap / expansion caps and per-vertex visit limits.
48const MAX_KBEST_PATH_EDGES: usize = 4_096;

50#[derive(Debug, Clone)]
51struct PathState {
52    start: usize,
53    edges: Vec<(usize, usize)>,
54    last: usize,
55    score: f64,
56    is_reference: bool,
57    /// Cached `path_bases(..).len()` — avoid rebuilding the haplotype string on every heap push.
58    bases_len: usize,
59}
60
61impl PathState {
62    fn new(graph: &AssemblyGraph, start: usize) -> Self {
63        Self {
64            start,
65            edges: Vec::new(),
66            last: start,
67            score: 0.0,
68            is_reference: false,
69            bases_len: graph.nodes()[start].kmer.len(),
70        }
71    }
72
73    fn extend(
74        &self,
75        graph: &AssemblyGraph,
76        to: usize,
77        edge_support: u32,
78        total_outgoing: u32,
79    ) -> Self {
80        // Avoid `Vec::clone` on the hot k-best frontier (OOM amplifier on bushy/cyclic graphs).
81        let mut edges = Vec::with_capacity(self.edges.len() + 1);
82        edges.extend_from_slice(&self.edges);
83        edges.push((self.last, to));
84        let penalty = log_penalty(edge_support, total_outgoing);
85        // Each edge appends the last base of the destination kmer (`AssemblyGraph::path_bases`).
86        let add = usize::from(graph.nodes()[to].kmer.last().is_some());
87        Self {
88            start: self.start,
89            edges,
90            last: to,
91            score: self.score + penalty,
92            is_reference: self.is_reference && graph.edge_is_ref(self.last, to),
93            bases_len: self.bases_len + add,
94        }
95    }
96}

98pub(crate) fn log_penalty(edge_multiplicity: u32, total_outgoing_multiplicity: u32) -> f64 {
99    if total_outgoing_multiplicity == 0 {
100        return 0.0;
101    }
102    (edge_multiplicity.max(1) as f64).log10() - (total_outgoing_multiplicity.max(1) as f64).log10()
103}

105struct HeapItem {
106    score_bits: u64,
107    tie: usize,
108    path: PathState,
109}
110
111impl Ord for HeapItem {
112    fn cmp(&self, other: &Self) -> Ordering {
113        self.score_bits
114            .cmp(&other.score_bits)
115            .then_with(|| other.tie.cmp(&self.tie))
116    }
117}
118
119impl PartialOrd for HeapItem {
120    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
121        Some(self.cmp(other))
122    }
123}
124
125impl PartialEq for HeapItem {
126    fn eq(&self, other: &Self) -> bool {
127        self.score_bits == other.score_bits && self.tie == other.tie
128    }
129}

133/// Remove cycle edges / dead vertices (GATK `KBestHaplotypeFinder.removeCyclesIfNecessary`).
134///
135/// On failure returns the **original** graph in [`Err`] so callers can fall back to a
136/// preserving search without `graph.clone()` (Peak-RSS on bushy cyclic loci).
137///
138/// Observable contract: same cycle-edge / non-reaching-vertex removal as GATK; Rust-native
139/// cuts avoid per-edge `Vec` allocs and fold isolated cleanup into one remapping pass.
140///
141/// # Complexity
142/// `find_cycle_guilty` must memoize per-vertex `reaches_sink`. Without memoization a bushy
143/// DAG (many diamond merges) is exponential in path count — NA12878 `20:47131737-47131971`
144/// hung for the full 6h CI job after `rt_graph_built` and before `kbest_begin`.
145pub fn graph_for_kbest(mut graph: AssemblyGraph) -> Result<AssemblyGraph, AssemblyGraph> {
146    if !graph.has_cycle() {
147        return Ok(graph);
148    }
149    let Some(source) = graph.reference_source_vertex() else {
150        return Err(graph);
151    };
152    let Some(sink) = graph.reference_sink_vertex() else {
153        return Err(graph);
154    };
155    let n = graph.node_count();
156    let mut is_sink = vec![false; n];
157    if sink < n {
158        is_sink[sink] = true;
159    }
160    let mut edges_to_remove = HashSet::new();
161    let mut vertices_to_remove = HashSet::new();
162    let mut on_path = vec![false; n];
163    // Memo: None = unvisited as a finished subtree; Some(reaches_sink).
164    let mut reaches_sink_memo: Vec<Option<bool>> = vec![None; n];
165    let found_path = find_cycle_guilty(
166        &graph,
167        source,
168        &is_sink,
169        &mut edges_to_remove,
170        &mut vertices_to_remove,
171        &mut on_path,
172        &mut reaches_sink_memo,
173    );
174    if !found_path || (edges_to_remove.is_empty() && vertices_to_remove.is_empty()) {
175        return Err(graph);
176    }
177    for (f, t) in edges_to_remove {
178        graph.remove_edge(f, t);
179    }
180    // Fold isolated cleanup into one remapping pass (was remove_nodes + cleanup_isolated).
181    graph.remove_nodes_and_isolated(vertices_to_remove);
182    Ok(graph)
183}

185fn find_cycle_guilty(
186    graph: &AssemblyGraph,
187    current: usize,
188    is_sink: &[bool],
189    edges_to_remove: &mut HashSet<(usize, usize)>,
190    vertices_to_remove: &mut HashSet<usize>,
191    on_path: &mut [bool],
192    reaches_sink_memo: &mut [Option<bool>],
193) -> bool {
194    if current < is_sink.len() && is_sink[current] {
195        return true;
196    }
197    if current < reaches_sink_memo.len() {
198        if let Some(cached) = reaches_sink_memo[current] {
199            return cached;
200        }
201    }
202    if current < on_path.len() {
203        on_path[current] = true;
204    }
205    let mut reaches_sink = false;
206    for to in graph.outgoing_targets(current) {
207        if to < on_path.len() && on_path[to] {
208            edges_to_remove.insert((current, to));
209        } else if find_cycle_guilty(
210            graph,
211            to,
212            is_sink,
213            edges_to_remove,
214            vertices_to_remove,
215            on_path,
216            reaches_sink_memo,
217        ) {
218            reaches_sink = true;
219        }
220    }
221    if !reaches_sink {
222        vertices_to_remove.insert(current);
223    }
224    if current < on_path.len() {
225        on_path[current] = false;
226    }
227    if current < reaches_sink_memo.len() {
228        reaches_sink_memo[current] = Some(reaches_sink);
229    }
230    reaches_sink
231}

249/// Production assembly k-best: prefer an acyclic graph. Cyclic graphs previously kept
250/// full topology (P12) and could grow the k-best frontier without bound on bushy
251/// loci (e.g. NA12878 `20:10098169`); try cycle removal first, then fall back to the
252/// capped preserving search.
253///
254/// Takes ownership so cycle stripping does not `graph.clone()` (Peak-RSS on bushy loci).
255pub fn find_best_haplotypes_for_assembly(
256    graph: AssemblyGraph,
257    max_number_of_haplotypes: usize,
258) -> GatkResult<(Vec<KBestPath>, AssemblyGraph)> {
259    // Observable contract: prefer acyclic k-best; when cycle stripping cannot produce a
260    // source→sink DAG, fall back to capped preserving search (pre-Peak behavior). Dropping
261    // that fallback and running `find_best_haplotypes_inner` on the cyclic graph left L2
262    // `g2-subset-live` at haplotype_count=1 (ref only).
263    crate::runtime_config::rss_trace_checkpoint(
264        "kbest_cycle_strip_begin",
265        &format!("nodes={}", graph.nodes().len()),
266    );
267    let stripped = graph_for_kbest(graph);
268    crate::runtime_config::rss_trace_checkpoint("kbest_cycle_strip_done", "");
269    let (paths, graph) = match stripped {
270        Ok(acyclic) => {
271            crate::runtime_config::rss_trace_checkpoint(
272                "kbest_begin",
273                &format!(
274                    "nodes={} max_haps={} mode=acyclic",
275                    acyclic.nodes().len(),
276                    max_number_of_haplotypes
277                ),
278            );
279            let paths = find_best_haplotypes_inner(&acyclic, max_number_of_haplotypes, false)?;
280            (paths, acyclic)
281        }
282        Err(cyclic) => {
283            // Peak-RSS: cyclic-preserve k-best on bushy graphs (≤8k nodes still) is the
284            // multi-GiB amplifier. Keep preserve for small cyclic L2 fixtures only.
285            const MAX_CYCLIC_PRESERVE_NODES: usize = 4_000;
286            if cyclic.nodes().len() > MAX_CYCLIC_PRESERVE_NODES {
287                crate::runtime_config::rss_trace_checkpoint(
288                    "kbest_cyclic_skip_huge",
289                    &format!(
290                        "nodes={} cap={MAX_CYCLIC_PRESERVE_NODES}",
291                        cyclic.nodes().len()
292                    ),
293                );
294                (Vec::new(), cyclic)
295            } else {
296                crate::runtime_config::rss_trace_checkpoint(
297                    "kbest_cyclic_preserve",
298                    &format!(
299                        "nodes={} max_haps={}",
300                        cyclic.nodes().len(),
301                        max_number_of_haplotypes
302                    ),
303                );
304                let paths =
305                    find_best_haplotypes_preserving_cycles(&cyclic, max_number_of_haplotypes)?;
306                (paths, cyclic)
307            }
308        }
309    };
310    crate::runtime_config::rss_trace_checkpoint("kbest_done", &format!("paths={}", paths.len()));
311    Ok((paths, graph))
312}

314fn find_best_haplotypes_inner(
315    graph: &AssemblyGraph,
316    max_number_of_haplotypes: usize,
317    strip_cycles: bool,
318) -> GatkResult<Vec<KBestPath>> {
319    if max_number_of_haplotypes == 0 {
320        return Ok(Vec::new());
321    }
322    // Borrow when possible; only clone for cycle stripping (find_best_haplotypes public API).
323    let owned;
324    let graph: &AssemblyGraph = if strip_cycles {
325        owned = match graph_for_kbest(graph.clone()) {
326            Ok(g) => g,
327            Err(g) => g, // preserve cyclic topology when strip fails
328        };
329        &owned
330    } else {
331        graph
332    };
333    let source = graph
334        .reference_source_vertex()
335        .ok_or_else(|| GatkError::algorithm("kbest: no reference source vertex"))?;
336    let sink = graph
337        .reference_sink_vertex()
338        .ok_or_else(|| GatkError::algorithm("kbest: no reference sink vertex"))?;
339    let sources = [source];
340    let sinks: HashSet<usize> = HashSet::from([sink]);
341
342    let mut result = Vec::new();
343    let mut heap: BinaryHeap<HeapItem> = BinaryHeap::new();
344    for &s in &sources {
345        let path = PathState::new(graph, s);
346        heap.push(HeapItem {
347            score_bits: path.score.to_bits(),
348            tie: path.bases_len,
349            path,
350        });
351    }
352    let mut vertex_counts: HashMap<usize, usize> =
353        graph.nodes().iter().map(|n| (n.id, 0)).collect();
354
355    // Bound total expansions: at the heap cap, a bushy/cyclic graph otherwise spins
356    // forever in pop/extend/push, fragmenting the allocator into multi-GiB Peak-RSS.
357    // 50k was still enough for ≤8k-node dense GIAB shards to climb to ~15 GiB.
358    const MAX_KBEST_EXPANSIONS: usize = 12_000;
359    let mut expansions = 0usize;
360    while !heap.is_empty() && result.len() < max_number_of_haplotypes {
361        if crate::runtime_config::hc_rss_abort_triggered() {
362            crate::runtime_config::rss_trace_checkpoint(
363                "kbest_rss_abort",
364                &format!("expansions={expansions} results={}", result.len()),
365            );
366            break;
367        }
368        let item = heap.pop().expect("non-empty");
369        let path = item.path;
370        if sinks.contains(&path.last) {
371            result.push(KBestPath {
372                start: path.start,
373                edges: path.edges,
374                score: path.score,
375                is_reference: path.is_reference,
376            });
377            continue;
378        }
379        if path.edges.len() >= MAX_KBEST_PATH_EDGES || expansions >= MAX_KBEST_EXPANSIONS {
380            continue;
381        }
382        if expansions > 0 && expansions % 5_000 == 0 {
383            crate::runtime_config::rss_trace_checkpoint(
384                "kbest_expand",
385                &format!(
386                    "expansions={expansions} heap={} results={}",
387                    heap.len(),
388                    result.len()
389                ),
390            );
391        }
392        // If the frontier is already saturated, only accept sinks (above); do not expand.
393        if heap.len() >= MAX_KBEST_HEAP_PATHS {
394            continue;
395        }
396        let count = vertex_counts.get_mut(&path.last).expect("vertex");
397        if *count < max_number_of_haplotypes {
398            *count += 1;
399            expansions += 1;
400            let outs = graph.outgoing_nodes(path.last);
401            let total: u32 = outs
402                .iter()
403                .filter_map(|&t| graph.edge_support(path.last, t))
404                .sum();
405            for to in outs {
406                if let Some(support) = graph.edge_support(path.last, to) {
407                    if heap.len() >= MAX_KBEST_HEAP_PATHS {
408                        break;
409                    }
410                    let extended = path.extend(graph, to, support, total);
411                    heap.push(HeapItem {
412                        score_bits: extended.score.to_bits(),
413                        tie: extended.bases_len,
414                        path: extended,
415                    });
416                }
417            }
418        }
419    }
420
421    result.sort_by(|a, b| {
422        b.score
423            .partial_cmp(&a.score)
424            .unwrap_or(Ordering::Equal)
425            .then_with(|| b.bases(graph).cmp(&a.bases(graph)))
426    });
427    Ok(result)
428}

481    /// Bushy diamond ladder + one back-edge: without per-vertex memoization,
482    /// `find_cycle_guilty` is exponential in diamond count (CI hang class).
483    #[test]
484    fn graph_for_kbest_memoizes_bushy_cyclic_dag() {
485        let mut g = AssemblyGraph::new(3).unwrap();
486        let kmer = |i: usize| -> [u8; 3] {
487            [
488                b'A' + ((i / 16) % 4) as u8,
489                b'A' + ((i / 4) % 4) as u8,
490                b'A' + (i % 4) as u8,
491            ]
492        };
493        // layers=14 → 2^14 path expansions without memo (~16k leaves of the recursion tree
494        // per diamond level compounds far beyond that).
495        let layers = 14usize;
496        let source = g.ensure_node(&kmer(0));
497        let mut prev = source;
498        let mut next_id = 1usize;
499        let mut ref_chain: Vec<(usize, usize)> = Vec::new();
500        for _ in 0..layers {
501            let left = g.ensure_node(&kmer(next_id));
502            next_id += 1;
503            let right = g.ensure_node(&kmer(next_id));
504            next_id += 1;
505            let merge = g.ensure_node(&kmer(next_id));
506            next_id += 1;
507            g.add_edge_support(prev, left, 1);
508            g.add_edge_support(prev, right, 1);
509            g.add_edge_support(left, merge, 1);
510            g.add_edge_support(right, merge, 1);
511            // Ref spine uses left branch.
512            ref_chain.push((prev, left));
513            ref_chain.push((left, merge));
514            prev = merge;
515        }
516        let sink = prev;
517        for &(a, b) in &ref_chain {
518            g.ref_edges.insert((a, b));
519            g.ref_nodes.insert(a);
520            g.ref_nodes.insert(b);
521        }
522        g.ref_source_kmer = Some(std::sync::Arc::from(kmer(0).as_slice()));
523        // Back-edge so has_cycle() is true and we enter find_cycle_guilty.
524        let mid = g.ensure_node(&kmer(next_id));
525        g.add_edge_support(sink, mid, 1);
526        g.add_edge_support(mid, source, 1);
527
528        assert!(g.has_cycle(), "test setup must enter cycle-strip path");
529        assert!(g.reference_source_vertex().is_some());
530        assert!(g.reference_sink_vertex().is_some());
531        let t0 = std::time::Instant::now();
532        let _ = graph_for_kbest(g);
533        assert!(
534            t0.elapsed().as_secs() < 2,
535            "cycle-strip on bushy diamond ladder must finish quickly (memoization)"
536        );
537    }
