8/// One haplotype path with GATK log-penalty score.
9/// # Invariants
10/// `edges` form an ordered walk from `start` through the [`AssemblyGraph`].
11/// `score` is the GATK log-penalty path score used for k-best ranking.
12/// # Ownership
13/// Owns edge list; borrows graph in [`KBestPath::bases`].
14/// # Mutation
15/// Immutable path after k-best search.
16/// # Biological assumptions
17/// Candidate haplotype sequence over the local assembly graph.
18/// # Java equivalence
19/// GATK `KBestHaplotype` / `GraphBasedKBestHaplotypeFinder` path.
20#[derive(Debug, Clone)]
21pub struct KBestPath {
22    pub start: usize,
23    pub edges: Vec<(usize, usize)>,
24    pub score: f64,
25    pub is_reference: bool,
26}
27
28impl KBestPath {
29    pub fn bases(&self, graph: &AssemblyGraph) -> String {
30        graph.path_bases(self.start, &self.edges)
31    }
32}
33
34#[derive(Debug, Clone)]
35struct PathState {
36    start: usize,
37    edges: Vec<(usize, usize)>,
38    last: usize,
39    score: f64,
40    is_reference: bool,
41}
42
43impl PathState {
44    fn new(start: usize) -> Self {
45        Self {
46            start,
47            edges: Vec::new(),
48            last: start,
49            score: 0.0,
50            is_reference: false,
51        }
52    }
53
54    fn extend(
55        &self,
56        graph: &AssemblyGraph,
57        to: usize,
58        edge_support: u32,
59        total_outgoing: u32,
60    ) -> Self {
61        let mut edges = self.edges.clone();
62        edges.push((self.last, to));
63        let penalty = log_penalty(edge_support, total_outgoing);
64        Self {
65            start: self.start,
66            edges,
67            last: to,
68            score: self.score + penalty,
69            is_reference: self.is_reference && graph.edge_is_ref(self.last, to),
70        }
71    }
72}
73
74pub(crate) fn log_penalty(edge_multiplicity: u32, total_outgoing_multiplicity: u32) -> f64 {
75    if total_outgoing_multiplicity == 0 {
76        return 0.0;
77    }
78    (edge_multiplicity.max(1) as f64).log10() - (total_outgoing_multiplicity.max(1) as f64).log10()
79}

81struct HeapItem {
82    score_bits: u64,
83    tie: usize,
84    path: PathState,
85}
86
87impl Ord for HeapItem {
88    fn cmp(&self, other: &Self) -> Ordering {
89        self.score_bits
90            .cmp(&other.score_bits)
91            .then_with(|| other.tie.cmp(&self.tie))
92    }
93}
94
95impl PartialOrd for HeapItem {
96    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
97        Some(self.cmp(other))
98    }
99}
100
101impl PartialEq for HeapItem {
102    fn eq(&self, other: &Self) -> bool {
103        self.score_bits == other.score_bits && self.tie == other.tie
104    }
105}
106
107impl Eq for HeapItem {}

109/// Remove cycle edges / dead vertices (GATK `KBestHaplotypeFinder.removeCyclesIfNecessary`).
110pub fn graph_for_kbest(mut graph: AssemblyGraph) -> GatkResult<AssemblyGraph> {
111    if !graph.has_cycle() {
112        return Ok(graph);
113    }
114    let sources: Vec<usize> = graph.reference_source_vertex().into_iter().collect();
115    let sinks: Vec<usize> = graph.reference_sink_vertex().into_iter().collect();
116    if sources.is_empty() || sinks.is_empty() {
117        return Err(GatkError::algorithm(
118            "kbest: missing reference source or sink for cycle removal",
119        ));
120    }
121    let sink_set: HashSet<usize> = sinks.into_iter().collect();
122    let mut edges_to_remove = HashSet::new();
123    let mut vertices_to_remove = HashSet::new();
124    let mut found_path = false;
125    for source in sources {
126        let mut parents = HashSet::new();
127        if find_cycle_guilty(
128            &graph,
129            source,
130            &sink_set,
131            &mut edges_to_remove,
132            &mut vertices_to_remove,
133            &mut parents,
134        ) {
135            found_path = true;
136        }
137    }
138    if !found_path {
139        return Err(GatkError::algorithm(
140            "kbest: no path from source to sink after cycle analysis",
141        ));
142    }
143    if edges_to_remove.is_empty() && vertices_to_remove.is_empty() {
144        return Err(GatkError::algorithm("kbest: cannot remove cycles"));
145    }
146    for (f, t) in edges_to_remove {
147        graph.remove_edge(f, t);
148    }
149    graph.remove_nodes(&vertices_to_remove);
150    graph.cleanup_isolated_nodes();
151    Ok(graph)
152}
153
154fn find_cycle_guilty(
155    graph: &AssemblyGraph,
156    current: usize,
157    sinks: &HashSet<usize>,
158    edges_to_remove: &mut HashSet<(usize, usize)>,
159    vertices_to_remove: &mut HashSet<usize>,
160    parent_vertices: &mut HashSet<usize>,
161) -> bool {
162    if sinks.contains(&current) {
163        return true;
164    }
165    parent_vertices.insert(current);
166    let outs = graph.outgoing_nodes(current);
167    let mut reaches_sink = false;
168    for to in outs {
169        if parent_vertices.contains(&to) {
170            edges_to_remove.insert((current, to));
171        } else if find_cycle_guilty(
172            graph,
173            to,
174            sinks,
175            edges_to_remove,
176            vertices_to_remove,
177            parent_vertices,
178        ) {
179            reaches_sink = true;
180        }
181    }
182    if !reaches_sink {
183        vertices_to_remove.insert(current);
184    }
185    parent_vertices.remove(&current);
186    reaches_sink
187}

189/// GATK `GraphBasedKBestHaplotypeFinder.findBestHaplotypes`.
190pub fn find_best_haplotypes(
191    graph: &AssemblyGraph,
192    max_number_of_haplotypes: usize,
193) -> GatkResult<Vec<KBestPath>> {
194    find_best_haplotypes_inner(graph, max_number_of_haplotypes, true)
195}

197/// K-best on the graph as-is (no cycle stripping). Parity diagnostics only.
198pub fn find_best_haplotypes_preserving_cycles(
199    graph: &AssemblyGraph,
200    max_number_of_haplotypes: usize,
201) -> GatkResult<Vec<KBestPath>> {
202    find_best_haplotypes_inner(graph, max_number_of_haplotypes, false)
203}

205/// Production assembly: strip cycles only when the graph is acyclic (P12 cyclic regions keep topology).
206pub fn find_best_haplotypes_for_assembly(
207    graph: &AssemblyGraph,
208    max_number_of_haplotypes: usize,
209) -> GatkResult<Vec<KBestPath>> {
210    if graph.has_cycle() {
211        find_best_haplotypes_preserving_cycles(graph, max_number_of_haplotypes)
212    } else {
213        find_best_haplotypes(graph, max_number_of_haplotypes)
214    }
215}
216
217fn find_best_haplotypes_inner(
218    graph: &AssemblyGraph,
219    max_number_of_haplotypes: usize,
220    strip_cycles: bool,
221) -> GatkResult<Vec<KBestPath>> {
222    if max_number_of_haplotypes == 0 {
223        return Ok(Vec::new());
224    }
225    let graph = if strip_cycles {
226        graph_for_kbest(graph.clone())?
227    } else {
228        graph.clone()
229    };
230    let source = graph
231        .reference_source_vertex()
232        .ok_or_else(|| GatkError::algorithm("kbest: no reference source vertex"))?;
233    let sink = graph
234        .reference_sink_vertex()
235        .ok_or_else(|| GatkError::algorithm("kbest: no reference sink vertex"))?;
236    let sources = [source];
237    let sinks: HashSet<usize> = HashSet::from([sink]);
238
239    let mut result = Vec::new();
240    let mut heap: BinaryHeap<HeapItem> = BinaryHeap::new();
241    for &s in &sources {
242        let path = PathState::new(s);
243        heap.push(HeapItem {
244            score_bits: path.score.to_bits(),
245            tie: 0,
246            path,
247        });
248    }
249    let mut vertex_counts: HashMap<usize, usize> =
250        graph.nodes().iter().map(|n| (n.id, 0)).collect();
251
252    while !heap.is_empty() && result.len() < max_number_of_haplotypes {
253        let item = heap.pop().expect("non-empty");
254        let path = item.path;
255        if sinks.contains(&path.last) {
256            result.push(KBestPath {
257                start: path.start,
258                edges: path.edges,
259                score: path.score,
260                is_reference: path.is_reference,
261            });
262        } else {
263            let count = vertex_counts.get_mut(&path.last).expect("vertex");
264            if *count < max_number_of_haplotypes {
265                *count += 1;
266                let outs = graph.outgoing_nodes(path.last);
267                let total: u32 = outs
268                    .iter()
269                    .filter_map(|&t| graph.edge_support(path.last, t))
270                    .sum();
271                for to in outs {
272                    if let Some(support) = graph.edge_support(path.last, to) {
273                        let extended = path.extend(&graph, to, support, total);
274                        let tie = graph.path_bases(extended.start, &extended.edges).len();
275                        heap.push(HeapItem {
276                            score_bits: extended.score.to_bits(),
277                            tie,
278                            path: extended,
279                        });
280                    }
281                }
282            }
283        }
284    }
285
286    result.sort_by(|a, b| {
287        b.score
288            .partial_cmp(&a.score)
289            .unwrap_or(Ordering::Equal)
290            .then_with(|| b.bases(&graph).cmp(&a.bases(&graph)))
291    });
292    Ok(result)
293}
