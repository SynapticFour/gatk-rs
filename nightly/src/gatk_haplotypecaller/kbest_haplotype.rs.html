8/// One haplotype path with GATK log-penalty score.
9/// # Invariants
10/// `edges` form an ordered walk from `start` through the [`AssemblyGraph`].
11/// `score` is the GATK log-penalty path score used for k-best ranking.
12/// # Ownership
13/// Owns edge list; borrows graph in [`KBestPath::bases`].
14/// # Mutation
15/// Immutable path after k-best search.
16/// # Biological assumptions
17/// Candidate haplotype sequence over the local assembly graph.
18/// # Java equivalence
19/// GATK `KBestHaplotype` / `GraphBasedKBestHaplotypeFinder` path.
20#[derive(Debug, Clone)]
21pub struct KBestPath {
22    pub start: usize,
23    pub edges: Vec<(usize, usize)>,
24    pub score: f64,
25    pub is_reference: bool,
26}
27
28impl KBestPath {
29    pub fn bases(&self, graph: &AssemblyGraph) -> Vec<u8> {
30        graph.path_bases(self.start, &self.edges)
31    }
32}

34/// Cap on in-flight k-best frontier paths. Bushy/cyclic local graphs (NA12878
35/// `20:10098169-10098441`) otherwise fill this heap with long `PathState`s and reach
36/// multi-GiB Peak-RSS — the realistic-window failure mode.
37///
38/// Keep well below `haplotype_budget × max_path_edges × 16B` ≈ tens of MiB.
39const MAX_KBEST_HEAP_PATHS: usize = 1_024;
40
41/// Cap on edges in a single in-flight path. Cyclic graphs otherwise let `edges` grow
42/// without bound → GiB-scale PathStates.
43///
44/// Must clear padded HC spans (active + ~500 bp each side → often 1–2 kb of k-mers,
45/// i.e. ≫256 edges). Cap 256 zeroed k-best on normal GIAB windows (chr21 recall
46/// collapse: `paths=0` everywhere, dangling fragments only). Cyclic wander remains
47/// bounded by heap / expansion caps and per-vertex visit limits.
48const MAX_KBEST_PATH_EDGES: usize = 4_096;
49
50#[derive(Debug, Clone)]
51struct PathState {
52    start: usize,
53    edges: Vec<(usize, usize)>,
54    last: usize,
55    score: f64,
56    is_reference: bool,
57    /// Cached `path_bases(..).len()` — avoid rebuilding the haplotype string on every heap push.
58    bases_len: usize,
59}
60
61impl PathState {
62    fn new(graph: &AssemblyGraph, start: usize) -> Self {
63        Self {
64            start,
65            edges: Vec::new(),
66            last: start,
67            score: 0.0,
68            is_reference: false,
69            bases_len: graph.nodes()[start].kmer.len(),
70        }
71    }
72
73    fn extend(
74        &self,
75        graph: &AssemblyGraph,
76        to: usize,
77        edge_support: u32,
78        total_outgoing: u32,
79    ) -> Self {
80        // Avoid `Vec::clone` on the hot k-best frontier (OOM amplifier on bushy/cyclic graphs).
81        let mut edges = Vec::with_capacity(self.edges.len() + 1);
82        edges.extend_from_slice(&self.edges);
83        edges.push((self.last, to));
84        let penalty = log_penalty(edge_support, total_outgoing);
85        // Each edge appends the last base of the destination kmer (`AssemblyGraph::path_bases`).
86        let add = usize::from(graph.nodes()[to].kmer.last().is_some());
87        Self {
88            start: self.start,
89            edges,
90            last: to,
91            score: self.score + penalty,
92            is_reference: self.is_reference && graph.edge_is_ref(self.last, to),
93            bases_len: self.bases_len + add,
94        }
95    }
96}

98pub(crate) fn log_penalty(edge_multiplicity: u32, total_outgoing_multiplicity: u32) -> f64 {
99    if total_outgoing_multiplicity == 0 {
100        return 0.0;
101    }
102    (edge_multiplicity.max(1) as f64).log10() - (total_outgoing_multiplicity.max(1) as f64).log10()
103}

105struct HeapItem {
106    score_bits: u64,
107    tie: usize,
108    path: PathState,
109}
110
111impl Ord for HeapItem {
112    fn cmp(&self, other: &Self) -> Ordering {
113        self.score_bits
114            .cmp(&other.score_bits)
115            .then_with(|| other.tie.cmp(&self.tie))
116    }
117}
118
119impl PartialOrd for HeapItem {
120    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
121        Some(self.cmp(other))
122    }
123}
124
125impl PartialEq for HeapItem {
126    fn eq(&self, other: &Self) -> bool {
127        self.score_bits == other.score_bits && self.tie == other.tie
128    }
129}
130
131impl Eq for HeapItem {}

133/// Remove cycle edges / dead vertices (GATK `KBestHaplotypeFinder.removeCyclesIfNecessary`).
134///
135/// On failure returns the **original** graph in [`Err`] so callers can fall back to a
136/// preserving search without `graph.clone()` (Peak-RSS on bushy cyclic loci).
137pub fn graph_for_kbest(mut graph: AssemblyGraph) -> Result<AssemblyGraph, AssemblyGraph> {
138    if !graph.has_cycle() {
139        return Ok(graph);
140    }
141    let sources: Vec<usize> = graph.reference_source_vertex().into_iter().collect();
142    let sinks: Vec<usize> = graph.reference_sink_vertex().into_iter().collect();
143    if sources.is_empty() || sinks.is_empty() {
144        return Err(graph);
145    }
146    let sink_set: HashSet<usize> = sinks.into_iter().collect();
147    let mut edges_to_remove = HashSet::new();
148    let mut vertices_to_remove = HashSet::new();
149    let mut found_path = false;
150    for source in sources {
151        let mut parents = HashSet::new();
152        if find_cycle_guilty(
153            &graph,
154            source,
155            &sink_set,
156            &mut edges_to_remove,
157            &mut vertices_to_remove,
158            &mut parents,
159        ) {
160            found_path = true;
161        }
162    }
163    if !found_path || (edges_to_remove.is_empty() && vertices_to_remove.is_empty()) {
164        return Err(graph);
165    }
166    for (f, t) in edges_to_remove {
167        graph.remove_edge(f, t);
168    }
169    graph.remove_nodes(&vertices_to_remove);
170    graph.cleanup_isolated_nodes();
171    Ok(graph)
172}
173
174fn find_cycle_guilty(
175    graph: &AssemblyGraph,
176    current: usize,
177    sinks: &HashSet<usize>,
178    edges_to_remove: &mut HashSet<(usize, usize)>,
179    vertices_to_remove: &mut HashSet<usize>,
180    parent_vertices: &mut HashSet<usize>,
181) -> bool {
182    if sinks.contains(&current) {
183        return true;
184    }
185    parent_vertices.insert(current);
186    let outs = graph.outgoing_nodes(current);
187    let mut reaches_sink = false;
188    for to in outs {
189        if parent_vertices.contains(&to) {
190            edges_to_remove.insert((current, to));
191        } else if find_cycle_guilty(
192            graph,
193            to,
194            sinks,
195            edges_to_remove,
196            vertices_to_remove,
197            parent_vertices,
198        ) {
199            reaches_sink = true;
200        }
201    }
202    if !reaches_sink {
203        vertices_to_remove.insert(current);
204    }
205    parent_vertices.remove(&current);
206    reaches_sink
207}

217/// K-best on the graph as-is (no cycle stripping). Parity diagnostics only.

225/// Production assembly k-best: prefer an acyclic graph. Cyclic graphs previously kept
226/// full topology (P12) and could grow the k-best frontier without bound on bushy
227/// loci (e.g. NA12878 `20:10098169`); try cycle removal first, then fall back to the
228/// capped preserving search.
229///
230/// Takes ownership so cycle stripping does not `graph.clone()` (Peak-RSS on bushy loci).
231pub fn find_best_haplotypes_for_assembly(
232    graph: AssemblyGraph,
233    max_number_of_haplotypes: usize,
234) -> GatkResult<(Vec<KBestPath>, AssemblyGraph)> {
235    // Observable contract: prefer acyclic k-best; when cycle stripping cannot produce a
236    // source→sink DAG, fall back to capped preserving search (pre-Peak behavior). Dropping
237    // that fallback and running `find_best_haplotypes_inner` on the cyclic graph left L2
238    // `g2-subset-live` at haplotype_count=1 (ref only).
239    let (paths, graph) = match graph_for_kbest(graph) {
240        Ok(acyclic) => {
241            crate::runtime_config::rss_trace_checkpoint(
242                "kbest_begin",
243                &format!(
244                    "nodes={} max_haps={} mode=acyclic",
245                    acyclic.nodes().len(),
246                    max_number_of_haplotypes
247                ),
248            );
249            let paths = find_best_haplotypes_inner(&acyclic, max_number_of_haplotypes, false)?;
250            (paths, acyclic)
251        }
252        Err(cyclic) => {
253            crate::runtime_config::rss_trace_checkpoint(
254                "kbest_cyclic_preserve",
255                &format!(
256                    "nodes={} max_haps={}",
257                    cyclic.nodes().len(),
258                    max_number_of_haplotypes
259                ),
260            );
261            let paths = find_best_haplotypes_preserving_cycles(&cyclic, max_number_of_haplotypes)?;
262            (paths, cyclic)
263        }
264    };
265    crate::runtime_config::rss_trace_checkpoint("kbest_done", &format!("paths={}", paths.len()));
266    Ok((paths, graph))
267}

269fn find_best_haplotypes_inner(
270    graph: &AssemblyGraph,
271    max_number_of_haplotypes: usize,
272    strip_cycles: bool,
273) -> GatkResult<Vec<KBestPath>> {
274    if max_number_of_haplotypes == 0 {
275        return Ok(Vec::new());
276    }
277    // Borrow when possible; only clone for cycle stripping (find_best_haplotypes public API).
278    let owned;
279    let graph: &AssemblyGraph = if strip_cycles {
280        owned = match graph_for_kbest(graph.clone()) {
281            Ok(g) => g,
282            Err(g) => g, // preserve cyclic topology when strip fails
283        };
284        &owned
285    } else {
286        graph
287    };
288    let source = graph
289        .reference_source_vertex()
290        .ok_or_else(|| GatkError::algorithm("kbest: no reference source vertex"))?;
291    let sink = graph
292        .reference_sink_vertex()
293        .ok_or_else(|| GatkError::algorithm("kbest: no reference sink vertex"))?;
294    let sources = [source];
295    let sinks: HashSet<usize> = HashSet::from([sink]);
296
297    let mut result = Vec::new();
298    let mut heap: BinaryHeap<HeapItem> = BinaryHeap::new();
299    for &s in &sources {
300        let path = PathState::new(graph, s);
301        heap.push(HeapItem {
302            score_bits: path.score.to_bits(),
303            tie: path.bases_len,
304            path,
305        });
306    }
307    let mut vertex_counts: HashMap<usize, usize> =
308        graph.nodes().iter().map(|n| (n.id, 0)).collect();
309
310    // Bound total expansions: at the heap cap, a bushy/cyclic graph otherwise spins
311    // forever in pop/extend/push, fragmenting the allocator into multi-GiB Peak-RSS.
312    const MAX_KBEST_EXPANSIONS: usize = 50_000;
313    let mut expansions = 0usize;
314    while !heap.is_empty() && result.len() < max_number_of_haplotypes {
315        let item = heap.pop().expect("non-empty");
316        let path = item.path;
317        if sinks.contains(&path.last) {
318            result.push(KBestPath {
319                start: path.start,
320                edges: path.edges,
321                score: path.score,
322                is_reference: path.is_reference,
323            });
324            continue;
325        }
326        if path.edges.len() >= MAX_KBEST_PATH_EDGES || expansions >= MAX_KBEST_EXPANSIONS {
327            continue;
328        }
329        if expansions > 0 && expansions % 5_000 == 0 {
330            crate::runtime_config::rss_trace_checkpoint(
331                "kbest_expand",
332                &format!(
333                    "expansions={expansions} heap={} results={}",
334                    heap.len(),
335                    result.len()
336                ),
337            );
338        }
339        // If the frontier is already saturated, only accept sinks (above); do not expand.
340        if heap.len() >= MAX_KBEST_HEAP_PATHS {
341            continue;
342        }
343        let count = vertex_counts.get_mut(&path.last).expect("vertex");
344        if *count < max_number_of_haplotypes {
345            *count += 1;
346            expansions += 1;
347            let outs = graph.outgoing_nodes(path.last);
348            let total: u32 = outs
349                .iter()
350                .filter_map(|&t| graph.edge_support(path.last, t))
351                .sum();
352            for to in outs {
353                if let Some(support) = graph.edge_support(path.last, to) {
354                    if heap.len() >= MAX_KBEST_HEAP_PATHS {
355                        break;
356                    }
357                    let extended = path.extend(graph, to, support, total);
358                    heap.push(HeapItem {
359                        score_bits: extended.score.to_bits(),
360                        tie: extended.bases_len,
361                        path: extended,
362                    });
363                }
364            }
365        }
366    }
367
368    result.sort_by(|a, b| {
369        b.score
370            .partial_cmp(&a.score)
371            .unwrap_or(Ordering::Equal)
372            .then_with(|| b.bases(graph).cmp(&a.bases(graph)))
373    });
374    Ok(result)
375}

392        let reference = read("ACGTT", 30);

394            read("ACGTT", 30),

397            read("ACGTA", 30),
398            read("ACGTA", 30),

401            kmer_size: crate::bio_ids::KmerSize::try_new(3).unwrap(),
402            min_base_quality: 10,
403            ..Default::default()
404        };
405        let mut graph =
406            assembly_graph_from_ref_and_reads_threading(&reference, &reads, &params).unwrap();
407        let mut pruning =
408            crate::assembly::AssemblyGraphPruningParams::gatk_haplotype_caller_defaults();
409        pruning.min_prune_factor = 2;
410        graph.apply_pruning(&pruning);
411        let paths = find_best_haplotypes(&graph, 128).unwrap();
412        assert!(!paths.is_empty());
413        let seqs: HashSet<_> = paths.iter().map(|p| p.bases(&graph)).collect();
414        assert!(seqs.contains(&b"ACGTT".to_vec()));
415    }

417    #[test]
418    fn kbest_bounds_are_finite_and_tight() {
419        assert!(MAX_KBEST_HEAP_PATHS <= 4_096);
420        assert!(MAX_KBEST_PATH_EDGES <= 8_192);
421        // Worst-case edge storage alone stays well under 100 MiB.
422        let worst_edge_bytes = MAX_KBEST_HEAP_PATHS
423            .saturating_mul(MAX_KBEST_PATH_EDGES)
424            .saturating_mul(std::mem::size_of::<(usize, usize)>());
425        assert!(worst_edge_bytes < 128 * 1024 * 1024);
426    }
