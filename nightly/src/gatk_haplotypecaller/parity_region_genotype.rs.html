17/// Build per-read haplotype log10 rows like Java `assemblyRegionGenotypeSubset`.
18pub fn parity_java_aligned_read_rows(
19    region_reads: &[rust_htslib::bam::Record],
20    haplotypes: &[Haplotype],
21) -> GatkResult<Vec<ReadLikelihoodRow>> {
22    let mut records: Vec<&rust_htslib::bam::Record> = region_reads.iter().collect();
23    records.sort_by(|a, b| a.qname().cmp(b.qname()).then_with(|| a.pos().cmp(&b.pos())));
24
25    let n_haps = haplotypes.len();
26    let mut rows = Vec::with_capacity(records.len());
27    for rec in records {
28        let bases = rec.seq().as_bytes();
29        let quals = rec.qual();
30        let mut ll = vec![f64::NEG_INFINITY; n_haps];
31        for (hi, hap) in haplotypes.iter().enumerate() {
32            ll[hi] = log10_pairhmm_likelihood_parity_defaults(&bases, quals, hap.bases.as_slice())?;
33        }
34        let read_id = String::from_utf8_lossy(rec.qname()).into_owned();
35        rows.push(ReadLikelihoodRow {
36            read_id,
37            haplotype_log10_likelihoods: ll,
38        });
39    }
40    Ok(rows)
41}

50/// Java `HcParityRegionGenotype.GenotypeDump` from per-read likelihood rows.
51/// # Invariants
52/// When `genotyped` is false, `genotype_log10` is empty and `format` is `None`.
53/// `ref_hap_idx` / `alt_hap_idx` index into the haplotype list used for biallelic GLs.
54/// # Ownership
55/// Owns genotype vectors and optional [`GenotypeFormatFields`]; rows are consumed upstream.
56/// # Mutation
57/// Immutable snapshot for parity dumps after genotype computation.
58/// # Biological assumptions
59/// Biallelic ref/alt genotype from summed read×haplogtype log10 likelihoods (parity gate path).
60/// # Java equivalence
61/// Java `HcParityRegionGenotype.GenotypeDump` / `assemblyRegionGenotypeSubset` live gates.
62pub struct ParityRegionGenotypeDump {
63    pub haplotype_count: usize,
64    pub read_count: usize,
65    pub genotyped: bool,
66    pub ref_hap_idx: usize,
67    pub alt_hap_idx: usize,
68    pub best_hap_idx: usize,
69    pub genotype_log10: Vec<f64>,
70    pub format: Option<GenotypeFormatFields>,
71}

80/// `legacy_gl`: frozen `g2-region` Java dumps use `2×lr` / `lr+la` / `2×la` (no ploidy denominator).
81pub fn parity_region_genotype_from_rows_with_gl_mode(
82    rows: &[ReadLikelihoodRow],
83    is_reference: &[bool],
84    legacy_gl: bool,
85) -> GatkResult<ParityRegionGenotypeDump> {
86    let hap_count = is_reference.len();
87    let read_count = rows.len();
88    if hap_count == 0 || read_count == 0 {
89        return Ok(ParityRegionGenotypeDump {
90            haplotype_count: hap_count,
91            read_count,
92            genotyped: false,
93            ref_hap_idx: 0,
94            alt_hap_idx: 0,
95            best_hap_idx: 0,
96            genotype_log10: Vec::new(),
97            format: None,
98        });
99    }
100    let mut sums = vec![0.0_f64; hap_count];
101    for row in rows {
102        for (i, s) in sums.iter_mut().enumerate() {
103            *s += row.haplotype_log10_likelihoods[i];
104        }
105    }
106    let mut best = 0usize;
107    for i in 1..hap_count {
108        if sums[i] > sums[best] {
109            best = i;
110        }
111    }
112    let ref_idx = is_reference.iter().position(|&r| r).unwrap_or(0);
113    let mut alt_idx = ref_idx;
114    let mut alt_sum = f64::NEG_INFINITY;
115    for (i, &is_ref) in is_reference.iter().enumerate() {
116        if i != ref_idx && !is_ref && sums[i] > alt_sum {
117            alt_sum = sums[i];
118            alt_idx = i;
119        }
120    }
121    let mut ref_ad = 0i32;
122    let mut alt_ad = 0i32;
123    for row in rows {
124        let lr = row.haplotype_log10_likelihoods[ref_idx];
125        let la = row.haplotype_log10_likelihoods[alt_idx];
126        if lr >= la {
127            ref_ad += 1;
128        } else {
129            alt_ad += 1;
130        }
131    }
132    if ref_ad == 0 && alt_ad == 0 {
133        ref_ad = 1;
134    }
135    let gls = if legacy_gl {
136        biallelic_genotype_log10_likelihoods_parity_legacy(rows, ref_idx, alt_idx)
137    } else {
138        biallelic_genotype_log10_likelihoods_gatk(rows, ref_idx, alt_idx)
139    };
140    let format = emit_genotype_format_fields(&gls, &[ref_ad, alt_ad])?;
141    Ok(ParityRegionGenotypeDump {
142        haplotype_count: hap_count,
143        read_count,
144        genotyped: true,
145        ref_hap_idx: ref_idx,
146        alt_hap_idx: alt_idx,
147        best_hap_idx: best,
148        genotype_log10: gls,
149        format: Some(format),
150    })
151}

153pub fn write_parity_region_genotype_dump(
154    out: &mut impl Write,
155    dump: &ParityRegionGenotypeDump,
156) -> GatkResult<()> {
157    writeln!(out, "haplotype_count\t{}", dump.haplotype_count)
158        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
159    writeln!(out, "read_count\t{}", dump.read_count)
160        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
161    writeln!(out, "genotyped\t{}", dump.genotyped)
162        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
163    let Some(format) = &dump.format else {
164        return Ok(());
165    };
166    writeln!(out, "ref_haplotype_index\t{}", dump.ref_hap_idx)
167        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
168    writeln!(out, "alt_haplotype_index\t{}", dump.alt_hap_idx)
169        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
170    writeln!(out, "best_haplotype_index\t{}", dump.best_hap_idx)
171        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
172    for (i, gl) in dump.genotype_log10.iter().enumerate() {
173        writeln!(out, "genotype_{i}_log10\t{gl}")
174            .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
175    }
176    writeln!(
177        out,
178        "pl\t{}",
179        format
180            .pl
181            .iter()
182            .map(|p| p.to_string())
183            .collect::<Vec<_>>()
184            .join(",")
185    )
186    .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
187    writeln!(out, "gq\t{}", format.gq)
188        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
189    writeln!(
190        out,
191        "ad\t{}",
192        format
193            .ad
194            .iter()
195            .map(|a| a.to_string())
196            .collect::<Vec<_>>()
197            .join(",")
198    )
199    .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
200    writeln!(out, "dp\t{}", format.dp)
201        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
202    Ok(())
203}

205#[cfg(test)]
206mod tests {
207    use super::*;
208    #[test]
209    fn parity_region_genotype_dump_writes_genotyped_row() {
210        let rows = vec![ReadLikelihoodRow {
211            read_id: "r1".into(),
212            haplotype_log10_likelihoods: vec![-1.0],
213        }];
214        let dump = parity_region_genotype_from_rows(&rows, &[true]).expect("dump");
215        assert!(dump.genotyped);
216        let mut buf = Vec::new();
217        write_parity_region_genotype_dump(&mut buf, &dump).expect("write");
218        let s = String::from_utf8(buf).expect("utf8");
219        assert!(s.contains("genotyped\ttrue"), "output:\n{s}");
220    }
221}
