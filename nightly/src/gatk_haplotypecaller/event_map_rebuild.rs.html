1//! Pure EventMap rebuild helpers (L7-B2).
2//! Keeps genotyping EventMap construction as data transforms over haplotypes + prior events,
3//! rather than only mutating [`crate::assembly_result_set::AssemblyResultSet`] in place.
4
5use crate::event_map::{collect_variation_events, VariationEvent};
6use crate::haplotype::Haplotype;
7
8/// Options for [`rebuild_variation_events`].
9/// # Invariants
10/// `event_map_only` ignores prior/supplement lists (CIGAR EventMap only).
11/// When supplements are enabled, `merge_read_supplements` selects supplement vs prior source.
12/// # Ownership
13/// [`Copy`] option flags.
14/// # Mutation
15/// Immutable per rebuild call.
16/// # Biological assumptions
17/// Controls whether read-discovered alleles merge into haplotype EventMap sites.
18/// # Java equivalence
19/// Rust-native L7-B2 rebuild knobs (algorithm parity with Java EventMap regeneration).
20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub struct RebuildVariationEventsOpts {
22    /// When true, ignore prior/supplement lists (CIGAR/EventMap only).
23    pub event_map_only: bool,
24    /// Merge `preserved_supplement` instead of `prior_events` when supplements are enabled.
25    pub merge_read_supplements: bool,
26}
27
28/// Rebuild variation events from haplotype CIGARs, optionally merging prior/supplement alleles.
29pub fn rebuild_variation_events(
30    haplotypes: &[Haplotype],
31    full_ref: &[u8],
32    full_pad_1based: u64,
33    contig: &str,
34    max_mnp_distance: usize,
35    prior_events: &[VariationEvent],
36    preserved_supplement: &[VariationEvent],
37    opts: RebuildVariationEventsOpts,
38) -> Vec<VariationEvent> {
39    let mut events = collect_variation_events(
40        haplotypes,
41        full_ref,
42        full_pad_1based,
43        contig,
44        max_mnp_distance,
45    );
46    if !opts.event_map_only {
47        let source = if opts.merge_read_supplements {
48            preserved_supplement
49        } else {
50            prior_events
51        };
52        for e in source {
53            if !events.iter().any(|x| {
54                x.start_1based == e.start_1based
55                    && x.ref_allele == e.ref_allele
56                    && x.alt_allele == e.alt_allele
57            }) {
58                // CLONE: needed because owned element into collection.
59                events.push(e.clone());
60            }
61        }
62    }
63    crate::event_map::prefer_indel_over_colocated_snps(&mut events);
64    events.sort_by_key(|e| e.start_1based);
65    events.dedup_by(|a, b| {
66        a.start_1based == b.start_1based
67            && a.ref_allele == b.ref_allele
68            && a.alt_allele == b.alt_allele
69    });
70    if !opts.event_map_only {
71        crate::read_event_discovery::scrub_p12_cluster_phantom_alleles(&mut events);
72    }
73    events
74}
