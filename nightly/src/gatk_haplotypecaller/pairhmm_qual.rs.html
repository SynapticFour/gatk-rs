1//! GATK `PairHMMLikelihoodCalculationEngine` base-quality capping.
2
3use gatk_core::BaseQuality;
4
5/// GATK `QualityUtils.MIN_USABLE_Q_SCORE`.
6pub const MIN_USABLE_Q_SCORE: u8 = 6;
7
8/// GATK `setToFixedValueIfTooLow`: values below `min_qual` become `fixed_qual`.
9#[inline]
10pub fn set_to_fixed_value_if_too_low(current: u8, min_qual: u8, fixed_qual: u8) -> u8 {
11    if current < min_qual {
12        fixed_qual
13    } else {
14        current
15    }
16}
17
18/// Cap base qualities by mapping quality, then enforce `base_quality_score_threshold`.
19/// Each byte is first normalized through [`BaseQuality::new`] (SAM max 93) so PairHMM
20/// never sees uncapped Phred values at this boundary.
21pub fn cap_read_base_qualities(
22    quals: &mut [u8],
23    mapq: u8,
24    base_quality_score_threshold: u8,
25    disable_cap_read_qualities_to_mapq: bool,
26) {
27    for q in quals.iter_mut() {
28        let mut v = BaseQuality::new(*q).value();
29        if !disable_cap_read_qualities_to_mapq {
30            v = v.min(mapq);
31        }
32        *q = set_to_fixed_value_if_too_low(v, base_quality_score_threshold, MIN_USABLE_Q_SCORE);
33    }
34}

36#[cfg(test)]

40    #[test]

42        let mut q = vec![5u8, 18, 30];
43        cap_read_base_qualities(&mut q, 60, 18, true);
44        assert_eq!(q, vec![6, 18, 30]);
45    }

47    #[test]
