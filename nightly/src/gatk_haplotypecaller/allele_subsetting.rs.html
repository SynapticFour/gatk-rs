9/// Pick alleles to retain when genotype enumeration would explode (G.3 + G-D05).
10pub fn subset_alleles_for_genotyping(
11    haplotypes: &[Haplotype],
12    aggregation: &HaplotypeLikelihoodAggregation,
13    max_allele_count: usize,
14) -> GatkResult<Vec<usize>> {
15    if haplotypes.len() <= max_allele_count {
16        return Ok((0..haplotypes.len()).collect());
17    }
18    // One score per haplotype/allele — borrow as `&[f64]` rows (no `Vec<Vec<f64>>`).
19    let scores: Vec<&[f64]> = aggregation
20        .haplotype_log10_sums
21        .iter()
22        .map(std::slice::from_ref)
23        .collect();
24    let is_ref: Vec<bool> = haplotypes.iter().map(|h| h.is_reference).collect();
25    Ok(which_alleles_to_keep_by_haplotype_scores_with_ref(
26        &scores,
27        max_allele_count,
28        Some(&is_ref),
29    ))
30}
