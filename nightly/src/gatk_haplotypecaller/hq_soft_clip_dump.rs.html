1//! HQ soft-clip mean dump.
2//! Emits the same **RCM** running mean as `ReferenceConfidenceModel#calcGenotypeLikelihoodsOfRefVsAny`,
3//! not a read-level heuristic.
4
5use crate::activity_scoring::{
6    hq_soft_clip_running_mean_rcm_path, HaplotypeCallerActivityScoringParams,
7};
8use crate::assembly_regions_dump::load_contig_records_linear;
9use crate::locus_iterator::LocusPileupWalker;
10use crate::read_model::ReadFilterParams;
11use gatk_common::{GatkError, GatkResult};
12use gatk_core::reference::{parse_intervals_cli_string, ReferenceWindowCache, SequenceDictionary};
13use std::io::Write;
14use std::path::Path;
15
16/// TSV: `contig`, `pos`, `hq_soft_clip_mean` (8 decimal places, `0` when none).
17pub fn dump_hq_soft_clip_mean_tsv(
18    reference_fasta: &Path,
19    bam_path: &Path,
20    intervals_cli: &str,
21    out: &mut impl Write,
22    read_filters: &ReadFilterParams,
23) -> GatkResult<()> {
24    writeln!(out, "contig\tpos\thq_soft_clip_mean")
25        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
26
27    let dict = SequenceDictionary::from_fasta_path(reference_fasta)?;
28    let specs = parse_intervals_cli_string(&dict, intervals_cli)?;
29    let mut records_by_contig = std::collections::HashMap::new();
30    let mut ref_cache = ReferenceWindowCache::new(reference_fasta.to_path_buf(), 4);
31    let mut rng = crate::read_downsample::GatkJavaRng::reset_gatk_default();
32    let scoring = HaplotypeCallerActivityScoringParams::default();
33
34    for spec in &specs {
35        let (c, s, e) = spec
36            .resolve_closed_ends(&dict)
37            .map_err(|e| GatkError::argument(e.to_string()))?;
38        let ref_window = ref_cache
39            .get_interval_bytes(&dict, &c, s, e)
40            .map_err(|e| GatkError::generic(e.to_string()))?;
41        if !records_by_contig.contains_key(&c) {
42            records_by_contig.insert(
43                c.clone(),
44                load_contig_records_linear(bam_path, &c, read_filters, &mut rng)?,
45            );
46        }
47        let (header, records) = records_by_contig.get(&c).expect("cache");
48        let mut walker = LocusPileupWalker::new(records, header, &c, read_filters);
49        for pos1 in s..=e {
50            let ref_base = *ref_window
51                .get((pos1 - s) as usize)
52                .ok_or_else(|| GatkError::argument("reference window index out of range"))?;
53            let pile = walker.pileup_at(pos1, ref_base)?;
54            let mean = hq_soft_clip_running_mean_rcm_path(&pile, &scoring);
55            let formatted = if mean == 0.0 {
56                "0".to_string()
57            } else {
58                format!("{mean:.8}")
59            };
60            writeln!(out, "{c}\t{pos1}\t{formatted}")
61                .map_err(|e| GatkError::generic(format!("write: {e}")))?;
62        }
63    }
64    Ok(())
65}
