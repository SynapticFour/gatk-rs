1//! Region-scoped RT extract cache: avoid rebuilding the same k-mer graph twice.
2//!
3//! # Invariants
4//! - Active only between [`begin_assemble_region`] and [`end_assemble_region`].
5//! - Cache key includes kmer + allow_lc/nu + before_remove mode (wrong key ⇒ wrong alleles).
6//! - Stores haplotype lists only (graphs dropped after extract) so Peak-RSS stays lean.
7//! - `empty_configured_kmers` records production configured k-mers whose before_remove
8//!   extract already yielded no alts (RT-first miss) so supplement/merge_rt skip re-probe.
9//!
10//! # Java equivalence
11//! Java pays one graph family per k-mer per region; this cache restores that cost shape
12//! when Rust orchestration would otherwise rebuild the same before_remove extract.
13
14use crate::haplotype::Haplotype;
15use std::cell::RefCell;
16use std::collections::{HashMap, HashSet};
17use std::sync::Arc;
18
19/// Extract identity for one RT before/after-remove haplotype pull.
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
21pub(crate) struct RtExtractKey {
22    pub kmer_size: usize,
23    pub allow_low_complexity: bool,
24    pub allow_non_unique_ref: bool,
25    pub before_remove_paths: bool,
26}
27
28thread_local! {
29    static ACTIVE: RefCell<bool> = const { RefCell::new(false) };
30    static EXTRACTS: RefCell<HashMap<RtExtractKey, Arc<Vec<Haplotype>>>> =
31        RefCell::new(HashMap::new());
32    static EMPTY_CONFIGURED: RefCell<HashSet<usize>> = RefCell::new(HashSet::new());
33}
34
35/// Start a fresh per-region assemble scope (call at `assemble_from_ref_and_reads` entry).
36pub(crate) fn begin_assemble_region() {
37    ACTIVE.with(|a| *a.borrow_mut() = true);
38    EXTRACTS.with(|c| c.borrow_mut().clear());
39    EMPTY_CONFIGURED.with(|s| s.borrow_mut().clear());
40}
41
42/// Drop cache before PairHMM / next region (Peak-RSS + correctness).
43pub(crate) fn end_assemble_region() {
44    EXTRACTS.with(|c| c.borrow_mut().clear());
45    EMPTY_CONFIGURED.with(|s| s.borrow_mut().clear());
46    ACTIVE.with(|a| *a.borrow_mut() = false);
47}
48
49fn is_active() -> bool {
50    ACTIVE.with(|a| *a.borrow())
51}
52
53/// Return a cached extract when present.
54pub(crate) fn get_cached(key: &RtExtractKey) -> Option<Vec<Haplotype>> {
55    if !is_active() {
56        return None;
57    }
58    EXTRACTS.with(|c| {
59        c.borrow().get(key).map(|arc| {
60            // CLONE: owned haplotype list for caller merge/dedup.
61            Vec::clone(arc)
62        })
63    })
64}
65
66/// Store extract result for the remainder of this region assemble; returns `haps` by move.
67pub(crate) fn put_cached(key: RtExtractKey, haps: Vec<Haplotype>) -> Vec<Haplotype> {
68    if is_active() {
69        let n = haps.len();
70        EXTRACTS.with(|c| {
71            // CLONE: cache retains a copy; caller keeps the original list.
72            c.borrow_mut().insert(key, Arc::new(Vec::clone(&haps)));
73        });
74        crate::runtime_config::rss_trace_checkpoint(
75            "rt_extract_cache_store",
76            &format!(
77                "kmer={} before_remove={} haps={n}",
78                key.kmer_size, key.before_remove_paths
79            ),
80        );
81    }
82    haps
83}
84
85/// Mark a configured k-mer whose before_remove extract had no usable alts (RT-first miss).
86pub(crate) fn mark_configured_kmer_empty(kmer_size: usize) {
87    if !is_active() {
88        return;
89    }
90    EMPTY_CONFIGURED.with(|s| {
91        s.borrow_mut().insert(kmer_size);
92    });
93    crate::runtime_config::rss_trace_checkpoint(
94        "rt_configured_empty",
95        &format!("kmer={kmer_size}"),
96    );
97}
98
99/// True when RT-first (or equivalent) already proved this configured k-mer empty.
100pub(crate) fn configured_kmer_already_empty(kmer_size: usize) -> bool {
101    if !is_active() {
102        return false;
103    }
104    EMPTY_CONFIGURED.with(|s| s.borrow().contains(&kmer_size))
105}
