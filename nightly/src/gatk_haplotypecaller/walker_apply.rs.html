1//! `AssemblyRegionWalker.apply` vs `HaplotypeCallerEngine.callRegion` disposition.
2//! Java: [`AssemblyRegionWalker`](https://github.com/broadinstitute/gatk/blob/master/src/main/java/org/broadinstitute/hellbender/engine/AssemblyRegionWalker.java)
3//! invokes `apply` once per region from [`AssemblyRegionIterator`](https://github.com/broadinstitute/gatk/blob/master/src/main/java/org/broadinstitute/hellbender/engine/AssemblyRegionIterator.java).
4//! [`HaplotypeCallerEngine.callRegion`](https://github.com/broadinstitute/gatk/blob/master/src/main/java/org/broadinstitute/hellbender/tools/walkers/haplotypecaller/HaplotypeCallerEngine.java)
5//! takes the **inactive fast path** (no graph / local assembly) when `!region.isActive`, and otherwise runs assembly + genotyping.
6
7use crate::assembly_region_iterator::AssemblyRegion;

9/// How HC-style `callRegion` would treat this region (assembly vs reference-only fast path).
10/// # Invariants
11/// Exactly one of active full vs inactive fast path per region (`region.is_active` bit).
12/// Inactive path skips `assembleReads` in Java parity semantics.
13/// # Ownership
14/// [`Copy`] enum derived from [`AssemblyRegion`] by [`call_disposition`].
15/// # Mutation
16/// Immutable tag computed per region at apply/call time.
17/// # Biological assumptions
18/// Inactive regions lack sufficient activity evidence for local re-assembly.
19/// # Java equivalence
20/// GATK `HaplotypeCallerEngine.callRegion` inactive guard vs full assembly path.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum AssemblyRegionCallDisposition {
23    /// `region.isActive` — full assembly + genotyping path (GATK `callRegion` after the inactive guard).
24    ActiveFull,
25    /// `!region.isActive` — `referenceModelForNoVariation` path; **no** `assembleReads` (GATK fast path).
26    InactiveReferenceFastPath,
27}
28
29#[inline]
30pub fn call_disposition(region: &AssemblyRegion) -> AssemblyRegionCallDisposition {
31    if region.is_active {
32        AssemblyRegionCallDisposition::ActiveFull
33    } else {
34        AssemblyRegionCallDisposition::InactiveReferenceFastPath
35    }
36}
37
38/// First region suitable for ASM-1 parity dumps: active if present, else inactive with reads.
39#[cfg(any(feature = "dev-dumps", test))]
40pub fn select_region_for_asm_dump(regions: &[AssemblyRegion]) -> Option<&AssemblyRegion> {
41    regions
42        .iter()
43        .find(|r| {
44            matches!(
45                call_disposition(r),
46                AssemblyRegionCallDisposition::ActiveFull
47            )
48        })
49        .or_else(|| {
50            regions.iter().find(|r| {
51                !r.reads.is_empty()
52                    && matches!(
53                        call_disposition(r),
54                        AssemblyRegionCallDisposition::InactiveReferenceFastPath
55                    )
56            })
57        })
58}

60/// Counts aligned with Java “one `apply` per iterator region” + inactive fast path in `callRegion`.
61/// # Invariants
62/// `total_apply == inactive_fast_path + active_full` (debug_assert in [`Self::from_regions`]).
63/// One count increment per [`AssemblyRegion`] in the iterator stream.
64/// # Ownership
65/// [`Copy`] aggregate stats; regions are not retained.
66/// # Mutation
67/// Immutable snapshot after traversal; built by scanning region dispositions.
68/// # Biological assumptions
69/// None documented (walker bookkeeping for parity dumps).
70/// # Java equivalence
71/// GATK `AssemblyRegionWalker.apply` per-region invocations + `callRegion` inactive fast-path tally.
72#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
73pub struct WalkerApplyStats {
74    /// Same as number of `AssemblyRegionWalker.apply` invocations for this shard/interval stream.
75    pub total_apply: usize,
76    pub inactive_fast_path: usize,
77    pub active_full: usize,
78}
79
80impl WalkerApplyStats {
81    pub fn from_regions(regions: &[AssemblyRegion]) -> Self {
82        let mut s = WalkerApplyStats::default();
83        s.total_apply = regions.len();
84        for r in regions {
85            match call_disposition(r) {
86                AssemblyRegionCallDisposition::InactiveReferenceFastPath => {
87                    s.inactive_fast_path += 1;
88                }
89                AssemblyRegionCallDisposition::ActiveFull => {
90                    s.active_full += 1;
91                }
92            }
93        }
94        debug_assert_eq!(s.total_apply, s.inactive_fast_path + s.active_full);
95        s
96    }
97}

121    #[test]
122    fn disposition_matches_is_active() {
123        assert_eq!(
124            call_disposition(&region(false)),
125            AssemblyRegionCallDisposition::InactiveReferenceFastPath
126        );
127        assert_eq!(
128            call_disposition(&region(true)),
129            AssemblyRegionCallDisposition::ActiveFull
130        );
131    }
132
133    #[test]
134    fn stats_sum_invariant() {
135        let v = vec![region(false), region(true), region(false)];
136        let st = WalkerApplyStats::from_regions(&v);
137        assert_eq!(st.total_apply, 3);
138        assert_eq!(st.inactive_fast_path, 2);
139        assert_eq!(st.active_full, 1);
