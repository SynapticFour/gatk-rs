1//! GATK `Log10PairHMM` (GAP-F-02).
2
3use crate::pairhmm_dump::load_pairhmm_cases_tsv;
4use crate::pairhmm_log10::log10_pairhmm_likelihood_parity_defaults;
5use gatk_common::{GatkError, GatkResult};
6use std::io::Write;
7use std::path::Path;
8
9pub fn dump_pairhmm_native_likelihoods_tsv(
10    cases_path: &Path,
11    out: &mut impl Write,
12) -> GatkResult<()> {
13    let cases = load_pairhmm_cases_tsv(cases_path)?;
14    writeln!(out, "case_id\tlog10_likelihood")
15        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
16    for row in &cases {
17        let read = row.read_bases.as_bytes();
18        let hap = row.haplotype.as_bytes();
19        let ll = log10_pairhmm_likelihood_parity_defaults(read, &row.read_base_quals, hap)?;
20        writeln!(out, "{}\t{ll:.17}", row.case_id)
21            .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
22    }
23    Ok(())
24}
