5/// Coarse production stages (stable JSON keys).
6#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
7pub enum Stage {
8    InputBamDecode,
9    ReadPreprocess,
10    ActiveRegionConstruction,
11    EventDiscovery,
12    AssemblyGraphConstruction,
13    GraphPruning,
14    HaplotypeGeneration,
15    SmithWaterman,
16    PairHmm,
17    LikelihoodProcessing,
18    GenotypeAssignment,
19    AdAnnotation,
20    VcfEmission,
21    Synchronization,
22    Allocations,
23    /// Catch-all for TRACE phases that do not map cleanly.
24    Other,
25}

27impl Stage {
28    pub fn as_str(self) -> &'static str {
29        match self {
30            Self::InputBamDecode => "input_bam_decode",
31            Self::ReadPreprocess => "read_preprocessing",
32            Self::ActiveRegionConstruction => "active_region_construction",
33            Self::EventDiscovery => "event_discovery",
34            Self::AssemblyGraphConstruction => "assembly_graph_construction",
35            Self::GraphPruning => "graph_pruning",
36            Self::HaplotypeGeneration => "haplotype_generation",
37            Self::SmithWaterman => "smith_waterman",
38            Self::PairHmm => "pairhmm",
39            Self::LikelihoodProcessing => "likelihood_processing",
40            Self::GenotypeAssignment => "genotype_assignment",
41            Self::AdAnnotation => "ad_annotation",
42            Self::VcfEmission => "vcf_emission",
43            Self::Synchronization => "synchronization_waiting",
44            Self::Allocations => "allocations",
45            Self::Other => "other",

47    }

66            Self::Other,
67        ]

70    /// Map `HC_RSS_TRACE phase=` names onto coarse stages.
71    pub fn from_trace_phase(phase: &str) -> Option<Stage> {
72        let p = phase;
73        if p.starts_with("rt_build")
74            || p.starts_with("rt_graph")
75            || p.starts_with("rt_first")
76            || p.starts_with("merge_rt")
77            || p.contains("graph_build")
78        {
79            return Some(Self::AssemblyGraphConstruction);
80        }
81        if p.starts_with("seq_") || p.contains("prune") || p.contains("simplify") {
82            return Some(Self::GraphPruning);
83        }
84        if p.starts_with("kbest") || p.contains("extract") {
85            return Some(Self::HaplotypeGeneration);
86        }
87        if p.starts_with("rt_dangling") || p.contains("dangling") {
88            return Some(Self::SmithWaterman);
89        }
90        if p.contains("pairhmm") || p == "before_pairhmm" || p == "after_pairhmm" {
91            return Some(Self::PairHmm);
92        }
93        if p.contains("realign") || p.contains("sw_") {
94            return Some(Self::SmithWaterman);
95        }
96        if p.contains("genotype") || p.contains("assign") {
97            return Some(Self::GenotypeAssignment);
98        }
99        if p.contains("spine")
100            || p.contains("event")
101            || p.contains("emap")
102            || p.contains("variation")
103            || p.starts_with("prep_trim")
104            || p.starts_with("prep_post")
105            || p.starts_with("prep_parity")
106            || p.starts_with("prep_early_allele")
107        {
108            return Some(Self::EventDiscovery);
109        }
110        if p.contains("emit") || p.contains("vcf") || p.contains("gvcf") {
111            return Some(Self::VcfEmission);
112        }
113        if p.contains("finalize") || p == "before_finalize" || p == "after_finalize" {
114            return Some(Self::ReadPreprocess);
115        }
116        if p == "after_assemble" || p.starts_with("assemble") {
117            return Some(Self::AssemblyGraphConstruction);
118        }
119        if p == "run_start" {
120            return None;
121        }
122        Some(Self::Other)
123    }
124}

126#[derive(Debug, Default, Clone)]
127pub struct StageStats {
128    pub calls: u64,
129    pub wall_ns: u64,
130    pub cpu_ns: u64,
131    pub cpu_samples: u64,
132    pub alloc_bytes: u64,
133    pub alloc_events: u64,
134}
135
136impl StageStats {
137    pub fn add(&mut self, wall: Duration, cpu: Option<Duration>) {
138        self.calls += 1;
139        self.wall_ns = self.wall_ns.saturating_add(wall.as_nanos() as u64);
140        if let Some(c) = cpu {
141            self.cpu_ns = self.cpu_ns.saturating_add(c.as_nanos() as u64);
142            self.cpu_samples += 1;
143        }
144    }
145
146    pub fn add_wall_only(&mut self, wall: Duration) {
147        self.calls += 1;
148        self.wall_ns = self.wall_ns.saturating_add(wall.as_nanos() as u64);
149    }
150
151    pub fn avg_wall_ns(&self) -> f64 {
152        if self.calls == 0 {
153            0.0
154        } else {
155            self.wall_ns as f64 / self.calls as f64
156        }
157    }
158}

160#[derive(Debug, Default)]
161pub struct StageAgg {
162    pub stats: [StageStats; 16],
163}
164
165impl StageAgg {
166    fn idx(stage: Stage) -> usize {
167        match stage {
168            Stage::InputBamDecode => 0,
169            Stage::ReadPreprocess => 1,
170            Stage::ActiveRegionConstruction => 2,
171            Stage::EventDiscovery => 3,
172            Stage::AssemblyGraphConstruction => 4,
173            Stage::GraphPruning => 5,
174            Stage::HaplotypeGeneration => 6,
175            Stage::SmithWaterman => 7,
176            Stage::PairHmm => 8,
177            Stage::LikelihoodProcessing => 9,
178            Stage::GenotypeAssignment => 10,
179            Stage::AdAnnotation => 11,
180            Stage::VcfEmission => 12,
181            Stage::Synchronization => 13,
182            Stage::Allocations => 14,
183            Stage::Other => 15,
184        }
185    }
186
187    pub fn add(&mut self, stage: Stage, wall: Duration, cpu: Option<Duration>) {
188        self.stats[Self::idx(stage)].add(wall, cpu);
189    }
190
191    pub fn add_wall_only(&mut self, stage: Stage, wall: Duration) {
192        self.stats[Self::idx(stage)].add_wall_only(wall);
193    }
194
195    pub fn add_alloc(&mut self, stage: Stage, bytes: u64, events: u64) {
196        let s = &mut self.stats[Self::idx(stage)];
197        s.alloc_bytes = s.alloc_bytes.saturating_add(bytes);
198        s.alloc_events = s.alloc_events.saturating_add(events);
199        // Also roll into Allocations bucket.
200        let a = &mut self.stats[Self::idx(Stage::Allocations)];
201        a.alloc_bytes = a.alloc_bytes.saturating_add(bytes);
202        a.alloc_events = a.alloc_events.saturating_add(events);
203    }
204
205    pub fn get(&self, stage: Stage) -> &StageStats {
206        &self.stats[Self::idx(stage)]
207    }
208
209    pub fn stats_mut(&mut self, stage: Stage) -> &mut StageStats {
210        &mut self.stats[Self::idx(stage)]
211    }
212}
