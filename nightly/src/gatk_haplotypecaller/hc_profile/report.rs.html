12pub fn write_reports(
13    json_path: &Path,
14    stages: &StageAgg,
15    pairhmm: &PairHmmAgg,
16    genotype: &GenotypeAgg,
17    run_wall: Duration,
18    run_cpu: Option<Duration>,
19    rayon_threads: usize,
20    regions: u64,
21) -> std::io::Result<()> {
22    if let Some(parent) = json_path.parent() {
23        if !parent.as_os_str().is_empty() {
24            fs::create_dir_all(parent)?;
25        }
26    }
27    let json = build_json(
28        stages,
29        pairhmm,
30        genotype,
31        run_wall,
32        run_cpu,
33        rayon_threads,
34        regions,
35    );
36    fs::write(json_path, &json)?;
37    let md_path = json_path.with_extension("md");
38    let md = build_markdown(
39        stages,
40        pairhmm,
41        genotype,
42        run_wall,
43        run_cpu,
44        rayon_threads,
45        regions,
46    );
47    fs::write(md_path, md)?;
48    Ok(())
49}

51fn ns_to_s(ns: u64) -> f64 {
52    ns as f64 / 1e9
53}

55fn build_json(
56    stages: &StageAgg,
57    pairhmm: &PairHmmAgg,
58    genotype: &GenotypeAgg,
59    run_wall: Duration,
60    run_cpu: Option<Duration>,
61    rayon_threads: usize,
62    regions: u64,
63) -> String {
64    let run_wall_s = run_wall.as_secs_f64();
65    let run_cpu_s = run_cpu.map(|d| d.as_secs_f64());
66    let parallel_eff = match run_cpu_s {
67        Some(cpu) if run_wall_s > 0.0 => Some(cpu / (run_wall_s * rayon_threads as f64)),
68        _ => None,
69    };
70    let occ = pairhmm.simd_occupancy();
71
72    let mut out = String::new();
73    out.push_str("{\n");
74    out.push_str("  \"schema\": \"gatk-rs.hc_profile.v1\",\n");
75    out.push_str("  \"honesty\": \"Observe-only production profile. Does not change genotype/emit. Stage walls may overlap TRACE dual-write; prefer PairHMM/genotype nested counters for leaf detail.\",\n");
76    out.push_str(&format!("  \"regions\": {regions},\n"));
77    out.push_str(&format!("  \"rayon_threads\": {rayon_threads},\n"));
78    out.push_str(&format!("  \"run_wall_s\": {run_wall_s:.6},\n"));
79    match run_cpu_s {
80        Some(c) => out.push_str(&format!("  \"run_cpu_s\": {c:.6},\n")),
81        None => out.push_str("  \"run_cpu_s\": null,\n"),
82    }
83    match parallel_eff {
84        Some(e) => out.push_str(&format!("  \"parallel_efficiency\": {e:.4},\n")),
85        None => out.push_str("  \"parallel_efficiency\": null,\n"),
86    }
87    out.push_str("  \"stages\": {\n");
88    for (i, stage) in Stage::all().iter().enumerate() {
89        let s = stages.get(*stage);
90        let comma = if i + 1 == Stage::all().len() { "" } else { "," };
91        out.push_str(&format!(
92            "    \"{}\": {{\n      \"calls\": {},\n      \"wall_s\": {:.6},\n      \"cpu_s\": {:.6},\n      \"cpu_samples\": {},\n      \"avg_wall_s\": {:.9},\n      \"alloc_bytes\": {},\n      \"alloc_events\": {}\n    }}{comma}\n",
93            stage.as_str(),
94            s.calls,
95            ns_to_s(s.wall_ns),
96            ns_to_s(s.cpu_ns),
97            s.cpu_samples,
98            s.avg_wall_ns() / 1e9,
99            s.alloc_bytes,
100            s.alloc_events
101        ));
102    }
103    out.push_str("  },\n");
104    out.push_str("  \"pairhmm\": {\n");
105    out.push_str(&format!("    \"calls\": {},\n", pairhmm.calls));
106    out.push_str(&format!(
107        "    \"reads_scored\": {},\n",
108        pairhmm.reads_scored
109    ));
110    out.push_str(&format!(
111        "    \"haplotypes_scored\": {},\n",
112        pairhmm.haplotypes_scored
113    ));
114    out.push_str(&format!(
115        "    \"read_haplotype_pairs\": {},\n",
116        pairhmm.read_hap_pairs
117    ));
118    out.push_str(&format!(
119        "    \"mean_haplotypes_per_read\": {:.4},\n",
120        pairhmm.mean_haps_per_read()
121    ));
122    out.push_str(&format!(
123        "    \"mean_read_len\": {:.2},\n",
124        if pairhmm.reads_scored == 0 {
125            0.0
126        } else {
127            pairhmm.read_len_sum as f64 / pairhmm.reads_scored as f64
128        }
129    ));
130    out.push_str(&format!(
131        "    \"mean_hap_len\": {:.2},\n",
132        if pairhmm.haplotypes_scored == 0 {
133            0.0
134        } else {
135            pairhmm.hap_len_sum as f64 / pairhmm.haplotypes_scored as f64
136        }
137    ));
138    out.push_str(&format!("    \"simd_pack_units\": {},\n", occ.pack_units));
139    out.push_str(&format!("    \"simd_pack_lanes\": {},\n", occ.pack_lanes));
140    out.push_str(&format!(
141        "    \"simd_pack_haps_est\": {},\n",
142        occ.pack_hap_est
143    ));
144    out.push_str(&format!(
145        "    \"simd_pack_occupancy_pct\": {:.2},\n",
146        occ.pack_occupancy_pct
147    ));
148    out.push_str(&format!(
149        "    \"prefix_reuse_haps\": {},\n",
150        pairhmm.prefix_reuse_haps
151    ));
152    out.push_str(&format!(
153        "    \"leftover_haps\": {},\n",
154        pairhmm.leftover_haps
155    ));
156    out.push_str(&format!(
157        "    \"prefix_reuse_pct\": {:.2},\n",
158        occ.prefix_reuse_pct
159    ));
160    out.push_str(&format!("    \"leftover_pct\": {:.2},\n", occ.leftover_pct));
161    out.push_str(&format!(
162        "    \"dp_cells_evaluated\": {},\n",
163        pairhmm.dp_cells_evaluated
164    ));
165    out.push_str(&format!(
166        "    \"dp_cells_avoided_prefix\": {},\n",
167        pairhmm.dp_cells_avoided_prefix
168    ));
169    out.push_str(&format!(
170        "    \"wall_s\": {:.6},\n",
171        ns_to_s(pairhmm.wall_ns)
172    ));
173    out.push_str("    \"read_len_hist_bucket25\": {\n");
174    write_hist(&mut out, &pairhmm.read_len_hist);
175    out.push_str("    },\n");
176    out.push_str("    \"hap_len_hist_bucket25\": {\n");
177    write_hist(&mut out, &pairhmm.hap_len_hist);
178    out.push_str("    }\n");
179    out.push_str("  },\n");
180    out.push_str("  \"genotype\": {\n");
181    out.push_str(&format!("    \"sites\": {},\n", genotype.sites));
182    out.push_str(&format!(
183        "    \"candidate_alleles_sum\": {},\n",
184        genotype.candidate_alleles_sum
185    ));
186    out.push_str(&format!(
187        "    \"genotype_states_sum\": {},\n",
188        genotype.genotype_states_sum
189    ));
190    out.push_str(&format!(
191        "    \"pl_vector_len_sum\": {},\n",
192        genotype.pl_vector_len_sum
193    ));
194    out.push_str(&format!("    \"samples_sum\": {},\n", genotype.samples_sum));
195    out.push_str(&format!(
196        "    \"mean_alleles_per_site\": {:.4},\n",
197        if genotype.sites == 0 {
198            0.0
199        } else {
200            genotype.candidate_alleles_sum as f64 / genotype.sites as f64
201        }
202    ));
203    out.push_str(&format!(
204        "    \"mean_states_per_site\": {:.4},\n",
205        if genotype.sites == 0 {
206            0.0
207        } else {
208            genotype.genotype_states_sum as f64 / genotype.sites as f64
209        }
210    ));
211    out.push_str(&format!(
212        "    \"time_per_site_s\": {:.9},\n",
213        genotype.time_per_site_ns() / 1e9
214    ));
215    out.push_str(&format!(
216        "    \"time_per_genotype_state_s\": {:.9},\n",
217        genotype.time_per_state_ns() / 1e9
218    ));
219    out.push_str(&format!(
220        "    \"wall_s\": {:.6},\n",
221        ns_to_s(genotype.wall_ns)
222    ));
223    out.push_str(&format!(
224        "    \"ad_wall_s\": {:.6},\n",
225        ns_to_s(genotype.ad_wall_ns)
226    ));
227    out.push_str(&format!(
228        "    \"allele_map_wall_s\": {:.6},\n",
229        ns_to_s(genotype.allele_map_wall_ns)
230    ));
231    out.push_str(&format!(
232        "    \"marginalize_wall_s\": {:.6},\n",
233        ns_to_s(genotype.marginalize_wall_ns)
234    ));
235    out.push_str(&format!(
236        "    \"genotype_enum_wall_s\": {:.6},\n",
237        ns_to_s(genotype.genotype_enum_wall_ns)
238    ));
239    out.push_str(&format!(
240        "    \"event_rebuild_wall_s\": {:.6}\n",
241        ns_to_s(genotype.event_rebuild_wall_ns)
242    ));
243    out.push_str("  }\n");
244    out.push_str("}\n");
245    out
246}

248fn write_hist(out: &mut String, hist: &std::collections::BTreeMap<u32, u64>) {
249    let items: Vec<_> = hist.iter().collect();
250    for (i, (k, v)) in items.iter().enumerate() {
251        let comma = if i + 1 == items.len() { "" } else { "," };
252        let _ = writeln!(out, "      \"{k}\": {v}{comma}");
253    }
254}

256fn build_markdown(
257    stages: &StageAgg,
258    pairhmm: &PairHmmAgg,
259    genotype: &GenotypeAgg,
260    run_wall: Duration,
261    run_cpu: Option<Duration>,
262    rayon_threads: usize,
263    regions: u64,
264) -> String {
265    let mut md = String::new();
266    md.push_str("# HC production profile\n\n");
267    md.push_str("Observe-only. Does **not** change genotype/emit.\n\n");
268    md.push_str(&format!("- Regions: **{regions}**\n"));
269    md.push_str(&format!("- Rayon threads: **{rayon_threads}**\n"));
270    md.push_str(&format!("- Run wall: **{:.3}s**\n", run_wall.as_secs_f64()));
271    if let Some(cpu) = run_cpu {
272        let eff = cpu.as_secs_f64() / (run_wall.as_secs_f64() * rayon_threads as f64).max(1e-9);
273        md.push_str(&format!(
274            "- Run CPU (user+sys): **{:.3}s** (parallel efficiency ≈ **{:.2}**)\n",
275            cpu.as_secs_f64(),
276            eff
277        ));
278    }
279    md.push_str("\n## Stages (by wall)\n\n");
280    md.push_str("| Stage | Calls | Wall s | CPU s | Avg wall s | Alloc bytes |\n");
281    md.push_str("|-------|------:|-------:|------:|-----------:|------------:|\n");
282    let mut rows: Vec<_> = Stage::all()
283        .iter()
284        .map(|s| (*s, stages.get(*s)))
285        .filter(|(_, st)| st.calls > 0 || st.alloc_bytes > 0)
286        .collect();
287    rows.sort_by(|a, b| b.1.wall_ns.cmp(&a.1.wall_ns));
288    for (stage, st) in rows {
289        md.push_str(&format!(
290            "| `{}` | {} | {:.3} | {:.3} | {:.6} | {} |\n",
291            stage.as_str(),
292            st.calls,
293            ns_to_s(st.wall_ns),
294            ns_to_s(st.cpu_ns),
295            st.avg_wall_ns() / 1e9,
296            st.alloc_bytes
297        ));
298    }
299    let occ = pairhmm.simd_occupancy();
300    md.push_str("\n## PairHMM\n\n");
301    md.push_str(&format!("- Reads scored: **{}**\n", pairhmm.reads_scored));
302    md.push_str(&format!(
303        "- Haplotypes scored: **{}**\n",
304        pairhmm.haplotypes_scored
305    ));
306    md.push_str(&format!(
307        "- Read×hap pairs: **{}**\n",
308        pairhmm.read_hap_pairs
309    ));
310    md.push_str(&format!(
311        "- Mean haps/read: **{:.2}**\n",
312        pairhmm.mean_haps_per_read()
313    ));
314    md.push_str(&format!(
315        "- SIMD pack units: **{}** ({}-wide ≈{} haps, occupancy **{:.1}%**)\n",
316        occ.pack_units, occ.pack_lanes, occ.pack_hap_est, occ.pack_occupancy_pct
317    ));
318    md.push_str(&format!(
319        "- Prefix reuse: **{:.1}%** ({} haps)\n",
320        occ.prefix_reuse_pct, pairhmm.prefix_reuse_haps
321    ));
322    md.push_str(&format!(
323        "- Leftover: **{:.1}%** ({} haps)\n",
324        occ.leftover_pct, pairhmm.leftover_haps
325    ));
326    md.push_str(&format!(
327        "- DP cells evaluated: **{}**; avoided via prefix: **{}**\n",
328        pairhmm.dp_cells_evaluated, pairhmm.dp_cells_avoided_prefix
329    ));
330    md.push_str(&format!(
331        "- PairHMM wall: **{:.3}s**\n",
332        ns_to_s(pairhmm.wall_ns)
333    ));
334
335    md.push_str("\n## Genotyping\n\n");
336    md.push_str(&format!("- Sites: **{}**\n", genotype.sites));
337    md.push_str(&format!(
338        "- Candidate alleles (sum): **{}**\n",
339        genotype.candidate_alleles_sum
340    ));
341    md.push_str(&format!(
342        "- Genotype states (sum): **{}** (diploid PL entries; typically 3/site)\n",
343        genotype.genotype_states_sum
344    ));
345    md.push_str(&format!(
346        "- PL vector lengths (sum): **{}**; samples (sum): **{}**\n",
347        genotype.pl_vector_len_sum, genotype.samples_sum
348    ));
349    md.push_str(&format!(
350        "- Time/site: **{:.6}s**; time/state: **{:.9}s**\n",
351        genotype.time_per_site_ns() / 1e9,
352        genotype.time_per_state_ns() / 1e9
353    ));
354    md.push_str(&format!(
355        "- Nested — AD: **{:.3}s**; allele-map: **{:.3}s**; marginalize: **{:.3}s**; PL-enum: **{:.3}s**; event-rebuild: **{:.3}s**; assign Σ: **{:.3}s**\n",
356        ns_to_s(genotype.ad_wall_ns),
357        ns_to_s(genotype.allele_map_wall_ns),
358        ns_to_s(genotype.marginalize_wall_ns),
359        ns_to_s(genotype.genotype_enum_wall_ns),
360        ns_to_s(genotype.event_rebuild_wall_ns),
361        ns_to_s(genotype.wall_ns)
362    ));
363    md.push('\n');
364    md
365}
