1use gatk_common::{GatkError, GatkResult};

3/// Typed SAM optional field value (parity subset: i, f, Z, A).
4/// # Invariants
5/// Tag identifiers are two alphanumeric ASCII characters at parse/format time.
6/// # Ownership
7/// Owns string/char payloads for `String`/`Char` variants; integers/floats are [`Copy`].
8/// # Mutation
9/// Immutable after parse; round-trips via [`format_optional_tag_field`].
10/// # Biological assumptions
11/// None — SAM tag transport encoding.
12/// # Java equivalence
13/// HTSJDK optional SAM tag typing (`SAMTag` / auxiliary field encoding).
14#[derive(Debug, Clone, PartialEq)]
15pub enum OptionalTagValue {
16    Int(i64),
17    Float(f32),
18    String(String),
19    Char(char),
20}
21
22pub fn parse_optional_tag_field(field: &str) -> GatkResult<(String, OptionalTagValue)> {
23    let mut parts = field.splitn(3, ':');
24    let tag = parts
25        .next()
26        .ok_or_else(|| GatkError::read("Malformed optional tag: missing tag"))?;
27    let type_code = parts
28        .next()
29        .ok_or_else(|| GatkError::read("Malformed optional tag: missing type code"))?;
30    let payload = parts
31        .next()
32        .ok_or_else(|| GatkError::read("Malformed optional tag: missing payload"))?;
33
34    if tag.len() != 2 || !tag.bytes().all(|b| b.is_ascii_alphanumeric()) {
35        return Err(GatkError::read(format!(
36            "Malformed optional tag: invalid tag identifier '{tag}'"
37        )));
38    }
39    if type_code.len() != 1 {
40        return Err(GatkError::read(format!(
41            "Malformed optional tag: invalid type code '{type_code}'"
42        )));
43    }
44
45    let value = match type_code.as_bytes()[0] {
46        b'i' => OptionalTagValue::Int(payload.parse::<i64>().map_err(|_| {
47            GatkError::read(format!(
48                "Malformed optional tag: invalid integer payload '{payload}'"
49            ))
50        })?),
51        b'f' => OptionalTagValue::Float(payload.parse::<f32>().map_err(|_| {
52            GatkError::read(format!(
53                "Malformed optional tag: invalid float payload '{payload}'"
54            ))
55        })?),
56        b'Z' => OptionalTagValue::String(payload.to_string()),
57        b'A' => {
58            let mut chars = payload.chars();
59            let c = chars.next().ok_or_else(|| {
60                GatkError::read("Malformed optional tag: empty char payload for A type")
61            })?;
62            if chars.next().is_some() {
63                return Err(GatkError::read(format!(
64                    "Malformed optional tag: expected single char payload for A type, got '{payload}'"
65                )));
66            }
67            OptionalTagValue::Char(c)
68        }
69        other => {
70            return Err(GatkError::read(format!(
71                "Malformed optional tag: unsupported type code '{}'",
72                other as char
73            )));
74        }
75    };
76
77    Ok((tag.to_string(), value))
78}

80pub fn format_optional_tag_field(tag: &str, value: &OptionalTagValue) -> GatkResult<String> {
81    if tag.len() != 2 || !tag.bytes().all(|b| b.is_ascii_alphanumeric()) {
82        return Err(GatkError::read(format!(
83            "Malformed optional tag: invalid tag identifier '{tag}'"
84        )));
85    }
86    let body = match value {
87        OptionalTagValue::Int(v) => format!("{tag}:i:{v}"),
88        OptionalTagValue::Float(v) => format!("{tag}:f:{v}"),
89        OptionalTagValue::String(v) => format!("{tag}:Z:{v}"),
90        OptionalTagValue::Char(v) => format!("{tag}:A:{v}"),
91    };
92    Ok(body)
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn optional_tag_roundtrip_common_types() {
101        let samples = ["NM:i:4", "AS:i:255", "rq:f:0.125", "RG:Z:group-a", "XT:A:R"];
102
103        for raw in samples {
104            let (tag, value) = parse_optional_tag_field(raw).unwrap();
105            let rendered = format_optional_tag_field(&tag, &value).unwrap();
106            let (_, reparsed) = parse_optional_tag_field(&rendered).unwrap();
107            assert_eq!(value, reparsed);
108        }
109    }
110
111    #[test]
112    fn optional_tag_rejects_malformed_values() {
113        for raw in ["N:i:1", "NM::1", "NM:i", "NM:q:1", "XT:A:AB"] {
114            assert!(parse_optional_tag_field(raw).is_err());
115        }
116    }
