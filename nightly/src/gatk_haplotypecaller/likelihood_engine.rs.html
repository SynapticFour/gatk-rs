16/// Per-read PairHMM input planes (BQ-capped quals + GOP/GCP). TLS avoids alloc-per-read
17/// on dense regions (observable contract: same fills as fresh `Vec`s each call).
18struct PairHmmReadScratch {
19    capped: Vec<u8>,
20    ins: Vec<u8>,
21    del: Vec<u8>,
22    gcp: Vec<u8>,
23}
24
25impl PairHmmReadScratch {
26    fn empty() -> Self {
27        Self {
28            capped: Vec::new(),
29            ins: Vec::new(),
30            del: Vec::new(),
31            gcp: Vec::new(),
32        }
33    }
34
35    fn ensure(&mut self, n: usize) {
36        if self.capped.len() < n {
37            self.capped.resize(n, 0);
38            self.ins.resize(n, 0);
39            self.del.resize(n, 0);
40            self.gcp.resize(n, 0);
41        }
42    }
43}
44
45thread_local! {
46    static PAIRHMM_READ_SCRATCH: RefCell<PairHmmReadScratch> =
47        RefCell::new(PairHmmReadScratch::empty());
48}

53/// GATK `ReadLikelihoodCalculationEngine.Implementation` slice for parity / production defaults.
54/// # Invariants
55/// Production `FastestAvailable` still scores via scalar `Log10PairHMM` until SIMD gates close
56/// (`pair_hmm_impl` defaults to [`PairHmmImpl::FastestAvailable`] — host SIMD when present).
57/// `FlowBased` is unused unless explicitly enabled.
58/// # Ownership
59/// [`Copy`] implementation discriminant.
60/// # Mutation
61/// Selected via [`HcLikelihoodEngineConfig`]; immutable during scoring.
62/// # Biological assumptions
63/// None beyond which PairHMM engine family scores reads.
64/// # Java equivalence
65/// GATK `ReadLikelihoodCalculationEngine.Implementation`.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub enum HcLikelihoodImplementation {
68    /// Standard HC likelihood path (PairHMM kernel from [`HcLikelihoodEngineConfig::pair_hmm_impl`]).
69    FastestAvailable,
70    /// Flow-based engine — not used unless explicitly enabled.
71    FlowBased,
72}

74/// DRAGSTR + stepwise filtering configuration (HC defaults: all off).
75/// # Invariants
76/// Production defaults: `FastestAvailable`, conservative PCR, BQ threshold 18, DRAGSTR off.
77/// Stepwise filtering only activates with a non-flow primary engine.
78/// # Ownership
79/// Cloneable config nested in [`crate::engine::CallRegionArgs`].
80/// # Mutation
81/// Snapshot for a region; prepare-quals helpers may mutate quality buffers, not this config.
82/// # Biological assumptions
83/// Controls PairHMM quality capping, PCR indel model, and optional DRAGSTR/stepwise paths.
84/// # Java equivalence
85/// GATK `PairHMMLikelihoodCalculationEngine` argument collection slice.
86#[derive(Debug, Clone)]
87pub struct HcLikelihoodEngineConfig {
88    pub implementation: HcLikelihoodImplementation,
89    /// PairHMM kernel (`--pair-hmm`). Default: Log10 for F.1/F.2 / P12 stability.
90    pub pair_hmm_impl: PairHmmImpl,
91    pub pcr_error_model: PcrErrorModel,
92    /// When true, a second FlowBased engine would score reads for allele filtering (Java stepwise).
93    pub stepwise_filtering: bool,
94    /// Parsed DRAGSTR params present (parity: path set in Java).
95    pub dragstr_params_loaded: bool,
96    pub dont_use_dragstr_pair_hmm: bool,
97    /// GATK `base-quality-score-threshold` before PairHMM.
98    pub base_quality_score_threshold: u8,
99    /// GATK `disable-cap-base-qualities-to-map-quality`.
100    pub disable_cap_read_qualities_to_mapq: bool,
101}

135/// Apply GATK production BQ/MQ capping before PairHMM (matches `PairHMMLikelihoodCalculationEngine`).
136pub fn prepare_read_quals_for_pairhmm(
137    read_quals: &[u8],
138    read_mapq: u8,
139    config: &HcLikelihoodEngineConfig,
140) -> Vec<u8> {
141    let mut quals = read_quals.to_vec();
142    prepare_read_quals_for_pairhmm_inplace(&mut quals, read_mapq, config);
143    quals
144}
145
146/// In-place BQ/MQ cap (avoids an extra allocation when the caller already owns a buffer).
147pub fn prepare_read_quals_for_pairhmm_inplace(
148    read_quals: &mut [u8],
149    read_mapq: u8,
150    config: &HcLikelihoodEngineConfig,
151) {
152    cap_read_base_qualities(
153        read_quals,
154        read_mapq,
155        config.base_quality_score_threshold,
156        config.disable_cap_read_qualities_to_mapq,
157    );
158}

160impl HcLikelihoodEngineConfig {
161    pub fn uses_dragstr_pair_hmm(&self) -> bool {
162        self.dragstr_params_loaded && !self.dont_use_dragstr_pair_hmm
163    }
164
165    pub fn filter_step_engine_active(&self) -> bool {
166        self.stepwise_filtering && self.implementation != HcLikelihoodImplementation::FlowBased
167    }
168
169    /// Stable label for parity dumps / logging.
170    ///
171    /// Reports the **configured** PairHMM selection (host-independent). Resolved
172    /// backends (AVX2 / NEON / scalar) vary by runner and must not appear in L2
173    /// frozen dumps.
174    pub fn primary_engine_label(&self) -> &'static str {
175        if self.implementation == HcLikelihoodImplementation::FlowBased {
176            return "FlowBased";
177        }
178        match self.pair_hmm_impl {
179            PairHmmImpl::Log10PairHmm => "Log10PairHMM",
180            PairHmmImpl::LoglessPairHmm => "LoglessPairHMM",
181            PairHmmImpl::Simd => "Simd",
182            PairHmmImpl::SimdF32 => "SimdF32",
183            PairHmmImpl::FastestAvailable => "FastestAvailable",
184        }
185    }
186}

188/// Score one read×haplotype with production `Log10PairHMM` (BQ cap + PCR + default indel/GCP).
189pub fn log10_read_haplotype_likelihood(
190    config: &HcLikelihoodEngineConfig,
191    read_bases: &[u8],
192    read_quals: &[u8],
193    read_mapq: u8,
194    haplotype_bases: &[u8],
195) -> GatkResult<f64> {
196    let scores = score_read_against_haplotypes(
197        config,
198        read_bases,
199        read_quals,
200        read_mapq,
201        &[haplotype_bases],
202    )?;
203    Ok(scores[0])
204}

206/// Score one read against many haplotypes (BQ/PCR prepared once; haplotypes in parallel).
207/// Result order matches `haplotype_bases` (algorithm-identical to sequential scoring).
208/// Under `GATK_RS_HC_SEQUENTIAL=1`, scalar Log10/Logless use a plain iterator (no rayon) so
209/// haplotype score order still matches sequential collect; SIMD path unchanged.
210///
211/// When already inside a Rayon worker (engine read-parallel), haplotype scoring stays
212/// sequential — one parallel axis only (Peak TLS + oversubscription).
213/// Dispatches on [`HcLikelihoodEngineConfig::implementation`] (FlowBased errors until enabled).
214pub fn score_read_against_haplotypes(
215    config: &HcLikelihoodEngineConfig,
216    read_bases: &[u8],
217    read_quals: &[u8],
218    read_mapq: u8,
219    haplotype_bases: &[&[u8]],
220) -> GatkResult<Vec<f64>> {
221    if config.implementation == HcLikelihoodImplementation::FlowBased {
222        return Err(gatk_common::GatkError::algorithm(
223            "FlowBased likelihood engine is not enabled in this Rust build (use standard HC)",
224        ));
225    }
226    if haplotype_bases.is_empty() {
227        return Ok(Vec::new());
228    }
229    let n = read_bases.len();
230    if n == 0 {
231        return Ok(vec![0.0; haplotype_bases.len()]);
232    }
233    let backend = config.resolved_pair_hmm_backend();
234
235    PAIRHMM_READ_SCRATCH.with(|cell| {
236        let mut scratch = cell.borrow_mut();
237        scratch.ensure(n);
238        scratch.capped[..n].copy_from_slice(read_quals);
239        prepare_read_quals_for_pairhmm_inplace(&mut scratch.capped[..n], read_mapq, config);
240        scratch.ins[..n].fill(GATK_PARITY_DEFAULT_INS_QUAL);
241        scratch.del[..n].fill(GATK_PARITY_DEFAULT_DEL_QUAL);
242        scratch.gcp[..n].fill(GATK_PARITY_DEFAULT_GCP);
243        if !config.uses_dragstr_pair_hmm() {
244            let PairHmmReadScratch {
245                ins,
246                del,
247                capped: _,
248                gcp: _,
249            } = &mut *scratch;
250            crate::pcr_error_model::apply_pcr_error_model(
251                read_bases,
252                &mut ins[..n],
253                &mut del[..n],
254                config.pcr_error_model,
255            );
256        }
257        let capped = &scratch.capped[..n];
258        let ins = &scratch.ins[..n];
259        let del = &scratch.del[..n];
260        let gcp = &scratch.gcp[..n];
261        match backend {
262            PairHmmBackend::Log10Scalar => haplotype_bases
263                .iter()
264                .map(|hap| {
265                    crate::pairhmm_log10::log10_pairhmm_likelihood(
266                        read_bases, capped, hap, ins, del, gcp,
267                    )
268                })
269                .collect(),
270            PairHmmBackend::LoglessScalar => haplotype_bases
271                .iter()
272                .map(|hap| {
273                    crate::pairhmm_logless::logless_pairhmm_likelihood(
274                        read_bases, capped, hap, ins, del, gcp,
275                    )
276                })
277                .collect(),
278            _ => {
279                score_read_haps_logless(backend, read_bases, capped, haplotype_bases, ins, del, gcp)
280            }
281        }
282    })
283}

289    #[test]
290    fn production_pairhmm_caps_low_base_qualities() {
291        let cfg = HcLikelihoodEngineConfig::default();
292        let mut quals = vec![5u8, 30, 30, 30];
293        let capped = prepare_read_quals_for_pairhmm(&quals, 60, &cfg);
294        assert_eq!(capped[0], crate::pairhmm_qual::MIN_USABLE_Q_SCORE);
295        cap_read_base_qualities(&mut quals, 60, cfg.base_quality_score_threshold, true);
296        assert_eq!(capped, quals);
297    }
