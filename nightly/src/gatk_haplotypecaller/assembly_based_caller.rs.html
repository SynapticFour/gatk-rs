23/// Assembly product plus the `finalizeRegion` BAM buffer for reuse by PairHMM.
24///
25/// # Ownership
26/// Owns haplotypes/events and the finalized read records from one assemble pass.
27/// # Observable contract
28/// Same haplotypes as [`assemble_reads`]; `finalized_reads` matches
29/// [`finalize_region_reads_for_assembly`] for the input region (softclip/adaptor applied once).
30pub struct AssembledRegion {
31    pub assembly: AssemblyResultSet,
32    pub finalized_reads: Vec<bam::Record>,
33}

35/// GATK `AssemblerArgumentCollection` read-error-correction slice.
36/// # Invariants
37/// HC defaults keep `error_correct_reads` false and flow HMER collapse disabled.
38/// `pileup_error_correction_log_odds == None` means −∞ (k-mer path when correction enabled).
39/// # Ownership
40/// Cloneable args nested in [`AssembleReadsArgs`].
41/// # Mutation
42/// Snapshot for one assemble pass.
43/// # Biological assumptions
44/// Optional pre-assembly error correction of reads (pileup LOD or nearby k-mer).
45/// # Java equivalence
46/// GATK `AssemblerArgumentCollection` error-correction / flow-collapse knobs.
47#[derive(Debug, Clone)]
48pub struct ReadErrorCorrectionArgs {
49    /// GATK `errorCorrectReads` (default false).
50    pub error_correct_reads: bool,
51    /// GATK `pileupErrorCorrectionLogOdds`; `None` = `Double.NEGATIVE_INFINITY` (use k-mer path when `error_correct_reads`).
52    pub pileup_error_correction_log_odds: Option<f64>,
53    /// GATK `flowAssemblyCollapseHKerSize` (0 = disabled for standard Illumina HC).
54    pub flow_assembly_collapse_hmer_size: usize,
55    pub flow_assembly_collapse_partial_mode: bool,
56}
57
58impl ReadErrorCorrectionArgs {
59    pub fn gatk_haplotype_caller_defaults() -> Self {
60        Self {
61            error_correct_reads: false,
62            pileup_error_correction_log_odds: None,
63            flow_assembly_collapse_hmer_size: 0,
64            flow_assembly_collapse_partial_mode: false,
65        }
66    }
67}

69/// HC defaults for the parity / E2E assembly slice (`HcFullParityGateDump` / `HaplotypeCallerEngine`).
70/// # Invariants
71/// Production/`strict_java_assembly` disables P12 materialize/inject at assemble time.
72/// Nested assembler + error-correction configs must be mutually consistent.
73/// # Ownership
74/// Owns nested assembler args, given alleles, and pileup-detection config.
75/// # Mutation
76/// Built once per `call_region` / assemble invocation; read immutably during assembly.
77/// # Biological assumptions
78/// Full assembleReads path: finalize → optional correct → thread → haplotypes.
79/// # Java equivalence
80/// GATK `AssemblyBasedCallerUtils.assembleReads` argument surface.
81#[derive(Debug, Clone)]
82pub struct AssembleReadsArgs {
83    pub assembler: ReadThreadingAssemblerArgs,
84    pub read_error_correction: ReadErrorCorrectionArgs,
85    /// GATK `!doNotCorrectOverlappingBaseQualities`.
86    pub correct_overlapping_base_qualities: bool,
87    /// GATK `givenAlleles` / `-alleles` (see [`crate::given_alleles`]).
88    pub given_alleles: Vec<crate::given_alleles::GatkGivenAllele>,
89    /// N1: read-pileup supplement when graph assembly is ref-only.
90    pub pileup_detection: PileupDetectionConfig,
91    /// `CallRegionMode::StrictJava` — no P12 materialize/inject at assemble time.
92    pub strict_java_assembly: bool,
93}

132pub(crate) fn apply_read_error_correction(
133    finalized: &[rust_htslib::bam::Record],
134    rec_args: &ReadErrorCorrectionArgs,
135    full_reference_with_padding: &[u8],
136) -> GatkResult<Vec<crate::assembly::AssemblyRead>> {
137    if let Some(log_odds) = rec_args.pileup_error_correction_log_odds {
138        let mut aligned: Vec<AlignedAssemblyRead> = finalized
139            .iter()
140            .map(|rec| {
141                let start1 = (rec.pos() as u64).saturating_add(1);
142                AlignedAssemblyRead {
143                    bases: rec.seq().as_bytes().to_vec(),
144                    base_quals: rec.qual().to_vec(),
145                    start1,
146                }
147            })
148            .collect();
149        correct_reads_pileup_log_odds(&mut aligned, log_odds)?;
150        return Ok(aligned
151            .into_iter()
152            .map(|a| crate::assembly::AssemblyRead {
153                bases: a.bases,
154                base_quals: a.base_quals,
155            })
156            .collect());
157    }
158    if rec_args.error_correct_reads {
159        let mut records = finalized.to_vec();
160        let kmer_cfg = NearbyKmerErrorCorrectorConfig::gatk_defaults();
161        correct_reads_nearby_kmer(&mut records, full_reference_with_padding, &kmer_cfg)?;
162        return Ok(records_to_assembly_reads(&records));
163    }
164    Ok(records_to_assembly_reads(finalized))
165}

167/// Replace full-padded ref stitches with the GATK `createReferenceHaplotype` slice on the padded span.
168fn normalize_production_haplotypes(
169    result: &mut AssemblyResult,
170    reference: &AssemblyRead,
171    region: &AssemblyRegion,
172    dictionary: &SequenceDictionary,
173    strict_java_assembly: bool,
174) {
175    let full_ref = reference.bases.as_slice();
176    let wrong_full_ref_stitch = |h: &crate::haplotype::Haplotype| {
177        h.is_reference && h.bases.len() == full_ref.len() && h.bases.as_slice() != full_ref
178    };
179    if matches!(result.status, AssemblyStatus::JustAssembledReference) {
180        result.haplotypes = vec![reference_haplotype_for_assembly_region(
181            reference, region, dictionary,
182        )];
183        return;
184    }
185    if result.haplotypes.len() == 1 && wrong_full_ref_stitch(&result.haplotypes[0]) {
186        result.haplotypes = vec![reference_haplotype_for_assembly_region(
187            reference, region, dictionary,
188        )];
189    }
190    let ref_hap = reference_haplotype_for_assembly_region(reference, region, dictionary);
191    crate::haplotype::prune_fragment_non_reference_haplotypes(
192        &mut result.haplotypes,
193        &ref_hap,
194        crate::read_threading_assembler::MIN_HAPLOTYPE_REFERENCE_LENGTH,
195    );
196    crate::haplotype::normalize_ref_equivalent_haplotypes(&mut result.haplotypes, &ref_hap.bases);
197    if !strict_java_assembly {
198        crate::haplotype::collapse_dangling_tail_alt_duplicates(&mut result.haplotypes, &ref_hap);
199        crate::haplotype::sort_haplotypes_assembly_result_order(&mut result.haplotypes);
200    }
201    result.status =
202        if result.haplotypes.iter().any(|h| !h.is_reference) && result.haplotypes.len() > 1 {
203            AssemblyStatus::AssembledSomeVariation
204        } else {
205            AssemblyStatus::JustAssembledReference
206        };
207}

220/// `assembleReads` keeping the `finalizeRegion` buffer for PairHMM reuse (A2).
221pub fn assemble_reads_with_finalized(
222    region: &AssemblyRegion,
223    dictionary: &SequenceDictionary,
224    ref_cache: &mut ReferenceWindowCache,
225    args: &AssembleReadsArgs,
226) -> GatkResult<AssembledRegion> {
227    let reference = assembly_reference_read(dictionary, ref_cache, region)?;
228    let finalized = finalize_region_reads_for_assembly(
229        &region.reads,
230        region,
231        args.correct_overlapping_base_qualities,
232        gatk_min_tail_quality_for_assembly(args.assembler.min_base_quality),
233        false,
234    );
235    let assembly_reads = apply_read_error_correction(
236        &finalized,
237        &args.read_error_correction,
238        reference.bases.as_slice(),
239    )?;
240    let reads = assembly_reads;
241    let (padded_loc_start, _) = padded_reference_loc(region, dictionary);
242    let mut assembler = args.assembler.clone();
243    if args.strict_java_assembly {
244        assembler.dangling_java_exact = true;
245        // P12 NA12878 cluster: RT-only assembly (SeqGraph zip drops coupled indels). General HC keeps SeqGraph.
246        if crate::read_threading_assembler::region_overlaps_p12_cluster(
247            region.start.get(),
248            region.end.get(),
249        ) {
250            assembler.use_seq_graph = false;
251            assembler.remove_paths_not_connected_to_ref = false;
252            assembler.skip_post_dangling_prune = true;
253        }
254        assembler.scoring = Some(crate::read_threading_assembler::AssemblyScoringContext {
255            padded_reference_start_1based: padded_loc_start,
256            active_start_1based: region.start.get(),
257            active_end_1based: region.end.get(),
258            // CLONE: needed because owned contig id for output record.
259            contig: region.contig.clone(),
260        });
261    } else if crate::read_threading_assembler::region_overlaps_p12_cluster(
262        region.start.get(),
263        region.end.get(),
264    ) {
265        assembler.scoring = Some(crate::read_threading_assembler::AssemblyScoringContext {
266            padded_reference_start_1based: padded_loc_start,
267            active_start_1based: region.start.get(),
268            active_end_1based: region.end.get(),
269            // CLONE: needed because owned contig id for output record.
270            contig: region.contig.clone(),
271        });
272    }
273    let mut result = assemble_from_ref_and_reads(&reference, &reads, &assembler)?;
274    normalize_production_haplotypes(
275        &mut result,
276        &reference,
277        region,
278        dictionary,
279        args.strict_java_assembly,
280    );
281    let full_ref = reference.bases.as_slice();
282    crate::read_event_discovery::refresh_alt_haplotype_indel_cigars(
283        &mut result.haplotypes,
284        full_ref,
285        padded_loc_start,
286        &args.assembler.haplotype_to_reference_sw,
287    );
288    let full_ref = reference.bases.as_slice();
289    result.haplotypes = collapse_haplotypes_if_configured(
290        result.haplotypes,
291        args.read_error_correction.flow_assembly_collapse_hmer_size,
292        args.read_error_correction
293            .flow_assembly_collapse_partial_mode,
294        full_ref,
295        padded_loc_start,
296    );
297    let mut assembly = AssemblyResultSet::from_assembly_for_calling(
298        &result,
299        reference.bases.as_slice(),
300        padded_loc_start,
301        &region.contig,
302        crate::assembly_result_set::DEFAULT_MAX_MNP_DISTANCE,
303    );
304    // Given alleles: Java `addGivenAlleles` runs in `call_region` after `trimTo`, not in `assembleReads`.
305    let sw = &args.assembler.haplotype_to_reference_sw;
306    let apply_bases = assembly.reference_bases_shared();
307    let apply_pad = assembly.padded_reference_start_1based();
308    let contig = region.contig.clone();
309    crate::read_event_discovery::sync_assembly_events_from_haplotype_cigars(
310        &mut assembly,
311        &contig,
312        sw,
313    );
314    if !args.strict_java_assembly {
315        if crate::read_event_discovery::assembly_cluster_inject_enabled() {
316            crate::read_event_discovery::ensure_assembly_cluster_indel_events(
317                &mut assembly,
318                &apply_bases,
319                apply_pad,
320                region.start.get(),
321                region.end.get(),
322                &contig,
323                sw,
324            )?;
325        } else if region.start.get()
326            <= crate::read_event_discovery::P12_CLUSTER_TTC_START.saturating_add(3)
327            && region.end.get() >= crate::read_event_discovery::P12_CLUSTER_TTC_START
328        {
329            crate::read_event_discovery::materialize_p12_cluster_from_assembly_cigars(
330                &mut assembly,
331                &apply_bases,
332                apply_pad,
333                region.start.get(),
334                region.end.get(),
335                &contig,
336                &region.reads,
337                sw,
338            )?;
339        }
340    }
341    Ok(AssembledRegion {
342        assembly,
343        finalized_reads: finalized,
344    })
345}

347/// Engine hook for `HaplotypeCallerEngine.callRegion` assembly slice.
348pub fn call_region_assemble(
349    region: &AssemblyRegion,
350    dictionary: &SequenceDictionary,
351    ref_cache: &mut ReferenceWindowCache,
352    args: &AssembleReadsArgs,
353) -> GatkResult<Option<AssembledRegion>> {
354    if !region.is_active {
355        return Ok(None);
356    }
357    Ok(Some(assemble_reads_with_finalized(
358        region, dictionary, ref_cache, args,
359    )?))
360}
