15/// H.2.1 — agreed `GATKVCFHeaderLines` subset.
16pub fn dump_gvcf_header_tsv(
17    contig: &str,
18    contig_length: u64,
19    out: &mut impl Write,
20) -> GatkResult<()> {
21    let lines = gatk_hc_gvcf_header_lines(contig, contig_length);
22    writeln!(out, "contig\t{contig}")?;
23    writeln!(out, "contig_length\t{contig_length}")?;
24    writeln!(out, "header_line_count\t{}", lines.len())?;
25    for (i, line) in lines.iter().enumerate() {
26        writeln!(out, "header_{i}\t{line}")?;
27    }
28    Ok(())
29}

31/// H.2.1 — build writer block records from p8-style locus fixture TSV.
32pub fn dump_gvcf_writer_from_loci_fixture_tsv(
33    fixture_path: &Path,
34    out: &mut impl Write,
35) -> GatkResult<()> {
36    let raw = std::fs::read_to_string(fixture_path)
37        .map_err(|e| GatkError::io(format!("read {}", fixture_path.display()), e))?;
38    let mut loci: Vec<ReferenceConfidenceLocus> = Vec::new();
39    let mut case_id = String::new();
40    for line in raw.lines() {
41        let t = line.trim();
42        if t.is_empty() || t.starts_with('#') {
43            continue;
44        }
45        let cols: Vec<&str> = t.split('\t').collect();
46        if cols.len() < 4 {
47            return Err(GatkError::argument(format!("bad fixture row: {t}")));
48        }
49        case_id = cols[0].to_string();
50        loci.push(ReferenceConfidenceLocus {
51            position_1based: cols[1]
52                .parse()
53                .map_err(|_| GatkError::argument("bad pos"))?,
54            gq: cols[2].parse().map_err(|_| GatkError::argument("bad gq"))?,
55            dp: cols[3].parse().map_err(|_| GatkError::argument("bad dp"))?,
56        });
57    }
58    let bands: Vec<i32> = GATK_HC_DEFAULT_GQB.to_vec();
59    let blocks = build_gvcf_blocks(&loci, &bands)?;
60    let mut writer = GvcfWriter::with_config(&GvcfWriterConfig {
61        sample_name: case_id.clone(),
62        ..GvcfWriterConfig::default()
63    });
64    for block in &blocks {
65        writer.records.push(gvcf_block_to_record_fields(block)?);
66    }
67    writeln!(out, "case_id\t{case_id}")?;
68    dump_gvcf_writer_state_tsv(&writer, out)
69}

71/// H-D01/D02 — merged gVCF block pseudo-VCF lines + joint-compat (L5 scaffold).
72pub fn dump_gvcf_l5_merged_tsv(fixture_path: &Path, out: &mut impl Write) -> GatkResult<()> {
73    let raw = std::fs::read_to_string(fixture_path)
74        .map_err(|e| GatkError::io(format!("read {}", fixture_path.display()), e))?;
75    let mut loci: Vec<ReferenceConfidenceLocus> = Vec::new();
76    let contig = "chr1";
77    for line in raw.lines() {
78        let t = line.trim();
79        if t.is_empty() || t.starts_with('#') {
80            continue;
81        }
82        let cols: Vec<&str> = t.split('\t').collect();
83        if cols.len() < 4 {
84            return Err(GatkError::argument(format!("bad fixture row: {t}")));
85        }
86        loci.push(ReferenceConfidenceLocus {
87            position_1based: cols[1]
88                .parse()
89                .map_err(|_| GatkError::argument("bad pos"))?,
90            gq: cols[2].parse().map_err(|_| GatkError::argument("bad gq"))?,
91            dp: cols[3].parse().map_err(|_| GatkError::argument("bad dp"))?,
92        });
93    }
94    let bands: Vec<i32> = GATK_HC_DEFAULT_GQB.to_vec();
95    let blocks = build_gvcf_blocks(&loci, &bands)?;
96    let records: Vec<_> = blocks
97        .iter()
98        .map(gvcf_block_to_record_fields)
99        .collect::<Result<_, _>>()?;
100    let compat = validate_joint_compatibility_gvcf_records(&records)?;
101    writeln!(out, "contig\t{contig}")?;
102    writeln!(out, "record_count\t{}", records.len())?;
103    writeln!(out, "joint_compatible\t{}", compat.compatible)?;
104    for (i, rec) in records.iter().enumerate() {
105        writeln!(
106            out,
107            "vcf_line_{i}\t{contig}\t{}\t.\t<NON_REF>\t.\t.\tEND={}\tMIN_DP={}\tMAX_DP={}\tGQ_BAND={}\tMIN_RGQ={}",
108            rec.start_1based,
109            rec.end_info,
110            rec.min_dp,
111            rec.max_dp,
112            rec.gq_band_upper,
113            rec.min_rgq,
114        )?;
115    }
116    Ok(())
