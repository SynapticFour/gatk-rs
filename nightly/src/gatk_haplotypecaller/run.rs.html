51/// Run the HaplotypeCaller tool: validate inputs, build traversal scaffold, then genotyping when implemented.
52pub fn run_haplotype_caller(config: &GatkConfig) -> GatkResult<()> {
53    if config.tool_config.tool_name != "HaplotypeCaller" {
54        return Err(GatkError::configuration(
55            "run_haplotype_caller requires a HaplotypeCaller GatkConfig",
56        ));
57    }
58
59    // Observe-only: optional NDJSON semantic checkpoints (`GATK_RS_SEMANTIC_TRACE`).
60    crate::semantic_trace::try_init_from_runtime(&RuntimeConfig::from_env());
61
62    let ref_path = config
63        .tool_config
64        .inputs
65        .reference
66        .as_deref()
67        .ok_or_else(|| GatkError::configuration("Reference FASTA (-R) is required"))?;
68
69    let reference = Path::new(ref_path);
70    if !reference.is_file() {
71        return Err(GatkError::argument(format!(
72            "Reference FASTA not found or not a file: {ref_path}"
73        )));
74    }
75
76    let dict = SequenceDictionary::from_fasta_path(reference).map_err(|e| {
77        GatkError::argument(format!("Failed to read reference FASTA {ref_path}: {e}"))
78    })?;
79    if dict.contig_count() == 0 {
80        return Err(GatkError::argument(format!(
81            "Reference FASTA contains no sequences: {ref_path}"
82        )));
83    }
84
85    if config.tool_config.inputs.input_files.is_empty() {
86        return Err(GatkError::configuration(
87            "At least one alignment input (-I) is required",
88        ));
89    }
90
91    let interval_specs = intervals_for_haplotype_caller(
92        &dict,
93        config.get_parameter("intervals").map(|s| s.as_str()),
94    )?;
95
96    let engine = HaplotypeCallerEngine::prepare_traversal_default(&dict, interval_specs)?;
97
98    let hc = config.get_haplotypecaller_config()?;
99    validate_haplotype_caller_input_policy(config)?;
100    reject_unsupported_haplotype_caller_cli(&hc, config)?;
101    let read_filters = ReadFilterParams::from_haplotype_caller(&hc);
102
103    for in_path in &config.tool_config.inputs.input_files {
104        let p = Path::new(in_path);
105        if !p.is_file() {
106            return Err(GatkError::argument(format!(
107                "Alignment input not found or not a file: {in_path}"
108            )));
109        }
110        let mut reader = bam::Reader::from_path(p).map_err(|e| {
111            GatkError::generic(format!("Failed to open alignment input {in_path}: {e}"))
112        })?;
113        let mut records = reader.records();
114        let seen = match records.next() {
115            None => false,
116            Some(res) => {
117                res.map_err(|e| GatkError::generic(format!("Failed reading {in_path}: {e}")))?;
118                true
119            }
120        };
121        if !seen {
122            return Err(GatkError::argument(format!(
123                "Alignment file contains no records: {in_path}"
124            )));
125        }
126    }
127
128    let out = config
129        .tool_config
130        .outputs
131        .output_vcf
132        .as_deref()
133        .ok_or_else(|| GatkError::configuration("Output VCF (-O) is required"))?;
134    let out_path = Path::new(out);
135    if let Some(parent) = out_path.parent() {
136        if !parent.as_os_str().is_empty() && !parent.is_dir() {
137            return Err(GatkError::argument(format!(
138                "Output directory does not exist: {}",
139                parent.display()
140            )));
141        }
142    }
143
144    // R4-1: default skips O(tiles×BAM) diagnostic scan (does not affect calling).
145    let debug_tile_overlaps = RuntimeConfig::from_env().debug.debug_tile_overlaps;
146    let read_tile_overlaps = if debug_tile_overlaps {
147        let mut n = 0usize;
148        for bam in &config.tool_config.inputs.input_files {
149            n += total_read_tile_overlaps(Path::new(bam), &engine.tiles, &read_filters)?;
150        }
151        Some(n)
152    } else {
153        None
154    };
155
156    match read_tile_overlaps {
157        Some(n) => info!(
158            "HaplotypeCaller traversal scaffold: {} input interval(s), {} traversal tile(s); read–tile overlap count (linear scan, scaffold) = {}",
159            engine.interval_specs.len(),
160            engine.tile_count(),
161            n
162        ),
163        None => info!(
164            "HaplotypeCaller traversal scaffold: {} input interval(s), {} traversal tile(s)",
165            engine.interval_specs.len(),
166            engine.tile_count(),
167        ),
168    }
169
170    reject_legacy_provisional_output(config)?;
171    crate::parity_harness::warn_if_harness_flags_set();
172
173    let variant_output_enabled = hc_variant_output_enabled();
174
175    // Default: assembly-region-v1 (`call_region`). Opt-in header-only scaffold for Phase 9 tests:
176    // `GATK_RS_HC_SCAFFOLD_OUTPUT=1` (or legacy `GATK_RS_HC_ACTIVATE_OUTPUT=0`).
177    let mut header = VcfHeader::default();
178    header.source = Some(if variant_output_enabled {
179        format!("gatk-rs HaplotypeCaller {HC_PIPELINE_ASSEMBLY_REGION_V1}")
180    } else {
181        format!("gatk-rs HaplotypeCaller {HC_PIPELINE_SCAFFOLD}")
182    });
183    header.reference = Some(ref_path.to_string());
184    header.contigs = dict
185        .contig_records()
186        .iter()
187        .map(|c| Contig {
188            id: c.name.clone(),
189            length: Some(c.length),
190            md5: None,
191            assembly: None,
192            species: None,
193            uri: None,
194        })
195        .collect();
196    header.other_headers.push((
197        "GATK_RS_HC_PIPELINE".to_string(),
198        if variant_output_enabled {
199            HC_PIPELINE_ASSEMBLY_REGION_V1.to_string()
200        } else {
201            HC_PIPELINE_SCAFFOLD.to_string()
202        },
203    ));
204    if variant_output_enabled {
205        if let Some(mode) = config.get_parameter("output_mode") {
206            header
207                .other_headers
208                // CLONE: needed because owned element into collection.
209                .push(("GATK_RS_HC_OUTPUT_MODE".to_string(), mode.clone()));
210        }
211    }
212    let emit_mode = emit_mode_from_output_mode(&config.tool_config.outputs.output_mode);
213    let sample_name = sample_name_from_bam_inputs(&config.tool_config.inputs.input_files)?;
214    if variant_output_enabled {
215        // CLONE: needed because owned element into collection.
216        header.samples.push(sample_name.clone());
217        if emit_mode == EmitMode::Gvcf {
218            let primary = dict
219                .contig_records()
220                .first()
221                .map(|c| (c.name.as_str(), c.length))
222                .unwrap_or(("chr1", 1_000_000));
223            for line in gatk_hc_gvcf_header_lines(primary.0, primary.1) {
224                if line.starts_with("##reference=") {
225                    continue;
226                }
227                if let Some((k, v)) = line.strip_prefix("##").and_then(|s| s.split_once('=')) {
228                    header.other_headers.push((k.to_string(), v.to_string()));
229                }
230            }
231            header
232                .other_headers
233                .push(("GATK_RS_HC_GVCF".to_string(), "1".to_string()));
234        }
235    }
236
237    let mut writer = VcfWriter::new(out_path, header)?;
238    writer.write_header()?;
239    let mut variant_records = 0usize;
240    if variant_output_enabled {
241        let pair_hmm_impl = pair_hmm_impl_from_config(&hc, config)?;
242        let records = assembly_region_variant_records(
243            reference,
244            &config.tool_config.inputs.input_files,
245            &engine.interval_specs,
246            &read_filters,
247            hc.stand_emit_confidence,
248            emit_mode,
249            &ReferenceConfidenceConfig::default(),
250            &sample_name,
251            config
252                .get_parameter("alleles")
253                .or_else(|| config.get_parameter("given_alleles"))
254                .cloned(),
255            pair_hmm_impl,
256        )?;
257        for rec in records {
258            writer.write_record(&rec)?;
259            variant_records += 1;
260        }
261    }
262    HaplotypeCallerEngine::shutdown();
263    info!(
264        "HaplotypeCaller wrote VCF to {} (variant records: {}, variant_output_enabled={}, scaffold_only={})",
265        out_path.display(),
266        variant_records,
267        variant_output_enabled,
268        !variant_output_enabled
269    );
270    Ok(())
271}

273/// VCF sample column: sole `@RG SM:` from the first input BAM, else `"SAMPLE"` (Java default).
274fn sample_name_from_bam_inputs(input_files: &[String]) -> GatkResult<String> {
275    let Some(bam_path) = input_files.first() else {
276        return Ok("SAMPLE".to_string());
277    };
278    let reader = bam::Reader::from_path(Path::new(bam_path)).map_err(|e| {
279        GatkError::argument(format!(
280            "Failed to open BAM for sample name ({bam_path}): {e}"
281        ))
282    })?;
283    let semantics = ReadHeaderSemantics::from_bam_header_view(reader.header())?;
284    Ok(semantics
285        .primary_sample_name()
286        .unwrap_or_else(|| "SAMPLE".to_string()))
287}

289/// H2-5: document multi-BAM policy — sequential per-BAM traversal, merged VCF rows (not Java joint calling).
290fn validate_haplotype_caller_input_policy(config: &GatkConfig) -> GatkResult<()> {
291    let n = config.tool_config.inputs.input_files.len();
292    if n > 1 {
293        tracing::warn!(
294            "Multiple -I alignment inputs ({n}): gatk-rs runs HaplotypeCaller independently on each \
295             BAM and merges variant rows into one VCF. Java merges read sets for joint calling; \
296             multi-sample / multi-library joint HC is not supported."
297        );
298    }
299    Ok(())
300}

302const HC_DEFAULT_MAX_ALTERNATE_ALLELES: u32 = 6;

304/// H2-3: fail fast on CLI flags parsed but not wired on the strict production path.
305fn reject_unsupported_haplotype_caller_cli(
306    hc: &HaplotypeCallerConfig,
307    config: &GatkConfig,
308) -> GatkResult<()> {
309    // Validate `--pair-hmm` early (wired into likelihood engine).
310    if let Some(raw) = hc
311        .pair_hmm
312        .as_deref()
313        .or_else(|| config.get_parameter("pair_hmm").map(|s| s.as_str()))
314    {
315        crate::pairhmm_simd::parse_pair_hmm_impl(raw)?;
316    }
317    if hc.max_alternate_alleles != HC_DEFAULT_MAX_ALTERNATE_ALLELES {
318        return Err(GatkError::configuration(format!(
319            "--max-alternate-alleles ({}) is not wired; gatk-rs uses the Java default ({HC_DEFAULT_MAX_ALTERNATE_ALLELES})",
320            hc.max_alternate_alleles
321        )));
322    }
323    if hc.original_base_qualities {
324        tracing::warn!(
325            "--original-base-qualities is accepted but not yet wired to PairHMM / activity inputs"
326        );
327    }
328    if hc.dont_use_soft_clipped_bases {
329        tracing::warn!(
330            "--dont-use-soft-clipped-bases is accepted but not yet wired on the strict callRegion path"
331        );
332    }
333    Ok(())
334}

336fn pair_hmm_impl_from_config(
337    hc: &HaplotypeCallerConfig,
338    config: &GatkConfig,
339) -> GatkResult<crate::pairhmm_simd::PairHmmImpl> {
340    let raw = hc
341        .pair_hmm
342        .as_deref()
343        .or_else(|| config.get_parameter("pair_hmm").map(|s| s.as_str()));
344    match raw {
345        Some(s) => crate::pairhmm_simd::parse_pair_hmm_impl(s),
346        None => Ok(crate::pairhmm_simd::PairHmmImpl::Log10PairHmm),
347    }
348}

372fn reject_legacy_provisional_output(config: &GatkConfig) -> GatkResult<()> {
373    let env_on = crate::runtime_config::RuntimeConfig::from_env()
374        .execution
375        .legacy_provisional;
376    let param_on = config
377        .get_parameter("legacy_provisional_output")
378        .is_some_and(|s| s == "1" || s.eq_ignore_ascii_case("true"));
379    if env_on || param_on {
380        return Err(GatkError::configuration(
381            "legacy provisional-output-v1 was removed (Sprint B); use default assembly-region-v1. \
382             For header-only scaffold tests set GATK_RS_HC_SCAFFOLD_OUTPUT=1.",
383        ));
384    }
385    Ok(())
386}

388fn assembly_region_variant_records(
389    reference_fasta: &Path,
390    input_bams: &[String],
391    interval_specs: &[gatk_core::reference::IntervalSpec],
392    read_filters: &ReadFilterParams,
393    stand_emit_confidence: f64,
394    emit_mode: EmitMode,
395    ref_confidence_config: &ReferenceConfidenceConfig,
396    sample_name: &str,
397    given_alleles_vcf: Option<String>,
398    pair_hmm_impl: crate::pairhmm_simd::PairHmmImpl,
399) -> GatkResult<Vec<VcfRecord>> {
400    let dict = SequenceDictionary::from_fasta_path(reference_fasta)?;
401    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(
402        GATK_DEFAULT_ASSEMBLY_REGION_PADDING,
403    );
404    let mut args = CallRegionArgs::default();
405    args.likelihood.pair_hmm_impl = pair_hmm_impl;
406    if let Some(path) = given_alleles_vcf.as_deref() {
407        args.given_alleles =
408            crate::given_alleles::load_given_alleles_from_vcf_path(Path::new(path))?;
409        args.assemble.given_alleles = args.given_alleles.clone();
410    }
411    let mut records = Vec::new();
412    let mut seen = std::collections::BTreeSet::new();
413    let mut ref_cache = ReferenceWindowCache::new(reference_fasta.to_path_buf(), 4);
414    let gvcf_global = emit_mode == EmitMode::Gvcf;
415
416    // GVCF path: sequential streaming (interval-wide collector + P12 reconcile on shells).
417    // Non-GVCF: stream regions in bounded rayon batches — never retain all region.reads at once.
418    if gvcf_global {
419        let mut gvcf_collector = GvcfIntervalCollector::new();
420        let mut region_shells: Vec<AssemblyRegion> = Vec::new();
421        for bam in input_bams {
422            let bam_path = Path::new(bam);
423            let header = bam::Reader::from_path(bam_path)
424                .map_err(|e| GatkError::generic(format!("open BAM for inactive ref model: {e}")))?
425                .header()
426                .clone();
427            for_each_assembly_region(
428                &dict,
429                interval_specs,
430                reference_fasta,
431                bam_path,
432                read_filters,
433                &cfg,
434                |_idx, mut region| {
435                    process_one_region_gvcf(
436                        &region,
437                        &header,
438                        &dict,
439                        reference_fasta,
440                        &args,
441                        read_filters,
442                        stand_emit_confidence,
443                        emit_mode,
444                        ref_confidence_config,
445                        &mut ref_cache,
446                        &mut gvcf_collector,
447                        &mut records,
448                        &mut seen,
449                        sample_name,
450                    )?;
451                    // Keep span metadata for P12 reconcile; drop owned BAM/ref payloads immediately.
452                    strip_region_payloads(&mut region);
453                    region_shells.push(region);
454                    Ok(())
455                },
456            )?;
457        }
458        if let Some(bam) = input_bams.last() {
459            let bam_path = Path::new(bam);
460            if p12_cluster_rcm_band_fragmented(&region_shells) {
461                reconcile_p12_cluster_rcm_band(
462                    &mut gvcf_collector,
463                    reference_fasta,
464                    bam_path,
465                    &dict,
466                    &mut ref_cache,
467                    read_filters,
468                    ref_confidence_config,
469                    &args,
470                    stand_emit_confidence,
471                    &region_shells,
472                )?;
473            }
474            reconcile_fragmented_dense_cluster_bands(
475                &mut gvcf_collector,
476                reference_fasta,
477                bam_path,
478                &dict,
479                &mut ref_cache,
480                read_filters,
481                ref_confidence_config,
482                &args,
483                stand_emit_confidence,
484                &region_shells,
485            )?;
486            gvcf_collector.fill_interval_gaps_from_pileup(
487                interval_specs,
488                &dict,
489                bam_path,
490                read_filters,
491                ref_confidence_config,
492                &mut ref_cache,
493            )?;
494            gvcf_collector.remove_cluster_core_excluded_hom_ref(P12_CLUSTER_RCM_CONTIG);
495            for &pos in P12_L5_JAVA_EXTRA_VARIANT_NO_HOM_REF {
496                gvcf_collector.remove_hom_ref_locus(P12_CLUSTER_RCM_CONTIG, pos);
497            }
498        }
499        for rec in gvcf_collector.into_block_records(&dict, &mut ref_cache, sample_name)? {
500            push_deduped_vcf(&mut records, &mut seen, rec);
501        }
502    } else {
503        // H2-5: each `-I` BAM is traversed independently; variant rows are merged.
504        // Bound in-flight regions to the Rayon pool size so Peak-RSS cannot grow with
505        // interval length × cloned BAM records (the 2 Mb / ~60 GiB failure mode).
506        let mut all_batches: Vec<RegionEmitBatch> = Vec::new();
507        let batch_limit = rayon::current_num_threads().max(1);
508        for bam in input_bams {
509            let bam_path = Path::new(bam);
510            let _ = bam::Reader::from_path(bam_path)
511                .map_err(|e| GatkError::generic(format!("open BAM for inactive ref model: {e}")))?;
512            let ref_path = reference_fasta.to_path_buf();
513            let bam_path_owned = bam_path.to_path_buf();
514            let sample = sample_name.to_string();
515            let mut pending: Vec<(usize, AssemblyRegion)> = Vec::with_capacity(batch_limit);
516
517            let flush_batch = |pending: &mut Vec<(usize, AssemblyRegion)>,
518                               all_batches: &mut Vec<RegionEmitBatch>|
519             -> GatkResult<()> {
520                if pending.is_empty() {
521                    return Ok(());
522                }
523                let chunk = std::mem::take(pending);
524                let batches: Vec<RegionEmitBatch> = chunk
525                    .into_par_iter()
526                    .map(|(region_index, region)| {
527                        let reader = bam::Reader::from_path(&bam_path_owned).map_err(|e| {
528                            GatkError::generic(format!("open BAM for region parallel worker: {e}"))
529                        })?;
530                        let header = reader.header().clone();
531                        let mut local_cache = ReferenceWindowCache::new(ref_path.clone(), 4);
532                        process_one_region_vcf(
533                            region_index,
534                            region,
535                            &header,
536                            &dict,
537                            &ref_path,
538                            &args,
539                            *read_filters,
540                            stand_emit_confidence,
541                            emit_mode,
542                            ref_confidence_config,
543                            &mut local_cache,
544                            &sample,
545                        )
546                    })
547                    .collect::<GatkResult<Vec<_>>>()?;
548                all_batches.extend(batches);
549                Ok(())
550            };
551
552            for_each_assembly_region(
553                &dict,
554                interval_specs,
555                reference_fasta,
556                bam_path,
557                read_filters,
558                &cfg,
559                |region_index, region| {
560                    pending.push((region_index, region));
561                    if pending.len() >= batch_limit {
562                        flush_batch(&mut pending, &mut all_batches)?;
563                    }
564                    Ok(())
565                },
566            )?;
567            flush_batch(&mut pending, &mut all_batches)?;
568        }
569        merge_region_emit_batches(&mut all_batches, &mut records, &mut seen);
570    }
571
572    records.sort_by(|a, b| {
573        a.chromosome
574            .cmp(&b.chromosome)
575            .then(a.position.cmp(&b.position))
576    });
577    Ok(records)
578}

580/// Process one AssemblyRegion for standard VCF emit (no shared GVCF collector).
581fn process_one_region_vcf(
582    region_index: usize,
583    region: AssemblyRegion,
584    header: &bam::HeaderView,
585    dict: &SequenceDictionary,
586    reference_fasta: &Path,
587    args: &CallRegionArgs,
588    read_filters: ReadFilterParams,
589    stand_emit_confidence: f64,
590    emit_mode: EmitMode,
591    ref_confidence_config: &ReferenceConfidenceConfig,
592    ref_cache: &mut ReferenceWindowCache,
593    sample_name: &str,
594) -> GatkResult<RegionEmitBatch> {
595    let contig = region.contig.clone();
596    let start = region.start.get();
597    let mut local_records = Vec::new();
598    let mut local_seen = std::collections::BTreeSet::new();
599
600    match call_disposition(&region) {
601        AssemblyRegionCallDisposition::InactiveReferenceFastPath => {
602            let outcome = HaplotypeCallerEngine::call_region_inactive_reference(
603                &region,
604                header,
605                dict,
606                reference_fasta,
607                emit_mode,
608                &read_filters,
609                ref_confidence_config,
610            )?;
611            crate::semantic_trace::emit_inactive_rcm(&region, outcome.loci.len());
612            let mut batch = crate::semantic_trace::is_enabled().then(Vec::new);
613            for rec in inactive_reference_model_to_vcf_records(
614                &outcome,
615                emit_mode,
616                dict,
617                ref_cache,
618                sample_name,
619            )? {
620                if let Some(b) = batch.as_mut() {
621                    // CLONE: needed because owned element into collection.
622                    b.push(rec.clone());
623                }
624                push_deduped_vcf(&mut local_records, &mut local_seen, rec);
625            }
626            if let Some(b) = batch.as_ref() {
627                crate::semantic_trace::emit_vcf_emission(Some(&region), b);
628            }
629        }
630        AssemblyRegionCallDisposition::ActiveFull => {
631            if let Some(outcome) =
632                HaplotypeCallerEngine::call_region(&region, dict, reference_fasta, args)?
633            {
634                let mut emitted = crate::semantic_trace::is_enabled().then(Vec::new);
635                for rec in try_emit_call_region_variants(
636                    &region,
637                    &outcome,
638                    sample_name,
639                    stand_emit_confidence,
640                )? {
641                    if let Some(e) = emitted.as_mut() {
642                        // CLONE: needed because owned element into collection.
643                        e.push(rec.clone());
644                    }
645                    push_deduped_vcf(&mut local_records, &mut local_seen, rec);
646                }
647                if let Some(e) = emitted.as_ref() {
648                    crate::semantic_trace::emit_vcf_emission(Some(&region), e);
649                }
650                for rec in crate::reference_vcf_emit::active_region_gvcf_reference_records(
651                    &region,
652                    &outcome,
653                    emit_mode,
654                    stand_emit_confidence,
655                    header,
656                    dict,
657                    ref_cache,
658                    &read_filters,
659                    ref_confidence_config,
660                    sample_name,
661                )? {
662                    push_deduped_vcf(&mut local_records, &mut local_seen, rec);
663                }
664            }
665        }
666    }
667
668    Ok(RegionEmitBatch {
669        region_index,
670        contig,
671        start,
672        records: local_records,
673    })
674}

676/// Sequential GVCF region processing (shared collector; not parallelized in v1).
677fn process_one_region_gvcf(
678    region: &AssemblyRegion,
679    header: &bam::HeaderView,
680    dict: &SequenceDictionary,
681    reference_fasta: &Path,
682    args: &CallRegionArgs,
683    read_filters: &ReadFilterParams,
684    stand_emit_confidence: f64,
685    emit_mode: EmitMode,
686    ref_confidence_config: &ReferenceConfidenceConfig,
687    ref_cache: &mut ReferenceWindowCache,
688    gvcf_collector: &mut GvcfIntervalCollector,
689    records: &mut Vec<VcfRecord>,
690    seen: &mut std::collections::BTreeSet<(String, u64, String, String)>,
691    sample_name: &str,
692) -> GatkResult<()> {
693    match call_disposition(region) {
694        AssemblyRegionCallDisposition::InactiveReferenceFastPath => {
695            let outcome = HaplotypeCallerEngine::call_region_inactive_reference(
696                region,
697                header,
698                dict,
699                reference_fasta,
700                emit_mode,
701                read_filters,
702                ref_confidence_config,
703            )?;
704            crate::semantic_trace::emit_inactive_rcm(region, outcome.loci.len());
705            gvcf_collector.ingest_loci(&outcome.region_contig, &outcome.loci);
706        }
707        AssemblyRegionCallDisposition::ActiveFull => {
708            let Some(outcome) =
709                HaplotypeCallerEngine::call_region(region, dict, reference_fasta, args)?
710            else {
711                let loci = reference_confidence_loci_for_active_call_none(
712                    region,
713                    header,
714                    ref_confidence_config,
715                    read_filters,
716                    ref_cache,
717                    dict,
718                )?;
719                gvcf_collector.ingest_loci(&region.contig, &loci);
720                return Ok(());
721            };
722            let mut emitted = crate::semantic_trace::is_enabled().then(Vec::new);
723            for rec in
724                try_emit_call_region_variants(region, &outcome, sample_name, stand_emit_confidence)?
725            {
726                gvcf_collector.add_variant_position(&rec.chromosome, rec.position);
727                if let Some(e) = emitted.as_mut() {
728                    // CLONE: needed because owned element into collection.
729                    e.push(rec.clone());
730                }
731                push_deduped_vcf(records, seen, rec);
732            }
733            if let Some(e) = emitted.as_ref() {
734                crate::semantic_trace::emit_vcf_emission(Some(region), e);
735            }
736            let loci = active_region_reference_confidence_loci(
737                region,
738                &outcome,
739                stand_emit_confidence,
740                header,
741                dict,
742                ref_cache,
743                read_filters,
744                ref_confidence_config,
745                crate::ref_confidence::ClusterRcmEvidenceMode::Production,
746            )?;
747            gvcf_collector.ingest_loci(&region.contig, &loci);
748        }
749    }
750    Ok(())
751}

762/// Deterministic merge: sort batches by genomic position / region_index, then global dedup.
763fn merge_region_emit_batches(
764    batches: &mut [RegionEmitBatch],
765    records: &mut Vec<VcfRecord>,
766    seen: &mut std::collections::BTreeSet<(String, u64, String, String)>,
767) {
768    batches.sort_by(|a, b| {
769        a.contig
770            .cmp(&b.contig)
771            .then(a.start.cmp(&b.start))
772            .then(a.region_index.cmp(&b.region_index))
773    });
774    for batch in batches.iter_mut() {
775        // Stable within-batch order; global first-wins via BTreeSet key.
776        for rec in std::mem::take(&mut batch.records) {
777            push_deduped_vcf(records, seen, rec);
778        }
779    }
780}
781
782fn push_deduped_vcf(
783    records: &mut Vec<VcfRecord>,
784    seen: &mut std::collections::BTreeSet<(String, u64, String, String)>,
785    rec: VcfRecord,
786) {
787    let alt = rec.alternate.first().cloned().unwrap_or_default();
788    // CLONE: needed because owned composite key for dedup/lookup.
789    let key = (
790        rec.chromosome.clone(),
791        rec.position,
792        rec.reference.clone(),
793        alt,
794    );
795    if seen.insert(key) {
796        records.push(rec);
797    }
798}
