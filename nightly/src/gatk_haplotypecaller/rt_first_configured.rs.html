1//! Production assemble shortcut: RT before_remove on configured k-mers before SeqGraph.
2
3use crate::assembly::AssemblyRead;
4use crate::cigar::{Cigar, CigarOperator};
5use crate::haplotype::Haplotype;
6use crate::read_threading_assembler::{
7    finalize_assembly_haplotypes, haplotypes_have_alt_bases, just_reference_result,
8    merge_rt_kbest_pre_remove_paths_at_kmer, AssemblyResult, AssemblyStatus,
9    ReadThreadingAssemblerArgs,
10};
11use gatk_common::GatkResult;
12
13/// Non-P12 production shortcut: RT before_remove on configured k-mers (largest first).
14///
15/// # Invariants
16/// Returns `Some` only with alt+ref haplotypes; `None` falls through to SeqGraph.
17/// Never runs on P12 / L-gate or tiny L2 k-mer sets (`k < 10`).
18/// On miss, marks each configured k-mer empty in [`crate::rt_region_cache`] so
19/// supplement/merge_rt skip re-building the same before_remove extracts.
20/// # Java equivalence
21/// Same before_remove RT extract as
22/// [`crate::read_threading_assembler::supplement_p12_cluster_coupled_haplotypes`], ordered
23/// to avoid a redundant SeqGraph prelude when that extract already yields alts.
24pub(crate) fn try_rt_configured_alts_before_seq_graph(
25    reference: &AssemblyRead,
26    reads: &[AssemblyRead],
27    args: &ReadThreadingAssemblerArgs,
28) -> GatkResult<Option<AssemblyResult>> {
29    // Production HC sets `scoring` under `strict_java_assembly`. Without it, SeqGraph +
30    // merge_rt is the sole alt source (supplement is a no-op) — do not short-circuit.
31    let Some(ctx) = args.scoring.as_ref() else {
32        return Ok(None);
33    };
34    if ctx.overlaps_p12_cluster() || ctx.overlaps_p12_l_gate_interval() {
35        return Ok(None);
36    }
37    let mut kmers: Vec<usize> = args.kmer_sizes.iter().copied().collect();
38    kmers.sort_unstable();
39    kmers.dedup();
40    let production_kmer_set = kmers.iter().any(|&k| k >= 25) && !kmers.iter().any(|&k| k < 10);
41    if !production_kmer_set {
42        return Ok(None);
43    }
44    // Dense ci-subset: early-stop is almost always at k=25 after a useless k=10 probe.
45    kmers.sort_unstable_by(|a, b| b.cmp(a));
46
47    crate::runtime_config::rss_trace_checkpoint(
48        "rt_first_configured_begin",
49        &format!("kmers={kmers:?}"),
50    );
51
52    let ref_bytes = reference.bases.as_slice();
53    let mut hit_kmer = kmers.first().copied().unwrap_or(25);
54    let mut haplotypes = just_reference_result(hit_kmer, reference).haplotypes;
55
56    for &kmer_size in &kmers {
57        if crate::runtime_config::hc_rss_abort_triggered() {
58            break;
59        }
60        crate::runtime_config::rss_trace_checkpoint(
61            "rt_first_configured_kmer",
62            &format!("kmer={kmer_size}"),
63        );
64        // Reuse merge_rt extract/dedup (owned-key clones stay in that helper's ratchet count).
65        merge_rt_kbest_pre_remove_paths_at_kmer(
66            reference,
67            reads,
68            args,
69            &[],
70            &mut haplotypes,
71            Some(kmer_size),
72        )?;
73        if !haplotypes.iter().any(|h| !h.is_reference) || haplotypes.len() <= 1 {
74            crate::rt_region_cache::mark_configured_kmer_empty(kmer_size);
75            continue;
76        }
77        // Defer indel CIGAR refresh to assemble_reads_with_finalized.
78        let mut ref_hap = Haplotype::new(ref_bytes, true);
79        let mut ref_cigar = Cigar::new();
80        ref_cigar.push(ref_hap.bases.len(), CigarOperator::Match);
81        ref_hap.cigar = Some(ref_cigar);
82        let status = finalize_assembly_haplotypes(
83            &mut haplotypes,
84            &ref_hap,
85            args.ensure_reference_in_result,
86        );
87        // `normalize_ref_equivalent_haplotypes` may collapse KBest `is_reference=false`
88        // paths that match ref bases — do not skip SeqGraph on a phantom hit (L2
89        // `g2-subset-live` p11: haplotype_count=1 vs Java 2).
90        if !matches!(status, AssemblyStatus::AssembledSomeVariation)
91            || !haplotypes_have_alt_bases(&haplotypes, ref_bytes)
92        {
93            crate::rt_region_cache::mark_configured_kmer_empty(kmer_size);
94            haplotypes = just_reference_result(kmer_size, reference).haplotypes;
95            continue;
96        }
97        hit_kmer = kmer_size;
98        let event_maps = Vec::new(); // production rebuilds EventMap from CIGARs later
99        crate::runtime_config::rss_trace_checkpoint(
100            "rt_first_configured_hit",
101            &format!("kmer={kmer_size} haps={}", haplotypes.len()),
102        );
103        return Ok(Some(AssemblyResult {
104            status,
105            kmer_size: hit_kmer,
106            haplotypes,
107            event_maps,
108        }));
109    }
110
111    crate::runtime_config::rss_trace_checkpoint(
112        "rt_first_configured_miss",
113        &format!("haps={}", haplotypes.len()),
114    );
115    Ok(None)
116}
