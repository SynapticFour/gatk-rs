22/// Aggregate per-read PairHMM rows (F.1 cases) into haplotype sums — G.1 gate.
23pub fn dump_genotyping_aggregate_tsv(
24    pairhmm_cases_path: &Path,
25    out: &mut impl Write,
26) -> GatkResult<()> {
27    let cases = load_pairhmm_cases_tsv(pairhmm_cases_path)?;
28    let params = PairHmmParams::default();
29    // IndexMap: first-seen read order (PairHMM case TSV order) — HashMap iteration
30    // would make aggregate dump / best-haplotype selection order-dependent.
31    let mut by_read: IndexMap<String, Vec<f64>> = IndexMap::new();
32    let mut haplotype_labels: Vec<String> = Vec::new();
33    for row in &cases {
34        let ll = pairhmm_log10_likelihood(
35            &PairHmmInput {
36                read_bases: row.read_bases.clone(),
37                read_base_quals: row.read_base_quals.clone(),
38                read_mapping_quality: row.read_mapq,
39                haplotype_bases: row.haplotype.clone(),
40            },
41            &params,
42        )?;
43        // CLONE: needed because owned HashMap entry key.
44        let entry = by_read.entry(row.read_bases.clone()).or_default();
45        if !haplotype_labels.contains(&row.haplotype) {
46            // CLONE: needed because owned element into collection.
47            haplotype_labels.push(row.haplotype.clone());
48        }
49        let hap_idx = haplotype_labels
50            .iter()
51            .position(|h| h == &row.haplotype)
52            .expect("hap label");
53        if entry.len() <= hap_idx {
54            entry.resize(hap_idx + 1, f64::NEG_INFINITY);
55        }
56        entry[hap_idx] = ll;
57    }
58
59    let rows: Vec<ReadLikelihoodRow> = by_read
60        .into_iter()
61        .enumerate()
62        .map(|(i, (read_id, haps))| ReadLikelihoodRow {
63            read_index: i,
64            read_id: format!("read_{i}_{read_id}"),
65            haplotype_log10_likelihoods: haps,
66        })
67        .collect();
68
69    let agg = aggregate_haplotype_log10_likelihoods(&rows)?;
70    let best = best_haplotype_index(&agg)
71        .unwrap_or(crate::bio_ids::HaplotypeIndex::new(0))
72        .get();
73
74    writeln!(out, "haplotype_count\t{}", agg.haplotype_log10_sums.len())
75        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
76    writeln!(out, "read_count\t{}", agg.read_count)
77        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
78    for (i, sum) in agg.haplotype_log10_sums.iter().enumerate() {
79        writeln!(out, "haplotype_{i}_log10_sum\t{sum}")
80            .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
81    }
82    writeln!(out, "best_haplotype_index\t{best}")
83        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
84    Ok(())
85}

87/// G.2.2 — PL/GQ/AD/DP from fixture genotype log10 likelihoods (p7 contract).
88pub fn dump_genotype_format_tsv(fixture_path: &Path, out: &mut impl Write) -> GatkResult<()> {
89    let raw = std::fs::read_to_string(fixture_path)
90        .map_err(|e| GatkError::io(format!("read fixture {}", fixture_path.display()), e))?;
91    writeln!(out, "# case_id\tpl\tgq\tad\tdp")
92        .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
93    for line in raw.lines() {
94        let t = line.trim();
95        if t.is_empty() || t.starts_with('#') {
96            continue;
97        }
98        let cols: Vec<&str> = t.split('\t').collect();
99        if cols.len() != 3 {
100            return Err(GatkError::argument(format!(
101                "genotype-format fixture row needs 3 cols: {t}"
102            )));
103        }
104        let case_id = cols[0];
105        let gls: Vec<f64> = cols[1]
106            .split(',')
107            .map(|s| s.parse::<f64>())
108            .collect::<Result<_, _>>()
109            .map_err(|e| GatkError::argument(format!("parse gl for {case_id}: {e}")))?;
110        let ads: Vec<i32> = cols[2]
111            .split(',')
112            .map(|s| s.parse::<i32>())
113            .collect::<Result<_, _>>()
114            .map_err(|e| GatkError::argument(format!("parse ad for {case_id}: {e}")))?;
115        let fields = emit_genotype_format_fields(&gls, &ads)?;
116        let pl = fields
117            .pl
118            .iter()
119            .map(|v| v.to_string())
120            .collect::<Vec<_>>()
121            .join(",");
122        let ad = fields
123            .ad
124            .iter()
125            .map(|v| v.to_string())
126            .collect::<Vec<_>>()
127            .join(",");
128        writeln!(out, "{case_id}\t{pl}\t{}\t{ad}\t{}", fields.gq, fields.dp)
129            .map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
130    }
131    Ok(())
132}

134/// G.2.3 — AF / EM posteriors from per-sample GL fixtures (`g2-af`).
135pub fn dump_af_em_tsv(fixture_path: &Path, out: &mut impl Write) -> GatkResult<()> {
136    let raw = std::fs::read_to_string(fixture_path)
137        .map_err(|e| GatkError::io(format!("read {}", fixture_path.display()), e))?;
138    for line in raw.lines() {
139        let t = line.trim();
140        if t.is_empty() || t.starts_with('#') {
141            continue;
142        }
143        let cols: Vec<&str> = t.split('\t').collect();
144        if cols.len() < 2 {
145            return Err(GatkError::argument(format!("g2-af row needs gl col: {t}")));
146        }
147        let case_id = cols[0];
148        let gls: Vec<f64> = cols[1]
149            .split(',')
150            .map(|s| s.parse())
151            .collect::<Result<_, _>>()
152            .map_err(|e| GatkError::argument(format!("parse gl: {e}")))?;
153        let gl_roundtrip =
154            crate::activity_scoring::genotype_log10_likelihoods_after_java_genotype_pl_roundtrip(
155                &gls,
156            );
157        let gl_ref = gl_roundtrip.as_slice();
158        let result = calculate_biallelic_af_em(&[gl_ref], &AfCalculatorConfig::default())?;
159        writeln!(out, "{case_id}\talt_ac\t{}", result.alt_allele_count)?;
160        writeln!(out, "{case_id}\taf\t{:.6}", result.af)?;
161        writeln!(
162            out,
163            "{case_id}\tlog10_p_no_variant\t{:.6}",
164            result.log10_posterior_no_variant
165        )?;
166        writeln!(out, "{case_id}\tem_iterations\t{}", result.em_iterations)?;
167    }
168    Ok(())
169}

184/// G.4.1 — physical phasing FORMAT fields (`g4`).
185pub fn dump_genotype_phasing_tsv(
186    alleles_csv: &str,
187    phasing_enabled: bool,
188    phase_set: Option<i32>,
189    out: &mut impl Write,
190) -> GatkResult<()> {
191    let alleles: Vec<i32> = alleles_csv
192        .split(',')
193        .map(|s| s.parse())
194        .collect::<Result<_, _>>()
195        .map_err(|e| GatkError::argument(format!("parse alleles: {e}")))?;
196    let fields = emit_genotype_phasing_fields(&alleles, phasing_enabled, phase_set)?;
197    writeln!(out, "gt\t{}", fields.gt)?;
198    writeln!(out, "phased\t{}", fields.phased)?;
199    writeln!(out, "pgt\t{}", fields.pgt.as_deref().unwrap_or("-"))?;
200    writeln!(out, "pid\t{}", fields.pid.as_deref().unwrap_or("-"))?;
201    writeln!(
202        out,
203        "ps\t{}",
204        fields
205            .ps
206            .map(|v| v.to_string())
207            .unwrap_or_else(|| "-".to_string())
208    )?;
209    Ok(())
210}

239/// G-D05 — allele subsetting indices for haplotype log10 sums fixture.
240pub fn dump_allele_subsetting_tsv(
241    haplotype_log10_sums_csv: &str,
242    is_reference_csv: &str,
243    max_allele_count: usize,
244    out: &mut impl Write,
245) -> GatkResult<()> {
246    use crate::genotyping::HaplotypeLikelihoodAggregation;
247    use crate::haplotype::Haplotype;
248    let sums: Vec<f64> = haplotype_log10_sums_csv
249        .split(',')
250        .map(|s| s.parse())
251        .collect::<Result<_, _>>()
252        .map_err(|e| GatkError::argument(format!("parse sums: {e}")))?;
253    let is_ref: Vec<bool> = is_reference_csv
254        .split(',')
255        .map(|s| s == "1" || s.eq_ignore_ascii_case("true"))
256        .collect();
257    let haps: Vec<Haplotype> = is_ref.iter().map(|&r| Haplotype::new(b"A", r)).collect();
258    let agg = HaplotypeLikelihoodAggregation {
259        haplotype_log10_sums: sums,
260        read_count: 1,
261    };
262    let kept = subset_alleles_for_genotyping(&haps, &agg, max_allele_count)?;
263    let (ref_i, alt_i) = biallelic_ref_alt_indices(&agg, &haps);
264    writeln!(out, "haplotype_count\t{}", haps.len())?;
265    writeln!(
266        out,
267        "kept_indices\t{}",
268        kept.iter()
269            .map(|i| i.to_string())
270            .collect::<Vec<_>>()
271            .join(",")
272    )?;
273    writeln!(out, "ref_haplotype_index\t{ref_i}")?;
274    writeln!(out, "alt_haplotype_index\t{alt_i}")?;
275    Ok(())
276}

278/// G-D05 — `AlleleSubsettingUtils.subsetAlleles` PL/AD (`g-subset-pl`).
279pub fn dump_subset_alleles_pl_tsv(fixture_path: &Path, out: &mut impl Write) -> GatkResult<()> {
280    let raw = std::fs::read_to_string(fixture_path)
281        .map_err(|e| GatkError::io(format!("read {}", fixture_path.display()), e))?;
282    for line in raw.lines() {
283        let t = line.trim();
284        if t.is_empty() || t.starts_with('#') {
285            continue;
286        }
287        let cols: Vec<&str> = t.split('\t').collect();
288        if cols.len() < 3 {
289            return Err(GatkError::argument(format!(
290                "g-subset-pl row needs 3 cols: {t}"
291            )));
292        }
293        let case_id = cols[0];
294        let log10_pl: Vec<f64> = cols[1]
295            .split(',')
296            .map(|s| s.parse())
297            .collect::<Result<_, _>>()
298            .map_err(|e| GatkError::argument(format!("parse pl: {e}")))?;
299        let ad: Vec<i32> = cols[2]
300            .split(',')
301            .map(|s| s.parse())
302            .collect::<Result<_, _>>()
303            .map_err(|e| GatkError::argument(format!("parse ad: {e}")))?;
304        let result = match case_id {
305            "trim_acg_to_ag" => subset_trim_acg_to_ag(&log10_pl, &ad)?,
306            _ => subset_trim_acg_to_ac(&log10_pl, &ad)?,
307        };
308        write_subset_vc_rows(case_id, &result, out)?;
309    }
310    Ok(())
311}

313fn write_subset_vc_rows(
314    case_id: &str,
315    result: &crate::allele_subsetting_pl::SubsetAllelesPlResult,
316    out: &mut impl Write,
317) -> GatkResult<()> {
318    writeln!(
319        out,
320        "{case_id}\tallele_count_before\t{}",
321        result.allele_count_before
322    )?;
323    writeln!(
324        out,
325        "{case_id}\tallele_count_after\t{}",
326        result.allele_count_after
327    )?;
328    writeln!(out, "{case_id}\tpl_length\t{}", result.pl.len())?;
329    for (i, pl) in result.pl.iter().enumerate() {
330        writeln!(out, "{case_id}\tpl_{i}\t{pl}")?;
331    }
332    writeln!(
333        out,
334        "{case_id}\tad\t{}",
335        result
336            .ad
337            .iter()
338            .map(|v| v.to_string())
339            .collect::<Vec<_>>()
340            .join(",")
341    )?;
342    if let Some(gq) = result.gq {
343        writeln!(out, "{case_id}\tgq\t{gq}")?;
344    }
345    if let Some(sac) = &result.sac {
346        writeln!(
347            out,
348            "{case_id}\tsac\t{}",
349            sac.iter()
350                .map(|v| v.to_string())
351                .collect::<Vec<_>>()
352                .join(",")
353        )?;
354    }
355    Ok(())
356}

358/// G-D05 — `AlleleSubsettingUtils.subsetAlleles` with SAC/GQ (`g-subset-vc`).
359pub fn dump_subset_alleles_vc_tsv(fixture_path: &Path, out: &mut impl Write) -> GatkResult<()> {
360    let raw = std::fs::read_to_string(fixture_path)
361        .map_err(|e| GatkError::io(format!("read {}", fixture_path.display()), e))?;
362    for line in raw.lines() {
363        let t = line.trim();
364        if t.is_empty() || t.starts_with('#') {
365            continue;
366        }
367        let cols: Vec<&str> = t.split('\t').collect();
368        if cols.len() < 4 {
369            return Err(GatkError::argument(format!(
370                "g-subset-vc row needs 4 cols: {t}"
371            )));
372        }
373        let case_id = cols[0];
374        let log10_pl: Vec<f64> = cols[1]
375            .split(',')
376            .map(|s| s.parse())
377            .collect::<Result<_, _>>()
378            .map_err(|e| GatkError::argument(format!("parse pl: {e}")))?;
379        let ad: Vec<i32> = cols[2]
380            .split(',')
381            .map(|s| s.parse())
382            .collect::<Result<_, _>>()
383            .map_err(|e| GatkError::argument(format!("parse ad: {e}")))?;
384        let sac: Vec<i32> = cols[3]
385            .split(',')
386            .map(|s| s.parse())
387            .collect::<Result<_, _>>()
388            .map_err(|e| GatkError::argument(format!("parse sac: {e}")))?;
389        let result = subset_het_ac_sac(&log10_pl, &ad, &sac)?;
390        write_subset_vc_rows(case_id, &result, out)?;
391    }
392    Ok(())
393}
