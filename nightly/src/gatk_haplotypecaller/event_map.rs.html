1//! GATK `Event` / `EventMap` — genomic events implied by haplotype vs reference.
2//! # Coordinate invariants (B5)
3//! [`Event::start`] is a **0-based offset into the padded reference haplotype bytes**, not a
4//! genome locus. Convert with the pad start before emit.
5//! [`VariationEvent`] carries **1-based inclusive** VCF coordinates; use
6//! [`VariationEvent::start_pos`] / [`VariationEvent::interval`] for typed access.

19/// One variant event on the reference haplotype byte axis (pad-relative).
20/// `start` is a [`PadOffset0`] into the reference haplotype sequence used to build the map.
21/// # Invariants
22/// `start` is **0-based** into padded reference haplotype bytes (not a genome locus).
23/// `ref_bases` / `alt_bases` are non-empty differing allele strings at that offset.
24/// # Ownership
25/// Owns allele byte vectors; collected into [`EventMap`].
26/// # Mutation
27/// Immutable once inserted; EventMap rebuild replaces events wholesale.
28/// # Biological assumptions
29/// Localized REF→ALT difference implied by haplotype CIGAR vs reference haplotype.
30/// # Java equivalence
31/// GATK `Event` inside `EventMap` (pad-relative coordinates).
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub struct Event {
34    pub start: PadOffset0,
35    pub ref_bases: Vec<u8>,
36    pub alt_bases: Vec<u8>,
37}
38
39/// Events for one haplotype relative to the reference haplotype.
40/// # Invariants
41/// Internal [`Event::start`] values are **0-based offsets on the reference haplotype bytes**, not genome loci.
42/// Events are ordered by `start` after construction/normalization.
43/// # Ownership
44/// Owns `events` vector; built from haplotype/reference pairs, consumed to emit [`VariationEvent`] lists.
45/// # Mutation
46/// Event lists may be rebuilt or replaced when haplotype CIGAR changes.
47/// # Biological assumptions
48/// Each event is a localized REF/ALT difference on the padded reference coordinate axis.
49/// # Java equivalence
50/// GATK `EventMap` / `Event` on haplotype vs reference (`AssemblyBasedCallerUtils`).
51#[derive(Debug, Clone, Default)]
52pub struct EventMap {
53    pub events: Vec<Event>,
54}

56/// VCF-ready variation event (sorted unique across haplotypes).
57/// # Invariants
58/// `start_1based` / `end_1based` are **1-based inclusive** VCF coordinates on `contig`.
59/// REF/ALT strings are display alleles after left-alignment for emit.
60/// # Ownership
61/// Owns contig and allele strings; referenced by genotyping and emit policies.
62/// # Mutation
63/// Treated as immutable site keys once emitted into genotype walks; normalization creates new values.
64/// # Biological assumptions
65/// Represents one alternate allele hypothesis at a genomic locus (SNP/MNP/indel).
66/// # Java equivalence
67/// GATK `Event` / `VariantContext` allele geometry at HC emit (converted from pad-relative events).
68#[derive(Debug, Clone, PartialEq, Eq)]
69pub struct VariationEvent {
70    pub contig: String,
71    pub start_1based: GenomePosition,
72    pub end_1based: GenomePosition,
73    pub ref_allele: String,
74    pub alt_allele: String,
75}
76
77/// Allele-length indel span `|len(REF) − len(ALT)|` (phenotype, not genomic end−start).
78/// L11-E2: consolidates magic `4` / `10` thresholds used by long-INS discovery and
79/// fragment-nest suppress.
80#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
81pub struct IndelSpan(usize);
82
83impl IndelSpan {
84    /// Max span treated as a short fragment nest (L10/L11 spray phenotype).
85    pub const SHORT_FRAGMENT_MAX: Self = Self(4);
86    /// Min span for production long-insertion discovery / long-allele nest filter.
87    pub const LONG_INSERTION_MIN: Self = Self(10);
88    /// Genomic window (bp) for fragment nests beside a long allele.
89    pub const FRAGMENT_WINDOW_BP: u64 = 60;
90
91    #[inline]
92    pub const fn new(span: usize) -> Self {
93        Self(span)
94    }
95
96    #[inline]
97    pub fn from_alleles(ref_allele: &str, alt_allele: &str) -> Self {
98        Self(ref_allele.len().abs_diff(alt_allele.len()))
99    }
100
101    #[inline]
102    pub fn from_event(event: &VariationEvent) -> Self {
103        Self::from_alleles(&event.ref_allele, &event.alt_allele)
104    }
105
106    #[inline]
107    pub const fn get(self) -> usize {
108        self.0
109    }
110
111    #[inline]
112    pub fn is_short_fragment(self) -> bool {
113        self.0 > 0 && self.0 <= Self::SHORT_FRAGMENT_MAX.get()
114    }
115
116    #[inline]
117    pub fn is_long_insertion_span(self) -> bool {
118        self.0 >= Self::LONG_INSERTION_MIN.get()
119    }
120
121    /// True when a short event at `short_pos` nests beside a long allele at `long_pos`.
122    #[inline]
123    pub fn nests_beside_long(short_pos: u64, long_pos: u64, long_span: Self) -> bool {
124        let win = Self::FRAGMENT_WINDOW_BP.max(long_span.get() as u64);
125        short_pos.abs_diff(long_pos) <= win
126    }
127
128    /// SNP in the upstream genomic flank of a long insertion (motif-bleed FP phenotype).
129    /// Window is **allele span only** (not [`Self::FRAGMENT_WINDOW_BP`]): holdout
130    /// `15001873`/`880` sit 14–21 bp upstream of +36 INS; a 60 bp SNP window would
131    /// falsely drop truth SNPs near unrelated long alleles (chr20 `10009795`/`871`).
132    #[inline]
133    pub fn snp_in_long_insertion_upstream_flank(
134        snp_pos: u64,
135        long_ins_pos: u64,
136        long_ins_span: Self,
137    ) -> bool {
138        snp_pos < long_ins_pos && long_ins_pos - snp_pos <= long_ins_span.get() as u64
139    }
140}

142impl VariationEvent {
143    pub fn is_indel(&self) -> bool {
144        self.ref_allele_bases().len() != self.alt_allele_bases().len()
145    }
146
147    /// Biallelic SNP (single-base REF and ALT).
148    pub fn is_snp(&self) -> bool {
149        self.ref_allele_bases().len() == 1 && self.alt_allele_bases().len() == 1
150    }
151
152    /// Indel allele-length span (0 for SNPs / equal-length alleles).
153    #[inline]
154    pub fn indel_span(&self) -> IndelSpan {
155        IndelSpan::from_event(self)
156    }
157
158    /// 1-based VCF start as a typed coordinate.
159    #[inline]
160    pub fn start_pos(&self) -> GenomePosition {
161        self.start_1based
162    }
163
164    /// 1-based inclusive VCF end as a typed coordinate.
165    #[inline]
166    pub fn end_pos(&self) -> GenomePosition {
167        self.end_1based
168    }
169
170    /// Inclusive genomic span on the contig (`start_1based..=end_1based`).
171    #[inline]
172    pub fn interval(&self) -> GenomeLoc {
173        GenomeLoc::try_new(self.start_1based, self.end_1based)
174            .expect("VariationEvent interval must satisfy end >= start")
175    }
176
177    /// REF allele bases (UTF-8 allele string as bytes).
178    #[inline]
179    pub fn ref_allele_bases(&self) -> &[u8] {
180        self.ref_allele.as_bytes()
181    }
182
183    /// ALT allele bases (UTF-8 allele string as bytes).
184    #[inline]
185    pub fn alt_allele_bases(&self) -> &[u8] {
186        self.alt_allele.as_bytes()
187    }
188}

190impl PartialOrd for VariationEvent {
191    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
192        Some(self.cmp(other))
193    }
194}

196impl Ord for VariationEvent {
197    fn cmp(&self, other: &Self) -> Ordering {
198        self.start_1based
199            .cmp(&other.start_1based)
200            .then_with(|| self.ref_allele.len().cmp(&other.ref_allele.len()))
201            .then_with(|| self.ref_allele.cmp(&other.ref_allele))
202            .then_with(|| self.alt_allele.cmp(&other.alt_allele))
203    }
204}

206fn is_regular_base(b: u8) -> bool {
207    matches!(b, b'A' | b'C' | b'G' | b'T' | b'a' | b'c' | b'g' | b't')
208}
209
210fn is_all_regular(bases: &[u8]) -> bool {
211    bases.iter().all(|&b| is_regular_base(b))
212}

214impl EventMap {
215    /// Build events from REF/ALT CIGAR walk (GATK `EventMap.processCigarForInitialEvents`).
216    pub fn from_haplotype_and_reference(
217        haplotype: &Haplotype,
218        reference: &Haplotype,
219        ref_bytes: &[u8],
220        ref_loc_start_1based: u64,
221        max_mnp_distance: usize,
222    ) -> Self {
223        let Some(cigar) = &haplotype.cigar else {
224            return Self::default();
225        };
226        let alignment = &haplotype.bases;
227        // GATK `EventMap.processCigarForInitialEvents`: start at haplotype offset in padded ref.
228        let mut ref_pos = haplotype.alignment_start_hap_wrt_ref;
229        if ref_pos >= ref_bytes.len() && !ref_bytes.is_empty() {
230            return Self::default();
231        }
232
233        let mut proposed = Vec::new();
234        let mut alignment_pos = 0usize;
235
236        for (cigar_index, el) in cigar.elements.iter().enumerate() {
237            let element_length = el.length;
238            match el.operator {
239                CigarOperator::Insertion => {
240                    if ref_pos > 0 && ref_pos <= ref_bytes.len() {
241                        let insertion_start = ref_loc_start_1based + ref_pos as u64 - 1;
242                        let ref_byte = ref_bytes[ref_pos - 1];
243                        let is_edge = cigar_index == 0 || cigar_index == cigar.elements.len() - 1;
244                        if is_regular_base(ref_byte) && !is_edge {
245                            let mut insertion_bases = vec![ref_byte];
246                            let end = alignment_pos
247                                .saturating_add(element_length)
248                                .min(alignment.len());
249                            insertion_bases.extend_from_slice(&alignment[alignment_pos..end]);
250                            if insertion_bases.len() >= 2 && is_all_regular(&insertion_bases) {
251                                let ref_allele = String::from_utf8(vec![ref_byte])
252                                    .unwrap_or_else(|_| "N".into());
253                                let alt_allele = String::from_utf8(insertion_bases)
254                                    .unwrap_or_else(|_| "N".into());
255                                proposed.push(VariationEvent {
256                                    contig: String::new(),
257                                    start_1based: GenomePosition::new_1based(insertion_start),
258                                    end_1based: GenomePosition::new_1based(insertion_start),
259                                    ref_allele,
260                                    alt_allele,
261                                });
262                            }
263                        }
264                    }
265                    alignment_pos += element_length;
266                }
267                CigarOperator::SoftClip => {
268                    alignment_pos += element_length;
269                }
270                CigarOperator::Deletion => {
271                    if ref_pos > 0 && ref_pos + element_length <= ref_bytes.len() {
272                        let mut deletion_bases = vec![ref_bytes[ref_pos - 1]];
273                        deletion_bases
274                            .extend_from_slice(&ref_bytes[ref_pos..ref_pos + element_length]);
275                        let deletion_start = ref_loc_start_1based + ref_pos as u64 - 1;
276                        let ref_byte = ref_bytes[ref_pos - 1];
277                        if is_regular_base(ref_byte) && is_all_regular(&deletion_bases) {
278                            let ref_allele =
279                                String::from_utf8(deletion_bases).unwrap_or_else(|_| "N".into());
280                            let alt_allele =
281                                String::from_utf8(vec![ref_byte]).unwrap_or_else(|_| "N".into());
282                            proposed.push(VariationEvent {
283                                contig: String::new(),
284                                start_1based: GenomePosition::new_1based(deletion_start),
285                                end_1based: GenomePosition::new_1based(
286                                    deletion_start + element_length as u64,
287                                ),
288                                ref_allele,
289                                alt_allele,
290                            });
291                        }
292                    }
293                    ref_pos += element_length;
294                }
295                CigarOperator::Match => {
296                    let mut mismatch_offsets = std::collections::VecDeque::new();
297                    for offset in 0..element_length {
298                        if ref_pos + offset >= ref_bytes.len() {
299                            break;
300                        }
301                        let ref_byte = ref_bytes[ref_pos + offset];
302                        let alt_byte = alignment
303                            .get(alignment_pos + offset)
304                            .copied()
305                            .unwrap_or(b'N');
306                        if ref_byte != alt_byte
307                            && is_regular_base(ref_byte)
308                            && is_regular_base(alt_byte)
309                        {
310                            mismatch_offsets.push_back(offset);
311                        }
312                    }
313                    while let Some(start) = mismatch_offsets.pop_front() {
314                        let mut end = start;
315                        while mismatch_offsets
316                            .front()
317                            .is_some_and(|next| *next - end <= max_mnp_distance)
318                        {
319                            end = mismatch_offsets.pop_front().unwrap_or(end);
320                        }
321                        let ref_allele =
322                            String::from_utf8(ref_bytes[ref_pos + start..=ref_pos + end].to_vec())
323                                .unwrap_or_else(|_| "N".into());
324                        let alt_allele = String::from_utf8(
325                            alignment[alignment_pos + start..=alignment_pos + end].to_vec(),
326                        )
327                        .unwrap_or_else(|_| "N".into());
328                        let start_pos = ref_loc_start_1based + ref_pos as u64 + start as u64;
329                        let end_pos = ref_loc_start_1based + ref_pos as u64 + end as u64;
330                        proposed.push(VariationEvent {
331                            contig: String::new(),
332                            start_1based: GenomePosition::new_1based(start_pos),
333                            end_1based: GenomePosition::new_1based(end_pos),
334                            ref_allele,
335                            alt_allele,
336                        });
337                    }
338                    ref_pos += element_length;
339                    alignment_pos += element_length;
340                }
341                _ => {
342                    ref_pos += element_length;
343                    alignment_pos += element_length;
344                }
345            }
346        }
347        let _ = reference;
348        Self {
349            events: proposed
350                .into_iter()
351                .map(|v| Event {
352                    start: PadOffset0::new(
353                        v.start_1based.get().saturating_sub(ref_loc_start_1based) as usize,
354                    ),
355                    ref_bases: v.ref_allele.into_bytes(),
356                    alt_bases: v.alt_allele.into_bytes(),
357                })
358                .collect(),
359        }
360    }
361
362    pub fn variation_events(&self, contig: &str, ref_loc_start_1based: u64) -> Vec<VariationEvent> {
363        self.events
364            .iter()
365            .map(|e| VariationEvent {
366                contig: contig.to_string(),
367                start_1based: GenomePosition::new_1based(
368                    ref_loc_start_1based + e.start.get() as u64,
369                ),
370                end_1based: GenomePosition::new_1based(
371                    ref_loc_start_1based
372                        + e.start.get() as u64
373                        + e.ref_bases.len().max(e.alt_bases.len()).saturating_sub(1) as u64,
374                ),
375                // CLONE: needed because multi-owner or ownership transfer into new structure.
376                ref_allele: String::from_utf8(e.ref_bases.clone()).unwrap_or_else(|_| "N".into()),
377                // CLONE: needed because owned allele string for seen-set / haplotype.
378                alt_allele: String::from_utf8(e.alt_bases.clone()).unwrap_or_else(|_| "N".into()),
379            })
380            .collect()
381    }

384/// Drop implausible mega-alleles from bad SW alignments.
385/// R4-2: raised from 4 → 40 so dense GIAB indels (Java max allele len often >4) are not
386/// discarded before genotyping. P12 cluster events remain short and still pass.
387pub const MAX_VARIATION_EVENT_ALLELE_LENGTH: usize = 40;
388
389fn cigar_has_indel(cigar: &Cigar) -> bool {
390    cigar.elements.iter().any(|e| e.operator.is_indel())
391}
392
393/// SOFTCLIP assembly cigars can be all-`M` with embedded mismatches; INDEL SW often recovers true indels (P12 92307327).
394fn supplemental_indel_variation_events(
395    haplotype: &Haplotype,
396    ref_hap: &Haplotype,
397    ref_bytes: &[u8],
398    ref_loc_start_1based: u64,
399    max_mnp_distance: usize,
400    contig: &str,
401) -> Vec<VariationEvent> {
402    if haplotype.is_reference {
403        return Vec::new();
404    }
405    let Some(cigar) = &haplotype.cigar else {
406        return Vec::new();
407    };
408    if cigar_has_indel(cigar) {
409        return Vec::new();
410    }
411    // Equal-length alts are SNP/MNP — Indel-strategy SW cannot add I/D events and was the
412    // dominant cost inside every `collect_variation_events` on dense NA12878.
413    if haplotype.bases.len() == ref_bytes.len() {
414        return Vec::new();
415    }
416    let sw = SwParameters::gatk_haplotype_to_reference();
417    let Some(indel_cigar) = calculate_haplotype_cigar_with_strategy(
418        ref_bytes,
419        &haplotype.bases,
420        &sw,
421        SwOverhangStrategy::Indel,
422    ) else {
423        return Vec::new();
424    };
425    if !cigar_has_indel(&indel_cigar) {
426        return Vec::new();
427    }
428    let mut alt = haplotype.clone();
429    alt.cigar = Some(indel_cigar);
430    alt.alignment_start_hap_wrt_ref =
431        effective_alignment_start_for_full_ref(haplotype, ref_hap, ref_loc_start_1based);
432    let map = EventMap::from_haplotype_and_reference(
433        &alt,
434        ref_hap,
435        ref_bytes,
436        ref_loc_start_1based,
437        max_mnp_distance,
438    );
439    map.variation_events(contig, ref_loc_start_1based)
440        .into_iter()
441        .filter(|v| {
442            v.ref_allele.len() <= MAX_VARIATION_EVENT_ALLELE_LENGTH
443                && v.alt_allele.len() <= MAX_VARIATION_EVENT_ALLELE_LENGTH
444        })
445        .collect()
446}

448/// When an indel and SNP share the same start, keep the indel (Java VCF alleles at P12 cluster).
449pub fn prefer_indel_over_colocated_snps(events: &mut Vec<VariationEvent>) {
450    let indel_starts: BTreeSet<GenomePosition> = events
451        .iter()
452        .filter(|e| e.is_indel())
453        .map(|e| e.start_1based)
454        .collect();
455    if indel_starts.is_empty() {
456        return;
457    }
458    events.retain(|e| {
459        e.is_indel()
460            || !indel_starts.contains(&e.start_1based)
461            || crate::read_event_discovery::is_p12_phase_e_gap_event(e)
462            || is_cluster_anchor_snp(e)
463    });
464}

466fn effective_alignment_start_for_full_ref(
467    haplotype: &Haplotype,
468    ref_hap: &Haplotype,
469    ref_loc_start_1based: u64,
470) -> usize {
471    let trim_offset = ref_hap
472        .genome_loc
473        .map(|g| g.start_1based().saturating_sub(ref_loc_start_1based) as usize)
474        .unwrap_or(0);
475    let a = haplotype.alignment_start_hap_wrt_ref;
476    if trim_offset > 0 && a < trim_offset {
477        trim_offset.saturating_add(a)
478    } else {
479        a
480    }
481}

483pub fn variation_events_for_haplotype(
484    haplotype: &Haplotype,
485    ref_hap: &Haplotype,
486    ref_bytes: &[u8],
487    ref_loc_start_1based: u64,
488    max_mnp_distance: usize,
489    contig: &str,
490) -> Vec<VariationEvent> {
491    if haplotype.cigar.is_none() {
492        return Vec::new();
493    }
494    let align_start =
495        effective_alignment_start_for_full_ref(haplotype, ref_hap, ref_loc_start_1based);
496    // Avoid cloning haplotype bases solely to lift trim→pad alignment start.
497    let map = if align_start == haplotype.alignment_start_hap_wrt_ref {
498        EventMap::from_haplotype_and_reference(
499            haplotype,
500            ref_hap,
501            ref_bytes,
502            ref_loc_start_1based,
503            max_mnp_distance,
504        )
505    } else {
506        let mut hap_for_map = haplotype.clone();
507        hap_for_map.alignment_start_hap_wrt_ref = align_start;
508        EventMap::from_haplotype_and_reference(
509            &hap_for_map,
510            ref_hap,
511            ref_bytes,
512            ref_loc_start_1based,
513            max_mnp_distance,
514        )
515    };
516    let mut events: Vec<VariationEvent> = map
517        .variation_events(contig, ref_loc_start_1based)
518        .into_iter()
519        .filter(|v| {
520            v.ref_allele.len() <= MAX_VARIATION_EVENT_ALLELE_LENGTH
521                && v.alt_allele.len() <= MAX_VARIATION_EVENT_ALLELE_LENGTH
522        })
523        .collect();
524    events.extend(supplemental_indel_variation_events(
525        haplotype,
526        ref_hap,
527        ref_bytes,
528        ref_loc_start_1based,
529        max_mnp_distance,
530        contig,
531    ));
532    prefer_indel_over_colocated_snps(&mut events);
533    events
534}

536/// GATK `AssemblyResultSet.getVariationEvents` — union of per-haplotype event maps.
537pub fn collect_variation_events(
538    haplotypes: &[Haplotype],
539    ref_bytes: &[u8],
540    ref_loc_start_1based: u64,
541    contig: &str,
542    max_mnp_distance: usize,
543) -> Vec<VariationEvent> {
544    let owned_ref;
545    let ref_hap = if let Some(h) = haplotypes.iter().find(|h| h.is_reference) {
546        h
547    } else {
548        owned_ref = Haplotype::new(ref_bytes, true);
549        &owned_ref
550    };
551    let mut set = BTreeSet::new();
552    for h in haplotypes {
553        for mut v in variation_events_for_haplotype(
554            h,
555            ref_hap,
556            ref_bytes,
557            ref_loc_start_1based,
558            max_mnp_distance,
559            contig,
560        ) {
561            v.contig = contig.to_string();
562            set.insert(v);
563        }
564    }
565    let mut out: Vec<VariationEvent> = set.into_iter().collect();
566    prefer_indel_over_colocated_snps(&mut out);
567    prefer_dominant_spanning_indels(&mut out);
568    out
569}

571/// GATK `EventMap.buildEventMapsForHaplotypes` — sorted 1-based start positions with variation.
572pub fn build_event_start_positions_1based(
573    haplotypes: &[Haplotype],
574    ref_bytes: &[u8],
575    ref_loc_start_1based: u64,
576    max_mnp_distance: usize,
577) -> BTreeSet<u64> {
578    let owned_ref;
579    let ref_hap = if let Some(h) = haplotypes.iter().find(|x| x.is_reference) {
580        h
581    } else {
582        owned_ref = Haplotype::new(ref_bytes, true);
583        &owned_ref
584    };
585    let mut positions = BTreeSet::new();
586    for h in haplotypes {
587        for v in variation_events_for_haplotype(
588            h,
589            ref_hap,
590            ref_bytes,
591            ref_loc_start_1based,
592            max_mnp_distance,
593            "",
594        ) {
595            positions.insert(v.start_1based.get());
596        }
597    }
598    positions
599}

601/// GATK `getVariantContextsFromActiveHaplotypes` → `VariationEvent` list.
602pub fn variation_events_at_position(
603    haplotypes: &[Haplotype],
604    ref_bytes: &[u8],
605    ref_loc_start_1based: u64,
606    loc_1based: u64,
607    include_spanning_events: bool,
608    max_mnp_distance: usize,
609    contig: &str,
610) -> Vec<VariationEvent> {
611    let loc = GenomePosition::new_1based(loc_1based);
612    let ref_hap = haplotypes
613        .iter()
614        .find(|h| h.is_reference)
615        .cloned()
616        .unwrap_or_else(|| Haplotype::new(ref_bytes, true));
617    let mut seen = BTreeSet::new();
618    let mut out = Vec::new();
619    for h in haplotypes {
620        for v in variation_events_for_haplotype(
621            h,
622            &ref_hap,
623            ref_bytes,
624            ref_loc_start_1based,
625            max_mnp_distance,
626            contig,
627        ) {
628            let overlaps = v.end_1based >= loc && v.start_1based <= loc;
629            if !overlaps {
630                continue;
631            }
632            if !include_spanning_events && v.start_1based != loc {
633                continue;
634            }
635            // CLONE: needed because owned composite key for dedup/lookup.
636            let key = (
637                v.start_1based.get(),
638                v.ref_allele.clone(),
639                v.alt_allele.clone(),
640            );
641            if seen.insert(key) {
642                out.push(v);
643            }
644        }
645    }
646    out
647}

649/// Remap a shorter-REF allele onto a longer REF that shares the same start pad
650/// (GATK `makeMergedVariantContext` longest-allele behavior for nested STR dels).
651/// Example: short `ATGTGTGTG>A` on long `ATGTGTGTGTGTGTGTGTG` → alt `ATGTGTGTGTG`.
652/// Typed REF/ALT pair for longest-REF remap and fragment predicates (L12-D3).
653#[derive(Debug, Clone, PartialEq, Eq, Hash)]
654pub struct AllelePair {
655    ref_allele: String,
656    alt_allele: String,
657}
658
659impl AllelePair {
660    #[inline]
661    pub fn new(ref_allele: impl Into<String>, alt_allele: impl Into<String>) -> Self {
662        Self {
663            ref_allele: ref_allele.into(),
664            alt_allele: alt_allele.into(),
665        }
666    }
667
668    #[inline]
669    pub fn from_event(event: &VariationEvent) -> Self {
670        Self::new(&event.ref_allele, &event.alt_allele)
671    }
672
673    #[inline]
674    pub fn ref_allele(&self) -> &str {
675        &self.ref_allele
676    }
677
678    #[inline]
679    pub fn alt_allele(&self) -> &str {
680        &self.alt_allele
681    }
682
683    /// Remap a shorter-REF allele onto `long_ref` (nested STR deletions).
684    pub fn remap_onto_longer_ref(&self, long_ref: &str) -> Option<Self> {
685        if self.ref_allele == long_ref {
686            // CLONE: needed because graph fork needs owned duplicate for speculative path.
687            return Some(self.clone());
688        }
689        if self.ref_allele.is_empty() || !long_ref.starts_with(&self.ref_allele) {
690            return None;
691        }
692        let mut out =
693            String::with_capacity(self.alt_allele.len() + long_ref.len() - self.ref_allele.len());
694        out.push_str(&self.alt_allele);
695        out.push_str(&long_ref[self.ref_allele.len()..]);
696        if out == long_ref || out.is_empty() {
697            return None;
698        }
699        Some(Self::new(long_ref, out))
700    }
701}
702
703pub fn remap_alt_onto_longer_ref(
704    short_ref: &str,
705    short_alt: &str,
706    long_ref: &str,
707) -> Option<String> {
708    AllelePair::new(short_ref, short_alt)
709        .remap_onto_longer_ref(long_ref)
710        .map(|p| p.alt_allele)
711}

713/// GATK `makeMergedVariantContext` lite: one biallelic site per unique ALT at `loc`.
714/// L10: shorter-REF colocated alleles are remapped onto the longest REF so nested
715/// STR deletions become a single multi-allelic site (holdout `20:15031984`).
716pub fn merged_biallelic_sites_at_position(
717    events: &[VariationEvent],
718    loc_1based: u64,
719) -> Vec<VariationEvent> {
720    let loc = GenomePosition::new_1based(loc_1based);
721    let at_loc: Vec<&VariationEvent> = events.iter().filter(|e| e.start_1based == loc).collect();
722    if at_loc.is_empty() {
723        return events
724            .iter()
725            .filter(|e| e.start_1based <= loc && e.end_1based >= loc)
726            .take(1)
727            .cloned()
728            .collect();
729    }
730    let ref_allele = at_loc
731        .iter()
732        .map(|e| e.ref_allele.as_str())
733        .max_by_key(|r| r.len())
734        .unwrap_or("N")
735        .to_string();
736    let mut alts = BTreeSet::new();
737    // CLONE: needed because multi-owner or ownership transfer into new structure.
738    let contig = at_loc.first().map(|e| e.contig.clone()).unwrap_or_default();
739    for e in &at_loc {
740        if e.ref_allele == ref_allele {
741            if e.alt_allele != ref_allele {
742                // CLONE: needed because owned HashMap/BTree/HashSet key or value.
743                alts.insert(e.alt_allele.clone());
744            }
745            continue;
746        }
747        if let Some(remapped) = AllelePair::from_event(e).remap_onto_longer_ref(&ref_allele) {
748            if remapped.alt_allele() != ref_allele {
749                alts.insert(remapped.alt_allele().to_string());
750            }
751        }
752    }
753    alts.into_iter()
754        .map(|alt| VariationEvent {
755            // CLONE: needed because owned contig id for output record.
756            contig: contig.clone(),
757            start_1based: loc,
758            end_1based: GenomePosition::new_1based(
759                loc.get()
760                    .saturating_add(ref_allele.len().saturating_sub(1) as u64),
761            ),
762            ref_allele: ref_allele.clone(),
763            alt_allele: alt,
764        })
765        .collect()
766}
767
768/// Drop short indels / SNPs nested inside a longer indel’s genomic span.
769/// Complements [`prefer_indel_over_colocated_snps`] (same-start only). Used after
770/// EventMap union so fragment alleles do not genotype beside a dominant long allele.
771/// Same-start shorter alleles are **kept** — they feed
772/// [`merged_biallelic_sites_at_position`] longest-REF remapping (STR multi-allelics).
773pub fn prefer_dominant_spanning_indels(events: &mut Vec<VariationEvent>) {
774    let long_indels: Vec<(GenomePosition, GenomePosition, usize)> = events
775        .iter()
776        .filter(|e| e.is_indel())
777        .map(|e| {
778            (
779                e.start_1based,
780                e.end_1based,
781                e.ref_allele.len().abs_diff(e.alt_allele.len()),
782            )
783        })
784        .filter(|(_, _, span)| *span >= 5)
785        .collect();
786    if long_indels.is_empty() {
787        return;
788    }
789    events.retain(|e| {
790        let e_span = e.ref_allele.len().abs_diff(e.alt_allele.len());
791        !long_indels.iter().any(|(ls, le, lspan)| {
792            // Keep colocated alleles at the long indel’s start (multi-allelic merge).
793            if e.start_1based == *ls {
794                return false;
795            }
796            // Nested fragment / SNP strictly inside the long allele's *reference* span.
797            // Insertions are zero-width on the reference (end == start); do not project
798            // alt-length onto the genome (would falsely nest distant true indels).
799            let ref_end = if le.get() > ls.get() { *le } else { *ls };
800            e.start_1based > *ls
801                && e.start_1based <= ref_end
802                && e_span < *lspan
803                && !(crate::read_event_discovery::is_p12_phase_e_gap_event(e)
804                    || is_cluster_anchor_snp(e))
805        })
806    });
807}
