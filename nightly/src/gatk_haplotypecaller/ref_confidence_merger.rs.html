6fn diploid_genotype_pairs(allele_count: usize) -> Vec<(usize, usize)> {
7    let mut pairs = Vec::new();
8    for j in 0..allele_count {
9        for i in 0..=j {
10            pairs.push((i, j));
11        }
12    }
13    pairs
14}

121fn allele_key(a: &MergeAllele) -> String {
122    if a.bases == NON_REF_ALLELE {
123        return NON_REF_ALLELE.to_string();
124    }
125    if a.bases == SPAN_DEL_ALLELE {
126        return SPAN_DEL_ALLELE.to_string();
127    }
128    a.bases.clone()
129}

131fn is_non_ref(a: &MergeAllele) -> bool {
132    a.bases == NON_REF_ALLELE
133}

135fn is_span_del(a: &MergeAllele) -> bool {
136    a.bases == SPAN_DEL_ALLELE
137}

139fn replace_with_no_calls_and_dels(vc: &MergeVcInput) -> Vec<MergeAllele> {
140    let mut out = Vec::with_capacity(vc.alleles.len());
141    out.push(MergeAllele {
142        bases: ".".to_string(),
143        is_reference: true,
144    });
145    for a in vc.alleles.iter().skip(1) {
146        if is_non_ref(a) {
147            // CLONE: needed because owned element into collection.
148            out.push(a.clone());
149        } else if a.bases.len() < vc.alleles[0].bases.len() {
150            out.push(MergeAllele {
151                bases: SPAN_DEL_ALLELE.to_string(),
152                is_reference: false,
153            });
154        } else {
155            out.push(MergeAllele {
156                bases: ".".to_string(),
157                is_reference: false,
158            });
159        }
160    }
161    out
162}

164fn extend_allele(allele: &MergeAllele, extra: usize, ref_bases: &[u8]) -> MergeAllele {
165    if extra == 0 || is_non_ref(allele) || is_span_del(allele) || allele.bases == "." {
166        return allele.clone();
167    }
168    let mut bytes = allele.bases.as_bytes().to_vec();
169    bytes.extend_from_slice(&ref_bases[ref_bases.len() - extra..]);
170    MergeAllele {
171        bases: String::from_utf8_lossy(&bytes).into_owned(),
172        is_reference: false,
173    }
174}

176fn remap_alleles(vc: &MergeVcInput, ref_allele: &MergeAllele) -> GatkResult<Vec<MergeAllele>> {
177    let vc_ref = &vc.alleles[0];
178    let ref_bases = ref_allele.bases.as_bytes();
179    let extra = ref_bases.len().saturating_sub(vc_ref.bases.len());
180    if ref_bases.len() < vc_ref.bases.len() {
181        return Err(GatkError::argument(
182            "remapAlleles: wrong reference selected",
183        ));
184    }
185    let mut out = vec![ref_allele.clone()];
186    for a in vc.alleles.iter().skip(1) {
187        if is_non_ref(a) || is_span_del(a) || a.bases == "." {
188            // CLONE: needed because owned element into collection.
189            out.push(a.clone());
190        } else {
191            out.push(extend_allele(a, extra, ref_bases));
192        }
193    }
194    Ok(out)
195}

197fn determine_reference_allele(
198    vcs: &[MergeVcInput],
199    loc_start: u64,
200    ref_base: Option<u8>,
201) -> GatkResult<Option<MergeAllele>> {
202    let mut ref_allele: Option<MergeAllele> = None;
203    for vc in vcs {
204        if vc.start != loc_start {
205            continue;
206        }
207        // CLONE: needed because MergeAllele is owned by the chosen-reference result.
208        let my_ref = vc.alleles[0].clone();
209        ref_allele = Some(match ref_allele {
210            None => my_ref,
211            Some(prev) => {
212                if prev.bases.len() < my_ref.bases.len() {
213                    my_ref
214                } else if my_ref.bases.len() < prev.bases.len() {
215                    prev
216                } else if prev.bases != my_ref.bases {
217                    return Err(GatkError::argument(format!(
218                        "inconsistent references at {loc_start}: {} vs {}",
219                        prev.bases, my_ref.bases
220                    )));
221                } else {
222                    prev
223                }
224            }
225        });
226    }
227    Ok(ref_allele.or_else(|| {
228        ref_base.map(|b| MergeAllele {
229            bases: String::from(b as char),
230            is_reference: true,
231        })
232    }))
233}

235fn collect_target_alleles(
236    pairs: &[RemappedVc],
237    ref_allele: &MergeAllele,
238    remove_non_ref: bool,
239) -> Vec<MergeAllele> {
240    let mut ordered: Vec<String> = vec![allele_key(ref_allele)];
241    let mut saw_span_del = false;
242    let mut saw_non_spanning = false;
243
244    for p in pairs {
245        for a in &p.remapped {
246            let key = allele_key(a);
247            if a.is_reference || is_non_ref(a) || a.bases == "." {
248                continue;
249            }
250            if is_span_del(a) {
251                saw_span_del = true;
252                continue;
253            }
254            saw_non_spanning = true;
255            if !ordered.contains(&key) {
256                ordered.push(key);
257            }
258        }
259        if p.is_spanning
260            && p.vc
261                .alleles
262                .iter()
263                .any(|a| !is_non_ref(a) && !a.is_reference)
264        {
265            saw_span_del = true;
266        }
267    }
268
269    let mut alleles: Vec<MergeAllele> = ordered
270        .iter()
271        .map(|k| {
272            if k == &allele_key(ref_allele) {
273                ref_allele.clone()
274            } else if k == SPAN_DEL_ALLELE {
275                MergeAllele {
276                    bases: SPAN_DEL_ALLELE.to_string(),
277                    is_reference: false,
278                }
279            } else {
280                MergeAllele {
281                    bases: k.clone(),
282                    is_reference: false,
283                }
284            }
285        })
286        .collect();
287
288    if saw_span_del && (saw_non_spanning || !remove_non_ref) {
289        alleles.push(MergeAllele {
290            bases: SPAN_DEL_ALLELE.to_string(),
291            is_reference: false,
292        });
293    }
294    if !remove_non_ref {
295        alleles.push(MergeAllele {
296            bases: NON_REF_ALLELE.to_string(),
297            is_reference: false,
298        });
299    }
300    alleles
301}

303/// GATK `AlleleSubsettingUtils.getIndexesOfRelevantAllelesForGVCF` (diploid, non-somatic).
304fn indexes_of_relevant_alleles_for_gvcf(
305    remapped: &[MergeAllele],
306    target: &[MergeAllele],
307) -> GatkResult<Vec<usize>> {
308    let non_ref_idx = remapped
309        .iter()
310        .position(is_non_ref)
311        .ok_or_else(|| GatkError::argument("remapped alleles must contain <NON_REF>"))?;
312    let mut map = vec![0usize; target.len()];
313    for (i, t) in target.iter().enumerate().skip(1) {
314        let key = allele_key(t);
315        let idx = remapped
316            .iter()
317            .position(|a| allele_key(a) == key)
318            .unwrap_or(non_ref_idx);
319        map[i] = idx;
320    }
321    Ok(map)
322}

324fn genotype_index_map_diploid(per_sample_map: &[usize], old_allele_count: usize) -> Vec<usize> {
325    let new_ac = per_sample_map.len();
326    let old_pairs = diploid_genotype_pairs(old_allele_count);
327    let new_pairs = diploid_genotype_pairs(new_ac);
328    let old_index: HashMap<(usize, usize), usize> = old_pairs
329        .iter()
330        .enumerate()
331        .map(|(idx, &(i, j))| ((i, j), idx))
332        .collect();
333    new_pairs
334        .iter()
335        .map(|&(ni, nj)| {
336            let oi = per_sample_map[ni];
337            let oj = per_sample_map[nj];
338            let (oi, oj) = if oi <= oj { (oi, oj) } else { (oj, oi) };
339            *old_index.get(&(oi, oj)).expect("genotype index")
340        })
341        .collect()
342}
343
344fn remap_pl(old_pl: &[i32], old_allele_count: usize, per_sample_map: &[usize]) -> Vec<i32> {
345    let map = genotype_index_map_diploid(per_sample_map, old_allele_count);
346    map.iter().map(|&i| old_pl[i]).collect()
347}

349/// GATK `AlleleSubsettingUtils.generateAD` / `remapRLengthList`.
350fn remap_ad(old_ad: &[i32], per_sample_map: &[usize]) -> Vec<i32> {
351    per_sample_map
352        .iter()
353        .map(|&old_index| {
354            if old_index >= old_ad.len() {
355                0
356            } else {
357                old_ad[old_index]
358            }
359        })
360        .collect()
361}

363fn merge_genotypes(
364    vc: &RemappedVc,
365    target: &[MergeAllele],
366    samples_are_uniquified: bool,
367) -> GatkResult<Vec<MergeGenotypeOut>> {
368    let mut out = Vec::new();
369    let per_sample = indexes_of_relevant_alleles_for_gvcf(&vc.remapped, target)?;
370    for g in &vc.vc.genotypes {
371        let name = if samples_are_uniquified {
372            format!("{}.{}", g.sample, vc.vc.source)
373        } else {
374            // CLONE: needed because owned sample id for carry/map.
375            g.sample.clone()
376        };
377        let pl =
378            g.pl.as_ref()
379                .map(|pl| remap_pl(pl, vc.remapped.len(), &per_sample));
380        let ad = g.ad.as_ref().map(|ad| remap_ad(ad, &per_sample));
381        out.push(MergeGenotypeOut { name, pl, ad });
382    }
383    Ok(out)
384}

386/// Production merge for gVCF records at one locus (H-D02).
387pub fn merge_reference_confidence(
388    contig: &str,
389    loc_start: u64,
390    vcs: &[MergeVcInput],
391    ref_base: Option<u8>,
392    remove_non_ref: bool,
393    samples_are_uniquified: bool,
394) -> GatkResult<Option<RefConfidenceMergeResult>> {
395    if vcs.is_empty() {
396        return Err(GatkError::argument("merge: empty inputs"));
397    }
398    let ref_allele = match determine_reference_allele(vcs, loc_start, ref_base)? {
399        Some(a) => a,
400        None => return Ok(None),
401    };
402
403    let mut pairs = Vec::with_capacity(vcs.len());
404    for vc in vcs {
405        let is_spanning = loc_start != vc.start;
406        let remapped = if is_spanning {
407            replace_with_no_calls_and_dels(vc)
408        } else {
409            remap_alleles(vc, &ref_allele)?
410        };
411        pairs.push(RemappedVc {
412            // CLONE: needed because owned variant/locus record in output map.
413            vc: vc.clone(),
414            remapped,
415            is_spanning,
416        });
417    }
418
419    let target = collect_target_alleles(&pairs, &ref_allele, remove_non_ref);
420    let mut genotypes = Vec::new();
421    for p in &pairs {
422        genotypes.extend(merge_genotypes(p, &target, samples_are_uniquified)?);
423    }
424
425    let alleles: Vec<String> = target.iter().map(|a| a.bases.clone()).collect();
426    Ok(Some(RefConfidenceMergeResult {
427        contig: contig.to_string(),
428        pos: loc_start,
429        alleles,
430        genotypes,
431        has_non_ref: target.iter().any(is_non_ref),
432    }))
433}

438// Diploid VCF PL order for allele indices 0..N-1 (htsjdk / GATK):
439// for j in 0..N { for i in 0..=j { genotype i/j } }
440// So for alleles [REF, A1, A2, <NON_REF>] (N=4) the 10 PLs are:
441// 0/0, 0/1, 1/1, 0/2, 1/2, 2/2, 0/3, 1/3, 2/3, 3/3
442// Remap contract (GATK `generatePL` + `getIndexesOfRelevantAllelesForGVCF`):
443// per_sample_map[new_allele] = old index of same allele, else old <NON_REF> index
444// new_PL[k] = old_PL[ old_genotype(map[a], map[b]) ] for new genotype a/b at slot k
445// No re-normalization after remap (min need not be recomputed — already 0 in source).

584    fn pl04_three_samples_three_alts_hand_indices() {
585        let a = snp_vc("A", "C", &[40, 0, 40, 40, 40, 40], &[4, 4, 0]);
586        let b = snp_vc("B", "G", &[50, 0, 50, 50, 50, 50], &[5, 5, 0]);
587        let c = snp_vc("C", "T", &[60, 0, 60, 60, 60, 60], &[6, 6, 0]);
588        // Java-style last→first discovery order among these three.
589        let merged = merge_reference_confidence("chr1", 10, &[c, b, a], None, false, false)
590            .unwrap()
591            .expect("merged");
592        assert_eq!(alts(&merged), vec!["T", "G", "C", NON_REF_ALLELE]);
593        let n = 1 + alts(&merged).len(); // 5 alleles → 15 PLs
594        assert_eq!(n * (n + 1) / 2, 15);
595
596        let pl_a = pl_of(&merged, "A");
597        let pl_b = pl_of(&merged, "B");
598        let pl_c = pl_of(&merged, "C");
599        assert_eq!(pl_a.len(), 15);
600        assert_eq!(pl_a.iter().position(|&x| x == 0), Some(6)); // 0/3
601        assert_eq!(pl_b.iter().position(|&x| x == 0), Some(3)); // 0/2
602        assert_eq!(pl_c.iter().position(|&x| x == 0), Some(1)); // 0/1
603                                                                // Non-zero cells stay at the constant filler from the source gVCF.
604        assert!(pl_a.iter().all(|&x| x == 0 || x == 40));
605        assert!(pl_b.iter().all(|&x| x == 0 || x == 50));
606        assert!(pl_c.iter().all(|&x| x == 0 || x == 60));
607    }
