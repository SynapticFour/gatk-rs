16pub(crate) const AFC_EMIT_EPSILON: f64 = 1e-9;
17
18pub(crate) fn qual_to_error_prob_log10(phred_threshold: f64) -> f64 {
19    -(phred_threshold / 10.0)
20}

22/// Minimum alt AD to emit hom-alt (cuts weak 1/1 rust-only).
23pub const MIN_HOM_ALT_AD_FOR_EMIT: AlleleDepth = AlleleDepth::new(2);

25/// Java sparse-BAM: single alt read sufficient at cluster anchor SNPs.
26/// `alt_ad` / `ref_ad` are non-negative pileup depths (VCF FORMAT AD wire values).
27pub fn passes_cluster_anchor_read_support(alt_ad: i32, ref_ad: i32) -> bool {
28    alt_ad >= 1 && alt_ad >= ref_ad
29}
30
31/// Read-style VCF emit: cluster anchors @ AD≥1; other SNPs need ≥2 alt reads (cuts rust-only).
32/// `alt_ad` / `ref_ad` are non-negative pileup depths (VCF FORMAT AD wire values).
33pub fn passes_read_style_sparse_emit(event: &VariationEvent, alt_ad: i32, ref_ad: i32) -> bool {
34    if event.ref_allele.len() != 1 || event.alt_allele.len() != 1 {
35        return false;
36    }
37    if is_cluster_anchor_snp(event) {
38        return passes_cluster_anchor_read_support(alt_ad, ref_ad);
39    }
40    alt_ad >= 2 && alt_ad >= ref_ad
41}

43/// GL vector fed to `AlleleFrequencyCalculator` after Java `GenotypeBuilder.PL` round-trip.
44pub fn gl_for_java_af_calculation(genotype_log10_likelihoods: &[f64]) -> Vec<f64> {
45    genotype_log10_likelihoods_after_java_genotype_pl_roundtrip(genotype_log10_likelihoods)
46}

48pub(crate) fn passes_hc_variant_emit_biallelic_inner(
49    genotype_log10_likelihoods: &[f64],
50    stand_emit_confidence: f64,
51) -> GatkResult<bool> {
52    if genotype_log10_likelihoods.len() < 3 {
53        return Ok(false);
54    }
55    let gl_java = gl_for_java_af_calculation(genotype_log10_likelihoods);
56    let af = calculate_biallelic_af_em(&[&gl_java], &AfCalculatorConfig::default())?;
57    let call_conf_log10 = qual_to_error_prob_log10(stand_emit_confidence);
58    let alt_plausible = af.log10_posterior_no_variant + AFC_EMIT_EPSILON < call_conf_log10;
59    let site_is_monomorphic = !alt_plausible;
60    let log10_vc_confidence = if !site_is_monomorphic {
61        af.log10_posterior_no_variant
62    } else {
63        log10_one_minus_pow10(af.log10_posterior_no_variant)
64    };
65    let phred_scaled = (-10.0 * log10_vc_confidence).max(0.0);
66    Ok(!site_is_monomorphic && phred_scaled >= stand_emit_confidence)
67}

69/// GATK `GenotypingEngine.passesEmitThreshold` for default `EMIT_VARIANTS_ONLY` (single sample).
70pub fn passes_hc_variant_emit_biallelic(
71    genotype_log10_likelihoods: &[f64],
72    stand_emit_confidence: f64,
73) -> GatkResult<bool> {
74    passes_hc_variant_emit_biallelic_inner(genotype_log10_likelihoods, stand_emit_confidence)
75}

77/// Java `GenotypingEngine.passesEmitThreshold`: emit when `!bestGuessIsRef` (after PL round-trip).
78pub fn passes_java_emit_not_hom_ref(
79    genotype_log10_likelihoods: &[f64],
80    format: &GenotypeFormatFields,
81) -> bool {
82    let gl_rt = gl_for_java_af_calculation(genotype_log10_likelihoods);
83    let ad = format.ad_as_i32();
84    best_biallelic_diploid_genotype_index(&gl_rt, &ad) != 0
85}

87/// Whether Java `calculateGenotypes` would return non-null (`passesEmitThreshold` on site AFC).
88/// **L14-D1:** coupled/CTC use [`is_coupled_indel_for_genotyping`]
89/// [`is_ctc_del_for_genotyping`] with `region_events` (phenotype when partners present;
90/// empty slice → absolute oracle for fixtures only).
91pub fn java_emit_would_pass(
92    event: &VariationEvent,
93    genotype_log10_likelihoods: &[f64],
94    format: &GenotypeFormatFields,
95    stand_emit_confidence: f64,
96    region_events: &[VariationEvent],
97) -> GatkResult<bool> {
98    if is_coupled_indel_for_genotyping(event, region_events)
99        || is_ctc_del_for_genotyping(event, region_events)
100    {
101        let best = biallelic_genotype_index_from_pl(&format.pl).as_usize();
102        return Ok(best != 0 || format.ad.get(1).copied().map(|d| d.as_i32()).unwrap_or(0) >= 1);
103    }
104    if !passes_java_emit_not_hom_ref(genotype_log10_likelihoods, format) {
105        return Ok(false);
106    }
107    let gl = gl_for_java_af_calculation(genotype_log10_likelihoods);
108    passes_hc_variant_emit_biallelic(&gl, stand_emit_confidence)
109}

111/// Genotyping + VCF emit confidence (biallelic diploid).
112/// **L14-D1:** coupled/CTC recognition threads `region_events` (same contract as
113/// [`java_emit_would_pass`]).
114pub fn passes_emit_for_variation_event(
115    event: &VariationEvent,
116    genotype_log10_likelihoods: &[f64],
117    format: &GenotypeFormatFields,
118    stand_emit_confidence: f64,
119    region_events: &[VariationEvent],
120) -> GatkResult<bool> {
121    if passes_hc_variant_emit_biallelic(genotype_log10_likelihoods, stand_emit_confidence)? {
122        return Ok(true);
123    }
124    // P12 cluster: emit coupled indels when genotype is not hom-ref (Java calls both sites).
125    if is_coupled_indel_for_genotyping(event, region_events)
126        || is_ctc_del_for_genotyping(event, region_events)
127    {
128        let best = biallelic_genotype_index_from_pl(&format.pl).as_usize();
129        if best != 0 {
130            return Ok(true);
131        }
132        let alt_ad = format.ad.get(1).copied().map(|d| d.as_i32()).unwrap_or(0);
133        if alt_ad >= 1 {
134            return Ok(true);
135        }
136    }
137    // Legacy N1 bridge only (parity uses Java AF threshold exclusively).
138    if is_cluster_anchor_snp(event) {
139        let ref_ad = format.ad.first().copied().map(|d| d.as_i32()).unwrap_or(0);
140        let alt_ad = format.ad.get(1).copied().map(|d| d.as_i32()).unwrap_or(0);
141        if passes_cluster_anchor_read_support(alt_ad, ref_ad) {
142            return Ok(true);
143        }
144    }
145    Ok(false)
146}
