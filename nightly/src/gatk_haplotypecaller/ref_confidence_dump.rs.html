29/// H.1.1 — per-locus RCM GL + GQ + DP + activity (C.4 GL walk + reference confidence fields).
30pub fn dump_reference_confidence_locus_tsv(
31    reference_fasta: &Path,
32    bam_path: &Path,
33    interval_cli: &str,
34    assembly_region_padding: u64,
35    out: &mut impl Write,
36    read_filters: &ReadFilterParams,
37) -> GatkResult<()> {
38    let config = ReferenceConfidenceConfig::default();
39    let scoring = &config.scoring;
40    let ploidy = scoring.sample_ploidy.as_u32();
41    let mut header_line = String::from("contig\tpos");
42    for i in 0..=ploidy {
43        header_line.push_str(&format!("\tgl{i}"));
44    }
45    header_line.push_str("\tgq\tdp\tactive_prob");
46    writeln!(out, "{header_line}").map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
47
48    let dict = SequenceDictionary::from_fasta_path(reference_fasta)?;
49    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
50    let mut ref_cache = ReferenceWindowCache::new(reference_fasta.to_path_buf(), 4);
51    let mut rng = crate::read_downsample::GatkJavaRng::reset_gatk_default();
52    let pipeline = ShardReadPipelineConfig::gatk_haplotype_caller_production();
53
54    for spec in &specs {
55        let (c, s, e) = spec
56            .resolve_closed_ends(&dict)
57            .map_err(|e| GatkError::argument(e.to_string()))?;
58        let shard = make_read_shards(&dict, std::slice::from_ref(spec), assembly_region_padding)?
59            .into_iter()
60            .find(|sh| sh.contig == c)
61            .ok_or_else(|| GatkError::argument(format!("no shard for {c}")))?;
62        let (header, mut records) =
63            crate::assembly_region_iterator::load_records_for_shard_raw(bam_path, &shard)?;
64        apply_shard_read_pipeline(
65            &mut records,
66            Some(&header),
67            read_filters,
68            &pipeline,
69            &mut rng,
70        )?;
71
72        let ref_window = ref_cache
73            .get_interval_bytes(&dict, &c, s, e)
74            .map_err(|e| GatkError::generic(e.to_string()))?;
75        let mut pileup_state = LocusPileupState::from_records(&records, &header, &c, read_filters);
76        for pos1 in IntervalLocusIterator::from_closed_interval(s, e) {
77            let ref_base = *ref_window
78                .get((pos1 - s) as usize)
79                .ok_or_else(|| GatkError::argument("reference window index out of range"))?;
80            let pile = pileup_state.pileup_at(&records, read_filters, pos1, ref_base)?;
81            let gl = capped_genotype_likelihoods_by_hom_ref(&pile, &config);
82            let active_prob =
83                calculate_single_sample_ref_vs_any_active_state_profile_value(&gl, scoring);
84            let detail = reference_confidence_locus_from_pileup(pos1 as usize, &pile, &config);
85            let mut row = format!("{c}\t{pos1}");
86            for g in &gl {
87                row.push('\t');
88                row.push_str(&format_activity_prob(*g));
89            }
90            row.push_str(&format!("\t{}\t{}", detail.locus.gq, detail.locus.dp));
91            row.push('\t');
92            row.push_str(&format_activity_prob(active_prob));
93            writeln!(out, "{row}").map_err(|e| GatkError::generic(format!("write tsv: {e}")))?;
94        }
95    }
96    Ok(())
97}

101/// Active `call_region` hybrid RCM loci (Java `getPileupsOverReference` path).
102pub fn dump_call_region_active_rcm_loci_tsv(
103    ref_fasta: &Path,
104    bam_path: &Path,
105    interval_cli: &str,
106    padding: u64,
107    out: &mut impl Write,
108) -> GatkResult<()> {
109    writeln!(out, "contig\tpos\tgq\tdp").map_err(|e| GatkError::generic(format!("write: {e}")))?;
110    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
111    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
112    let filters = ReadFilterParams::gatk_standard_hc();
113    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
114    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
115    let regions = flatten_assembly_regions(&walk);
116    let args = CallRegionArgs::default();
117    let config = ReferenceConfidenceConfig::default();
118    let header = rust_htslib::bam::Reader::from_path(bam_path)
119        .map_err(|e| GatkError::generic(format!("open bam: {e}")))?
120        .header()
121        .clone();
122    let mut ref_cache = ReferenceWindowCache::new(ref_fasta.to_path_buf(), 4);
123
124    for region in regions.iter().filter(|r| {
125        matches!(
126            call_disposition(r),
127            AssemblyRegionCallDisposition::ActiveFull
128        )
129    }) {
130        let Some(outcome) = HaplotypeCallerEngine::call_region(region, &dict, ref_fasta, &args)?
131        else {
132            continue;
133        };
134        let loci = active_region_reference_confidence_loci(
135            region,
136            &outcome,
137            DEFAULT_STAND_EMIT_CONFIDENCE,
138            &header,
139            &dict,
140            &mut ref_cache,
141            &filters,
142            &config,
143            crate::ref_confidence::ClusterRcmEvidenceMode::Production,
144        )?;
145        writeln!(
146            out,
147            "region\t{}\t{}\t{}\t{}",
148            region.contig,
149            region.start.get(),
150            region.end.get(),
151            loci.len()
152        )
153        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
154        for locus in &loci {
155            writeln!(
156                out,
157                "{}\t{}\t{}\t{}",
158                region.contig, locus.position_1based, locus.gq, locus.dp
159            )
160            .map_err(|e| GatkError::generic(format!("write: {e}")))?;
161        }
162        let blocks = build_gvcf_blocks_hc_emit(&loci, GATK_HC_DEFAULT_GQB)?;
163        writeln!(
164            out,
165            "block_header\tstart\tend\tmin_dp\tmax_dp\tgq_band\tmin_rgq"
166        )
167        .map_err(|e| GatkError::generic(format!("write: {e}")))?;
168        for block in &blocks {
169            let fields = gvcf_block_to_record_fields(block)?;
170            writeln!(
171                out,
172                "block\t{}\t{}\t{}\t{}\t{}\t{}",
173                fields.start_1based,
174                fields.end_info,
175                fields.min_dp,
176                fields.max_dp,
177                fields.gq_band_upper,
178                fields.min_rgq,
179            )
180            .map_err(|e| GatkError::generic(format!("write: {e}")))?;
181        }
182    }
183    Ok(())
184}

186/// H.1.2 — first inactive region `referenceModelForNoVariation` summary.
187pub fn dump_inactive_reference_model_tsv(
188    ref_fasta: &Path,
189    bam_path: &Path,
190    interval_cli: &str,
191    padding: u64,
192    emit_mode: EmitMode,
193    out: &mut impl Write,
194) -> GatkResult<()> {
195    let dict = SequenceDictionary::from_fasta_path(ref_fasta)?;
196    let specs = parse_intervals_cli_string(&dict, interval_cli)?;
197    let filters = ReadFilterParams::gatk_standard_hc();
198    let cfg = WalkerTraversalConfig::gatk_haplotype_caller_production(padding);
199    let walk = traverse_assembly_region_walker(&dict, &specs, ref_fasta, bam_path, &filters, &cfg)?;
200    let regions = flatten_assembly_regions(&walk);
201    let region = regions
202        .iter()
203        .find(|r| {
204            matches!(
205                call_disposition(r),
206                AssemblyRegionCallDisposition::InactiveReferenceFastPath
207            )
208        })
209        .ok_or_else(|| GatkError::argument("no inactive assembly region in interval"))?;
210
211    let shard = crate::walker::make_read_shards(&dict, &specs, padding)?
212        .into_iter()
213        .find(|s| s.contig == region.contig)
214        .ok_or_else(|| GatkError::argument("no shard for inactive region contig"))?;
215    let (header, _) =
216        crate::assembly_region_iterator::load_records_for_shard_raw(bam_path, &shard)?;
217
218    let mut ref_cache = ReferenceWindowCache::new(ref_fasta.to_path_buf(), 4);
219    let config = ReferenceConfidenceConfig::default();
220    let outcome = reference_model_for_no_variation_region(
221        region,
222        &header,
223        &config,
224        &filters,
225        &mut ref_cache,
226        &dict,
227        emit_mode,
228    )?;
229
230    writeln!(out, "region_contig\t{}", outcome.region_contig)?;
231    writeln!(out, "region_start\t{}", outcome.region_start)?;
232    writeln!(out, "region_end\t{}", outcome.region_end)?;
233    writeln!(out, "is_active\tfalse")?;
234    writeln!(out, "path\treferenceModelForNoVariation")?;
235    let emit_label = match outcome.emit_mode {
236        EmitMode::Gvcf => "GVCF",
237        EmitMode::Vcf => "VCF",
238        EmitMode::BpResolution => "BP_RESOLUTION",
239    };
240    writeln!(out, "emit_mode\t{emit_label}")?;
241    writeln!(out, "locus_count\t{}", outcome.loci.len())?;
242    writeln!(
243        out,
244        "reference_blocks_emitted\t{}",
245        outcome.summary.reference_blocks_emitted
246    )?;
247    writeln!(
248        out,
249        "reference_sites_emitted\t{}",
250        outcome.summary.reference_sites_emitted
251    )?;
252    if let Some(first) = outcome.loci.first() {
253        writeln!(out, "first_gq\t{}", first.gq)?;
254        writeln!(out, "first_dp\t{}", first.dp)?;
255    }
256    if let Some(last) = outcome.loci.last() {
257        writeln!(out, "last_gq\t{}", last.gq)?;
258        writeln!(out, "last_dp\t{}", last.dp)?;
259    }
260    if outcome.emit_mode == EmitMode::Gvcf && !outcome.loci.is_empty() {
261        let blocks = build_gvcf_blocks_hc_emit(&outcome.loci, GATK_HC_DEFAULT_GQB)?;
262        writeln!(out, "gvcf_block_count\t{}", blocks.len())?;
263        writeln!(out, "locus_header\tpos\tgq\tdp")?;
264        for locus in &outcome.loci {
265            writeln!(
266                out,
267                "locus\t{}\t{}\t{}",
268                locus.position_1based, locus.gq, locus.dp
269            )?;
270        }
271        writeln!(
272            out,
273            "block_header\tstart\tend\tmin_dp\tmax_dp\tgq_band\tmin_rgq"
274        )?;
275        for block in &blocks {
276            let fields = gvcf_block_to_record_fields(block)?;
277            writeln!(
278                out,
279                "block\t{}\t{}\t{}\t{}\t{}\t{}",
280                fields.start_1based,
281                fields.end_info,
282                fields.min_dp,
283                fields.max_dp,
284                fields.gq_band_upper,
285                fields.min_rgq,
286            )?;
287        }
288    }
289    Ok(())
