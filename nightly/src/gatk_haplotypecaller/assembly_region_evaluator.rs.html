19/// Shared HC activity path used by the iterator and TSV dump.
20pub fn evaluate_hc_activity_state(
21    contig: &str,
22    pos1: u64,
23    pile: &[PileupObservation],
24    scoring: &HaplotypeCallerActivityScoringParams,
25) -> ActivityProfileState {
26    let mut pile = pile.to_vec();
27    if scoring.contamination_fraction_to_filter > 0.0 {
28        let mut rng = Well19937c::reset_gatk_default();
29        apply_contamination_to_pileup(
30            &mut pile,
31            scoring.contamination_fraction_to_filter,
32            &mut rng,
33        );
34    }
35    haplotype_caller_activity_profile_state_single_sample(contig, pos1, &pile, scoring)
36}

38/// Feed one 1-based locus into the band-pass profile (iterator + activity TSV dumps).
39/// When `pileup_state` is `Some`, uses LIBS-style incremental pileup (GAP-B-02); otherwise scans all records.
40/// `force_active` mirrors GATK `--alleles` force-calling overlap at this locus.
41pub fn add_locus_for_smoothed_activity(
42    prof: &mut BandPassActivityProfile,
43    records: &[bam::Record],
44    header: &bam::HeaderView,
45    header_semantics: &ReadHeaderSemantics,
46    contig: &str,
47    pos1: u64,
48    read_filters: &ReadFilterParams,
49    ref_base: u8,
50    scoring: &HaplotypeCallerActivityScoringParams,
51    pileup_state: Option<&mut LocusPileupState>,
52    force_active: bool,
53) -> GatkResult<()> {
54    let st = match pileup_state {
55        Some(st) => {
56            st.advance_to(records, read_filters, pos1)?;
57            hc_activity_after_locus_advance(
58                contig,
59                pos1,
60                st,
61                records,
62                header_semantics,
63                scoring,
64                ref_base,
65                force_active,
66            )?
67        }
68        None => {
69            let pile = pileup_at_locus(records, header, contig, pos1, read_filters, ref_base)?;
70            evaluate_hc_activity_state(contig, pos1, &pile, scoring)
71        }
72    };
73
74    prof.add(st)?;
75    Ok(())
76}

78/// Incremental pileup → activity state (LIBS path). `force_active` sets active_prob=1.0.
79pub(crate) fn hc_activity_after_locus_advance(
80    contig: &str,
81    pos1: u64,
82    pileup_state: &mut LocusPileupState,
83    records: &[bam::Record],
84    header_semantics: &ReadHeaderSemantics,
85    scoring: &HaplotypeCallerActivityScoringParams,
86    ref_base: u8,
87    force_active: bool,
88) -> GatkResult<ActivityProfileState> {
89    if force_active {
90        return Ok(ActivityProfileState {
91            contig: std::sync::Arc::from(contig),
92            pos: pos1,
93            active_prob: 1.0,
94            original_active_prob: 1.0,
95            evidence: crate::activity_profile::ActivityEvidence::Plain,
96        });
97    }
98    let mut pile = pileup_state.pileup_observations(records, ref_base)?;
99    if pile.is_empty() {
100        return Ok(ActivityProfileState::new(contig, pos1, 0.0));
101    }
102    if scoring.contamination_fraction_to_filter > 0.0 {
103        let mut rng = Well19937c::reset_gatk_default();
104        apply_contamination_to_pileup(
105            &mut pile,
106            scoring.contamination_fraction_to_filter,
107            &mut rng,
108        );
109    }
110
111    // R4-1: single-sample HC never uses stratified piles — skip the second observation walk.
112    if header_semantics.is_single_sample_header() {
113        return Ok(haplotype_caller_activity_profile_state_minimal_genotyping(
114            contig, pos1, &pile, scoring,
115        ));
116    }
117
118    let mut stratified = pileup_state.nonempty_stratified_sample_pileups_ordered(
119        records,
120        header_semantics,
121        ref_base,
122    )?;
123    if scoring.contamination_fraction_to_filter > 0.0 {
124        let mut rng = Well19937c::reset_gatk_default();
125        for sample_pile in &mut stratified {
126            apply_contamination_to_pileup(
127                sample_pile,
128                scoring.contamination_fraction_to_filter,
129                &mut rng,
130            );
131        }
132    }
133
134    Ok(if stratified.len() <= 1 {
135        haplotype_caller_activity_profile_state_minimal_genotyping(contig, pos1, &pile, scoring)
136    } else {
137        {
138            let strata: Vec<&[_]> = stratified.iter().map(|p| p.as_slice()).collect();
139            crate::hc_joint_is_active::haplotype_caller_joint_multisample_is_active_activity_state(
140                contig, pos1, &strata, scoring,
141            )
142        }
143    })
144}

146fn pileup_at_locus(
147    records: &[bam::Record],
148    header: &bam::HeaderView,
149    contig: &str,
150    pos1: u64,
151    filters: &ReadFilterParams,
152    ref_base: u8,
153) -> GatkResult<Vec<PileupObservation>> {
154    let ref_pos0 = pos1.saturating_sub(1) as i64;
155    let mut obs = Vec::new();
156
157    for rec in records {
158        if !passes_hc_read_filters_with_header(rec, header, filters) {
159            continue;
160        }
161        let rn = String::from_utf8_lossy(header.tid2name(rec.tid() as u32)).into_owned();
162        if rn != contig {
163            continue;
164        }
165        if let Some(o) = pileup_observation_from_record(rec, ref_pos0, ref_base, false)? {
166            obs.push(o);
167        }
168    }
169
170    Ok(obs)
171}
