15/// Assembled haplotype with optional CIGAR vs padded reference.
16/// # Invariants
17/// Exactly one haplotype should be marked `is_reference` before genotyping filters run.
18/// Non-reference haplotypes sort by descending `score` (base-sequence tie-break), reference last.
19/// When present, `genome_loc` uses 1-based inclusive coordinates ([`GenomeLoc`]).
20/// # Ownership
21/// Owns base bytes and optional [`Cigar`]; borrowed by EventMap and PairHMM stages.
22/// # Mutation
23/// Assembly/finalize paths mutate `cigar`, `genome_loc`, and `alignment_start_hap_wrt_ref`; trim produces new instances.
24/// # Biological assumptions
25/// Bases are haplotype sequence over the padded active region; CIGAR is vs reference haplotype.
26/// # Java equivalence
27/// GATK `Haplotype` (assembly result, CIGAR, `alignmentStartHapwrtRef`, trim).
28#[derive(Debug, Clone, PartialEq)]
29pub struct Haplotype {
30    pub bases: Vec<u8>,
31    pub is_reference: bool,
32    pub score: f64,
33    pub kmer_size: usize,
34    pub cigar: Option<Cigar>,
35    /// GATK `genomeLocation` (required for [`Self::trim`]).
36    pub genome_loc: Option<GenomeLoc>,
37    /// GATK `alignmentStartHapwrtRef` (offset of haplotype vs reference haplotype).
38    pub alignment_start_hap_wrt_ref: usize,
39}
40
41impl Haplotype {
42    pub fn new(bases: impl Into<Vec<u8>>, is_reference: bool) -> Self {
43        Self {
44            bases: bases.into(),
45            is_reference,
46            score: 0.0,
47            kmer_size: 0,
48            cigar: None,
49            genome_loc: None,
50            alignment_start_hap_wrt_ref: 0,
51        }
52    }
53
54    pub fn sequence_string(&self) -> String {
55        String::from_utf8_lossy(&self.bases).into_owned()
56    }
57
58    /// Tag all haplotypes with the padded reference span (GATK `activeRegionExtendedLocation`).
59    pub fn tag_padded_reference_span(haplotypes: &mut [Haplotype], pad_start_1based: u64) {
60        let ref_len = haplotypes
61            .iter()
62            .find(|h| h.is_reference)
63            .and_then(|h| h.cigar.as_ref().map(|c| c.reference_length()))
64            .or_else(|| {
65                haplotypes
66                    .iter()
67                    .find(|h| h.is_reference)
68                    .map(|h| h.bases.len())
69            })
70            .unwrap_or(0);
71        let end = pad_start_1based
72            .saturating_add(ref_len as u64)
73            .saturating_sub(1);
74        let loc = GenomeLoc::new(pad_start_1based, end);
75        for h in haplotypes {
76            h.genome_loc = Some(loc);
77            if h.alignment_start_hap_wrt_ref == 0 && !h.is_reference {
78                h.alignment_start_hap_wrt_ref = 0;
79            }
80        }
81    }
82
83    /// GATK `Haplotype.trim(loc, ignoreRefState)`.
84    pub fn trim(&self, loc: &GenomeLoc, ignore_ref_state: bool) -> Option<Haplotype> {
85        let genome = self.genome_loc?;
86        if !genome.contains(loc) {
87            return None;
88        }
89        let cigar = self.cigar.as_ref()?;
90        let new_start = loc.start_1based().saturating_sub(genome.start_1based()) as usize;
91        let new_stop = new_start + loc.reference_span_length().saturating_sub(1) as usize;
92        let new_bases =
93            get_bases_covering_ref_interval(new_start, new_stop, &self.bases, 0, cigar)?;
94        if new_bases.is_empty() {
95            return None;
96        }
97        let trimmed = trim_cigar_by_reference(cigar, new_start, new_stop);
98        let new_cigar = trimmed.cigar;
99        let leading_insertion = new_cigar
100            .elements
101            .first()
102            .is_some_and(|e| !e.operator.consumes_reference_bases());
103        let trailing_insertion = new_cigar
104            .elements
105            .last()
106            .is_some_and(|e| !e.operator.consumes_reference_bases());
107        let first_keep = if leading_insertion { 1 } else { 0 };
108        let last_keep_exclusive = new_cigar
109            .elements
110            .len()
111            .saturating_sub(if trailing_insertion { 1 } else { 0 });
112        if last_keep_exclusive <= first_keep {
113            return None;
114        }
115        let final_cigar = if leading_insertion || trailing_insertion {
116            let mut b = CigarBuilder::new(false);
117            for e in &new_cigar.elements[first_keep..last_keep_exclusive] {
118                b.add(CigarElement {
119                    length: e.length,
120                    operator: e.operator,
121                });
122            }
123            b.make_and_record().cigar
124        } else {
125            new_cigar
126        };
127        let mut ret = Haplotype::new(new_bases, !ignore_ref_state && self.is_reference);
128        ret.cigar = Some(final_cigar);
129        ret.genome_loc = Some(*loc);
130        ret.score = self.score;
131        ret.kmer_size = self.kmer_size;
132        ret.alignment_start_hap_wrt_ref = new_start + self.alignment_start_hap_wrt_ref;
133        Some(ret)
134    }
135}

137/// Sort key matching GATK `Haplotype.SIZE_AND_BASE_ORDER`.
138pub fn haplotype_size_and_base_order(a: &Haplotype, b: &Haplotype) -> std::cmp::Ordering {
139    a.bases
140        .len()
141        .cmp(&b.bases.len())
142        .then_with(|| a.bases.cmp(&b.bases))
143}

145/// Collapse base-identical haplotypes before `variationPresent` / assembly status (GATK ref path).
146/// KBest can label a ref-spine path `is_reference=false` while bases match the reference haplotype;
147/// Java does not treat that as a distinct alt allele.
148pub fn normalize_ref_equivalent_haplotypes(haplotypes: &mut Vec<Haplotype>, ref_bytes: &[u8]) {
149    for h in haplotypes.iter_mut() {
150        if h.bases.as_slice() == ref_bytes {
151            h.is_reference = true;
152        }
153    }
154    // Preserve assembly discovery order (Java `getHaplotypeList`) for parity genotype indexing.
155    let mut out: Vec<Haplotype> = Vec::with_capacity(haplotypes.len());
156    let mut index_by_bases: std::collections::HashMap<Vec<u8>, usize> =
157        std::collections::HashMap::new();
158    for h in haplotypes.drain(..) {
159        // Lifetime: `h` is owned for this iteration; look up by `&h.bases` so duplicates
160        // never allocate a key. Clone bases only when inserting a new HashMap entry that
161        // must outlive `h` after `out.push(h)` moves the haplotype.
162        let is_ref = h.is_reference;
163        let score = h.score;
164        if let Some(&idx) = index_by_bases.get(&h.bases) {
165            if is_ref {
166                out[idx] = h;
167            } else if !out[idx].is_reference && score > out[idx].score {
168                out[idx] = h;
169            }
170        } else {
171            let idx = out.len();
172            // CLONE: needed because owned HashMap/BTree/HashSet key or value.
173            index_by_bases.insert(h.bases.clone(), idx);
174            out.push(h);
175        }
176    }
177    *haplotypes = out;
178}

180/// Collapse alt haplotypes that share a prefix and differ only in the last 8 bp (dangling tail recovery).
181pub fn collapse_dangling_tail_alt_duplicates(haplotypes: &mut Vec<Haplotype>, ref_hap: &Haplotype) {
182    let n = ref_hap.bases.len();
183    if n < 8 {
184        return;
185    }
186    let tail_len = 8;
187    let mut best_by_prefix: std::collections::HashMap<Vec<u8>, usize> =
188        std::collections::HashMap::new();
189    let mut keep = vec![true; haplotypes.len()];
190    for (i, h) in haplotypes.iter().enumerate() {
191        if h.is_reference || h.bases.len() != n {
192            continue;
193        }
194        let prefix = h.bases[..n - tail_len].to_vec();
195        if let Some(&prev) = best_by_prefix.get(&prefix) {
196            let ref_tail = &ref_hap.bases[n - tail_len..];
197            let prev_tail = &haplotypes[prev].bases[n - tail_len..];
198            let cur_tail = &h.bases[n - tail_len..];
199            let prefer = if prev_tail == ref_tail && cur_tail != ref_tail {
200                prev
201            } else if cur_tail == ref_tail && prev_tail != ref_tail {
202                i
203            } else if haplotypes[prev].score >= h.score {
204                prev
205            } else {
206                i
207            };
208            if prefer == prev {
209                keep[i] = false;
210            } else {
211                keep[prev] = false;
212                best_by_prefix.insert(prefix, i);
213            }
214        } else {
215            best_by_prefix.insert(prefix, i);
216        }
217    }
218    let ref_tail = &ref_hap.bases[n - tail_len..];
219    let mut out = Vec::with_capacity(haplotypes.len());
220    for (i, h) in haplotypes.drain(..).enumerate() {
221        if !keep[i] {
222            continue;
223        }
224        // Dangling recovery can emit a ref-spine path with only a tail mismatch (e.g. ATGT vs ACGT).
225        if !h.is_reference
226            && h.bases.len() == n
227            && h.bases[..n - tail_len] == ref_hap.bases[..n - tail_len]
228            && h.bases[n - tail_len..] != *ref_tail
229        {
230            continue;
231        }
232        out.push(h);
233    }
234    *haplotypes = out;
235}

237/// GATK assembly list order: non-reference by descending score, reference last.
238/// Invariant: after sort, if any `is_reference` haplotype exists, it is at the end;
239/// preceding entries are ordered by `score` desc, then `bases` asc.
240pub fn sort_haplotypes_assembly_result_order(haplotypes: &mut Vec<Haplotype>) {
241    haplotypes.sort_by(|a, b| match (a.is_reference, b.is_reference) {
242        (true, false) => std::cmp::Ordering::Greater,
243        (false, true) => std::cmp::Ordering::Less,
244        _ => b
245            .score
246            .partial_cmp(&a.score)
247            .unwrap_or(std::cmp::Ordering::Equal)
248            .then_with(|| a.bases.cmp(&b.bases)),
249    });
250}

252/// Drop non-reference k-best fragments (GATK `findBestPaths` / full-length alt paths only).
253pub fn prune_fragment_non_reference_haplotypes(
254    haplotypes: &mut Vec<Haplotype>,
255    ref_hap: &Haplotype,
256    min_bases: usize,
257) {
258    let ref_len = ref_hap.bases.len();
259    if ref_len == 0 {
260        return;
261    }
262    let min_alt_len = ref_len.saturating_mul(3).saturating_div(4).max(min_bases);
263    haplotypes.retain(|h| h.is_reference || h.bases.len() >= min_alt_len);
264}

266fn hash_hap_bases(bases: &[u8]) -> u64 {
267    let mut hasher = std::collections::hash_map::DefaultHasher::new();
268    bases.hash(&mut hasher);
269    hasher.finish()
270}

272/// Sequence-identity set that does not clone haplotype bases into the key.
273/// Hash collisions fall back to byte equality against already-accepted haplotypes.
274pub(crate) struct HapSeqSet {
275    by_hash: HashMap<(u64, bool), Vec<usize>>,
276}
277
278impl HapSeqSet {
279    pub(crate) fn from_haps(haps: &[Haplotype]) -> Self {
280        let mut s = Self {
281            by_hash: HashMap::new(),
282        };
283        for (i, h) in haps.iter().enumerate() {
284            s.by_hash
285                .entry((hash_hap_bases(&h.bases), h.is_reference))
286                .or_default()
287                .push(i);
288        }
289        s
290    }
291
292    pub(crate) fn new() -> Self {
293        Self {
294            by_hash: HashMap::new(),
295        }
296    }
297
298    /// Push `h` onto `haps` if no existing haplotype has the same bases + ref flag.
299    pub(crate) fn insert(&mut self, haps: &mut Vec<Haplotype>, h: Haplotype) -> bool {
300        let key = (hash_hap_bases(&h.bases), h.is_reference);
301        if let Some(idxs) = self.by_hash.get(&key) {
302            if idxs.iter().any(|&i| haps[i].bases == h.bases) {
303                return false;
304            }
305        }
306        self.by_hash.entry(key).or_default().push(haps.len());
307        haps.push(h);
308        true
309    }
310
311    pub(crate) fn contains(&self, haps: &[Haplotype], bases: &[u8], is_reference: bool) -> bool {
312        let key = (hash_hap_bases(bases), is_reference);
313        self.by_hash
314            .get(&key)
315            .is_some_and(|idxs| idxs.iter().any(|&i| haps[i].bases == bases))
316    }
317}

326        let ref_bases = b"TGCATGACTGATCGTACGATTCGAGCTAGTCGATCGATGCTAGCTAGGCTAACGTTAGCTAGTAACTGATCGATCGATACGTACGT";
327        let ref_hap = Haplotype::new(ref_bases.to_vec(), true);

330        let mut haps = vec![
331            Haplotype::new(ref_bases.to_vec(), true),
