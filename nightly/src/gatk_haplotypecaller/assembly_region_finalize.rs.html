19/// `AssemblyBasedCallerUtils.REFERENCE_PADDING_FOR_ASSEMBLY`.
20pub const GATK_REFERENCE_PADDING_FOR_ASSEMBLY: u64 = 500;
21
22/// Padded reference bytes for `assembleReads` / `runLocalAssembly` (±500 bp around extended span).
23pub fn assembly_reference_read(
24    dictionary: &SequenceDictionary,
25    ref_cache: &mut ReferenceWindowCache,
26    region: &AssemblyRegion,
27) -> GatkResult<AssemblyRead> {
28    let contig_len = dictionary
29        .contig(&region.contig)
30        .map(|c| c.length)
31        .ok_or_else(|| GatkError::argument(format!("unknown contig {}", region.contig)))?;
32    let pad_start = region
33        .extended_start
34        .get()
35        .saturating_sub(GATK_REFERENCE_PADDING_FOR_ASSEMBLY)
36        .max(1);
37    let pad_end = (region.extended_end.get() + GATK_REFERENCE_PADDING_FOR_ASSEMBLY).min(contig_len);
38    let bases: Vec<u8> = if pad_start > pad_end {
39        Vec::new()
40    } else {
41        ref_cache
42            .get_interval_bytes(dictionary, &region.contig, pad_start, pad_end)?
43            .to_vec()
44    };
45    Ok(AssemblyRead {
46        bases: String::from_utf8_lossy(&bases).into_owned(),
47        base_quals: vec![30; bases.len()],
48    })
49}

56/// GATK `AssemblyBasedCallerUtils.finalizeRegion` (soft-clip, low-qual tails, adaptor, padded-span clip, overlap).
57pub fn finalize_region_reads_for_assembly(
58    reads: &[SharedBamRecord],
59    region: &AssemblyRegion,
60    correct_overlapping_base_qualities: bool,
61    min_tail_quality: u8,
62    soft_clip_low_quality_ends: bool,
63) -> Vec<bam::Record> {
64    let policy = HcSoftclipPolicy::haplotype_caller_defaults();
65    let ref_start = region.extended_start.get();
66    let ref_stop = region.extended_end.get();
67    let mut out: Vec<bam::Record> = Vec::new();
68    for original in reads {
69        let mut read = apply_hc_softclip_pre_step(original, &policy).0;
70        if read.tid() < 0 || read.is_unmapped() || read.pos() < 0 {
71            continue;
72        }
73        if unclipped_read_length(&read) == 0 {
74            continue;
75        }
76        read = if soft_clip_low_quality_ends {
77            soft_clip_low_qual_ends(&read, min_tail_quality)
78        } else {
79            hard_clip_low_qual_ends(&read, min_tail_quality)
80        };
81        if read.is_unmapped() || !read_has_positive_cigar_length(&read) {
82            continue;
83        }
84        let adaptor_clipped = hard_clip_adaptor_sequence(&read);
85        if adaptor_clipped.is_unmapped() || !read_has_positive_cigar_length(&adaptor_clipped) {
86            continue;
87        }
88        if !read_overlaps_padded_span(&adaptor_clipped, ref_start, ref_stop) {
89            continue;
90        }
91        read = hard_clip_to_region(&adaptor_clipped, ref_start, ref_stop);
92        if read.is_unmapped() || !read_has_positive_cigar_length(&read) {
93            continue;
94        }
95        let aln_start = read.pos() + 1;
96        let aln_end = i64::from(alignment_end_1based(&read));
97        if aln_start > aln_end || aln_start > ref_stop as i64 || aln_end < ref_start as i64 {
98            continue;
99        }
100        normalize_record_cigar(&mut read);
101        out.push(read);
102    }
103    out.sort_by(|a, b| {
104        a.tid()
105            .cmp(&b.tid())
106            .then_with(|| a.pos().cmp(&b.pos()))
107            .then_with(|| a.qname().cmp(b.qname()))
108    });
109    if correct_overlapping_base_qualities {
110        // INVARIANT: `out` was sorted by (tid, pos, qname) immediately above.
111        #[allow(clippy::expect_used)]
112        clean_overlapping_read_pairs(&mut out, true)
113            .expect("reads sorted by alignment start before overlap correction");
114    }
115    out
116}

118fn read_overlaps_padded_span(rec: &bam::Record, ref_start: u64, ref_stop: u64) -> bool {
119    if rec.tid() < 0 || rec.is_unmapped() || rec.pos() < 0 {
120        return false;
121    }
122    let aln_start = rec.pos() + 1;
123    let aln_end = i64::from(alignment_end_1based(rec));
124    aln_start <= ref_stop as i64 && aln_end >= ref_start as i64
125}
126
127fn alignment_end_1based(rec: &bam::Record) -> i32 {
128    crate::read_unclip::alignment_end_1based(rec)
129}

131/// GATK `getPaddedReferenceLoc` span (±500 bp around the assembly region padded span).
132pub fn padded_reference_loc(
133    region: &AssemblyRegion,
134    dictionary: &SequenceDictionary,
135) -> (u64, u64) {
136    let contig_len = dictionary
137        .contig(&region.contig)
138        .map(|c| c.length)
139        .unwrap_or(u64::MAX);
140    let left = region
141        .extended_start
142        .get()
143        .saturating_sub(GATK_REFERENCE_PADDING_FOR_ASSEMBLY)
144        .max(1);
145    let right = (region.extended_end.get() + GATK_REFERENCE_PADDING_FOR_ASSEMBLY).min(contig_len);
146    (left, right)
147}

149/// Reference hap aligned to the assembly region padded span (GATK `createReferenceHaplotype`).
150pub fn reference_haplotype_for_assembly_region(
151    reference: &AssemblyRead,
152    region: &AssemblyRegion,
153    dictionary: &SequenceDictionary,
154) -> Haplotype {
155    let (loc_start, _) = padded_reference_loc(region, dictionary);
156    let alignment_start = region.extended_start.get().saturating_sub(loc_start) as usize;
157    let span_len = (region.extended_end.get() - region.extended_start.get() + 1) as usize;
158    let ref_bytes = reference.bases.as_bytes();
159    let end = (alignment_start + span_len).min(ref_bytes.len());
160    let bases = ref_bytes[alignment_start..end].to_vec();
161    // CLONE: needed because haplotype constructor takes owned bases.
162    let mut h = Haplotype::new(bases.clone(), true);
163    h.alignment_start_hap_wrt_ref = alignment_start;
164    let mut cigar = Cigar::new();
165    cigar.push(bases.len(), CigarOperator::Match);
166    h.cigar = Some(cigar);
167    h
168}

203pub fn records_to_assembly_reads<R: std::borrow::Borrow<bam::Record>>(
204    records: &[R],
205) -> Vec<AssemblyRead> {
206    records
207        .iter()
208        .map(|rec| {
209            let rec = rec.borrow();
210            let bases: Vec<u8> = rec.seq().as_bytes().to_vec();
211            AssemblyRead {
212                bases: String::from_utf8_lossy(&bases).into_owned(),
213                base_quals: rec.qual().to_vec(),
214            }
215        })
216        .collect()
217}
