19/// `AssemblyBasedCallerUtils.REFERENCE_PADDING_FOR_ASSEMBLY`.
20pub const GATK_REFERENCE_PADDING_FOR_ASSEMBLY: u64 = 500;
21
22/// Padded reference bytes for `assembleReads` / `runLocalAssembly` (±500 bp around extended span).
23pub fn assembly_reference_read(
24    dictionary: &SequenceDictionary,
25    ref_cache: &mut ReferenceWindowCache,
26    region: &AssemblyRegion,
27) -> GatkResult<AssemblyRead> {
28    let contig_len = dictionary
29        .contig(&region.contig)
30        .map(|c| c.length)
31        .ok_or_else(|| GatkError::argument(format!("unknown contig {}", region.contig)))?;
32    let pad_start = region
33        .extended_start
34        .get()
35        .saturating_sub(GATK_REFERENCE_PADDING_FOR_ASSEMBLY)
36        .max(1);
37    let pad_end = (region.extended_end.get() + GATK_REFERENCE_PADDING_FOR_ASSEMBLY).min(contig_len);
38    let bases: Vec<u8> = if pad_start > pad_end {
39        Vec::new()
40    } else {
41        ref_cache
42            .get_interval_bytes(dictionary, &region.contig, pad_start, pad_end)?
43            .to_vec()
44    };
45    let n = bases.len();
46    Ok(AssemblyRead {
47        bases,
48        base_quals: vec![30; n],
49    })
50}

94/// Softclip / adaptor / overlap finalize on already-owned BAM records.
95pub(crate) fn finalize_owned_bam_records(
96    reads: Vec<bam::Record>,
97    region: &AssemblyRegion,
98    correct_overlapping_base_qualities: bool,
99    min_tail_quality: u8,
100    soft_clip_low_quality_ends: bool,
101) -> Vec<bam::Record> {
102    let _prof = crate::hc_profile::begin(crate::hc_profile::Stage::ReadPreprocess);
103    let policy = HcSoftclipPolicy::haplotype_caller_defaults();
104    let ref_start = region.extended_start.get();
105    let ref_stop = region.extended_end.get();
106    let mut out: Vec<bam::Record> = Vec::with_capacity(reads.len());
107    for original in reads {
108        let mut read = apply_hc_softclip_pre_step(&original, &policy).0;
109        drop(original);
110        if read.tid() < 0 || read.is_unmapped() || read.pos() < 0 {
111            continue;
112        }
113        if unclipped_read_length(&read) == 0 {
114            continue;
115        }
116        read = if soft_clip_low_quality_ends {
117            soft_clip_low_qual_ends(&read, min_tail_quality)
118        } else {
119            hard_clip_low_qual_ends(&read, min_tail_quality)
120        };
121        if read.is_unmapped() || !read_has_positive_cigar_length(&read) {
122            continue;
123        }
124        let adaptor_clipped = hard_clip_adaptor_sequence(&read);
125        if adaptor_clipped.is_unmapped() || !read_has_positive_cigar_length(&adaptor_clipped) {
126            continue;
127        }
128        if !read_overlaps_padded_span(&adaptor_clipped, ref_start, ref_stop) {
129            continue;
130        }
131        read = hard_clip_to_region(&adaptor_clipped, ref_start, ref_stop);
132        if read.is_unmapped() || !read_has_positive_cigar_length(&read) {
133            continue;
134        }
135        let aln_start = read.pos() + 1;
136        let aln_end = i64::from(alignment_end_1based(&read));
137        if aln_start > aln_end || aln_start > ref_stop as i64 || aln_end < ref_start as i64 {
138            continue;
139        }
140        normalize_record_cigar(&mut read);
141        out.push(read);
142    }
143    out.sort_by(|a, b| {
144        a.tid()
145            .cmp(&b.tid())
146            .then_with(|| a.pos().cmp(&b.pos()))
147            .then_with(|| a.qname().cmp(b.qname()))
148    });
149    if correct_overlapping_base_qualities {
150        // INVARIANT: `out` was sorted by (tid, pos, qname) immediately above.
151        #[allow(clippy::expect_used)]
152        clean_overlapping_read_pairs(&mut out, true)
153            .expect("reads sorted by alignment start before overlap correction");
154    }
155    out
156}

158/// Re-clip already-finalized records to a (possibly trimmed) region's padded span.
159///
160/// Softclip / adaptor / overlap-quality correction are **not** re-applied — those ran in
161/// [`finalize_region_reads_for_assembly`]. Only hard-clip-to-region + overlap/unmapped filters + sort.
162pub fn clip_finalized_reads_to_region(
163    reads: &[bam::Record],
164    region: &AssemblyRegion,
165) -> Vec<bam::Record> {
166    let mut owned = reads.to_vec();
167    clip_finalized_reads_in_place(&mut owned, region);
168    owned
169}

171/// Consume-and-clip path for the assemble `finalizeRegion` buffer (no second full copy when
172/// most reads already lie inside the genotyping/padded span).
173pub fn clip_finalized_reads_in_place(reads: &mut Vec<bam::Record>, region: &AssemblyRegion) {
174    let ref_start = region.extended_start.get();
175    let ref_stop = region.extended_end.get();
176    reads.retain_mut(|original| {
177        if original.tid() < 0 || original.is_unmapped() || original.pos() < 0 {
178            return false;
179        }
180        if !read_overlaps_padded_span(original, ref_start, ref_stop) {
181            return false;
182        }
183        let aln_start = original.pos() + 1;
184        let aln_end = i64::from(alignment_end_1based(original));
185        let needs_clip = aln_start < ref_start as i64 || aln_end > ref_stop as i64;
186        if needs_clip {
187            let clipped = hard_clip_to_region(original, ref_start, ref_stop);
188            if clipped.is_unmapped() || !read_has_positive_cigar_length(&clipped) {
189                return false;
190            }
191            let aln_start = clipped.pos() + 1;
192            let aln_end = i64::from(alignment_end_1based(&clipped));
193            if aln_start > aln_end || aln_start > ref_stop as i64 || aln_end < ref_start as i64 {
194                return false;
195            }
196            *original = clipped;
197        }
198        normalize_record_cigar(original);
199        true
200    });
201    reads.sort_by(|a, b| {
202        a.tid()
203            .cmp(&b.tid())
204            .then_with(|| a.pos().cmp(&b.pos()))
205            .then_with(|| a.qname().cmp(b.qname()))
206    });
207}

209fn read_overlaps_padded_span(rec: &bam::Record, ref_start: u64, ref_stop: u64) -> bool {
210    if rec.tid() < 0 || rec.is_unmapped() || rec.pos() < 0 {
211        return false;
212    }
213    let aln_start = rec.pos() + 1;
214    let aln_end = i64::from(alignment_end_1based(rec));
215    aln_start <= ref_stop as i64 && aln_end >= ref_start as i64
216}

218fn alignment_end_1based(rec: &bam::Record) -> i32 {
219    crate::read_unclip::alignment_end_1based(rec)
220}

222/// GATK `getPaddedReferenceLoc` span (±500 bp around the assembly region padded span).
223pub fn padded_reference_loc(
224    region: &AssemblyRegion,
225    dictionary: &SequenceDictionary,
226) -> (u64, u64) {
227    let contig_len = dictionary
228        .contig(&region.contig)
229        .map(|c| c.length)
230        .unwrap_or(u64::MAX);
231    let left = region
232        .extended_start
233        .get()
234        .saturating_sub(GATK_REFERENCE_PADDING_FOR_ASSEMBLY)
235        .max(1);
236    let right = (region.extended_end.get() + GATK_REFERENCE_PADDING_FOR_ASSEMBLY).min(contig_len);
237    (left, right)
238}

240/// Reference hap aligned to the assembly region padded span (GATK `createReferenceHaplotype`).
241pub fn reference_haplotype_for_assembly_region(
242    reference: &AssemblyRead,
243    region: &AssemblyRegion,
244    dictionary: &SequenceDictionary,
245) -> Haplotype {
246    let (loc_start, _) = padded_reference_loc(region, dictionary);
247    let alignment_start = region.extended_start.get().saturating_sub(loc_start) as usize;
248    let span_len = (region.extended_end.get() - region.extended_start.get() + 1) as usize;
249    let ref_bytes = reference.bases.as_slice();
250    let end = (alignment_start + span_len).min(ref_bytes.len());
251    let bases = ref_bytes[alignment_start..end].to_vec();
252    // CLONE: needed because haplotype constructor takes owned bases.
253    let mut h = Haplotype::new(bases.clone(), true);
254    h.alignment_start_hap_wrt_ref = alignment_start;
255    let mut cigar = Cigar::new();
256    cigar.push(bases.len(), CigarOperator::Match);
257    h.cigar = Some(cigar);
258    h
259}

294pub fn records_to_assembly_reads<R: std::borrow::Borrow<bam::Record>>(
295    records: &[R],
296) -> Vec<AssemblyRead> {
297    records
298        .iter()
299        .map(|rec| {
300            let rec = rec.borrow();
301            let bases: Vec<u8> = rec.seq().as_bytes().to_vec();
302            AssemblyRead {
303                bases,
304                base_quals: rec.qual().to_vec(),
305            }
306        })
307        .collect()
