5use gatk_common::{GatkError, GatkResult};
6use rust_htslib::bam;
7use std::collections::{BTreeMap, BTreeSet};

9/// Parsed SAM/BAM header semantics for read-group and program validation.
10/// # Invariants
11/// Each `@RG ID` maps to at most one optional `SM` sample name.
12/// `@PG ID` values are unique in the parsed header.
13/// # Ownership
14/// Owns RG→sample map and PG id set; built once per header text/view.
15/// # Mutation
16/// Immutable after construction; [`Self::validate_record_links`] returns per-read resolution.
17/// # Biological assumptions
18/// Sample identity comes from `@RG SM:` tags (multi-sample vs single-sample HC paths).
19/// # Java equivalence
20/// GATK read header validation / `ReadUtils.alignmentAgreesWithHeader` inputs.
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub struct ReadHeaderSemantics {
23    read_group_to_sample: BTreeMap<String, Option<String>>,
24    program_ids: BTreeSet<String>,
25}
26
27/// Resolved `@RG` / `@PG` links for one read record against a parsed header.
28/// # Invariants
29/// When `read_group_id` is `Some`, it existed in the header map used to resolve `sample_name`.
30/// When `program_id` is `Some`, it was present in the header `@PG` set.
31/// # Ownership
32/// Owns optional tag strings; produced by [`ReadHeaderSemantics::validate_record_links`].
33/// # Mutation
34/// Immutable per-read snapshot.
35/// # Biological assumptions
36/// Sample name derives from read group's `SM` when declared.
37/// # Java equivalence
38/// GATK header-aware read validation (`SAMRecord` RG/PG linkage).
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct ResolvedReadHeaderSemantics {
41    pub read_group_id: Option<String>,
42    pub sample_name: Option<String>,
43    pub program_id: Option<String>,
44}
45
46impl ReadHeaderSemantics {
47    pub fn from_sam_header_text(header_text: &str) -> GatkResult<Self> {
48        let mut read_group_to_sample = BTreeMap::new();
49        let mut program_ids = BTreeSet::new();
50
51        for line in header_text.lines() {
52            if line.starts_with("@RG\t") {
53                let fields = parse_key_value_fields(line);
54                let rg_id = fields.get("ID").cloned().ok_or_else(|| {
55                    GatkError::validation("Malformed SAM header: @RG line missing ID")
56                })?;
57                let sample = fields.get("SM").cloned();
58                // Lifetime: detect duplicates with a borrow, then move `rg_id` into the map.
59                if read_group_to_sample.contains_key(&rg_id) {
60                    return Err(GatkError::validation(format!(
61                        "Malformed SAM header: duplicate @RG ID '{rg_id}'"
62                    )));
63                }
64                read_group_to_sample.insert(rg_id, sample);
65            } else if line.starts_with("@PG\t") {
66                let fields = parse_key_value_fields(line);
67                let pg_id = fields.get("ID").cloned().ok_or_else(|| {
68                    GatkError::validation("Malformed SAM header: @PG line missing ID")
69                })?;
70                // Lifetime: `contains` borrows; move `pg_id` into the set on success.
71                if program_ids.contains(&pg_id) {
72                    return Err(GatkError::validation(format!(
73                        "Malformed SAM header: duplicate @PG ID '{pg_id}'"
74                    )));
75                }
76                program_ids.insert(pg_id);
77            }
78        }
79
80        Ok(Self {
81            read_group_to_sample,
82            program_ids,
83        })
84    }
85
86    pub fn from_bam_header_view(header: &bam::HeaderView) -> GatkResult<Self> {
87        let text = std::str::from_utf8(header.as_bytes()).map_err(|e| {
88            GatkError::validation(format!("Malformed BAM header: non-UTF8 header text ({e})"))
89        })?;
90        Self::from_sam_header_text(text)
91    }
92
93    pub fn sample_for_read_group(&self, rg_id: &str) -> Option<&str> {
94        self.read_group_to_sample
95            .get(rg_id)
96            .and_then(|s| s.as_deref())
97    }
98
99    /// Distinct `@RG` SM tags (GATK `samplesList.numberOfSamples` for HC `isActive` shortcut).
100    pub fn unique_sample_count(&self) -> usize {
101        self.read_group_to_sample
102            .values()
103            .filter_map(|s| s.as_deref())
104            .collect::<BTreeSet<_>>()
105            .len()
106    }
107
108    /// Single `@RG SM:` when exactly one distinct sample is declared; else `None`.
109    pub fn primary_sample_name(&self) -> Option<String> {
110        let samples: BTreeSet<_> = self
111            .read_group_to_sample
112            .values()
113            .filter_map(|s| s.as_deref())
114            .collect();
115        if samples.len() == 1 {
116            samples.into_iter().next().map(|s| s.to_string())
117        } else {
118            None
119        }
120    }
121
122    /// True when the header declares at most one sample (Java single-sample `isActive` fast path).
123    pub fn is_single_sample_header(&self) -> bool {
124        self.unique_sample_count() <= 1
125    }
126
127    pub fn validate_record_links(
128        &self,
129        rg_tag: Option<&str>,
130        pg_tag: Option<&str>,
131    ) -> GatkResult<ResolvedReadHeaderSemantics> {
132        let sample_name = match rg_tag {
133            Some(rg) => {
134                let sample = self.read_group_to_sample.get(rg).ok_or_else(|| {
135                    GatkError::validation(format!(
136                        "Read header mismatch: record RG '{rg}' not found in header @RG IDs"
137                    ))
138                })?;
139                // CLONE: needed because owned sample id for carry/map.
140                sample.clone()
141            }
142            None => None,
143        };
144
145        if let Some(pg) = pg_tag {
146            if !self.program_ids.contains(pg) {
147                return Err(GatkError::validation(format!(
148                    "Read header mismatch: record PG '{pg}' not found in header @PG IDs"
149                )));
150            }
151        }
152
153        Ok(ResolvedReadHeaderSemantics {
154            read_group_id: rg_tag.map(ToOwned::to_owned),
155            sample_name,
156            program_id: pg_tag.map(ToOwned::to_owned),
157        })
158    }
159}
160
161fn parse_key_value_fields(line: &str) -> BTreeMap<String, String> {
162    let mut fields = BTreeMap::new();
163    for field in line.split('\t').skip(1) {
164        if let Some((k, v)) = field.split_once(':') {
165            fields.insert(k.to_string(), v.to_string());
166        }
167    }
168    fields
169}

171#[cfg(test)]
172#[allow(clippy::unwrap_used, clippy::expect_used, clippy::panic)]
173mod tests {
174    use super::*;
175
176    #[test]
177    fn parses_rg_pg_and_resolves_sample() {
178        let header = "@HD\tVN:1.6\n@RG\tID:rg1\tSM:sampleA\n@PG\tID:pg1\tPN:tool\n";
179        let semantics = ReadHeaderSemantics::from_sam_header_text(header).unwrap();
180        assert_eq!(semantics.sample_for_read_group("rg1"), Some("sampleA"));
181        let resolved = semantics
182            .validate_record_links(Some("rg1"), Some("pg1"))
183            .unwrap();
184        assert_eq!(resolved.sample_name.as_deref(), Some("sampleA"));
185    }
186
187    #[test]
188    fn rejects_duplicate_rg_id() {
189        let header = "@RG\tID:rg1\tSM:s1\n@RG\tID:rg1\tSM:s2\n";
190        let err = ReadHeaderSemantics::from_sam_header_text(header).unwrap_err();
191        match err {
192            GatkError::Validation { message, .. } => assert!(message.contains("duplicate @RG ID")),
193            other => panic!("expected Validation error, got {other:?}"),
194        }
195    }
196
197    #[test]
198    fn rejects_unknown_rg_and_pg_links() {
199        let header = "@RG\tID:rg1\tSM:s1\n@PG\tID:pg1\n";
200        let semantics = ReadHeaderSemantics::from_sam_header_text(header).unwrap();
201        let rg_err = semantics
202            .validate_record_links(Some("missing-rg"), Some("pg1"))
203            .unwrap_err();
204        let pg_err = semantics
205            .validate_record_links(Some("rg1"), Some("missing-pg"))
206            .unwrap_err();
207        match rg_err {
208            GatkError::Validation { message, .. } => assert!(message.contains("record RG")),
209            other => panic!("expected Validation error, got {other:?}"),
210        }
211        match pg_err {
212            GatkError::Validation { message, .. } => assert!(message.contains("record PG")),
213            other => panic!("expected Validation error, got {other:?}"),
214        }
215    }
