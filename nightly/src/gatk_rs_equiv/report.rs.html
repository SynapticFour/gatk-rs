10pub fn report(args: ReportArgs) -> Result<i32> {
11    let mut results = run::load_results(&args.results_dir)?;
12    if let Some(thr) = args.f1_delta_threshold {
13        for d in &mut results.f1_deltas {
14            d.within_threshold = d.abs_delta <= thr;
15        }
16        results.manifest.f1_delta_threshold = thr;
17        results.max_abs_delta = results
18            .f1_deltas
19            .iter()
20            .map(|d| d.abs_delta)
21            .fold(0.0_f64, f64::max);
22        results.gate_passed = results.f1_deltas.iter().all(|d| d.within_threshold);
23    }
24    write_reports(&args.results_dir, &results)?;
25    // Also refresh results.json with recomputed gate if threshold overridden.
26    let json = serde_json::to_string_pretty(&results)?;
27    fs::write(args.results_dir.join("results.json"), json)?;
28    Ok(if results.gate_passed { 0 } else { 1 })
29}

31pub fn write_reports(out_dir: &Path, results: &EquivResults) -> Result<()> {
32    let md = render_markdown(results);
33    fs::write(out_dir.join("REPORT.md"), md)?;
34    // Compact JSON already in results.json; also emit report.json alias.
35    let json = serde_json::to_string_pretty(results)?;
36    fs::write(out_dir.join("report.json"), json)?;
37    Ok(())
38}

40fn render_markdown(r: &EquivResults) -> String {
41    let mut s = String::new();
42    s.push_str("# gatk-rs-equiv report\n\n");
43    s.push_str(&format!(
44        "- **Created:** {}\n- **Engine:** `{}`\n- **Interval:** {}\n- **F1 Δ threshold:** {:.4}\n- **Gate:** {}\n- **Max |ΔF1|:** {:.4}\n\n",
45        r.manifest.created_utc,
46        r.manifest.engine,
47        r.manifest
48            .interval
49            .as_deref()
50            .unwrap_or("(full BAM / BED)"),
51        r.manifest.f1_delta_threshold,
52        if r.gate_passed {
53            "**PASS**"
54        } else {
55            "**FAIL**"
56        },
57        r.max_abs_delta
58    ));
59
60    s.push_str("## Equivalence metric (Rust F1 − Java F1)\n\n");
61    s.push_str(
62        "Absolute F1 against truth can be high for both tools while they still disagree. \
63The gate uses **|ΔF1|** only.\n\n",
64    );
65    s.push_str(
66        "| Stratum | Class | Java F1 | Rust F1 | Δ (Rust−Java) | |Δ| | Within threshold |\n",
67    );
68    s.push_str(
69        "|---------|-------|--------:|--------:|--------------:|----:|:----------------:|\n",
70    );
71    for d in &r.f1_deltas {
72        s.push_str(&format!(
73            "| {} | {} | {:.4} | {:.4} | {:+.4} | {:.4} | {} |\n",
74            d.stratum,
75            d.class,
76            d.java_f1,
77            d.rust_f1,
78            d.delta,
79            d.abs_delta,
80            if d.within_threshold { "yes" } else { "NO" }
81        ));
82    }
83    s.push('\n');
84
85    s.push_str("## Truth evaluation — Java\n\n");
86    s.push_str(&metrics_table(&r.java_vs_truth));
87    s.push_str("\n## Truth evaluation — Rust\n\n");
88    s.push_str(&metrics_table(&r.rust_vs_truth));
89
90    s.push_str("\n## Direct Rust vs Java (no truth)\n\n");
91    let c = &r.direct_compare;
92    s.push_str(&format!(
93        "| Metric | Count |\n|--------|------:|\n\
94         | Java sites (CHROM+POS+REF+ALT) | {} |\n\
95         | Rust sites | {} |\n\
96         | **Exact identical** (POS+REF+ALT+GT) | {} |\n\
97         | Allele match, GT mismatch | {} |\n\
98         | Same GT, FORMAT fields differ | {} |\n\
99         | Java only | {} |\n\
100         | Rust only | {} |\n\n",
101        c.java_sites,
102        c.rust_sites,
103        c.identical_sites,
104        c.allele_match_gt_mismatch,
105        c.format_mismatch_same_gt,
106        c.java_only,
107        c.rust_only
108    ));
109
110    s.push_str("## Inputs\n\n");
111    s.push_str(&format!(
112        "- Reference: `{}`\n- BAM: `{}`\n- Truth VCF: `{}`\n- Confident BED: `{}`\n- Java VCF: `{}`\n- Rust VCF: `{}`\n",
113        r.manifest.reference.display(),
114        r.manifest.bam.display(),
115        r.manifest.truth_vcf.display(),
116        r.manifest.confident_regions.display(),
117        r.manifest.java_vcf.display(),
118        r.manifest.rust_vcf.display()
119    ));
120    if !r.manifest.stratification_beds.is_empty() {
121        s.push_str("\n### Stratification BEDs\n\n");
122        for (name, path) in &r.manifest.stratification_beds {
123            s.push_str(&format!("- `{name}`: `{}`\n", path.display()));
124        }
125    }
126
127    if !r.notes.is_empty() {
128        s.push_str("\n## Notes\n\n");
129        for n in &r.notes {
130            s.push_str(&format!("- {n}\n"));
131        }
132    }
133
134    s.push_str(
135        "\n## Limits\n\n\
136         This tool checks **output equivalence** via community truth engines and a direct site table. \
137It does **not** measure runtime, memory, or full FORMAT/QUAL bitwise identity. \
138See `gatk-rs-equiv/README.md`.\n",
139    );
140    s
141}

143fn metrics_table(rows: &[TruthMetrics]) -> String {
144    let mut s = String::new();
145    s.push_str("| Stratum | Class | Precision | Recall | F1 | TP | FN | FP |\n");
146    s.push_str("|---------|-------|----------:|-------:|---:|---:|---:|---:|\n");
147    for m in rows {
148        for (class, p) in [("SNP", &m.snp), ("INDEL", &m.indel)] {
149            s.push_str(&format!(
150                "| {} | {class} | {:.4} | {:.4} | {:.4} | {} | {} | {} |\n",
151                m.stratum, p.precision, p.recall, p.f1, p.truth_tp, p.truth_fn, p.query_fp
152            ));
153        }
154    }
155    s
156}
