13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum EngineKind {
15    Happy,
16    Vcfeval,
17}

19impl EngineKind {
20    pub fn as_str(self) -> &'static str {
21        match self {
22            Self::Happy => "hap.py",
23            Self::Vcfeval => "rtg-vcfeval",
24        }
25    }
26}

28pub struct EnginePaths {
29    pub happy_bin: Option<PathBuf>,
30    pub rtg_bin: Option<PathBuf>,
31}
32
33fn which(name: &str) -> Option<PathBuf> {
34    std::env::var_os("PATH").and_then(|paths| {
35        for dir in std::env::split_paths(&paths) {
36            let candidate = dir.join(name);
37            if candidate.is_file() {
38                return Some(candidate);
39            }
40        }
41        None
42    })
43}
44
45fn resolve_happy(explicit: Option<&Path>) -> Option<PathBuf> {
46    if let Some(p) = explicit {
47        return Some(p.to_path_buf());
48    }
49    if let Ok(p) = std::env::var("HAPPY_BIN") {
50        return Some(PathBuf::from(p));
51    }
52    which("hap.py").or_else(|| which("happy"))
53}
54
55fn resolve_rtg(explicit: Option<&Path>) -> Option<PathBuf> {
56    if let Some(p) = explicit {
57        return Some(p.to_path_buf());
58    }
59    if let Ok(p) = std::env::var("RTG_BIN") {
60        return Some(PathBuf::from(p));
61    }
62    which("rtg")
63}
64
65pub fn select_engine(choice: &EngineChoice, paths: &EnginePaths) -> Result<(EngineKind, PathBuf)> {
66    let happy = resolve_happy(paths.happy_bin.as_deref());
67    let rtg = resolve_rtg(paths.rtg_bin.as_deref());
68    match choice {
69        EngineChoice::Happy => {
70            let p = happy.context("hap.py not found (set --happy-bin or HAPPY_BIN)")?;
71            Ok((EngineKind::Happy, p))
72        }
73        EngineChoice::Vcfeval => {
74            let p = rtg.context("rtg not found (set --rtg-bin or RTG_BIN)")?;
75            Ok((EngineKind::Vcfeval, p))
76        }
77        EngineChoice::Auto => {
78            if let Some(p) = happy {
79                Ok((EngineKind::Happy, p))
80            } else if let Some(p) = rtg {
81                eprintln!("[gatk-rs-equiv] hap.py not found; falling back to RTG vcfeval");
82                Ok((EngineKind::Vcfeval, p))
83            } else {
84                bail!(
85                    "Neither hap.py nor rtg found. Install Illumina hap.py \
86                     (https://github.com/Illumina/hap.py) or RTG Tools, \
87                     or use Dockerfile.equiv."
88                );
89            }
90        }
91    }
92}

105/// Run the selected engine; return metrics including `*` (all) and named strata when available.
106pub fn evaluate(
107    kind: EngineKind,
108    engine_bin: &Path,
109    input: &EvalInput<'_>,
110    query_label: &str,
111) -> Result<Vec<TruthMetrics>> {
112    match kind {
113        EngineKind::Happy => happy::run_happy(engine_bin, input, query_label),
114        EngineKind::Vcfeval => vcfeval::run_vcfeval(engine_bin, input, query_label),
115    }
116}

118pub(crate) fn run_checked(cmd: &mut Command, label: &str) -> Result<()> {
119    eprintln!("[gatk-rs-equiv] {label}: {cmd:?}");
120    let status = cmd
121        .status()
122        .with_context(|| format!("failed to spawn {label}"))?;
123    if !status.success() {
124        bail!("{label} failed with {status}");
125    }
126    Ok(())
127}

129pub(crate) fn prf_from_counts(tp: u64, fn_: u64, fp: u64) -> Prf {
130    let precision = if tp + fp == 0 {
131        0.0
132    } else {
133        tp as f64 / (tp + fp) as f64
134    };
135    let recall = if tp + fn_ == 0 {
136        0.0
137    } else {
138        tp as f64 / (tp + fn_) as f64
139    };
140    let f1 = if precision + recall == 0.0 {
141        0.0
142    } else {
143        2.0 * precision * recall / (precision + recall)
144    };
145    Prf {
146        precision,
147        recall,
148        f1,
149        truth_tp: tp,
150        truth_fn: fn_,
151        query_fp: fp,
152    }
