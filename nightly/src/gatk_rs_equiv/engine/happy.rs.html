12/// Run hap.py for one query callset.
13pub fn run_happy(
14    happy_bin: &Path,
15    input: &EvalInput<'_>,
16    query_label: &str,
17) -> Result<Vec<TruthMetrics>> {
18    if let Some(parent) = input.out_prefix.parent() {
19        std::fs::create_dir_all(parent)?;
20    }
21
22    let mut cmd = Command::new(happy_bin);
23    cmd.arg(input.truth_vcf)
24        .arg(input.query_vcf)
25        .arg("-r")
26        .arg(input.reference)
27        .arg("-f")
28        .arg(input.confident_bed)
29        .arg("-o")
30        .arg(input.out_prefix)
31        .arg("--threads")
32        .arg(input.threads.to_string())
33        .arg("--verbose");
34
35    // Stratification: write a TSV for hap.py --stratification when beds are provided.
36    let strat_tsv = write_stratification_tsv(input)?;
37    if let Some(ref tsv) = strat_tsv {
38        cmd.arg("--stratification").arg(tsv);
39    }
40
41    run_checked(&mut cmd, "hap.py")?;
42
43    let summary = PathBuf::from(format!("{}.summary.csv", input.out_prefix.display()));
44    if !summary.is_file() {
45        // Some builds write without the intermediate dot form; try extended location.
46        let alt = input.out_prefix.with_extension("summary.csv");
47        if alt.is_file() {
48            return parse_happy_summary_csv(&alt, query_label);
49        }
50        bail!(
51            "hap.py did not produce summary.csv at {}",
52            summary.display()
53        );
54    }
55    parse_happy_summary_csv(&summary, query_label)
56}
57
58fn write_stratification_tsv(input: &EvalInput<'_>) -> Result<Option<PathBuf>> {
59    if input.stratification.is_empty() {
60        return Ok(None);
61    }
62    let path = PathBuf::from(format!("{}.stratification.tsv", input.out_prefix.display()));
63    let mut f = File::create(&path)?;
64    // hap.py stratification TSV: region_name <tab> bed_path
65    for (name, bed) in input.stratification {
66        if !bed.is_file() {
67            bail!(
68                "stratification BED not found for '{name}': {}",
69                bed.display()
70            );
71        }
72        writeln!(f, "{name}\t{}", bed.display())?;
73    }
74    Ok(Some(path))
75}

77/// Parse Illumina hap.py `*.summary.csv`.
78pub fn parse_happy_summary_csv(path: &Path, query_label: &str) -> Result<Vec<TruthMetrics>> {
79    let file = File::open(path).with_context(|| format!("open {}", path.display()))?;
80    let mut rdr = csv::ReaderBuilder::new()
81        .flexible(true)
82        .from_reader(BufReader::new(file));
83    let headers = rdr
84        .headers()
85        .context("read hap.py summary headers")?
86        .iter()
87        .map(|s| s.to_string())
88        .collect::<Vec<_>>();
89
90    let idx =
91        |name: &str| -> Option<usize> { headers.iter().position(|h| h.eq_ignore_ascii_case(name)) };
92
93    let i_type = idx("Type").context("summary.csv missing Type")?;
94    let i_filter = idx("Filter");
95    let i_subtype = idx("Subtype")
96        .or_else(|| idx("Subset"))
97        .or_else(|| idx("QQ"));
98    // Stratification column names vary; try common ones.
99    let i_strat = idx("Subset")
100        .or_else(|| idx("Stratifier"))
101        .or_else(|| idx("Region"));
102
103    let i_tp = idx("TRUTH.TP").context("summary.csv missing TRUTH.TP")?;
104    let i_fn = idx("TRUTH.FN").context("summary.csv missing TRUTH.FN")?;
105    let i_fp = idx("QUERY.FP").context("summary.csv missing QUERY.FP")?;
106    let i_prec = idx("METRIC.Precision");
107    let i_rec = idx("METRIC.Recall");
108    let i_f1 = idx("METRIC.F1_Score");
109
110    // Accumulate by (stratum, type)
111    let mut snp: BTreeMap<String, Prf> = BTreeMap::new();
112    let mut indel: BTreeMap<String, Prf> = BTreeMap::new();
113    let mut all: BTreeMap<String, Prf> = BTreeMap::new();
114
115    for rec in rdr.records() {
116        let rec = rec?;
117        let typ = rec.get(i_type).unwrap_or("").trim().to_ascii_uppercase();
118        if let Some(i) = i_filter {
119            let filt = rec.get(i).unwrap_or("").trim().to_ascii_uppercase();
120            // Prefer PASS / ALL rows; skip others when Filter column exists.
121            if !filt.is_empty() && filt != "PASS" && filt != "ALL" && filt != "*" {
122                continue;
123            }
124        }
125        // Skip per-genotype subtypes when Subtype is not * /.
126        if let Some(i) = i_subtype {
127            let sub = rec.get(i).unwrap_or("").trim();
128            if !sub.is_empty() && sub != "*" && sub != "." && !sub.eq_ignore_ascii_case("ALL") {
129                // Still allow stratification Subset column separately.
130                if i_strat != Some(i) {
131                    continue;
132                }
133            }
134        }
135
136        let stratum = i_strat
137            .and_then(|i| rec.get(i))
138            .map(|s| s.trim().to_string())
139            .filter(|s| !s.is_empty())
140            .unwrap_or_else(|| "*".to_string());
141
142        let tp: u64 = parse_u64(rec.get(i_tp));
143        let fn_: u64 = parse_u64(rec.get(i_fn));
144        let fp: u64 = parse_u64(rec.get(i_fp));
145        let mut prf = prf_from_counts(tp, fn_, fp);
146        if let Some(i) = i_prec {
147            if let Some(v) = parse_f64(rec.get(i)) {
148                prf.precision = v;
149            }
150        }
151        if let Some(i) = i_rec {
152            if let Some(v) = parse_f64(rec.get(i)) {
153                prf.recall = v;
154            }
155        }
156        if let Some(i) = i_f1 {
157            if let Some(v) = parse_f64(rec.get(i)) {
158                prf.f1 = v;
159            }
160        }
161
162        match typ.as_str() {
163            "SNP" => {
164                snp.insert(stratum, prf);
165            }
166            "INDEL" => {
167                indel.insert(stratum, prf);
168            }
169            "ALL" | "*" => {
170                all.insert(stratum, prf);
171            }
172            _ => {}
173        }
174    }
175
176    let mut strata: BTreeMap<String, ()> = BTreeMap::new();
177    for k in snp.keys().chain(indel.keys()).chain(all.keys()) {
178        strata.insert(k.clone(), ());
179    }
180    if strata.is_empty() {
181        strata.insert("*".into(), ());
182    }
183
184    let mut out = Vec::new();
185    for stratum in strata.keys() {
186        out.push(TruthMetrics {
187            query_label: query_label.to_string(),
188            stratum: stratum.clone(),
189            snp: snp.get(stratum).cloned().unwrap_or_else(Prf::zero),
190            indel: indel.get(stratum).cloned().unwrap_or_else(Prf::zero),
191            all: all.get(stratum).cloned(),
192        });
193    }
194    Ok(out)
195}
196
197fn parse_u64(s: Option<&str>) -> u64 {
198    s.and_then(|x| x.trim().parse().ok()).unwrap_or(0)
199}
200
201fn parse_f64(s: Option<&str>) -> Option<f64> {
202    s.and_then(|x| {
203        let t = x.trim();
204        if t.is_empty() || t == "." || t.eq_ignore_ascii_case("nan") {
205            None
206        } else {
207            t.parse().ok()
208        }
209    })
210}

212#[cfg(test)]
213mod tests {
214    use super::*;
215    use std::io::Write;
216    use tempfile::tempdir;
217
218    #[test]
219    fn parses_minimal_summary() {
220        let dir = tempdir().unwrap();
221        let path = dir.path().join("x.summary.csv");
222        let mut f = File::create(&path).unwrap();
223        writeln!(
224            f,
225            "Type,Filter,TRUTH.TP,TRUTH.FN,QUERY.FP,METRIC.Precision,METRIC.Recall,METRIC.F1_Score"
226        )
227        .unwrap();
228        writeln!(f, "SNP,PASS,10,2,1,0.909,0.833,0.869").unwrap();
229        writeln!(f, "INDEL,PASS,5,1,0,1.0,0.833,0.909").unwrap();
230        let m = parse_happy_summary_csv(&path, "rust").unwrap();
231        assert_eq!(m.len(), 1);
232        assert!((m[0].snp.f1 - 0.869).abs() < 1e-6);
233        assert!((m[0].indel.f1 - 0.909).abs() < 1e-6);
234    }
