10/// Run RTG vcfeval. Stratification BEDs are evaluated as separate vcfeval passes
11/// (intersection with confident regions left to the caller-provided BED when possible).
12pub fn run_vcfeval(
13    rtg_bin: &Path,
14    input: &EvalInput<'_>,
15    query_label: &str,
16) -> Result<Vec<TruthMetrics>> {
17    let out_dir = PathBuf::from(format!("{}.vcfeval", input.out_prefix.display()));
18    if out_dir.exists() {
19        fs::remove_dir_all(&out_dir)?;
20    }
21
22    // Ensure SDF for reference (cached next to out_prefix).
23    let sdf = PathBuf::from(format!("{}.sdf", input.reference.display()));
24    if !sdf.exists() {
25        let mut fmt = Command::new(rtg_bin);
26        fmt.arg("format").arg("-o").arg(&sdf).arg(input.reference);
27        run_checked(&mut fmt, "rtg format")?;
28    }
29
30    // RTG vcfeval requires block-gzipped VCFs (plain `.vcf` → "is not in bgzip format").
31    let query_gz = ensure_bgzip_vcf(
32        input.query_vcf,
33        &PathBuf::from(format!("{}.query.vcf.gz", input.out_prefix.display())),
34    )?;
35    let truth_gz = ensure_bgzip_vcf(
36        input.truth_vcf,
37        &PathBuf::from(format!("{}.truth.vcf.gz", input.out_prefix.display())),
38    )?;
39
40    let mut metrics = Vec::new();
41    metrics.push(run_one(
42        rtg_bin,
43        &truth_gz,
44        &query_gz,
45        &out_dir,
46        input.confident_bed,
47        "*",
48        query_label,
49        &sdf,
50        input.threads,
51    )?);
52
53    for (name, bed) in input.stratification {
54        let strat_out = PathBuf::from(format!("{}.strat_{name}", out_dir.display()));
55        if strat_out.exists() {
56            fs::remove_dir_all(&strat_out)?;
57        }
58        metrics.push(run_one(
59            rtg_bin,
60            &truth_gz,
61            &query_gz,
62            &strat_out,
63            bed,
64            name,
65            query_label,
66            &sdf,
67            input.threads,
68        )?);
69    }
70    Ok(metrics)
71}

73/// Return a path suitable for RTG `-b`/`-c`: already `.gz`, or a freshly bgzipped copy.
74fn ensure_bgzip_vcf(vcf: &Path, staging_gz: &Path) -> Result<PathBuf> {
75    if looks_gzip_path(vcf) {
76        ensure_vcf_index(vcf)?;
77        return Ok(vcf.to_path_buf());
78    }
79    if let Some(parent) = staging_gz.parent() {
80        fs::create_dir_all(parent)?;
81    }
82    // Prefer bcftools (already on GIAB finalize PATH); fall back to bgzip.
83    if which("bcftools").is_some() {
84        let mut view = Command::new("bcftools");
85        view.args(["view", "-Oz", "-o"]).arg(staging_gz).arg(vcf);
86        run_checked(&mut view, "bcftools view -Oz")?;
87    } else if which("bgzip").is_some() {
88        let out_file = fs::File::create(staging_gz)
89            .with_context(|| format!("create {}", staging_gz.display()))?;
90        let status = Command::new("bgzip")
91            .arg("-c")
92            .arg(vcf)
93            .stdout(out_file)
94            .status()
95            .with_context(|| format!("spawn bgzip for {}", vcf.display()))?;
96        if !status.success() {
97            bail!("bgzip failed for {} ({status})", vcf.display());
98        }
99    } else {
100        bail!(
101            "RTG needs bgzipped VCFs; install bcftools or bgzip to compress {}",
102            vcf.display()
103        );
104    }
105    ensure_vcf_index(staging_gz)?;
106    Ok(staging_gz.to_path_buf())
107}

109fn looks_gzip_path(path: &Path) -> bool {
110    path.extension()
111        .and_then(|e| e.to_str())
112        .is_some_and(|e| e.eq_ignore_ascii_case("gz"))
113}

115fn ensure_vcf_index(vcf_gz: &Path) -> Result<()> {
116    let tbi = PathBuf::from(format!("{}.tbi", vcf_gz.display()));
117    let csi = PathBuf::from(format!("{}.csi", vcf_gz.display()));
118    if tbi.is_file() || csi.is_file() {
119        return Ok(());
120    }
121    if which("bcftools").is_some() {
122        let mut idx = Command::new("bcftools");
123        idx.args(["index", "-f", "-t"]).arg(vcf_gz);
124        run_checked(&mut idx, "bcftools index -t")?;
125        return Ok(());
126    }
127    if which("tabix").is_some() {
128        let mut idx = Command::new("tabix");
129        idx.args(["-f", "-p", "vcf"]).arg(vcf_gz);
130        run_checked(&mut idx, "tabix -p vcf")?;
131        return Ok(());
132    }
133    // RTG often accepts bgzip without an index for small callsets; continue.
134    eprintln!(
135        "[gatk-rs-equiv] warning: no tabix/bcftools index for {} (continuing)",
136        vcf_gz.display()
137    );
138    Ok(())
139}

141fn which(name: &str) -> Option<PathBuf> {
142    std::env::var_os("PATH").and_then(|paths| {
143        for dir in std::env::split_paths(&paths) {
144            let candidate = dir.join(name);
145            if candidate.is_file() {
146                return Some(candidate);
147            }
148        }
149        None
150    })
151}

153#[allow(clippy::too_many_arguments)]
154fn run_one(
155    rtg_bin: &Path,
156    truth_vcf: &Path,
157    query_vcf: &Path,
158    out_dir: &Path,
159    bed: &Path,
160    stratum: &str,
161    query_label: &str,
162    sdf: &Path,
163    threads: u32,
164) -> Result<TruthMetrics> {
165    let mut cmd = Command::new(rtg_bin);
166    cmd.arg("vcfeval")
167        .arg("-b")
168        .arg(truth_vcf)
169        .arg("-c")
170        .arg(query_vcf)
171        .arg("-t")
172        .arg(sdf)
173        .arg("-e")
174        .arg(bed)
175        .arg("-o")
176        .arg(out_dir)
177        .arg("--threads")
178        .arg(threads.to_string());
179    run_checked(&mut cmd, "rtg vcfeval")?;
180
181    let summary = out_dir.join("summary.txt");
182    if !summary.is_file() {
183        bail!("rtg vcfeval missing summary.txt in {}", out_dir.display());
184    }
185    parse_vcfeval_summary(&summary, query_label, stratum)
186}

188/// Parse RTG `summary.txt` (SNP / INDEL / Total rows).
189pub fn parse_vcfeval_summary(
190    path: &Path,
191    query_label: &str,
192    stratum: &str,
193) -> Result<TruthMetrics> {
194    let text = fs::read_to_string(path).with_context(|| format!("read {}", path.display()))?;
195    let mut snp = Prf::zero();
196    let mut indel = Prf::zero();
197    let mut all = None;
198
199    for line in text.lines() {
200        let line = line.trim();
201        if line.is_empty() || line.starts_with('=') || line.starts_with("Threshold") {
202            continue;
203        }
204        // Typical: None SNP 10 2 1... precision recall f-measure
205        let cols: Vec<&str> = line.split_whitespace().collect();
206        if cols.len() < 6 {
207            continue;
208        }
209        // Find a type token.
210        let type_idx = cols.iter().position(|c| {
211            matches!(
212                c.to_ascii_uppercase().as_str(),
213                "SNP" | "INDEL" | "INDELS" | "TOTAL" | "ALL"
214            )
215        });
216        let Some(ti) = type_idx else {
217            continue;
218        };
219        let typ = cols[ti].to_ascii_uppercase();
220        // After type: baseline/TP-style counts vary by RTG version.
221        // Prefer trailing precision recall f-measure floats.
222        let floats: Vec<f64> = cols.iter().filter_map(|c| c.parse::<f64>().ok()).collect();
223        if floats.len() < 3 {
224            continue;
225        }
226        let n = floats.len();
227        let precision = floats[n - 3];
228        let recall = floats[n - 2];
229        let f1 = floats[n - 1];
230        // Best-effort TP/FN/FP from integer columns near the type token.
231        let ints: Vec<u64> = cols
232            .iter()
233            .skip(ti + 1)
234            .filter_map(|c| c.parse::<u64>().ok())
235            .collect();
236        let (tp, fn_, fp) = match ints.as_slice() {
237            [tp, fn_, fp, ..] => (*tp, *fn_, *fp),
238            _ => (0, 0, 0),
239        };
240        let mut prf = prf_from_counts(tp, fn_, fp);
241        prf.precision = precision;
242        prf.recall = recall;
243        prf.f1 = f1;
244        match typ.as_str() {
245            "SNP" => snp = prf,
246            "INDEL" | "INDELS" => indel = prf,
247            "TOTAL" | "ALL" => all = Some(prf),
248            _ => {}
249        }
250    }
251
252    Ok(TruthMetrics {
253        query_label: query_label.to_string(),
254        stratum: stratum.to_string(),
255        snp,
256        indel,
257        all,
258    })
