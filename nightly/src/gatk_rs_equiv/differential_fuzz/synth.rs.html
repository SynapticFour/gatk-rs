1//! Materialize a [`Scenario`] into FASTA + BAM (+ index) under a work directory.
2
3use super::scenario::Scenario;
4use anyhow::{bail, Context, Result};
5use std::fs;
6use std::io::Write;
7use std::path::{Path, PathBuf};
8use std::process::Command;
9
10const CONTIG: &str = "synth1";
11
12#[derive(Debug, Clone)]
13pub struct Materialized {
14    pub dir: PathBuf,
15    pub reference: PathBuf,
16    pub bam: PathBuf,
17    pub interval: String,
18}
19
20struct Rng(u64);
21impl Rng {
22    fn next_u32(&mut self) -> u32 {
23        // xorshift64*
24        let mut x = self.0;
25        x ^= x >> 12;
26        x ^= x << 25;
27        x ^= x >> 27;
28        self.0 = x;
29        ((x.wrapping_mul(0x2545F4914F6CDD1D)) >> 32) as u32
30    }
31    fn chance(&mut self, p255: u8) -> bool {
32        (self.next_u32() % 256) < u32::from(p255)
33    }
34    fn gen_range(&mut self, lo: u32, hi: u32) -> u32 {
35        if hi <= lo {
36            return lo;
37        }
38        lo + self.next_u32() % (hi - lo + 1)
39    }
40}
41
42fn base_at(rng: &mut Rng) -> u8 {
43    b"ACGT"[rng.gen_range(0, 3) as usize]
44}
45
46fn mutate_base(b: u8, rng: &mut Rng) -> u8 {
47    let bases = [b'A', b'C', b'G', b'T'];
48    let mut out = bases[rng.gen_range(0, 3) as usize];
49    if out == b {
50        out = bases[((rng.next_u32() as usize) + 1) % 4];
51    }
52    out
53}
54
55fn phred_char(mean: u8, jitter: u8, rng: &mut Rng) -> u8 {
56    let j = if jitter == 0 {
57        0
58    } else {
59        rng.gen_range(0, u32::from(jitter)) as i32 - (jitter as i32 / 2)
60    };
61    let q = (i32::from(mean) + j).clamp(2, 40) as u8;
62    q + 33
63}

65/// Write FASTA + BAM for `scenario` into `dir`.
66pub fn materialize_scenario(scenario: &Scenario, dir: &Path) -> Result<Materialized> {
67    fs::create_dir_all(dir)?;
68    let mut rng = Rng(scenario.seed ^ 0x9E37_79B9_7F4A_7C15);
69
70    // Reference with mild complexity (avoid pure homopolymer).
71    let mut reference = Vec::with_capacity(scenario.ref_len as usize);
72    for i in 0..scenario.ref_len {
73        if i > 0 && i % 17 == 0 {
74            reference.push(reference[i as usize - 1]); // short homopolymer run
75        } else {
76            reference.push(base_at(&mut rng));
77        }
78    }
79    if scenario.plant_snp && scenario.ref_len > 40 {
80        let mid = (scenario.ref_len / 2) as usize;
81        reference[mid] = mutate_base(reference[mid], &mut rng);
82    }
83    if scenario.plant_indel && scenario.ref_len > 60 {
84        let mid = (scenario.ref_len / 2 + 7) as usize;
85        // Delete one base in haplotype space by rewriting flanking — keep ref as-is;
86        // plant indel via reads instead (below).
87        let _ = mid;
88    }
89
90    let fa = dir.join("reference.fa");
91    {
92        let mut f = fs::File::create(&fa)?;
93        writeln!(f, ">{CONTIG}")?;
94        for chunk in reference.chunks(80) {
95            f.write_all(chunk)?;
96            f.write_all(b"\n")?;
97        }
98    }
99    // faidx via samtools when available; else write trivial.fai
100    if Command::new("samtools")
101        .args(["faidx", fa.to_str().unwrap()])
102        .status()
103        .map(|s| s.success())
104        .unwrap_or(false)
105    {
106        // ok
107    } else {
108        let fai = dir.join("reference.fa.fai");
109        fs::write(
110            fai,
111            format!(
112                "{CONTIG}\t{}\t{}\t80\t81\n",
113                scenario.ref_len,
114                CONTIG.len() + 2
115            ),
116        )?;
117    }
118
119    let sam = dir.join("reads.sam");
120    let mut sam_f = fs::File::create(&sam)?;
121    writeln!(sam_f, "@HD\tVN:1.6\tSO:coordinate")?;
122    writeln!(sam_f, "@SQ\tSN:{CONTIG}\tLN:{}", scenario.ref_len)?;
123    writeln!(sam_f, "@RG\tID:synth\tSM:SYNTH\tPL:ILLUMINA")?;
124
125    let max_start = scenario.ref_len.saturating_sub(scenario.read_len).max(1);
126    let n = scenario.n_reads.max(scenario.coverage);
127
128    for i in 0..n {
129        let start0 = rng.gen_range(0, max_start); // 0-based
130        let pos1 = start0 + 1; // SAM 1-based
131        let mapq =
132            scenario.mapq_min + (rng.gen_range(0, u32::from(scenario.mapq_span.max(1))) as u8);
133        let mapq = mapq.min(60);
134
135        let mut seq = reference[start0 as usize..(start0 + scenario.read_len) as usize].to_vec();
136        let mut cigar = format!("{}M", scenario.read_len);
137
138        // Soft-clip
139        if rng.chance(scenario.softclip_p) && scenario.softclip_max > 0 && scenario.read_len > 20 {
140            let sc = rng
141                .gen_range(1, u32::from(scenario.softclip_max))
142                .min(scenario.read_len / 4);
143            for b in seq.iter_mut().take(sc as usize) {
144                *b = base_at(&mut rng);
145            }
146            cigar = format!("{sc}S{}M", scenario.read_len - sc);
147        }
148
149        // Small indel in read vs reference
150        if rng.chance(scenario.indel_p) && scenario.indel_max > 0 && seq.len() > 20 {
151            let k = rng.gen_range(1, u32::from(scenario.indel_max)) as usize;
152            let at = seq.len() / 2;
153            if rng.chance(128) {
154                // insertion
155                for _ in 0..k {
156                    seq.insert(at, base_at(&mut rng));
157                }
158                cigar = format!("{}M{}I{}M", at, k, seq.len() - at - k);
159            } else if at + k < seq.len() {
160                seq.drain(at..at + k);
161                cigar = format!("{}M{}D{}M", at, k, seq.len() - at);
162            }
163        }
164
165        // Plant indel haplotype near mid for some reads
166        if scenario.plant_indel && i % 3 == 0 && seq.len() > 30 {
167            let at = seq.len() / 2;
168            seq.insert(at, b'A');
169            // crude CIGAR rebuild for planted insertion
170            cigar = format!("{}M1I{}M", at, seq.len() - at - 1);
171        }
172
173        let qual: String = (0..seq.len())
174            .map(|_| phred_char(scenario.bq_mean, scenario.bq_jitter, &mut rng) as char)
175            .collect();
176        let seq_s =
177            String::from_utf8(seq).unwrap_or_else(|_| "N".repeat(scenario.read_len as usize));
178
179        let flag = if rng.chance(scenario.mate_overlap_p) {
180            // paired, proper pair, mate reverse (simplified)
181            99u16
182        } else {
183            0u16
184        };
185        let qname = format!("r{i:04}");
186        let mate_pos = if flag != 0 {
187            (pos1 as i64 + i32::from(scenario.read_len as i16 / 2) as i64).max(1)
188        } else {
189            0
190        };
191        let tlen = if flag != 0 {
192            i32::from(scenario.read_len as i16)
193        } else {
194            0
195        };
196
197        writeln!(
198            sam_f,
199            "{qname}\t{flag}\t{CONTIG}\t{pos1}\t{mapq}\t{cigar}\t{}\t{mate_pos}\t{tlen}\t{seq_s}\t{qual}\tRG:Z:synth",
200            if flag != 0 { CONTIG } else { "*" }
201        )?;
202
203        // overlapping mate
204        if flag != 0 {
205            let mate_start = (start0 + scenario.read_len / 3).min(max_start);
206            let mpos1 = mate_start + 1;
207            let mut mseq = reference[mate_start as usize
208                ..(mate_start + scenario.read_len).min(scenario.ref_len) as usize]
209                .to_vec();
210            while mseq.len() < scenario.read_len as usize {
211                mseq.push(b'N');
212            }
213            let mqual: String = (0..mseq.len())
214                .map(|_| phred_char(scenario.bq_mean, scenario.bq_jitter, &mut rng) as char)
215                .collect();
216            let mseq_s = String::from_utf8_lossy(&mseq).into_owned();
217            let mcigar = format!("{}M", mseq.len());
218            writeln!(
219                sam_f,
220                "{qname}\t147\t{CONTIG}\t{mpos1}\t{mapq}\t{mcigar}\t{CONTIG}\t{pos1}\t{}\t{mseq_s}\t{mqual}\tRG:Z:synth",
221                -(tlen)
222            )?;
223        }
224    }
225    drop(sam_f);
226
227    let bam_unsorted = dir.join("reads.unsorted.bam");
228    let bam = dir.join("reads.bam");
229    let status = Command::new("samtools")
230        .args([
231            "view",
232            "-bS",
233            sam.to_str().unwrap(),
234            "-o",
235            bam_unsorted.to_str().unwrap(),
236        ])
237        .status()
238        .context("samtools view (SAM→BAM) — samtools required on PATH")?;
239    if !status.success() {
240        bail!("samtools view failed: {status}");
241    }
242    let status = Command::new("samtools")
243        .args([
244            "sort",
245            "-o",
246            bam.to_str().unwrap(),
247            bam_unsorted.to_str().unwrap(),
248        ])
249        .status()
250        .context("samtools sort")?;
251    if !status.success() {
252        bail!("samtools sort failed: {status}");
253    }
254    let _ = fs::remove_file(&bam_unsorted);
255    let status = Command::new("samtools")
256        .args(["index", bam.to_str().unwrap()])
257        .status()
258        .context("samtools index")?;
259    if !status.success() {
260        bail!("samtools index failed: {status}");
261    }
262
263    let interval = format!("{CONTIG}:1-{}", scenario.ref_len);
264    Ok(Materialized {
265        dir: dir.to_path_buf(),
266        reference: fa,
267        bam,
268        interval,
269    })
270}

272#[cfg(test)]
273mod tests {
274    use super::*;
275    use crate::differential_fuzz::scenario::scenario_from_bytes;
276    use tempfile::tempdir;
277
278    #[test]
279    fn materialize_smoke() {
280        if Command::new("samtools").arg("--version").output().is_err() {
281            eprintln!("skip: samtools missing");
282            return;
283        }
284        let dir = tempdir().unwrap();
285        let s = scenario_from_bytes(&[9, 8, 7, 6, 5, 4, 3, 2, 1, 0]);
286        let m = materialize_scenario(&s, dir.path()).unwrap();
287        assert!(m.bam.is_file());
288        assert!(m.reference.is_file());
289    }
