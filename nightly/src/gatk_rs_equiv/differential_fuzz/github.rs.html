1//! Open a GitHub issue for a minimized parity divergence (via `gh` CLI).
2
3use super::runner::Divergence;
4use super::scenario::Scenario;
5use anyhow::{bail, Context, Result};
6use std::path::Path;
7use std::process::Command;
8
9pub const LABEL: &str = "parity-divergence";
10
11fn ensure_label() -> Result<()> {
12    let check = Command::new("gh")
13        .args([
14            "label", "list", "--search", LABEL, "--json", "name", "-q", ".[].name",
15        ])
16        .output()
17        .context("gh label list (is `gh` installed and authenticated?)")?;
18    if !check.status.success() {
19        bail!(
20            "gh label list failed: {}",
21            String::from_utf8_lossy(&check.stderr)
22        );
23    }
24    let names = String::from_utf8_lossy(&check.stdout);
25    if names.lines().any(|l| l.trim() == LABEL) {
26        return Ok(());
27    }
28    let create = Command::new("gh")
29        .args([
30            "label",
31            "create",
32            LABEL,
33            "--description",
34            "Java GATK4 vs gatk-rs HaplotypeCaller divergence (differential fuzzer)",
35            "--color",
36            "D93F0B",
37        ])
38        .output()
39        .context("gh label create")?;
40    // Ignore "already exists" races.
41    if !create.status.success() {
42        let err = String::from_utf8_lossy(&create.stderr);
43        if !err.to_ascii_lowercase().contains("already exists") {
44            bail!("gh label create failed: {err}");
45        }
46    }
47    Ok(())
48}

50/// Create an issue with the fixture path and divergence summary. Returns the issue URL.
51pub fn open_parity_issue(
52    fixture_dir: &Path,
53    scenario: &Scenario,
54    div: &Divergence,
55) -> Result<String> {
56    ensure_label()?;
57
58    let title = format!(
59        "parity-divergence: {} (seed={}, reads={}, reflen={})",
60        div.kind, scenario.seed, scenario.n_reads, scenario.ref_len
61    );
62
63    let body = format!(
64        "Automated report from `gatk-rs-equiv differential-fuzz` / `fuzz/run_hc_differential.sh`.\n\n\
65         ## Summary\n\n\
66         - **Kind:** `{kind}`\n\
67         - **Details:** {summary}\n\
68         - **Seed:** `{seed}`\n\
69         - **Reads / ref_len / read_len:** {n_reads} / {ref_len} / {read_len}\n\
70         - **java_only / rust_only / gt_mismatch / format_mismatch:** \
71           {java_only} / {rust_only} / {gt_mismatch} / {format_mismatch}\n\n\
72         ## Minimal fixture\n\n\
73         Path (in-repo):\n\n\
74         ```\n{fixture}\n```\n\n\
75         Contents: `scenario.json`, `reference.fa`, `reads.bam`, `java.vcf`, `rust.vcf`, `diverge.json`.\n\n\
76         ## Reproduce\n\n\
77         ```bash\n\
78         cargo run -p gatk-rs-equiv -- differential-fuzz \\\n\
79           --replay-fixture {fixture} \\\n\
80           --rust-binary ./target/release/gatk-rs\n\
81         ```\n\n\
82         Label: `{label}`\n",
83        kind = div.kind,
84        summary = div.summary,
85        seed = scenario.seed,
86        n_reads = scenario.n_reads,
87        ref_len = scenario.ref_len,
88        read_len = scenario.read_len,
89        java_only = div.java_only_sites,
90        rust_only = div.rust_only_sites,
91        gt_mismatch = div.gt_mismatch,
92        format_mismatch = div.format_mismatch_same_gt,
93        fixture = fixture_dir.display(),
94        label = LABEL,
95    );
96
97    let output = Command::new("gh")
98        .args([
99            "issue", "create", "--title", &title, "--body", &body, "--label", LABEL,
100        ])
101        .output()
102        .context("gh issue create")?;
103    if !output.status.success() {
104        bail!(
105            "gh issue create failed: {}",
106            String::from_utf8_lossy(&output.stderr)
107        );
108    }
109    let url = String::from_utf8_lossy(&output.stdout).trim().to_string();
110    if url.is_empty() {
111        bail!("gh issue create succeeded but returned empty URL");
112    }
113    Ok(url)
114}
