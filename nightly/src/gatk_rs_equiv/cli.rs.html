1//! clap CLI for `gatk-rs-equiv`.

3use clap::{Parser, Subcommand, ValueEnum};
4use std::path::PathBuf;
5
6#[derive(Debug, Clone, ValueEnum)]
7pub enum EngineChoice {
8    /// Prefer hap.py; fall back to RTG vcfeval if missing.
9    Auto,
10    /// Require Illumina hap.py (https://github.com/Illumina/hap.py).
11    Happy,
12    /// Require RTG Tools `vcfeval`.
13    Vcfeval,
14}
15
16#[derive(Parser, Debug)]
17#[command(
18    name = "gatk-rs-equiv",
19    version,
20    about = "Scientific HaplotypeCaller equivalence: gatk-rs vs GATK4 via hap.py / RTG vcfeval",
21    long_about = "Runs Java GATK4 HaplotypeCaller and gatk-rs HaplotypeCaller on identical inputs, \
22evaluates both callsets with hap.py (or RTG vcfeval) against the same truth set, \
23and reports Rust−Java F1 deltas plus a direct site comparison.\n\n\
24Independent community tool — not affiliated with the Broad Institute. \
25See gatk-rs-equiv/README.md for scope limits."
26)]
27pub struct Cli {
28    #[command(subcommand)]
29    pub command: Command,
30}

32#[derive(Subcommand, Debug)]
33pub enum Command {
34    /// Run both callers + equivalence engines; write results under --out.
35    Run(RunArgs),
36    /// Render Markdown + JSON report from a previous --out directory.
37    Report(ReportArgs),
38    /// Differential fuzzer: synthetic BAM → Java vs Rust → shrink → fixture (+ optional gh issue).
39    DifferentialFuzz(DiffFuzzCliArgs),
40}

42#[derive(Parser, Debug)]
43pub struct DiffFuzzCliArgs {
44    /// Number of random scenarios to evaluate (M4 default: small).
45    #[arg(long, default_value_t = 8)]
46    pub iterations: u32,
47
48    /// Work directory for intermediate BAM/VCF per iteration.
49    #[arg(long = "out", default_value = "target/diff-fuzz")]
50    pub out: PathBuf,
51
52    /// Path to the gatk-rs binary.
53    #[arg(long = "rust-binary", default_value = "target/release/gatk-rs")]
54    pub rust_binary: PathBuf,
55
56    /// Path to GATK4 package JAR (e.g. gatk-package-4.4.0.0-local.jar).
57    #[arg(long = "java-gatk-jar")]
58    pub java_gatk_jar: Option<PathBuf>,
59
60    /// Alternative: `gatk` launcher script/binary on PATH or absolute path.
61    #[arg(long = "java-gatk-bin")]
62    pub java_gatk_bin: Option<PathBuf>,
63
64    /// Where minimized regression fixtures are written.
65    #[arg(
66        long = "fixture-root",
67        default_value = "gatk-haplotypecaller/tests/fixtures/regressions"
68    )]
69    pub fixture_root: PathBuf,
70
71    /// Max |AD/DP| difference still treated as FORMAT-equal (default 0 = exact).
72    #[arg(long = "format-ad-tol", default_value_t = 0)]
73    pub format_ad_tol: u32,
74
75    /// Create a GitHub issue with label `parity-divergence` for each fixture (`gh` required).
76    #[arg(long = "open-github-issue", default_value_t = false)]
77    pub open_github_issue: bool,
78
79    /// Max shrink evaluation steps per divergence.
80    #[arg(long = "shrink-steps", default_value_t = 24)]
81    pub shrink_steps: usize,
82
83    /// Fixed campaign seed (optional; default: wall-clock).
84    #[arg(long = "seed")]
85    pub seed: Option<u64>,
86
87    /// Abort if fewer than this many GiB are free near --out.
88    #[arg(long = "min-free-gb", default_value_t = crate::resources::DEFAULT_MIN_FREE_GB)]
89    pub min_free_gb: u64,
90
91    /// Replay a previously written fixture directory (skips campaign).
92    #[arg(long = "replay-fixture")]
93    pub replay_fixture: Option<PathBuf>,
94}

96#[derive(Parser, Debug)]
97pub struct RunArgs {
98    /// Path to GATK4 package JAR (e.g. gatk-package-4.4.0.0-local.jar).
99    #[arg(long = "java-gatk-jar")]
100    pub java_gatk_jar: Option<PathBuf>,
101
102    /// Alternative: `gatk` launcher script/binary on PATH or absolute path.
103    #[arg(long = "java-gatk-bin")]
104    pub java_gatk_bin: Option<PathBuf>,
105
106    /// Path to the gatk-rs binary.
107    #[arg(long = "rust-binary", required = true)]
108    pub rust_binary: PathBuf,
109
110    /// Reference FASTA (indexed.fai required by both callers / hap.py).
111    #[arg(long = "reference", required = true)]
112    pub reference: PathBuf,
113
114    /// Input BAM/CRAM (indexed).
115    #[arg(long = "bam", required = true)]
116    pub bam: PathBuf,
117
118    /// Truth VCF (e.g. GIAB benchmark).
119    #[arg(long = "truth-vcf", required = true)]
120    pub truth_vcf: PathBuf,
121
122    /// High-confidence / evaluation regions BED.
123    #[arg(long = "confident-regions", required = true)]
124    pub confident_regions: PathBuf,
125
126    /// Output directory for VCFs, engine outputs, and reports.
127    #[arg(long = "out", required = true)]
128    pub out: PathBuf,
129
130    /// Optional calling / evaluation interval (GATK `-L` syntax, e.g. 20:10000000-10050000).
131    #[arg(long = "interval")]
132    pub interval: Option<String>,
133
134    /// Max allowed |rust_f1 − java_f1| for gate pass (default 0.02).
135    #[arg(long = "f1-delta-threshold", default_value = "0.02")]
136    pub f1_delta_threshold: f64,
137
138    /// Equivalence engine selection.
139    #[arg(long = "engine", value_enum, default_value_t = EngineChoice::Auto)]
140    pub engine: EngineChoice,
141
142    /// Path to hap.py executable (default: search PATH / HAPPY_BIN).
143    #[arg(long = "happy-bin")]
144    pub happy_bin: Option<PathBuf>,
145
146    /// Path to `rtg` launcher (default: search PATH / RTG_BIN).
147    #[arg(long = "rtg-bin")]
148    pub rtg_bin: Option<PathBuf>,
149
150    /// Extra stratification BED: NAME=path (repeatable). Name appears in reports.
151    #[arg(long = "stratification-bed", value_name = "NAME=PATH")]
152    pub stratification_bed: Vec<String>,
153
154    /// Threads for hap.py / callers (clamped to 1..=4; default 2 for 16GB hosts).
155    #[arg(long = "threads", default_value_t = crate::resources::DEFAULT_THREADS)]
156    pub threads: u32,
157
158    /// Abort if fewer than this many GiB are free near --out (default 8).
159    #[arg(long = "min-free-gb", default_value_t = crate::resources::DEFAULT_MIN_FREE_GB)]
160    pub min_free_gb: u64,
161
162    /// Skip disk free-space check (not recommended on laptop-class hosts).
163    #[arg(long = "skip-disk-check", default_value_t = false)]
164    pub skip_disk_check: bool,
165
166    /// Skip re-running callers if java.vcf / rust.vcf already exist under --out.
167    #[arg(long = "reuse-vcfs", default_value_t = false)]
168    pub reuse_vcfs: bool,
169
170    /// Write report after run (default true).
171    #[arg(long = "write-report", default_value_t = true)]
172    pub write_report: bool,
173}
