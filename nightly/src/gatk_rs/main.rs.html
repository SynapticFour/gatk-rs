359/// Join repeatable `-L` / `--intervals` args the way GATK unions multiple `-L` tokens.
360/// A single path (interval list file) stays a one-element list; multiple literals become
361/// a `;`-separated string for [`parse_intervals_cli_string`].
362fn join_interval_cli_args(intervals: &[String]) -> Option<String> {
363    if intervals.is_empty() {
364        None
365    } else {
366        Some(intervals.join(";"))
367    }
368}

370/// Resolve HC worker threads and initialize Rayon's global pool (via haplotypecaller).
371/// Explicit `--threads N` / `--nt N` (N>0) wins over ambient `RAYON_NUM_THREADS`
372/// `--threads 0` → auto-detect (`num_cpus`)
373/// Omitted → honor `RAYON_NUM_THREADS` if set, else `num_cpus`
374fn init_haplotype_caller_threads(threads: Option<usize>) -> usize {
375    let n = match threads {
376        Some(0) => num_cpus::get().max(1),
377        Some(n) => n.max(1),
378        None => std::env::var("RAYON_NUM_THREADS")
379            .ok()
380            .and_then(|s| s.parse::<usize>().ok())
381            .filter(|&v| v > 0)
382            .unwrap_or_else(|| num_cpus::get().max(1)),
383    };
384    init_worker_threads(n, threads.is_some());
385    info!("HaplotypeCaller worker threads: {n}");
386    n
387}

389fn main() {
390    // Initialize logging
391    init_logging();
392
393    let cli = match Cli::try_parse() {
394        Ok(cli) => cli,
395        Err(err) => {
396            let _ = err.print();
397            let code = match err.kind() {
398                ErrorKind::DisplayHelp | ErrorKind::DisplayVersion => 0,
399                _ => 2,
400            };
401            process::exit(code);
402        }
403    };
404
405    // Compatibility shim: do not imply a JVM / Broad GATK launch.
406    if cli.java_options.is_some() {
407        eprintln!(
408            "note: --java-options is accepted for Broad GATK launcher familiarity only; \
409gatk-rs is a native Rust binary and does not start a JVM \
410(not affiliated with, endorsed by, or supported by the Broad Institute)."
411        );
412    }
413
414    let result: Result<()> = (|| {
415        match cli.tool {
416            Tool::HaplotypeCaller {
417                reference,
418                input,
419                output,
420                intervals,
421                output_mode,
422                min_base_quality_score,
423                emit_ref_confidence,
424                pair_hmm,
425                original_base_qualities,
426                dont_use_soft_clipped_bases,
427                max_alternate_alleles,
428                min_mapping_quality,
429                stand_call_confidence,
430                stand_emit_confidence,
431                alleles,
432                threads,
433            } => {
434                let hc_run: gatk_common::GatkResult<()> = (|| {
435                    let n_threads = init_haplotype_caller_threads(threads);
436                    let mut config = match &cli.java_options {
437                        Some(java_opts) => {
438                            GatkConfig::with_java_options("HaplotypeCaller".to_string(), java_opts)?
439                        }
440                        None => GatkConfig::new("HaplotypeCaller".to_string()),
441                    };
442                    config.global_config.num_threads = Some(n_threads);
443
444                    for input_file in &input {
445                        config.add_input_file(input_file.clone());
446                    }
447                    config.set_reference(reference.clone());
448                    config.set_output_vcf(output.clone());
449
450                    if let Some(intervals_val) = join_interval_cli_args(&intervals) {
451                        let dict = SequenceDictionary::from_fasta_path(&reference)?;
452                        let _specs = parse_intervals_cli_string(&dict, &intervals_val)?;
453                        config.set_parameter("intervals".to_string(), intervals_val);
454                    }
455                    config.set_output_mode(output_mode.clone());
456                    config.set_parameter("output_mode".to_string(), output_mode);
457                    config.set_parameter(
458                        "min_base_quality_score".to_string(),
459                        min_base_quality_score.to_string(),
460                    );
461                    config.set_parameter(
462                        "min_mapping_quality".to_string(),
463                        min_mapping_quality.to_string(),
464                    );
465                    config.set_parameter(
466                        "max_alternate_alleles".to_string(),
467                        max_alternate_alleles.to_string(),
468                    );
469                    config.set_parameter(
470                        "stand_call_confidence".to_string(),
471                        stand_call_confidence.to_string(),
472                    );
473                    config.set_parameter(
474                        "stand_emit_confidence".to_string(),
475                        stand_emit_confidence.to_string(),
476                    );
477
478                    if let Some(alleles_vcf) = alleles {
479                        config.set_parameter("alleles".to_string(), alleles_vcf);
480                    }
481
482                    if let Some(emit_ref_conf) = emit_ref_confidence {
483                        config.set_parameter(
484                            "emit_ref_confidence".to_string(),
485                            emit_ref_conf.clone(),
486                        );
487                        // GATK -ERC GVCF / BP_RESOLUTION implies matching output mode when not overridden.
488                        match emit_ref_conf.as_str() {
489                            "GVCF" => config.set_output_mode("GVCF".to_string()),
490                            "BP_RESOLUTION" => config.set_output_mode("BP_RESOLUTION".to_string()),
491                            _ => {}
492                        }
493                    }
494                    if let Some(pair_hmm_val) = pair_hmm {
495                        config.set_parameter("pair_hmm".to_string(), pair_hmm_val);
496                    }
497                    if original_base_qualities {
498                        config.set_parameter(
499                            "original_base_qualities".to_string(),
500                            "true".to_string(),
501                        );
502                    }
503                    if dont_use_soft_clipped_bases {
504                        config.set_parameter(
505                            "dont_use_soft_clipped_bases".to_string(),
506                            "true".to_string(),
507                        );
508                    }
509
510                    config.validate()?;
511
512                    info!("Starting HaplotypeCaller");
513                    info!("Reference: {}", reference);
514                    info!("Input files: {:?}", input);
515                    info!("Output: {}", output);
516
517                    if let Some(ref java_opts) = config.java_options {
518                        info!("Java memory: {:?}", java_opts.memory);
519                        info!("Java GC: {:?}", java_opts.garbage_collector);
520                        if !java_opts.additional_args.is_empty() {
521                            info!("Additional Java args: {:?}", java_opts.additional_args);
522                        }
523                    }
524
525                    run_haplotype_caller(&config)
526                })();
527
528                match hc_run {
529                    Ok(()) => Ok(()),
530                    Err(e) => {
531                        eprintln!("{}", e.display_with_context());
532                        process::exit(gatk_cli_exit_code(&e));
533                    }
534                }
535            }
536
537            Tool::CountBasesInReference {
538                reference,
539                intervals,
540            } => {
541                let dict = SequenceDictionary::from_fasta_path(&reference).map_err(|e| {
542                    anyhow::anyhow!("Failed to derive sequence dictionary from reference: {e}")
543                })?;
544                let specs = match &intervals {
545                    None => dict.whole_genome_interval_specs(),
546                    Some(s) => parse_intervals_cli_string(&dict, s)
547                        .map_err(|e| anyhow::anyhow!("Invalid intervals: {e}"))?,
548                };
549                let counts = count_acgtn_histogram_for_interval_specs(&reference, &dict, &specs)
550                    .map_err(|e| anyhow::anyhow!("Count failed: {e}"))?;
551                let labels = ['A', 'C', 'G', 'T', 'N'];
552                for (i, &n) in counts.iter().enumerate() {
553                    if i == 4 && n == 0 {
554                        continue;
555                    }
556                    println!("{} : {}", labels[i], n);
557                }
558                Ok(())
559            }
560
561            Tool::CountReadsInRegion { input, region } => {
562                let (contig, start, end) = parse_single_region(&region)?;
563                let count = count_records_in_region_indexed(
564                    std::path::Path::new(&input),
565                    &contig,
566                    start,
567                    end,
568                )
569                .map_err(|e| anyhow::anyhow!("Indexed region count failed: {e}"))?;
570                println!("COUNT : {}", count);
571                Ok(())
572            }
573
574            Tool::ListReadsInRegion { input, region } => {
575                let (contig, start, end) = parse_single_region(&region)?;
576                let qnames =
577                    qnames_in_region_indexed(std::path::Path::new(&input), &contig, start, end)
578                        .map_err(|e| anyhow::anyhow!("Indexed region list failed: {e}"))?;
579                for q in qnames {
580                    println!("{q}");
581                }
582                Ok(())
583            }
584
585            Tool::PrintReads { input, output } => {
586                info!("Starting PrintReads");
587                info!("Input: {}", input);
588                info!("Output: {}", output);
589
590                let n = copy_alignments_with_htslib(
591                    std::path::Path::new(&input),
592                    std::path::Path::new(&output),
593                )?;
594                info!("PrintReads copied {} records", n);

596            }
597
598            Tool::CombineGvcfs {
599                reference,
600                variant,
601                output,
602                intervals,
603            } => {
604                info!("Starting CombineGVCFs");
605                info!("Reference: {}", reference);
606                info!("Inputs: {:?}", variant);
607                info!("Output: {}", output);
608                run_combine_gvcfs(&CombineGvcfsArgs {
609                    reference: PathBuf::from(reference),
610                    variant_paths: variant.into_iter().map(PathBuf::from).collect(),
611                    output: PathBuf::from(output),
612                    intervals: join_interval_cli_args(&intervals),
613                })
614                .map_err(|e| anyhow::anyhow!("{e}"))?;
615                Ok(())
616            }
617
618            Tool::VariantFiltration {
619                variant,
620                output,
621                reference,
622                filter_expression,
623                filter_name,
624                preset,
625            } => {
626                info!("Starting VariantFiltration");
627                let mut filters: Vec<FilterSpec> = Vec::new();
628                if let Some(p) = preset.as_deref() {
629                    match p.to_ascii_lowercase().as_str() {
630                        "snp" | "snps" => filters.extend(gatk_snp_hard_filters()),
631                        "indel" | "indels" => filters.extend(gatk_indel_hard_filters()),
632                        other => {
633                            return Err(anyhow::anyhow!(
634                                "unknown --preset '{other}' (use snp or indel)"
635                            ));
636                        }
637                    }
638                }
639                if !filter_expression.is_empty() || !filter_name.is_empty() {
640                    filters.extend(
641                        zip_filter_pairs(&filter_expression, &filter_name)
642                            .map_err(|e| anyhow::anyhow!("{e}"))?,
643                    );
644                }
645                if filters.is_empty() {
646                    return Err(anyhow::anyhow!(
647                        "VariantFiltration requires --preset snp|indel and/or \
648                         matching --filter-expression / --filter-name pairs"
649                    ));
650                }
651                run_variant_filtration(&VariantFiltrationArgs {
652                    variant: PathBuf::from(variant),
653                    output: PathBuf::from(output),
654                    filters,
655                    reference: reference.map(PathBuf::from),
656                })
657                .map_err(|e| anyhow::anyhow!("{e}"))?;
658                Ok(())
659            }
660
661            Tool::GenotypeGvcfs {
662                reference,
663                variant,
664                output,
665                intervals,
666                stand_call_conf,
667            } => {
668                info!("Starting GenotypeGVCFs");
669                info!("Reference: {}", reference);
670                info!("Input: {}", variant);
671                info!("Output: {}", output);
672                run_genotype_gvcfs(&GenotypeGvcfsArgs {
673                    reference: PathBuf::from(reference),
674                    variant: PathBuf::from(variant),
675                    output: PathBuf::from(output),
676                    intervals: join_interval_cli_args(&intervals),
677                    stand_call_conf,
678                    include_non_variant_sites: false,
679                })
680                .map_err(|e| anyhow::anyhow!("{e}"))?;
681                Ok(())
682            }
683
684            Tool::DumpSmoothedActivity {
685                reference,
686                input,
687                intervals,
688                output,
689            } => {
690                info!("Starting DumpSmoothedActivity");
691                let p = ReadFilterParams::default();
692                dump_smoothed_activity_tsv(
693                    Path::new(&reference),
694                    Path::new(&input),
695                    &intervals,
696                    Path::new(&output),
697                    &p,
698                )
699                .map_err(|e| anyhow::anyhow!("{e}"))?;
700                info!("Wrote smoothed activity TSV to {}", output);
701                Ok(())
702            }
703
704            Tool::FilterReads {
705                input,
706                output,
707                interval,
708                min_mapping_quality,
709                include_duplicates,
710            } => {
711                info!("Starting FilterReads");
712                info!("Input: {}", input);
713                info!("Output: {}", output);
714                let _ = include_duplicates; // reserved; PrintReads parity always drops duplicates
715                let out_format = if output.to_ascii_lowercase().ends_with(".sam") {
716                    bam::Format::Sam
717                } else {
718                    bam::Format::Bam
719                };
720                let mut kept = 0usize;
721                if let Some(reg) = interval {
722                    let (contig, start_1, end_1) = parse_single_region(&reg)?;
723                    let mut reader = bam::IndexedReader::from_path(&input)
724                        .map_err(|e| anyhow::anyhow!("Indexed BAM required for -L: {e}"))?;
725                    let header = bam::Header::from_template(reader.header());
726                    let tid = reader.header().tid(contig.as_bytes()).ok_or_else(|| {
727                        anyhow::anyhow!("Contig {contig} not found in BAM header")
728                    })?;
729                    let start0 = start_1 - 1;
730                    let end0 = end_1;
731                    reader
732                        .fetch((tid, start0 as i64, end0 as i64))
733                        .map_err(|e| anyhow::anyhow!("fetch failed: {e}"))?;
734                    let hdr_view = reader.header().clone();
735                    let mut writer =
736                        bam::Writer::from_path(&output, &header, out_format).map_err(|e| {
737                            anyhow::anyhow!("Failed to create output alignment file: {e}")
738                        })?;
739                    for rec in reader.records() {
740                        let rec =
741                            rec.map_err(|e| anyhow::anyhow!("Failed reading input record: {e}"))?;
742                        if passes_printreads_parity_filters(&rec, &hdr_view, min_mapping_quality) {
743                            writer.write(&rec).map_err(|e| {
744                                anyhow::anyhow!("Failed writing output record: {e}")
745                            })?;
746                            kept += 1;
747                        }
748                    }
749                } else {
750                    let mut reader = bam::Reader::from_path(&input)
751                        .map_err(|e| anyhow::anyhow!("Failed to open input alignment file: {e}"))?;
752                    let header = bam::Header::from_template(reader.header());
753                    let hdr_view = reader.header().clone();
754                    let mut writer =
755                        bam::Writer::from_path(&output, &header, out_format).map_err(|e| {
756                            anyhow::anyhow!("Failed to create output alignment file: {e}")
757                        })?;
758                    for rec in reader.records() {
759                        let rec =
760                            rec.map_err(|e| anyhow::anyhow!("Failed reading input record: {e}"))?;
761                        if passes_printreads_parity_filters(&rec, &hdr_view, min_mapping_quality) {
762                            writer.write(&rec).map_err(|e| {
763                                anyhow::anyhow!("Failed writing output record: {e}")
764                            })?;
765                            kept += 1;
766                        }
767                    }
768                }
769                println!("FilterReads kept {} records", kept);
770                Ok(())
771            }

773            Tool::ListTools => {
774                println!("Product spine (germline short variants vs GATK 4.4):");
775                println!(
776                    "  HaplotypeCaller       - Call germline SNPs and indels via local re-assembly"
777                );
778                println!(
779                    "  CombineGVCFs          - Merge per-sample gVCFs into a multi-sample gVCF"

786                );

789                );

792                println!("  BQSR, VQSR, Mutect2, gCNV/SV, Funcotator — not planned; do not expect stubs.");
793                println!();
794                println!("Experimental utilities (hidden from --help; prefer samtools/bcftools):");
795                println!("  PrintReads, FilterReads, CountBasesInReference, CountReadsInRegion,");
796                println!("  ListReadsInRegion, Validate, Benchmark");
797                println!();

801            }

804                // Do not impersonate Broad GATK. Expose our version and the pinned
805                // Java oracle version used for differential parity testing only.
806                println!(
807                    "gatk-rs {} (independent community project — not Broad Institute GATK)",

818            Tool::Validate {
819                input,
820                file_type,

824                let dictionary = if let Some(reference_path) = &reference {
825                    Some(
826                        SequenceDictionary::from_fasta_path(reference_path).map_err(|e| {
827                            anyhow::anyhow!("Reference dictionary creation failed: {e}")
828                        })?,
829                    )
830                } else {
831                    None
832                };
833
834                match file_type.to_uppercase().as_str() {
835                    "BAM" => {
836                        let count =
837                            validate_bam_file(std::path::Path::new(&input), dictionary.as_ref())
838                                .map_err(|e| anyhow::anyhow!("BAM validation failed: {e}"))?;
839                        println!(
840                            "BAM validation passed for file: {} ({} records)",
841                            input, count
842                        );
843                    }
844                    "SAM" => {
845                        let mut reader =
846                            gatk_core::io::SamReader::from_file(&input).map_err(|e| {
847                                anyhow::anyhow!("SAM validation failed to open file: {e}")
848                            })?;
849                        if let Some(dict) = &dictionary {
850                            dict.validate_sam_header(reader.header()).map_err(|e| {
851                                anyhow::anyhow!("SAM/reference dictionary mismatch: {e}")
852                            })?;
853                        }
854                        let semantics = ReadHeaderSemantics::from_sam_header_text(
855                            &sam_header_to_text(reader.header()),
856                        )
857                        .map_err(|e| anyhow::anyhow!("SAM header semantics invalid: {e}"))?;
858
859                        let mut records_seen = 0usize;
860                        while let Some(record) = reader.read_next_record().map_err(|e| {
861                            anyhow::anyhow!("SAM validation failed while reading: {e}")
862                        })? {
863                            records_seen += 1;
864                            if record.cigar != "*" && record.seq != "*" {
865                                let expected_seq_bases =
866                                    query_bases_consumed_by_cigar(&record.cigar).map_err(|e| {
867                                        anyhow::anyhow!(
868                                            "SAM validation failed for record {}: {e}",
869                                            record.qname
870                                        )
871                                    })?;
872                                if expected_seq_bases != record.seq.len() {
873                                    return Err(anyhow::anyhow!(
874                                        "SAM validation failed for record {}: CIGAR/SEQ length mismatch (expected {} query bases from CIGAR '{}', observed SEQ length {})",
875                                        record.qname,
876                                        expected_seq_bases,
877                                        record.cigar,
878                                        record.seq.len()
879                                    ));
880                                }
881                            }
882                            let rg_tag = record.get_optional_value("RG");
883                            let pg_tag = record.get_optional_value("PG");
884                            semantics
885                                .validate_record_links(rg_tag.as_deref(), pg_tag.as_deref())
886                                .map_err(|e| {
887                                    anyhow::anyhow!(
888                                        "SAM read-header semantics failed for record {}: {e}",
889                                        record.qname
890                                    )
891                                })?;
892                        }
893                        if records_seen == 0 {
894                            return Err(anyhow::anyhow!(
895                                "SAM validation failed while reading: no alignment records found"
896                            ));
897                        }
898                        println!("SAM validation passed for file: {}", input);
899                    }
900                    "VCF" => {
901                        let mut reader = VcfReader::from_file(&input).map_err(|e| {
902                            anyhow::anyhow!("VCF validation failed to open file: {e}")
903                        })?;
904                        if let Some(dict) = &dictionary {
905                            dict.validate_vcf_header(reader.header()).map_err(|e| {
906                                anyhow::anyhow!("VCF/reference dictionary mismatch: {e}")
907                            })?;
908                        }
909                        let _ = reader.read_next_record().map_err(|e| {
910                            anyhow::anyhow!("VCF validation failed while reading: {e}")
911                        })?;
912                        println!("VCF validation passed for file: {}", input);
913                    }
914                    "FASTA" => {
915                        let mut reader = FastaReader::from_file(&input).map_err(|e| {
916                            anyhow::anyhow!("FASTA validation failed to open file: {e}")
917                        })?;
918                        let _ = reader.read_next_sequence().map_err(|e| {
919                            anyhow::anyhow!("FASTA validation failed while reading: {e}")
920                        })?;
921                        println!("FASTA validation passed for file: {}", input);
922                    }
923                    _ => {
924                        return Err(anyhow::anyhow!("Unsupported file type: {}", file_type));
925                    }
926                }
927
928                Ok(())
929            }
930
931            Tool::Benchmark { args } => {
932                info!("Running benchmarking command");
933                run_benchmarking_command(args)?;
934                Ok(())
935            }
936        }
937    })();
938
939    if let Err(e) = result {
940        eprintln!("{e}");
941        // Match GATK user-facing failure convention.
942        process::exit(2);
943    }
944}

955fn sam_header_to_text(header: &gatk_core::io::SamHeader) -> String {
956    let mut lines = Vec::new();
957    lines.push("@HD\tVN:1.6".to_string());
958    for rg in &header.read_groups {
959        let mut line = format!("@RG\tID:{}", rg.id);
960        if let Some(sm) = &rg.sample {
961            line.push_str(&format!("\tSM:{sm}"));
962        }
963        lines.push(line);
964    }
965    for pg in &header.programs {
966        lines.push(format!("@PG\tID:{}", pg.id));
967    }
968    lines.join("\n")
969}

971fn parse_single_region(region: &str) -> Result<(String, u64, u64)> {
972    let (contig, span) = region
973        .split_once(':')
974        .ok_or_else(|| anyhow::anyhow!("Invalid region '{}': expected contig:start-end", region))?;
975    let span = span.replace(',', "");
976    let (start_s, end_s) = span
977        .split_once('-')
978        .ok_or_else(|| anyhow::anyhow!("Invalid region '{}': expected start-end", region))?;
979    let start = start_s
980        .parse::<u64>()
981        .map_err(|_| anyhow::anyhow!("Invalid region start '{}'", start_s))?;
982    let end = end_s
983        .parse::<u64>()
984        .map_err(|_| anyhow::anyhow!("Invalid region end '{}'", end_s))?;
985    if start == 0 {
986        return Err(anyhow::anyhow!(
987            "Invalid region '{}': start must be >= 1",
988            region
989        ));
990    }
991    if start > end {
992        return Err(anyhow::anyhow!(
993            "Invalid region '{}': start must be <= end",
994            region
995        ));
996    }
997    Ok((contig.to_string(), start, end))
998}

1000fn query_bases_consumed_by_cigar(cigar: &str) -> Result<usize> {
1001    let mut total = 0usize;
1002    let mut num = String::new();
1003    for ch in cigar.chars() {
1004        if ch.is_ascii_digit() {
1005            num.push(ch);
1006            continue;
1007        }
1008        if num.is_empty() {
1009            return Err(anyhow::anyhow!(
1010                "Invalid CIGAR '{}': missing length before op '{}'",
1011                cigar,
1012                ch
1013            ));
1014        }
1015        let len: usize = num
1016            .parse()
1017            .map_err(|_| anyhow::anyhow!("Invalid CIGAR '{}': bad op length '{}'", cigar, num))?;
1018        match ch {
1019            'M' | 'I' | 'S' | '=' | 'X' => total += len,
1020            'D' | 'N' | 'H' | 'P' => {}
1021            _ => {
1022                return Err(anyhow::anyhow!(
1023                    "Invalid CIGAR '{}': unknown op '{}'",
1024                    cigar,
1025                    ch
1026                ))
1027            }
1028        }
1029        num.clear();
1030    }
1031    if !num.is_empty() {
1032        return Err(anyhow::anyhow!(
1033            "Invalid CIGAR '{}': trailing length without op '{}'",
1034            cigar,
1035            num
1036        ));
1037    }
1038    Ok(total)
